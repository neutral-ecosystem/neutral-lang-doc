6//! validated reader views. It must remain independent of the compiler, private

29/// Deterministic generic summary of one validated Neutral document.
30#[derive(Clone, Debug, Eq, PartialEq)]
31pub struct ProbeSummary {
32    /// Logical module name.
33    module: String,
34    /// Logical, source-map, derivation, resource, and identity metadata.
35    metadata: Vec<String>,
36    /// Exact captured vocabulary identity summary, when present.
37    vocabulary: Option<String>,
38    /// Vocabulary-owned type summaries in reader order.
39    vocabulary_types: Vec<String>,
40    /// Nominal record schema summaries in reader order.
41    record_types: Vec<String>,
42    /// Typed declaration summaries in reader order.
43    declarations: Vec<String>,
44    /// Original-byte source mappings in element order.
45    source_mappings: Vec<String>,
46    /// Root value-origin and normalization summaries in compiler order.
47    value_provenance: Vec<String>,
48    /// Explicit/default field-provenance summaries in compiler order.
49    field_provenance: Vec<String>,
50    /// Ordinary immutable-value reuse edges in compiler order.
51    reuse_provenance: Vec<String>,
52    /// Typed identity-reference edges in compiler order.
53    reference_provenance: Vec<String>,
54    /// Safe diagnostic-code summaries.
55    diagnostics: Vec<String>,
56}

132/// Traverses only immutable public reader views to summarize a document.
133#[must_use]
134pub fn summarize(document: &ValidatedDocument) -> ProbeSummary {
135    let artifacts = document.artifacts();
136    let (vocabulary, vocabulary_types) = summarize_vocabulary(document);
137    let metadata = summarize_metadata(document);
138    let record_types = summarize_record_types(document);
139    let declarations = summarize_declarations(document);
140    let source_mappings = summarize_source_mappings(document);
141    let value_provenance = artifacts
142        .provenance()
143        .iter()
144        .map(|record| {
145            format!(
146                "{}:{}:{}",
147                record.element_id().get(),
148                record.origin().as_str(),
149                record.normalization().as_str()
150            )
151        })
152        .collect();
153    let field_provenance = artifacts
154        .field_provenance()
155        .iter()
156        .map(|record| {
157            format!(
158                "{}:{}:{}",
159                record.element_id().get(),
160                record.field_path().join("."),
161                record.origin().as_str()
162            )
163        })
164        .collect();
165    let reuse_provenance = artifacts
166        .reuse_provenance()
167        .iter()
168        .map(|record| {
169            format!(
170                "{}:{}:{}",
171                record.element_id().get(),
172                record.value_path().join("."),
173                record.source_element_id().get()
174            )
175        })
176        .collect();
177    let reference_provenance = artifacts
178        .reference_provenance()
179        .iter()
180        .map(|record| {
181            format!(
182                "{}:{}:{}",
183                record.element_id().get(),
184                record.value_path().join("."),
185                record.target_element_id().get()
186            )
187        })
188        .collect();
189    ProbeSummary {
190        module: document.module_name().to_owned(),
191        metadata,
192        vocabulary,
193        vocabulary_types,
194        record_types,
195        declarations,
196        source_mappings,
197        value_provenance,
198        field_provenance,
199        reuse_provenance,
200        reference_provenance,
201        diagnostics: Vec::new(),
202    }
203}

205/// Summarizes version, source, derivation, resource, and identity metadata.
206fn summarize_metadata(document: &ValidatedDocument) -> Vec<String> {
207    let artifacts = document.artifacts();
208    let logical = artifacts.logical_document();
209    let source_map = artifacts.source_map();
210    let derivation = artifacts.derivation();
211    let acceptance = derivation.acceptance();
212    let resources = derivation.resource_facts();
213    let mut metadata = vec![
214        format!(
215            "language-behavior {}",
216            logical.module().language_behavior_version()
217        ),
218        format!(
219            "logical-ir-schema {}",
220            derivation.logical_ir_schema_version()
221        ),
222        format!("source-map-schema {}", derivation.source_map_version()),
223        format!("provenance-schema {}", derivation.provenance_version()),
224        format!("source-digest {}", source_map.source_digest()),
225        format!("source-bytes {}", source_map.source_byte_length()),
226        format!(
227            "module-span {}..{}",
228            source_map.module_span().start(),
229            source_map.module_span().end()
230        ),
231        format!(
232            "resource-facts source-bytes={} declarations={} diagnostics={} decoded-string-bytes={}",
233            resources.source_bytes(),
234            resources.declarations(),
235            resources.diagnostics(),
236            resources.decoded_string_bytes()
237        ),
238        format!(
239            "acceptance source-bytes={} diagnostics={} string-bytes={} numeric-digits={} numeric-scale={} declarations={} record-fields={} nesting-depth={} list-items={} traversal-nodes={}",
240            acceptance.source_byte_limit(),
241            acceptance.diagnostic_limit(),
242            acceptance.string_byte_limit(),
243            acceptance.numeric_digit_limit(),
244            acceptance.numeric_scale_limit(),
245            acceptance.declaration_limit(),
246            acceptance.record_field_limit(),
247            acceptance.nesting_depth_limit(),
248            acceptance.list_item_limit(),
249            acceptance.traversal_node_limit()
250        ),
251        format!(
252            "safe-bounded-output {}",
253            derivation.diagnostics().safe_bounded_output()
254        ),
255    ];
256    metadata.extend(document.record_types().iter().map(|record| {
257        format!(
258            "record-identity id={} symbol={}::{} fingerprint={}",
259            record.element_id().get(),
260            record.symbol_identity().module().module_name(),
261            record.symbol_identity().declaration_name(),
262            record.fingerprint().digest()
263        )
264    }));
265    metadata.extend(document.declarations().iter().map(|declaration| {
266        format!(
267            "declaration-identity id={} symbol={}::{} fingerprint={}",
268            declaration.element_id().get(),
269            declaration.symbol_identity().module().module_name(),
270            declaration.symbol_identity().declaration_name(),
271            declaration.fingerprint().digest()
272        )
273    }));
274    metadata
275}

277/// Summarizes nominal record schemas in deterministic reader order.
278fn summarize_record_types(document: &ValidatedDocument) -> Vec<String> {
279    document
280        .record_types()
281        .iter()
282        .map(|record| {
283            let fields = record
284                .fields()
285                .iter()
286                .map(|field| match field.default_value() {
287                    Some(default) => {
288                        format!("{}: {} = {}", field.name(), field.resolved_type(), default)
289                    }
290                    None => format!("{}: {}", field.name(), field.resolved_type()),
291                })
292                .collect::<Vec<_>>()
293                .join(", ");
294            format!("record {} {{ {fields} }}", record.name())
295        })
296        .collect()
297}

299/// Summarizes declarations with resolved types and final logical values.
300fn summarize_declarations(document: &ValidatedDocument) -> Vec<String> {
301    document
302        .declarations()
303        .iter()
304        .map(|declaration| {
305            format!(
306                "{}: {} = {}",
307                declaration.name(),
308                declaration.resolved_type(),
309                declaration.value()
310            )
311        })
312        .collect()
313}

315/// Summarizes every original-byte source-map entry in element order.
316fn summarize_source_mappings(document: &ValidatedDocument) -> Vec<String> {
317    document
318        .artifacts()
319        .source_map()
320        .entries()
321        .iter()
322        .map(|entry| {
323            format!(
324                "{}:declaration={}..{}:type={}..{}:name={}..{}:value={}..{}",
325                entry.element_id().get(),
326                entry.declaration_span().start(),
327                entry.declaration_span().end(),
328                entry.type_span().start(),
329                entry.type_span().end(),
330                entry.name_span().start(),
331                entry.name_span().end(),
332                entry.value_span().start(),
333                entry.value_span().end()
334            )
335        })
336        .collect()
337}

339/// Renders a summary as deterministic, categorized consumer observations.
340#[must_use]
341pub fn render_summary(summary: &ProbeSummary) -> Vec<String> {
342    let mut lines = vec![format!("module {}", summary.module())];
343    lines.extend(
344        summary
345            .metadata()
346            .iter()
347            .map(|value| format!("metadata {value}")),
348    );
349    if let Some(vocabulary) = summary.vocabulary() {
350        lines.push(format!("vocabulary {vocabulary}"));
351    }
352    lines.extend(
353        summary
354            .record_types()
355            .iter()
356            .map(|record| format!("record {record}")),
357    );
358    lines.extend(
359        summary
360            .vocabulary_types()
361            .iter()
362            .map(|record| format!("vocabulary-type {record}")),
363    );
364    lines.extend(
365        summary
366            .declarations()
367            .iter()
368            .map(|declaration| format!("declaration {declaration}")),
369    );
370    lines.extend(
371        summary
372            .source_mappings()
373            .iter()
374            .map(|record| format!("source-map {record}")),
375    );
376    lines.extend(
377        summary
378            .value_provenance()
379            .iter()
380            .map(|record| format!("value-provenance {record}")),
381    );
382    lines.extend(
383        summary
384            .field_provenance()
385            .iter()
386            .map(|record| format!("field-provenance {record}")),
387    );
388    lines.extend(
389        summary
390            .reuse_provenance()
391            .iter()
392            .map(|record| format!("reuse-provenance {record}")),
393    );
394    lines.extend(
395        summary
396            .reference_provenance()
397            .iter()
398            .map(|record| format!("reference-provenance {record}")),
399    );
400    lines.extend(
401        summary
402            .diagnostics()
403            .iter()
404            .map(|diagnostic| format!("diagnostic {diagnostic}")),
405    );
406    lines
407}

409/// Decodes hostile external bytes and returns only a validated generic summary.
410///
411/// # Errors
412///
413/// Returns the decoder's stable bounded failure when the artifact is malformed,
414/// unsupported, oversized, inconsistent, or cancelled.
415pub fn inspect_encoded(
416    bytes: &[u8],
417    limits: DecodeLimits,
418    cancellation: &CancellationToken,
419) -> Result<ProbeSummary, DecodeError> {
420    decode(bytes, limits, cancellation).map(|document| summarize(&document))
421}

423/// Summarizes exact vocabulary identity and schema data without interpretation.
424fn summarize_vocabulary(document: &ValidatedDocument) -> (Option<String>, Vec<String>) {
425    let vocabulary = document.vocabulary().map(|contract| {
426        let identity = contract.identity();
427        format!(
428            "{}@{} schema={} encoding={} digest={} features=[{}]",
429            identity.identity(),
430            identity.version(),
431            identity.schema_version(),
432            identity.encoding_version(),
433            identity.content_digest(),
434            identity.required_features().join(",")
435        )
436    });
437    let types = document.vocabulary().map_or_else(Vec::new, |contract| {
438        contract
439            .types()
440            .iter()
441            .map(|definition| {
442                let fields = definition
443                    .fields()
444                    .iter()
445                    .map(|field| match field.default_value() {
446                        Some(default) => {
447                            format!("{}: {} = {}", field.name(), field.resolved_type(), default)
448                        }
449                        None => format!("{}: {}", field.name(), field.resolved_type()),
450                    })
451                    .collect::<Vec<_>>()
452                    .join(", ");
453                format!("record {} {{ {fields} }}", definition.identity())
454            })
455            .collect()
456    });
457    (vocabulary, types)
458}

460/// Creates a consumer-owned diagnostic mapped through the public source map.
461///
462/// # Errors
463///
464/// Returns an error when `element_id` does not exist in the validated document.
465pub fn source_linked_diagnostic(
466    document: &ValidatedDocument,
467    element_id: ElementId,
468) -> Result<Diagnostic, ProbeError> {
469    let primary = document
470        .source_location(element_id)
471        .ok_or(ProbeError::UnknownElement)?;
472    let code =
473        DiagnosticCode::new(diagnostics::OBSERVATION).map_err(|_| ProbeError::InternalInvariant)?;
474    Ok(Diagnostic::new(
475        code,
476        DiagnosticLayer::Consumer,
477        DiagnosticSeverity::Note,
478        primary,
479        Vec::new(),
480        false,
481    ))
482}

484/// A bounded generic-probe traversal failure.
485#[derive(Clone, Copy, Debug, Eq, PartialEq)]
486pub enum ProbeError {
487    /// The requested graph-local element was absent.
488    UnknownElement,
489    /// A frozen project-owned diagnostic contract was internally invalid.
490    InternalInvariant,
491}
