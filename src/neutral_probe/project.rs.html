13/// Atomically classified project inspection failure, with no partial summary.
14#[derive(Clone, Copy, Debug, Eq, PartialEq)]
15pub enum ProjectProbeError {
16    /// External byte/schema/reader validation failed.
17    Decode(DecodeError),
18    /// Post-validation root selection failed.
19    View(ProjectReadError),
20    /// Bounded complete-project identity construction failed.
21    Identity(IdentityError),
22}
23
24/// Complete-project counts and a selected public view, never private provenance.
25#[derive(Clone, Debug, Eq, PartialEq)]
26pub struct ProjectProbeSummary {
27    /// Complete module count, not a list of private module names.
28    pub modules: u64,
29    /// Complete declaration count, independent of root selection.
30    pub declarations: u64,
31    /// Public-only interpretive dependency closure and materialized values.
32    pub view: ProjectView,
33    /// Complete public-interface fingerprint, not a complete project identity.
34    pub interface_fingerprint: String,
35    /// Complete IR contract identification.
36    pub schema: String,
37    /// Typed complete logical identity, never a public-view fingerprint.
38    pub logical_identity: LogicalProjectIdentity,
39    /// Frozen identity profile, independent of package and transport versions.
40    pub identity_profile: &'static str,
41}

43/// Inspects external project bytes without compiler linkage or host acquisition.
44///
45/// An absent selection means all public exports; an explicit empty selection
46/// means an empty view. Complete IR is validated regardless of selected roots.
47///
48/// # Errors
49/// Rejects malformed bytes or invalid/private/duplicate roots atomically.
50pub fn inspect_project_encoded(
51    bytes: &[u8],
52    wire_limits: DecodeLimits,
53    project_limits: ProjectLimits,
54    roots: Option<&[String]>,
55    cancellation: &CancellationToken,
56) -> Result<ProjectProbeSummary, ProjectProbeError> {
57    let project = decode_project(bytes, wire_limits, project_limits, cancellation)
58        .map_err(ProjectProbeError::Decode)?;
59    let ir = project.complete_ir();
60    let roots = if let Some(names) = roots {
61        if names.len() > ir.public_interface.exports().len() {
62            return Err(ProjectProbeError::View(ProjectReadError::Limit));
63        }
64        let index = ir
65            .public_interface
66            .exports()
67            .iter()
68            .map(|export| (symbol_name(export.identity()), export.identity()))
69            .collect::<BTreeMap<_, _>>();
70        names
71            .iter()
72            .map(|name| {
73                if cancellation.is_cancelled() {
74                    return Err(ProjectProbeError::View(ProjectReadError::Cancelled));
75                }
76                index
77                    .get(name)
78                    .map(|identity| (*identity).clone())
79                    .ok_or(ProjectProbeError::View(ProjectReadError::View))
80            })
81            .collect::<Result<Vec<_>, _>>()?
82    } else {
83        ir.public_interface
84            .exports()
85            .iter()
86            .map(|export| export.identity().clone())
87            .collect()
88    };
89    let view = project
90        .derive_view(
91            &ViewRequest {
92                schema: neutral_reader::PROJECT_VIEW_SCHEMA.to_owned(),
93                roots,
94            },
95            cancellation,
96        )
97        .map_err(ProjectProbeError::View)?;
98    let logical_identity = project
99        .logical_identity(
100            IdentityLimits {
101                bytes: MAX_TRANSCRIPT_BYTES,
102                nodes: MAX_TRANSCRIPT_NODES,
103            },
104            cancellation,
105        )
106        .map_err(ProjectProbeError::Identity)?
107        .identity();
108    Ok(summarize_project(&project, view, logical_identity))
109}

111/// Returns a bounded trusted projection; source IDs, spans and private roots are absent.
112fn summarize_project(
113    project: &ValidatedProject,
114    view: ProjectView,
115    logical_identity: LogicalProjectIdentity,
116) -> ProjectProbeSummary {
117    let ir = project.complete_ir();
118    ProjectProbeSummary {
119        modules: ir.resources.source_units,
120        declarations: ir.resources.declarations,
121        view,
122        interface_fingerprint: ir.public_interface.fingerprint().to_string(),
123        schema: ir.schema.clone(),
124        logical_identity,
125        identity_profile: IDENTITY_PROFILE,
126    }
127}

129/// Formats one public module-symbol identity for root selection and inspection.
130fn symbol_name(symbol: &neutral_reader::ModuleSymbolIdentity) -> String {
131    format!(
132        "{}::{}",
133        symbol.module().module_name(),
134        symbol.declaration_name()
135    )
136}

138/// Renders indented JSON with the same escaping rules as document inspection.
139#[must_use]
140pub fn render_project_summary_json(summary: &ProjectProbeSummary) -> String {
141    let roots = summary
142        .view
143        .roots()
144        .iter()
145        .map(symbol_name)
146        .collect::<Vec<_>>();
147    let exports = summary
148        .view
149        .exports()
150        .iter()
151        .map(|export| symbol_name(export.identity()))
152        .collect::<Vec<_>>();
153    let values = summary
154        .view
155        .values()
156        .iter()
157        .map(|(symbol, value)| format!("{} = {value:?}", symbol_name(symbol)))
158        .collect::<Vec<_>>();
159    let vocabularies = summary
160        .view
161        .vocabulary_records()
162        .iter()
163        .map(|record| format!("{}@{}::{}", record.identity, record.version, record.name))
164        .collect::<Vec<_>>();
165    let modules = summary.modules.to_string();
166    let declarations = summary.declarations.to_string();
167    let logical_identity = summary.logical_identity.to_string();
168    render_fields_json(&[
169        Field {
170            json_key: "schema",
171            text_prefix: "schema",
172            value: FieldValue::Text(&summary.schema),
173        },
174        Field {
175            json_key: "view_schema",
176            text_prefix: "view",
177            value: FieldValue::Text(summary.view.schema()),
178        },
179        Field {
180            json_key: "modules",
181            text_prefix: "modules",
182            value: FieldValue::Text(&modules),
183        },
184        Field {
185            json_key: "declarations",
186            text_prefix: "declarations",
187            value: FieldValue::Text(&declarations),
188        },
189        Field {
190            json_key: "interface_fingerprint",
191            text_prefix: "interface",
192            value: FieldValue::Text(&summary.interface_fingerprint),
193        },
194        Field {
195            json_key: "identity_profile",
196            text_prefix: "identity-profile",
197            value: FieldValue::Text(summary.identity_profile),
198        },
199        Field {
200            json_key: "logical_identity",
201            text_prefix: "logical-identity",
202            value: FieldValue::Text(&logical_identity),
203        },
204        Field {
205            json_key: "roots",
206            text_prefix: "root",
207            value: FieldValue::TextList(&roots),
208        },
209        Field {
210            json_key: "exports",
211            text_prefix: "export",
212            value: FieldValue::TextList(&exports),
213        },
214        Field {
215            json_key: "values",
216            text_prefix: "value",
217            value: FieldValue::TextList(&values),
218        },
219        Field {
220            json_key: "vocabulary_types",
221            text_prefix: "vocabulary",
222            value: FieldValue::TextList(&vocabularies),
223        },
224    ])
225}
