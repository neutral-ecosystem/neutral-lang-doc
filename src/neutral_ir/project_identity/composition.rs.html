22/// Frames complete successor meaning without accepting source evidence as logical content.
23///
24/// # Errors
25/// Rejects malformed canonical order, schema, bounds and cancellation. This is
26/// structural framing, not a substitute for independent semantic reader validation.
27pub fn canonical_composition_project(
28    ir: &CompositionProjectIr,
29    limits: IdentityLimits,
30    cancellation: &CancellationToken,
31) -> Result<IdentityTranscript<CompositionLogicalIdentity>, E> {
32    build(
33        ir,
34        limits,
35        cancellation,
36        false,
37        profile::LOGICAL_DOMAIN,
38        CompositionLogicalIdentity,
39    )
40}

42/// Recomputes the successor public interpretive projection, excluding private implementation facts.
43///
44/// # Errors
45/// Rejects malformed structure, bounds or cancellation; semantic public closure
46/// must be established by the caller's independent reader before trusting this digest.
47pub fn composition_interface(
48    ir: &CompositionProjectIr,
49    limits: IdentityLimits,
50    cancellation: &CancellationToken,
51) -> Result<IdentityTranscript<SemanticDigest>, E> {
52    build(
53        ir,
54        limits,
55        cancellation,
56        true,
57        profile::INTERFACE_DOMAIN,
58        |digest| digest,
59    )
60}

62/// Selects only versioned framing; package releases and host mappings never choose semantics.
63fn build<I>(
64    ir: &CompositionProjectIr,
65    limits: IdentityLimits,
66    cancellation: &CancellationToken,
67    public: bool,
68    domain: &str,
69    wrap: impl FnOnce(SemanticDigest) -> I,
70) -> Result<IdentityTranscript<I>, E> {
71    if ir.schema != profile::PROJECT_IR_SCHEMA || ir.modules.is_empty() {
72        return Err(E::InvalidInput);
73    }
74    inspect_composition_ir(ir, ir.limits, ir.composition_limits, cancellation).map_err(
75        |e| match e {
76            crate::composition::project::CompositionShapeError::Cancelled => E::Cancelled,
77            crate::composition::project::CompositionShapeError::Limit => E::Limit,
78        },
79    )?;
80    transcript_profile(
81        domain,
82        profile::IDENTITY_PROFILE,
83        limits,
84        cancellation,
85        |w| {
86            ordered(ir.modules.iter().map(|m| &m.identity))?;
87            ordered(ir.declarations.iter().map(|d| &d.identity))?;
88            ordered(
89                ir.vocabularies
90                    .iter()
91                    .map(|b| (b.identity.identity(), b.identity.version())),
92            )?;
93            w.leaf("schema", ir.schema.as_bytes())?;
94            w.leaf("profile", V1_SOURCE_PROFILE.as_bytes())?;
95            modules(w, ir)?;
96            declarations(w, ir, public)?;
97            vocabulary_types(w, ir, public)?;
98            catalogues(w, ir)?;
99            dependencies(w, ir)?;
100            // Borrow canonical declarations rather than allocating a visibility index.
101            let visible = |owner: &crate::ModuleSymbolIdentity| {
102                ir.declarations
103                    .binary_search_by(|d| d.identity.cmp(owner))
104                    .is_ok_and(|index| ir.declarations[index].public)
105            };
106            w.items(ir.provenance.len())?;
107            let mut edges = Vec::new();
108            for e in &ir.provenance {
109                w.check()?;
110                if visible(&e.from)
111                    && visible(&e.to)
112                    && matches!(
113                        e.kind,
114                        ProjectPublicEdgeKind::Type
115                            | ProjectPublicEdgeKind::ReferenceType
116                            | ProjectPublicEdgeKind::Reference
117                    )
118                {
119                    for owner in [&e.from, &e.to] {
120                        w.text(owner.module().module_name())?;
121                        w.text(owner.declaration_name())?;
122                    }
123                    edges.try_retain(1).map_err(|_| E::Limit)?;
124                    edges.push((&e.from, e.kind, &e.to));
125                }
126            }
127            // Multiple source occurrences contribute one semantic edge, unchanged from the set projection.
128            edges.sort_unstable();
129            edges.dedup();
130            w.check()?;
131            w.frame("public-edges", |w| {
132                for (from, kind, to) in edges {
133                    w.frame("edge", |w| {
134                        w.frame("from", |w| symbol(w, from))?;
135                        w.frame("to", |w| symbol(w, to))?;
136                        w.leaf("kind", kind.spelling().as_bytes())
137                    })?;
138                }
139                Ok(())
140            })
141        },
142        wrap,
143    )
144}

146/// Frames the canonical modules section under the selected projection.
147fn modules(w: &mut Writer<'_>, ir: &CompositionProjectIr) -> Result<(), E> {
148    w.frame("modules", |w| {
149        for m in &ir.modules {
150            if m.identity.language_behavior_version() != V1_SOURCE_PROFILE
151                || m.identity.module_name().is_empty()
152            {
153                return Err(E::InvalidInput);
154            }
155            ordered(&m.imports)?;
156            w.frame("module", |w| {
157                w.leaf("name", m.identity.module_name().as_bytes())?;
158                w.frame("imports", |w| {
159                    for i in &m.imports {
160                        w.leaf("target", i.as_bytes())?;
161                    }
162                    Ok(())
163                })
164            })?;
165        }
166        Ok(())
167    })
168}

170/// Frames the canonical declarations section under the selected projection.
171fn declarations(w: &mut Writer<'_>, ir: &CompositionProjectIr, public: bool) -> Result<(), E> {
172    w.frame("declarations", |w| {
173        for d in ir.declarations.iter().filter(|d| !public || d.public) {
174            if matches!(d.signature, S::Binding(_)) != d.value.is_some() {
175                return Err(E::InvalidInput);
176            }
177            w.frame("declaration", |w| {
178                symbol(w, &d.identity)?;
179                w.leaf("public", &[u8::from(d.public)])?;
180                w.frame("signature", |w| match &d.signature {
181                    S::Binding(t) => w.frame("binding", |w| type_transcript(w, t, 0)),
182                    S::Definition(b) => body(w, b),
183                })?;
184                w.frame("value", |w| match &d.value {
185                    Some(v) => value(w, v, 0),
186                    None => w.leaf("absent", &[]),
187                })
188            })?;
189        }
190        Ok(())
191    })
192}

194/// Frames the canonical vocabulary types section under the selected projection.
195fn vocabulary_types(w: &mut Writer<'_>, ir: &CompositionProjectIr, public: bool) -> Result<(), E> {
196    w.frame("vocabulary-types", |w| {
197        for bundle in &ir.vocabularies {
198            for d in bundle.definitions.iter().filter(|d| !public || d.public) {
199                w.frame("definition", |w| {
200                    w.leaf("identity", bundle.identity.identity().as_bytes())?;
201                    w.leaf("version", bundle.identity.version().as_bytes())?;
202                    w.leaf("name", d.name.as_bytes())?;
203                    w.leaf("public", &[u8::from(d.public)])?;
204                    w.frame("body", |w| body(w, &d.body))
205                })?;
206            }
207        }
208        Ok(())
209    })
210}

260/// Frames one source/vocabulary body with complete unused contract facts.
261fn body(w: &mut Writer<'_>, b: &B) -> Result<(), E> {
262    match b {
263        B::Record(fields) => {
264            ordered(fields.iter().map(|f| &f.name))?;
265            w.frame("record", |w| {
266                for f in fields {
267                    if (f.presence == FieldPresence::Defaulted) != f.default.is_some() {
268                        return Err(E::InvalidInput);
269                    }
270                    w.frame("field", |w| {
271                        w.leaf("name", f.name.as_bytes())?;
272                        type_transcript(w, &f.ty, 0)?;
273                        w.leaf(
274                            "presence",
275                            match f.presence {
276                                FieldPresence::Required => b"required",
277                                FieldPresence::Optional => b"optional",
278                                FieldPresence::Defaulted => b"defaulted",
279                            },
280                        )?;
281                        restrictions(w, &f.restrictions)?;
282                        w.frame("default", |w| match &f.default {
283                            Some(v) => value(w, v, 0),
284                            None => w.leaf("absent", &[]),
285                        })
286                    })?;
287                }
288                Ok(())
289            })
290        }
291        B::Variant(alternatives) => {
292            if alternatives.is_empty() || alternatives.iter().any(|a| a.tag.is_empty()) {
293                return Err(E::InvalidInput);
294            }
295            ordered(alternatives.iter().map(|a| &a.tag))?;
296            w.frame("variant", |w| {
297                for a in alternatives {
298                    w.frame("alternative", |w| {
299                        w.leaf("tag", a.tag.as_bytes())?;
300                        type_transcript(w, &a.ty, 0)
301                    })?;
302                }
303                Ok(())
304            })
305        }
306    }
307}

309/// Preserves absence of constraints separately from present finite or numeric bounds.
310fn restrictions(w: &mut Writer<'_>, r: &FieldRestrictions) -> Result<(), E> {
311    if let Some(choices) = &r.choices {
312        w.items(choices.len())?;
313        for choice in choices {
314            w.check()?;
315            match choice {
316                V::Number(n) => w.text(n.coefficient())?,
317                V::String(s) | V::Url(s) | V::Path(s) => w.text(s)?,
318                V::Bool(_) => {}
319                _ => return Err(E::InvalidInput),
320            }
321        }
322        for pair in choices.windows(2) {
323            if crate::composition::compare_composition_scalars(&pair[0], &pair[1])
324                != Some(std::cmp::Ordering::Less)
325            {
326                return Err(E::InvalidInput);
327            }
328        }
329    }
330    w.frame("restrictions", |w| {
331        w.frame("choices", |w| {
332            if let Some(values) = &r.choices {
333                for v in values {
334                    value(w, v, 0)?;
335                }
336            }
337            Ok(())
338        })?;
339        for (tag, bound) in [("minimum", &r.minimum), ("maximum", &r.maximum)] {
340            w.frame(tag, |w| match bound {
341                Some(n) => number(w, n),
342                None => w.leaf("absent", &[]),
343            })?;
344        }
345        for (tag, bound) in [("min-length", r.min_length), ("max-length", r.max_length)] {
346            w.frame(tag, |w| match bound {
347                Some(n) => w.number("length", n),
348                None => w.leaf("absent", &[]),
349            })?;
350        }
351        Ok(())
352    })
353}

355/// Checks successor wrapper invariants before using the unchanged frozen type framing grammar.
356fn type_transcript(w: &mut Writer<'_>, ty: &ProjectPublicType, depth: usize) -> Result<(), E> {
357    let mut inner = ty;
358    let mut current_depth = depth;
359    loop {
360        w.check()?;
361        if current_depth > PROJECT_MAX_DEPTH {
362            return Err(E::Limit);
363        }
364        match inner {
365            ProjectPublicType::Ref(target)
366                if !matches!(
367                    target.as_ref(),
368                    ProjectPublicType::Nominal(_) | ProjectPublicType::VocabularyNominal { .. }
369                ) =>
370            {
371                return Err(E::InvalidInput);
372            }
373            ProjectPublicType::Nullable(target)
374                if matches!(target.as_ref(), ProjectPublicType::Nullable(_)) =>
375            {
376                return Err(E::InvalidInput);
377            }
378            ProjectPublicType::List(target)
379            | ProjectPublicType::Ref(target)
380            | ProjectPublicType::Nullable(target) => {
381                inner = target;
382                current_depth += 1;
383            }
384            _ => break,
385        }
386    }
387    frozen_type_transcript(w, ty, depth)
388}

390/// Frames borrowed exact numbers without constructing an allocating temporary value.
391fn number(w: &mut Writer<'_>, n: &crate::ExactNumber) -> Result<(), E> {
392    w.frame("num", |w| {
393        w.leaf("negative", &[u8::from(n.is_negative())])?;
394        w.leaf("coefficient", n.coefficient().as_bytes())?;
395        w.leaf("scale", &n.scale().to_be_bytes())
396    })
397}

399/// Lets closed values share framing without permitting non-null default references.
400trait Reference {
401    /// Frames an identity edge without traversing its target.
402    fn write(&self, w: &mut Writer<'_>) -> Result<(), E>;
403}
404impl Reference for crate::ModuleSymbolIdentity {
405    /// Uses the unchanged stable symbol tuple.
406    fn write(&self, w: &mut Writer<'_>) -> Result<(), E> {
407        symbol(w, self)
408    }
409}
410impl Reference for std::convert::Infallible {
411    /// Closed defaults cannot construct this branch.
412    fn write(&self, _: &mut Writer<'_>) -> Result<(), E> {
413        match *self {}
414    }
415}

417/// Frames materialized values; optional absence is only emitted by a record field.
418fn value<R: Reference>(w: &mut Writer<'_>, v: &V<R>, depth: usize) -> Result<(), E> {
419    if depth > PROJECT_MAX_DEPTH {
420        return Err(E::Limit);
421    }
422    match v {
423        V::Null => w.leaf("null", &[]),
424        V::Bool(b) => w.leaf("bool", &[u8::from(*b)]),
425        V::Number(n) => number(w, n),
426        V::String(s) => w.leaf("string", s.as_bytes()),
427        V::Url(s) => w.leaf("url", s.as_bytes()),
428        V::Path(s) => w.leaf("path", s.as_bytes()),
429        V::Reference(r) => w.frame("Ref", |w| r.write(w)),
430        V::List(vs) => w.frame("List", |w| {
431            for v in vs {
432                value(w, v, depth + 1)?;
433            }
434            Ok(())
435        }),
436        V::Record(fields) => {
437            ordered(fields.iter().map(|(n, _)| n))?;
438            w.frame("record", |w| {
439                for (n, v) in fields {
440                    w.frame("field", |w| {
441                        w.leaf("name", n.as_bytes())?;
442                        match v {
443                            Some(v) => value(w, v, depth + 1),
444                            None => w.leaf("omitted", &[]),
445                        }
446                    })?;
447                }
448                Ok(())
449            })
450        }
451        V::Variant { tag, payload } => w.frame("variant", |w| {
452            w.leaf("tag", tag.as_bytes())?;
453            w.frame("payload", |w| value(w, payload, depth + 1))
454        }),
455    }
