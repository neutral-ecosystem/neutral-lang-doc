13/// Complete explicit derivation inputs, never ambient compiler/host state.
14pub struct DerivationContext<'a> {
15    /// Exact captured closure that produced the logical input.
16    pub captured: CapturedClosureIdentity,
17    /// Stable producer implementation identity.
18    pub producer: &'a str,
19    /// Explicit producer revision; not the identity schema version.
20    pub producer_version: &'a str,
21    /// Fourteen acceptance limits in `CAPTURE_LIMIT_TAGS` order.
22    pub capture_limits: [u64; CAPTURE_LIMIT_TAGS.len()],
23    /// Complete project producer bounds, separate from logical meaning.
24    pub project_limits: ProjectLimits,
25}
26
27/// Distinct complete publication and public selection artifact domains.
28#[derive(Clone, Copy, Debug, Eq, PartialEq)]
29pub enum ArtifactKind {
30    /// Complete project, including private content and companions.
31    Project,
32    /// Post-compilation public dependency closure.
33    View,
34}
35
36/// Explicit artifact selection/format context; no roots enter capture or logical meaning.
37pub struct ArtifactIdentityInput<'a> {
38    /// Complete project or selected public view.
39    pub kind: ArtifactKind,
40    /// Exact nonempty ASCII format identifier.
41    pub format: &'a str,
42    /// View roots, normalized before framing; empty for complete project artifacts.
43    pub roots: &'a [ModuleSymbolIdentity],
44    /// Sorted unique explicit format options, never implicit environment defaults.
45    pub options: &'a [(&'a str, &'a str)],
46}

48/// Derives processing identity without adding source or producer data to logical identity.
49///
50/// # Errors
51/// Rejects invalid/zero context, byte/node bounds, or cancellation.
52pub fn derivation_identity(
53    logical: LogicalProjectIdentity,
54    context: &DerivationContext<'_>,
55    limits: IdentityLimits,
56    cancellation: &CancellationToken,
57) -> Result<IdentityTranscript<ProjectDerivationIdentity>, IdentityError> {
58    transcript(
59        DERIVATION_DOMAIN,
60        limits,
61        cancellation,
62        |writer| {
63            if context.producer.is_empty() || context.producer_version.is_empty() {
64                return Err(IdentityError::InvalidInput);
65            }
66            writer.leaf("logical", &logical.as_bytes())?;
67            writer.leaf("captured", &context.captured.as_bytes())?;
68            writer.leaf("producer", context.producer.as_bytes())?;
69            writer.leaf("producer-version", context.producer_version.as_bytes())?;
70            writer.frame("capture-limits", |writer| {
71                for (tag, limit) in CAPTURE_LIMIT_TAGS.iter().zip(context.capture_limits) {
72                    if limit == 0 {
73                        return Err(IdentityError::InvalidInput);
74                    }
75                    writer.number(tag, limit)?;
76                }
77                Ok(())
78            })?;
79            let value = context.project_limits;
80            writer.frame("project-limits", |writer| {
81                for (tag, limit) in [
82                    ("modules", value.modules),
83                    ("declarations", value.declarations),
84                    ("import-edges", value.import_edges),
85                    ("nodes", value.nodes),
86                    ("text-bytes", value.text_bytes),
87                    ("artifact-bytes", value.artifact_bytes),
88                ] {
89                    if limit == 0 {
90                        return Err(IdentityError::InvalidInput);
91                    }
92                    writer.number(tag, limit)?;
93                }
94                Ok(())
95            })
96        },
97        ProjectDerivationIdentity,
98    )
99}

101/// Derives artifact identity by kind, exact format, normalized selection, and options.
102///
103/// # Errors
104/// Rejects duplicate roots/options, roots on a complete artifact, invalid format, bounds, or cancellation.
105pub fn artifact_identity(
106    derivation: ProjectDerivationIdentity,
107    input: &ArtifactIdentityInput<'_>,
108    limits: IdentityLimits,
109    cancellation: &CancellationToken,
110) -> Result<IdentityTranscript<ProjectArtifactIdentity>, IdentityError> {
111    transcript(
112        ARTIFACT_DOMAIN,
113        limits,
114        cancellation,
115        |writer| {
116            writer.text(input.format)?;
117            writer.items(input.roots.len())?;
118            writer.items(input.options.len())?;
119            for root in input.roots {
120                writer.text(root.module().language_behavior_version())?;
121                writer.text(root.module().module_name())?;
122                writer.text(root.declaration_name())?;
123            }
124            for (name, _) in input.options {
125                writer.text(name)?;
126            }
127            if input.format.is_empty()
128                || !input.format.is_ascii()
129                || input.format.bytes().any(|byte| byte.is_ascii_control())
130                || (input.kind == ArtifactKind::Project && !input.roots.is_empty())
131            {
132                return Err(IdentityError::InvalidInput);
133            }
134            let mut roots = Vec::new();
135            roots
136                .try_reserve(input.roots.len())
137                .map_err(|_| IdentityError::Limit)?;
138            roots.extend(input.roots);
139            roots.sort_unstable();
140            writer.check()?;
141            ordered(&roots)?;
142            ordered(input.options.iter().map(|(name, _)| name))?;
143            writer.leaf("derivation", &derivation.as_bytes())?;
144            writer.leaf(
145                "kind",
146                match input.kind {
147                    ArtifactKind::Project => b"project",
148                    ArtifactKind::View => b"view",
149                },
150            )?;
151            writer.leaf("format", input.format.as_bytes())?;
152            writer.frame("roots", |writer| {
153                for root in roots {
154                    writer.frame("root", |writer| symbol(writer, root))?;
155                }
156                Ok(())
157            })?;
158            writer.frame("options", |writer| {
159                for (name, value) in input.options {
160                    if name.is_empty() {
161                        return Err(IdentityError::InvalidInput);
162                    }
163                    writer.frame("option", |writer| {
164                        writer.leaf("name", name.as_bytes())?;
165                        writer.leaf("value", value.as_bytes())
166                    })?;
167                }
168                Ok(())
169            })
170        },
171        ProjectArtifactIdentity,
172    )
