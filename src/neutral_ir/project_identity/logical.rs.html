3//! Complete typed logical form; source/processing companions never enter this layer.
4//!
5//! Canonical traversal follows retained module, declaration, schema, and public
6//! edge order. Private and disconnected declarations are included; source maps,
7//! aliases, local allocation labels, resource facts, and view roots are not.
8//! Keep these exclusions explicit: adding a convenient companion field here
9//! would change complete logical identity even if program meaning stayed fixed.

23/// Builds bounded complete logical form for structurally canonical, semantically validated IR.
24///
25/// This function checks canonical structure needed by the transcript, not every
26/// semantic invariant of a project. For external artifacts, validate through the
27/// reader first. It never hashes a selected public view or signature snapshot.
28/// Profile/domain and field-order changes require a new versioned contract;
29/// changing a package release number must not change these bytes.
30///
31/// # Errors
32/// Rejects noncanonical/schema input, byte/node/depth bounds, or cancellation.
33/// Independent semantic reader validation remains mandatory before trusting an untrusted producer.
34pub fn canonical_logical_project(
35    ir: &ProjectIr,
36    limits: IdentityLimits,
37    cancellation: &CancellationToken,
38) -> Result<IdentityTranscript<LogicalProjectIdentity>, IdentityError> {
39    transcript(
40        LOGICAL_DOMAIN,
41        limits,
42        cancellation,
43        |writer| {
44            if ir.schema != PROJECT_IR_SCHEMA || ir.modules.is_empty() {
45                return Err(IdentityError::InvalidInput);
46            }
47            writer.items(ir.modules.len())?;
48            writer.items(ir.declarations.len())?;
49            writer.items(ir.vocabulary_records.len())?;
50            preflight_keys(writer, ir)?;
51            let schema_limits = crate::project::ProjectLimits {
52                modules: limits.nodes.min(super::MAX_TRANSCRIPT_NODES),
53                declarations: limits.nodes.min(super::MAX_TRANSCRIPT_NODES),
54                import_edges: limits.nodes.min(super::MAX_TRANSCRIPT_NODES),
55                nodes: limits.nodes.min(super::MAX_TRANSCRIPT_NODES),
56                text_bytes: limits.bytes.min(super::MAX_TRANSCRIPT_BYTES),
57                artifact_bytes: limits.bytes.min(super::MAX_TRANSCRIPT_BYTES),
58            };
59            if !ir.within_schema_limits(schema_limits) {
60                return Err(IdentityError::Limit);
61            }
62            ordered(ir.modules.iter().map(|module| &module.identity))?;
63            ordered(ir.declarations.iter().map(|decl| &decl.identity))?;
64            ordered(
65                ir.vocabulary_records
66                    .iter()
67                    .map(|record| (&record.identity, &record.version, &record.name)),
68            )?;
69            writer.leaf("schema", ir.schema.as_bytes())?;
70            writer.leaf("profile", V1_SOURCE_PROFILE.as_bytes())?;
71            writer.frame("modules", |writer| {
72                for module in &ir.modules {
73                    if module.identity.language_behavior_version() != V1_SOURCE_PROFILE {
74                        return Err(IdentityError::InvalidInput);
75                    }
76                    writer.items(module.imports.len())?;
77                    ordered(&module.imports)?;
78                    writer.frame("module", |writer| {
79                        writer.leaf("name", module.identity.module_name().as_bytes())?;
80                        writer.frame("imports", |writer| {
81                            for import in &module.imports {
82                                writer.leaf("target", import.as_bytes())?;
83                            }
84                            Ok(())
85                        })
86                    })?;
87                }
88                Ok(())
89            })?;
90            declarations(writer, ir)?;
91            vocabularies(writer, ir)?;
92            public_edges(writer, ir)
93        },
94        LogicalProjectIdentity,
95    )
96}

98/// Caps aggregate identity-key comparison work before comparing canonical root sequences.
99///
100/// Input is typed but can still come from an untrusted producer. Charge all
101/// comparison keys and aggregate import counts before order checks; otherwise a
102/// tiny output limit could still permit expensive comparisons of huge inputs.
103fn preflight_keys(writer: &mut Writer<'_>, ir: &ProjectIr) -> Result<(), IdentityError> {
104    let mut imports = 0_usize;
105    for module in &ir.modules {
106        writer.text(module.identity.module_name())?;
107        imports = imports
108            .checked_add(module.imports.len())
109            .ok_or(IdentityError::Limit)?;
110        writer.items(imports)?;
111        for target in &module.imports {
112            writer.text(target)?;
113        }
114    }
115    for declaration in &ir.declarations {
116        writer.text(declaration.identity.module().language_behavior_version())?;
117        writer.text(declaration.identity.module().module_name())?;
118        writer.text(declaration.identity.declaration_name())?;
119    }
120    for record in &ir.vocabulary_records {
121        writer.text(&record.identity)?;
122        writer.text(&record.version)?;
123        writer.text(&record.name)?;
124    }
125    Ok(())
126}

128/// Frames stable full module-symbol identity without graph-local labels.
129pub(super) fn symbol(
130    writer: &mut Writer<'_>,
131    identity: &ModuleSymbolIdentity,
132) -> Result<(), IdentityError> {
133    writer.leaf(
134        "profile",
135        identity.module().language_behavior_version().as_bytes(),
136    )?;
137    writer.leaf("module", identity.module().module_name().as_bytes())?;
138    writer.leaf("declaration", identity.declaration_name().as_bytes())
139}

141/// Frames complete public/private declarations and all unused closed defaults.
142fn declarations(writer: &mut Writer<'_>, ir: &ProjectIr) -> Result<(), IdentityError> {
143    writer.frame("declarations", |writer| {
144        for decl in &ir.declarations {
145            writer.frame("declaration", |writer| {
146                symbol(writer, &decl.identity)?;
147                writer.leaf("public", &[u8::from(decl.public)])?;
148                signature(writer, &decl.signature)?;
149                writer.frame("value", |writer| match &decl.value {
150                    Some(value) => value_transcript(writer, value, 0),
151                    None => writer.leaf("absent", &[]),
152                })?;
153                writer.frame("defaults", |writer| fields(writer, &decl.defaults, 0))
154            })?;
155        }
156        Ok(())
157    })
158}

160/// Frames the full locked interpretation schemas and public nominal catalogues.
161fn vocabularies(writer: &mut Writer<'_>, ir: &ProjectIr) -> Result<(), IdentityError> {
162    writer.frame("vocabulary-records", |writer| {
163        for record in &ir.vocabulary_records {
164            writer.frame("record", |writer| {
165                writer.leaf("identity", record.identity.as_bytes())?;
166                writer.leaf("version", record.version.as_bytes())?;
167                writer.leaf("name", record.name.as_bytes())?;
168                writer.leaf("public", &[u8::from(record.public)])?;
169                writer.items(record.fields.len())?;
170                for (name, _) in &record.fields {
171                    writer.text(name)?;
172                }
173                ordered(record.fields.iter().map(|(name, _)| name))?;
174                writer.frame("fields", |writer| {
175                    for (name, ty) in &record.fields {
176                        writer.frame("field", |writer| {
177                            writer.leaf("name", name.as_bytes())?;
178                            type_transcript(writer, ty, 0)
179                        })?;
180                    }
181                    Ok(())
182                })
183            })?;
184        }
185        Ok(())
186    })?;
187    writer.items(ir.public_interface.vocabularies().len())?;
188    for vocabulary in ir.public_interface.vocabularies() {
189        writer.text(vocabulary.identity())?;
190        writer.text(vocabulary.version())?;
191    }
192    ordered(
193        ir.public_interface
194            .vocabularies()
195            .iter()
196            .map(|vocabulary| (vocabulary.identity(), vocabulary.version())),
197    )?;
198    writer.frame("vocabulary-catalogues", |writer| {
199        for vocabulary in ir.public_interface.vocabularies() {
200            writer.frame("vocabulary", |writer| {
201                writer.leaf("identity", vocabulary.identity().as_bytes())?;
202                writer.leaf("version", vocabulary.version().as_bytes())?;
203                writer.items(vocabulary.public_types().len())?;
204                for name in vocabulary.public_types() {
205                    writer.text(name)?;
206                }
207                ordered(vocabulary.public_types())?;
208                writer.frame("public-types", |writer| {
209                    for name in vocabulary.public_types() {
210                        writer.leaf("name", name.as_bytes())?;
211                    }
212                    Ok(())
213                })
214            })?;
215        }
216        Ok(())
217    })
218}

220/// Preserves validated public dependency topology, excluding occurrence locations.
221fn public_edges(writer: &mut Writer<'_>, ir: &ProjectIr) -> Result<(), IdentityError> {
222    writer.items(ir.public_interface.edges().len())?;
223    for edge in ir.public_interface.edges() {
224        for identity in [edge.from(), edge.to()] {
225            writer.text(identity.module().language_behavior_version())?;
226            writer.text(identity.module().module_name())?;
227            writer.text(identity.declaration_name())?;
228        }
229    }
230    ordered(
231        ir.public_interface
232            .edges()
233            .iter()
234            .map(|edge| (edge.from(), edge.kind(), edge.to())),
235    )?;
236    writer.frame("public-edges", |writer| {
237        for edge in ir.public_interface.edges() {
238            writer.frame("edge", |writer| {
239                writer.frame("from", |writer| symbol(writer, edge.from()))?;
240                writer.frame("to", |writer| symbol(writer, edge.to()))?;
241                writer.leaf("kind", edge.kind().spelling().as_bytes())
242            })?;
243        }
244        Ok(())
245    })
246}

248/// Frames distinct binding and record signatures with canonical field order.
249fn signature(
250    writer: &mut Writer<'_>,
251    signature: &ProjectPublicSignature,
252) -> Result<(), IdentityError> {
253    writer.frame("signature", |writer| match signature {
254        ProjectPublicSignature::Binding(ty) => {
255            writer.frame("binding", |writer| type_transcript(writer, ty, 0))
256        }
257        ProjectPublicSignature::Record(fields) => {
258            writer.items(fields.len())?;
259            for field in fields {
260                writer.text(field.name())?;
261            }
262            ordered(
263                fields
264                    .iter()
265                    .map(crate::project_interface::ProjectPublicField::name),
266            )?;
267            writer.frame("record", |writer| {
268                for field in fields {
269                    writer.frame("field", |writer| {
270                        writer.leaf("name", field.name().as_bytes())?;
271                        type_transcript(writer, field.ty(), 0)
272                    })?;
273                }
274                Ok(())
275            })
276        }
277    })
278}

280/// Frames typed meaning without source aliases or recursive reference expansion.
281pub(super) fn type_transcript(
282    writer: &mut Writer<'_>,
283    ty: &ProjectPublicType,
284    depth: usize,
285) -> Result<(), IdentityError> {
286    if depth > PROJECT_MAX_DEPTH {
287        return Err(IdentityError::Limit);
288    }
289    match ty {
290        ProjectPublicType::Num => writer.leaf(language::NUM, &[]),
291        ProjectPublicType::String => writer.leaf(language::STRING, &[]),
292        ProjectPublicType::Bool => writer.leaf(language::BOOL, &[]),
293        ProjectPublicType::Url => writer.leaf(language::URL, &[]),
294        ProjectPublicType::Path => writer.leaf(language::PATH, &[]),
295        ProjectPublicType::Nominal(identity) => {
296            writer.frame("nominal", |writer| symbol(writer, identity))
297        }
298        ProjectPublicType::VocabularyNominal {
299            identity,
300            version,
301            name,
302        } => writer.frame("vocabulary", |writer| {
303            writer.leaf("identity", identity.as_bytes())?;
304            writer.leaf("version", version.as_bytes())?;
305            writer.leaf("name", name.as_bytes())
306        }),
307        ProjectPublicType::List(inner) => writer.frame(language::LIST, |writer| {
308            type_transcript(writer, inner, depth + 1)
309        }),
310        ProjectPublicType::Ref(inner) => writer.frame(language::REF_TYPE, |writer| {
311            type_transcript(writer, inner, depth + 1)
312        }),
313        ProjectPublicType::Nullable(inner) => writer.frame("nullable", |writer| {
314            type_transcript(writer, inner, depth + 1)
315        }),
316    }
317}

319/// Frames materialized typed values, preserving list order and exact rational normalization.
320fn value_transcript(
321    writer: &mut Writer<'_>,
322    value: &ProjectValue,
323    depth: usize,
324) -> Result<(), IdentityError> {
325    if depth > PROJECT_MAX_DEPTH {
326        return Err(IdentityError::Limit);
327    }
328    match value {
329        ProjectValue::Null => writer.leaf(language::NULL, &[]),
330        ProjectValue::Bool(value) => writer.leaf(language::BOOL, &[u8::from(*value)]),
331        ProjectValue::Number(value) => writer.frame(language::NUM, |writer| {
332            writer.leaf("negative", &[u8::from(value.is_negative())])?;
333            writer.leaf("coefficient", value.coefficient().as_bytes())?;
334            writer.leaf("scale", &value.scale().to_be_bytes())
335        }),
336        ProjectValue::String(value) => writer.leaf(language::STRING, value.as_bytes()),
337        ProjectValue::Url(value) => writer.leaf(language::URL, value.as_bytes()),
338        ProjectValue::Path(value) => writer.leaf(language::PATH, value.as_bytes()),
339        ProjectValue::Reference(identity) => {
340            writer.frame(language::REF_TYPE, |writer| symbol(writer, identity))
341        }
342        ProjectValue::List(values) => {
343            writer.items(values.len())?;
344            writer.frame(language::LIST, |writer| {
345                for value in values {
346                    value_transcript(writer, value, depth + 1)?;
347                }
348                Ok(())
349            })
350        }
351        ProjectValue::Record(values) => {
352            writer.frame(language::RECORD, |writer| fields(writer, values, depth + 1))
353        }
354    }
355}

357/// Frames sorted unique fields and defaults without depending on source field order.
358fn fields(
359    writer: &mut Writer<'_>,
360    fields: &[(String, ProjectValue)],
361    depth: usize,
362) -> Result<(), IdentityError> {
363    writer.items(fields.len())?;
364    for (name, _) in fields {
365        writer.text(name)?;
366    }
367    ordered(fields.iter().map(|(name, _)| name))?;
368    for (name, value) in fields {
369        writer.frame("field", |writer| {
370            writer.leaf("name", name.as_bytes())?;
371            value_transcript(writer, value, depth)
372        })?;
373    }
374    Ok(())
