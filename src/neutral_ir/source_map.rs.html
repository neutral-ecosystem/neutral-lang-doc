3//! Original-byte locations for a document and its declared elements.
4//!
5//! A source digest and byte length anchor declaration, type, name and value spans
6//! to one captured input. Element IDs connect entries to the logical document;
7//! they are not durable cross-document identities. Consumers must validate the
8//! map against its document rather than treating construction as proof of coverage.
9
10use crate::ElementId;
11use neutral_core::{ByteSpan, SourceContentDigest};
12
13/// Source-map entry for one declaration and its minimal child components.
14#[derive(Clone, Copy, Debug, Eq, PartialEq)]
15pub struct SourceMapEntry {
16    /// Graph-local declaration identifier.
17    element_id: ElementId,
18    /// Complete declaration source span.
19    declaration_span: ByteSpan,
20    /// Explicit type source span.
21    type_span: ByteSpan,
22    /// Declaration-name source span.
23    name_span: ByteSpan,
24    /// Explicit value source span.
25    value_span: ByteSpan,
26}
27
28impl SourceMapEntry {
29    /// Creates source accounting for one root declaration.
30    #[must_use]
31    pub const fn new(
32        element_id: ElementId,
33        declaration_span: ByteSpan,
34        type_span: ByteSpan,
35        name_span: ByteSpan,
36        value_span: ByteSpan,
37    ) -> Self {
38        Self {
39            element_id,
40            declaration_span,
41            type_span,
42            name_span,
43            value_span,
44        }
45    }
46
47    /// Returns the declaration element identifier.
48    #[must_use]
49    pub const fn element_id(self) -> ElementId {
50        self.element_id
51    }
52
53    /// Returns the complete declaration span.
54    #[must_use]
55    pub const fn declaration_span(self) -> ByteSpan {
56        self.declaration_span
57    }
58
59    /// Returns the explicit type span.
60    #[must_use]
61    pub const fn type_span(self) -> ByteSpan {
62        self.type_span
63    }
64
65    /// Returns the declaration-name span.
66    #[must_use]
67    pub const fn name_span(self) -> ByteSpan {
68        self.name_span
69    }
70
71    /// Returns the explicit value span.
72    #[must_use]
73    pub const fn value_span(self) -> ByteSpan {
74        self.value_span
75    }
76}
77
78/// Immutable original-byte source map for a validated document.
79#[derive(Clone, Debug, Eq, PartialEq)]
80pub struct SourceMap {
81    /// Exact captured source identity.
82    source_digest: SourceContentDigest,
83    /// Exact captured source byte length.
84    source_byte_length: u64,
85    /// Complete module-header source span.
86    module_span: ByteSpan,
87    /// Declaration mappings.
88    entries: Vec<SourceMapEntry>,
89}
90
91impl SourceMap {
92    /// Creates an immutable source map from validated compiler accounting.
93    #[must_use]
94    pub fn new(
95        source_digest: SourceContentDigest,
96        source_byte_length: u64,
97        module_span: ByteSpan,
98        entries: Vec<SourceMapEntry>,
99    ) -> Self {
100        Self {
101            source_digest,
102            source_byte_length,
103            module_span,
104            entries,
105        }
106    }
107
108    /// Returns the exact captured source identity.
109    #[must_use]
110    pub const fn source_digest(&self) -> SourceContentDigest {
111        self.source_digest
112    }
113
114    /// Returns the exact captured source byte length.
115    #[must_use]
116    pub const fn source_byte_length(&self) -> u64 {
117        self.source_byte_length
118    }
119
120    /// Returns the complete module-header source span.
121    #[must_use]
122    pub const fn module_span(&self) -> ByteSpan {
123        self.module_span
124    }
125
126    /// Returns deterministic declaration source mappings.
127    #[must_use]
128    pub fn entries(&self) -> &[SourceMapEntry] {
129        &self.entries
130    }
131
132    /// Finds source accounting for one graph-local element.
133    #[must_use]
134    pub fn entry(&self, element_id: ElementId) -> Option<&SourceMapEntry> {
135        self.entries
136            .iter()
137            .find(|entry| entry.element_id == element_id)
138    }
139}
