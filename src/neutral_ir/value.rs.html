16/// Resolved minimal Neutral type identity.
17#[derive(Clone, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
18pub enum ResolvedType {
19    /// Exact Neutral numeric type.
20    Num,
21    /// Unicode scalar-sequence string type.
22    String,
23    /// Exact Boolean type.
24    Bool,
25    /// Exact module-owned nominal record type.
26    Record(NominalTypeIdentity),
27    /// Exact nominal record type owned by a captured vocabulary.
28    VocabularyRecord(VocabularyTypeIdentity),
29    /// Ordered homogeneous list with one invariant element type.
30    List(Box<ResolvedType>),
31    /// Typed document-local identity reference with one invariant target type.
32    Ref(Box<ResolvedType>),
33    /// One outer nullable layer around an otherwise resolved type.
34    Nullable(Box<ResolvedType>),
35}
36
37impl ResolvedType {
38    /// Wraps a resolved non-null type in exactly one outer nullable layer.
39    #[must_use]
40    pub fn nullable(inner: Self) -> Self {
41        Self::Nullable(Box::new(inner))
42    }
43
44    /// Wraps one resolved type as an invariant ordered list element type.
45    #[must_use]
46    pub fn list(inner: Self) -> Self {
47        Self::List(Box::new(inner))
48    }
49
50    /// Wraps one exact resolved target type as a document-local identity reference.
51    #[must_use]
52    pub fn reference(inner: Self) -> Self {
53        Self::Ref(Box::new(inner))
54    }
55
56    /// Returns whether this type has an outer nullable layer.
57    #[must_use]
58    pub const fn is_nullable(&self) -> bool {
59        matches!(self, Self::Nullable(_))
60    }
61
62    /// Returns the inner resolved type when this type is nullable.
63    #[must_use]
64    pub fn nullable_inner(&self) -> Option<&Self> {
65        match self {
66            Self::Nullable(inner) => Some(inner),
67            _ => None,
68        }
69    }
70
71    /// Returns whether a logical value satisfies exact identity plus outer nullability.
72    #[must_use]
73    pub fn accepts_value(&self, value: &LogicalValue) -> bool {
74        match (self, value) {
75            (Self::Num, LogicalValue::Number(_))
76            | (Self::String, LogicalValue::String(_))
77            | (Self::Bool, LogicalValue::Boolean(_))
78            | (Self::Nullable(_), LogicalValue::Null) => true,
79            (Self::Record(expected), LogicalValue::Record(value)) => {
80                expected == value.nominal_type()
81            }
82            (Self::VocabularyRecord(expected), LogicalValue::VocabularyRecord(value)) => {
83                expected == value.nominal_type()
84            }
85            (Self::List(expected), LogicalValue::List(items)) => {
86                items.iter().all(|item| expected.accepts_value(item))
87            }
88            (Self::Ref(expected), LogicalValue::Reference(reference)) => {
89                expected.as_ref() == reference.target_type()
90            }
91            (Self::Nullable(inner), value) => inner.accepts_value(value),
92            _ => false,
93        }
94    }
95}

97impl fmt::Display for ResolvedType {
98    /// Formats the stable source-facing type name.
99    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
100        match self {
101            Self::Num => formatter.write_str("num"),
102            Self::String => formatter.write_str("string"),
103            Self::Bool => formatter.write_str("bool"),
104            Self::Record(identity) => formatter.write_str(identity.name()),
105            Self::VocabularyRecord(identity) => identity.fmt(formatter),
106            Self::List(inner) => write!(formatter, "List<{inner}>"),
107            Self::Ref(inner) => write!(formatter, "Ref<{inner}>"),
108            Self::Nullable(inner) => write!(formatter, "{inner}?"),
109        }
110    }
111}

113/// Immutable logical value in the active public IR slice.
114#[derive(Clone, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
115pub enum LogicalValue {
116    /// Exact normalized Neutral number.
117    Number(ExactNumber),
118    /// Exact decoded Unicode scalar sequence.
119    String(String),
120    /// Exact Boolean value.
121    Boolean(bool),
122    /// Explicit null value, valid only with a nullable resolved type.
123    Null,
124    /// Contextually typed nominal record value with canonical field order.
125    Record(RecordValue),
126    /// Contextually typed vocabulary-owned record value.
127    VocabularyRecord(VocabularyRecordValue),
128    /// Ordered homogeneous logical values.
129    List(Vec<LogicalValue>),
130    /// Typed document-local identity edge to one binding declaration.
131    Reference(IdentityReference),
132}
133
134impl fmt::Display for LogicalValue {
135    /// Formats a deterministic reader-facing logical value.
136    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
137        match self {
138            Self::Number(number) => number.fmt(formatter),
139            Self::String(value) => format_safe_string(value, formatter),
140            Self::Boolean(value) => value.fmt(formatter),
141            Self::Null => formatter.write_str("null"),
142            Self::Record(value) => value.fmt(formatter),
143            Self::VocabularyRecord(value) => value.fmt(formatter),
144            Self::List(items) => {
145                formatter.write_str("[")?;
146                for (index, item) in items.iter().enumerate() {
147                    if index > 0 {
148                        formatter.write_str(", ")?;
149                    }
150                    item.fmt(formatter)?;
151                }
152                formatter.write_str("]")
153            }
154            Self::Reference(reference) => {
155                write!(formatter, "ref(#{})", reference.target_element_id().get())
156            }
157        }
158    }
159}

328/// Formats a logical string with all hostile control text escaped.
329fn format_safe_string(value: &str, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
330    formatter.write_str("\"")?;
331    for character in value.chars() {
332        match character {
333            '"' => formatter.write_str("\\\"")?,
334            '\\' => formatter.write_str("\\\\")?,
335            '\n' => formatter.write_str("\\n")?,
336            '\r' => formatter.write_str("\\r")?,
337            '\t' => formatter.write_str("\\t")?,
338            '\0' => formatter.write_str("\\0")?,
339            character if character.is_control() => {
340                write!(formatter, "\\u{{{:x}}}", u32::from(character))?;
341            }
342            character => write!(formatter, "{character}")?,
343        }
344    }
345    formatter.write_str("\"")
346}

630        Self { identity, types }
631    }
632    /// Returns exact captured and logical identity facts.
633    #[must_use]
634    pub const fn identity(&self) -> &VocabularyIdentity {
635        &self.identity
636    }
637    /// Returns canonical nominal type contracts.
638    #[must_use]
639    pub fn types(&self) -> &[VocabularyTypeContract] {
640        &self.types
641    }
642    /// Finds a type without external acquisition.
643    #[must_use]
644    pub fn type_by_name(&self, name: &str) -> Option<&VocabularyTypeContract> {
645        self.types
646            .iter()
647            .find(|definition| definition.identity().name() == name)
648    }
649
650    /// Compares normalized logical vocabulary meaning without captured-byte facts.
651    #[must_use]
652    pub fn logically_equivalent(&self, other: &Self) -> bool {
653        self.identity.identity == other.identity.identity
654            && self.identity.version == other.identity.version
655            && self.identity.schema_version == other.identity.schema_version
656            && self.identity.required_features == other.identity.required_features
657            && self.types == other.types
658    }
