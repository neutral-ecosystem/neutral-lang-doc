8/// Domain separating public signature fingerprints from all other identities.
9pub const PROJECT_INTERFACE_FINGERPRINT_DOMAIN: &str = "neutral/project-interface/v1";
10/// Maximum nested type layers accepted by the public-interface contract.
11pub const MAX_PROJECT_INTERFACE_TYPE_DEPTH: usize = 64;
12
13/// A canonical public type signature with alias-independent nominal identity.
14#[derive(Clone, Debug, Eq, PartialEq)]
15pub enum ProjectPublicType {
16    /// Exact number type.
17    Num,
18    /// Unicode string type.
19    String,
20    /// Boolean type.
21    Bool,
22    /// Inert URL text, distinct from a string or path.
23    Url,
24    /// Inert path text, distinct from a string or URL.
25    Path,
26    /// Public nominal vocabulary type under its exact canonical lock.
27    VocabularyNominal {
28        /// Canonical vocabulary identity, not a source alias.
29        identity: String,
30        /// Exact semantic release from the captured lock.
31        version: String,
32        /// Public nominal type name.
33        name: String,
34    },
35    /// Public nominal record owned by one logical module.
36    Nominal(ModuleSymbolIdentity),
37    /// Invariant ordered list type.
38    List(Box<Self>),
39    /// Invariant identity-reference target type.
40    Ref(Box<Self>),
41    /// One explicit outer nullable layer.
42    Nullable(Box<Self>),
43}
44
45impl ProjectPublicType {
46    /// Frames one bounded canonical type for the interface transcript.
47    fn transcript(&self, depth: usize) -> Result<Vec<u8>, CoreError> {
48        if depth > MAX_PROJECT_INTERFACE_TYPE_DEPTH {
49            return Err(CoreError::TranscriptLengthExceeded);
50        }
51        match self {
52            Self::Num => nht_frame(crate::language::NUM, &[]),
53            Self::String => nht_frame(crate::language::STRING, &[]),
54            Self::Bool => nht_frame(crate::language::BOOL, &[]),
55            Self::Url => nht_frame(crate::language::URL, &[]),
56            Self::Path => nht_frame(crate::language::PATH, &[]),
57            Self::VocabularyNominal {
58                identity,
59                version,
60                name,
61            } => {
62                let mut payload = nht_frame("identity", identity.as_bytes())?;
63                payload.extend(nht_frame("version", version.as_bytes())?);
64                payload.extend(nht_frame("name", name.as_bytes())?);
65                nht_frame("vocabulary-nominal", &payload)
66            }
67            Self::Nominal(identity) => nht_frame("nominal", &identity_transcript(identity)?),
68            Self::List(inner) => nht_frame("list", &inner.transcript(depth + 1)?),
69            Self::Ref(inner) => nht_frame("ref", &inner.transcript(depth + 1)?),
70            Self::Nullable(inner) => nht_frame("nullable", &inner.transcript(depth + 1)?),
71        }
72    }
73}

75/// A source-authored inert location scalar retained as exact decoded text.
76///
77/// This value carries no resolver, filesystem handle, URL client, permission,
78/// or normalization policy. Its variants are intentionally not interchangeable.
79#[derive(Clone, Debug, Eq, PartialEq)]
80pub enum ProjectLocationValue {
81    /// Exact decoded text of a `url` value.
82    Url(String),
83    /// Exact decoded text of a `path` value.
84    Path(String),
85}
86
87impl ProjectLocationValue {
88    /// Returns the exact decoded text without interpretation.
89    #[must_use]
90    pub fn text(&self) -> &str {
91        match self {
92            Self::Url(text) | Self::Path(text) => text,
93        }
94    }
95}

97/// One public record field's name and canonical type.
98#[derive(Clone, Debug, Eq, PartialEq)]
99pub struct ProjectPublicField {
100    /// Exact canonical field name.
101    name: String,
102    /// Alias-independent field type.
103    ty: ProjectPublicType,
104}
105
106impl ProjectPublicField {
107    /// Constructs a field for a canonical public record signature.
108    #[must_use]
109    pub fn new(name: impl Into<String>, ty: ProjectPublicType) -> Self {
110        Self {
111            name: name.into(),
112            ty,
113        }
114    }
115
116    /// Returns the canonical field name.
117    #[must_use]
118    pub fn name(&self) -> &str {
119        &self.name
120    }
121
122    /// Returns the canonical field type.
123    #[must_use]
124    pub const fn ty(&self) -> &ProjectPublicType {
125        &self.ty
126    }
127
128    /// Frames one field without depending on source order or trivia.
129    fn transcript(&self) -> Result<Vec<u8>, CoreError> {
130        let mut payload = nht_frame("name", self.name.as_bytes())?;
131        payload.extend(nht_frame("type", &self.ty.transcript(0)?)?);
132        nht_frame("field", &payload)
133    }
134}

136/// One public declaration signature, excluding private implementation values.
137#[derive(Clone, Debug, Eq, PartialEq)]
138pub enum ProjectPublicSignature {
139    /// An immutable binding's explicit type.
140    Binding(ProjectPublicType),
141    /// A nominal record's fields in canonical name order.
142    Record(Vec<ProjectPublicField>),
143}
144
145impl ProjectPublicSignature {
146    /// Frames one public declaration signature.
147    fn transcript(&self) -> Result<Vec<u8>, CoreError> {
148        match self {
149            Self::Binding(ty) => nht_frame("binding", &ty.transcript(0)?),
150            Self::Record(fields) => {
151                let mut payload = Vec::new();
152                for field in fields {
153                    payload.extend(field.transcript()?);
154                }
155                nht_frame("record", &payload)
156            }
157        }
158    }
159}

200/// Public dependency kind retained after private provenance redaction.
201#[derive(Clone, Copy, Debug, Eq, Ord, PartialEq, PartialOrd)]
202pub enum ProjectPublicEdgeKind {
203    /// Public nominal type dependency.
204    Type,
205    /// Public nominal target under a reference type.
206    ReferenceType,
207    /// Immutable value reuse between public roots.
208    Value,
209    /// Identity-only reference between public roots.
210    Reference,
211}
212
213impl ProjectPublicEdgeKind {
214    /// Returns the stable transcript spelling of one typed edge.
215    #[must_use]
216    pub const fn spelling(self) -> &'static str {
217        match self {
218            Self::Type => "type",
219            Self::ReferenceType => "reference-type",
220            Self::Value => "value",
221            Self::Reference => "reference",
222        }
223    }

237/// One locked vocabulary's public, alias-free type catalogue.
238#[derive(Clone, Debug, Eq, PartialEq)]
239pub struct ProjectPublicVocabulary {
240    /// Canonical identity from the exact captured lock.
241    identity: String,
242    /// Exact semantic release from that lock.
243    version: String,
244    /// Source-authorable type names in canonical order; private names are absent.
245    public_types: Vec<String>,
246}
247
248impl ProjectPublicVocabulary {
249    /// Constructs one public-only locked vocabulary fact.
250    #[must_use]
251    pub fn new(
252        identity: impl Into<String>,
253        version: impl Into<String>,
254        public_types: Vec<String>,
255    ) -> Self {
256        Self {
257            identity: identity.into(),
258            version: version.into(),
259            public_types,
260        }
261    }
262
263    /// Returns the canonical identity, never a local source alias.
264    #[must_use]
265    pub fn identity(&self) -> &str {
266        &self.identity
267    }
268
269    /// Returns the exact locked semantic release.
270    #[must_use]
271    pub fn version(&self) -> &str {
272        &self.version
273    }
274
275    /// Returns only public source-authorable type names.
276    #[must_use]
277    pub fn public_types(&self) -> &[String] {
278        &self.public_types
279    }
280
281    /// Frames the complete public fact without alias or authoring metadata.
282    fn transcript(&self) -> Result<Vec<u8>, CoreError> {
283        let mut payload = nht_frame("identity", self.identity.as_bytes())?;
284        payload.extend(nht_frame("version", self.version.as_bytes())?);
285        for name in &self.public_types {
286            payload.extend(nht_frame("public-type", name.as_bytes())?);
287        }
288        nht_frame("vocabulary", &payload)
289    }
290}

314
315    /// Returns the public edge category.
316    #[must_use]
317    pub const fn kind(&self) -> ProjectPublicEdgeKind {
318        self.kind
319    }
320
321    /// Frames one alias-independent edge without source provenance.
322    fn transcript(&self) -> Result<Vec<u8>, CoreError> {
323        let mut payload = nht_frame("from", &identity_transcript(&self.from)?)?;

325        payload.extend(nht_frame("kind", self.kind.spelling().as_bytes())?);
326        nht_frame("edge", &payload)
327    }

330/// Public interface snapshot awaiting independent reader validation.
331#[derive(Clone, Debug, Eq, PartialEq)]
332pub struct ProjectInterface {
333    /// Locked public vocabulary facts in canonical identity order.
334    vocabularies: Vec<ProjectPublicVocabulary>,
335    /// Public exports in canonical module-symbol order.
336    exports: Vec<ProjectPublicExport>,
337    /// Public-to-public dependency edges in canonical order.
338    edges: Vec<ProjectPublicEdge>,
339    /// Declared public signature fingerprint.
340    fingerprint: SemanticDigest,
341}
342
343impl ProjectInterface {
344    /// Constructs a snapshot with a separately supplied declared fingerprint.
345    #[must_use]
346    pub const fn from_parts(
347        exports: Vec<ProjectPublicExport>,
348        edges: Vec<ProjectPublicEdge>,
349        fingerprint: SemanticDigest,
350    ) -> Self {
351        Self {
352            vocabularies: Vec::new(),
353            exports,
354            edges,
355            fingerprint,
356        }
357    }
358
359    /// Constructs a complete snapshot with a separately supplied fingerprint.
360    #[must_use]
361    pub const fn from_parts_with_vocabularies(
362        vocabularies: Vec<ProjectPublicVocabulary>,
363        exports: Vec<ProjectPublicExport>,
364        edges: Vec<ProjectPublicEdge>,
365        fingerprint: SemanticDigest,
366    ) -> Self {
367        Self {
368            vocabularies,
369            exports,
370            edges,
371            fingerprint,
372        }
373    }
374
375    /// Constructs a snapshot with its canonical public-interface fingerprint.
376    ///
377    /// # Errors
378    ///
379    /// Returns transcript framing failures for unrepresentable payloads.
380    pub fn with_computed_fingerprint(
381        exports: Vec<ProjectPublicExport>,
382        edges: Vec<ProjectPublicEdge>,
383    ) -> Result<Self, CoreError> {
384        Self::with_vocabularies(Vec::new(), exports, edges)
385    }
386
387    /// Constructs a complete public snapshot with canonical locked vocabularies.
388    ///
389    /// # Errors
390    ///
391    /// Returns transcript framing failures for unrepresentable payloads.
392    pub fn with_vocabularies(
393        vocabularies: Vec<ProjectPublicVocabulary>,
394        exports: Vec<ProjectPublicExport>,
395        edges: Vec<ProjectPublicEdge>,
396    ) -> Result<Self, CoreError> {
397        let fingerprint = Self::fingerprint_for(&vocabularies, &exports, &edges)?;
398        Ok(Self {
399            vocabularies,
400            exports,
401            edges,
402            fingerprint,
403        })
404    }
405
406    /// Returns canonical locked vocabulary facts without aliases or metadata.
407    #[must_use]
408    pub fn vocabularies(&self) -> &[ProjectPublicVocabulary] {
409        &self.vocabularies
410    }
411
412    /// Returns public exports in canonical order.
413    #[must_use]
414    pub fn exports(&self) -> &[ProjectPublicExport] {
415        &self.exports
416    }
417
418    /// Returns only public-to-public edges, with private provenance omitted.
419    #[must_use]
420    pub fn edges(&self) -> &[ProjectPublicEdge] {
421        &self.edges
422    }
423
424    /// Returns the declared public signature fingerprint.
425    #[must_use]
426    pub const fn fingerprint(&self) -> SemanticDigest {
427        self.fingerprint
428    }
429
430    /// Recomputes the fingerprint from the complete public signature surface.
431    ///
432    /// # Errors
433    ///
434    /// Returns transcript framing failures for unrepresentable payloads.
435    pub fn recompute_fingerprint(&self) -> Result<SemanticDigest, CoreError> {
436        Self::fingerprint_for(&self.vocabularies, &self.exports, &self.edges)
437    }
438
439    /// Hashes exports and public edges with one domain-separated transcript.
440    fn fingerprint_for(
441        vocabularies: &[ProjectPublicVocabulary],
442        exports: &[ProjectPublicExport],
443        edges: &[ProjectPublicEdge],
444    ) -> Result<SemanticDigest, CoreError> {
445        let mut payload = nht_frame("profile", V1_SOURCE_PROFILE.as_bytes())?;
446        for vocabulary in vocabularies {
447            payload.extend(vocabulary.transcript()?);
448        }
449        for export in exports {
450            payload.extend(export.transcript()?);
451        }
452        for edge in edges {
453            payload.extend(edge.transcript()?);
454        }
455        SemanticDigest::from_nht(PROJECT_INTERFACE_FINGERPRINT_DOMAIN, &payload)
456    }
457}

459/// Frames one full module-symbol identity with the exact profile.
460fn identity_transcript(identity: &ModuleSymbolIdentity) -> Result<Vec<u8>, CoreError> {
461    let mut payload = nht_frame(
462        "profile",
463        identity.module().language_behavior_version().as_bytes(),
464    )?;
465    payload.extend(nht_frame(
466        "module",
467        identity.module().module_name().as_bytes(),
468    )?);
469    payload.extend(nht_frame(
470        "declaration",
471        identity.declaration_name().as_bytes(),
472    )?);
473    Ok(payload)
