17/// Safe structural preflight failures without untrusted names or partial facts.
18#[derive(Clone, Copy, Debug, Eq, PartialEq)]
19pub enum CompositionShapeError {
20    /// An independent count, byte, depth, work or allocation bound was exceeded.
21    Limit,
22    /// Cancellation prevented complete inspection.
23    Cancelled,
24}
25
26/// Request-local retained-data inspection counters shared by all declarations and companions.
27struct Inspector<'a> {
28    /// Explicit project retention policy.
29    limits: ProjectLimits,
30    /// Explicit composition controls.
31    policy: CompositionPolicy,
32    /// Cooperative caller cancellation.
33    cancellation: &'a CancellationToken,
34    /// Charged inspection/key work, not asserted producer work.
35    work: u64,
36    /// Aggregate retained text, excluding advertised source byte lengths.
37    text: u64,
38    /// Independently recomputed retained facts.
39    facts: CompositionResourceFacts,
40}
41
42impl Inspector<'_> {
43    /// Charges checked inspection work before proportional traversal or allocation.
44    fn step(&mut self, n: u64) -> Result<(), CompositionShapeError> {
45        if self.cancellation.is_cancelled() {
46            return Err(CompositionShapeError::Cancelled);
47        }
48        self.work = self
49            .work
50            .checked_add(n)
51            .ok_or(CompositionShapeError::Limit)?;
52        if self.work > self.policy.work.min(profile::MAX_ITEMS) {
53            return Err(CompositionShapeError::Limit);
54        }
55        Ok(())
56    }
57
58    /// Charges aggregate retained key/text bytes before cloning or comparing them.
59    fn text(&mut self, text: &str) -> Result<(), CompositionShapeError> {
60        self.step(text.len() as u64 + 1)?;
61        self.text = self
62            .text
63            .checked_add(text.len() as u64)
64            .ok_or(CompositionShapeError::Limit)?;
65        if self.text > self.limits.text_bytes {
66            return Err(CompositionShapeError::Limit);
67        }
68        Ok(())
69    }
70
71    /// Bounds an input collection against both an explicit policy and the shared hard item ceiling.
72    fn count(&mut self, n: usize, limit: u64) -> Result<(), CompositionShapeError> {
73        self.step(1)?;
74        if n as u64 > limit.min(profile::MAX_ITEMS) {
75            return Err(CompositionShapeError::Limit);
76        }
77        Ok(())
78    }
79
80    /// Checks exact symbol key work without allocating an owner or source path.
81    fn symbol(&mut self, symbol: &ModuleSymbolIdentity) -> Result<(), CompositionShapeError> {
82        self.text(symbol.module().language_behavior_version())?;
83        self.text(symbol.module().module_name())?;
84        self.text(symbol.declaration_name())
85    }
86
87    /// Bounds every unary wrapper before recursive semantic/type comparison consumers.
88    fn ty(&mut self, mut ty: &T) -> Result<(), CompositionShapeError> {
89        let mut depth = 0;
90        loop {
91            self.step(1)?;
92            if depth > self.policy.type_depth.min(PROJECT_MAX_DEPTH as u64) {
93                return Err(CompositionShapeError::Limit);
94            }
95            match ty {
96                T::List(inner) | T::Ref(inner) | T::Nullable(inner) => {
97                    ty = inner;
98                    depth += 1;
99                }
100                T::Nominal(owner) => return self.symbol(owner),
101                T::VocabularyNominal {
102                    identity,
103                    version,
104                    name,
105                } => {
106                    self.text(identity)?;
107                    self.text(version)?;
108                    return self.text(name);
109                }
110                _ => return Ok(()),
111            }
112        }
113    }
114
115    /// Counts retained value nodes iteratively; references are keys, never embedded targets.
116    fn value<R>(
117        &mut self,
118        value: &CompositionValue<R>,
119        reference: impl Fn(&R) -> Option<&ModuleSymbolIdentity>,
120    ) -> Result<(), CompositionShapeError> {
121        self.step(1)?;
122        let mut stack = Vec::new();
123        stack
124            .try_retain(1)
125            .map_err(|_| CompositionShapeError::Limit)?;
126        stack.push((value, 0_u64));
127        while let Some((value, depth)) = stack.pop() {
128            self.step(1)?;
129            if depth > self.policy.value_depth.min(PROJECT_MAX_DEPTH as u64) {
130                return Err(CompositionShapeError::Limit);
131            }
132            self.facts.retained_value_nodes = self
133                .facts
134                .retained_value_nodes
135                .checked_add(1)
136                .ok_or(CompositionShapeError::Limit)?;
137            if self.facts.retained_value_nodes > self.limits.nodes.min(self.policy.value_nodes) {
138                return Err(CompositionShapeError::Limit);
139            }
140            match value {
141                CompositionValue::Number(number) => self.text(number.coefficient())?,
142                CompositionValue::String(text)
143                | CompositionValue::Url(text)
144                | CompositionValue::Path(text) => self.text(text)?,
145                CompositionValue::Reference(target) => {
146                    if let Some(target) = reference(target) {
147                        self.symbol(target)?;
148                    }
149                }
150                CompositionValue::List(items) => {
151                    self.count(items.len(), self.limits.nodes)?;
152                    stack
153                        .try_retain(items.len())
154                        .map_err(|_| CompositionShapeError::Limit)?;
155                    stack.extend(items.iter().map(|value| (value, depth + 1)));
156                }
157                CompositionValue::Record(fields) => {
158                    self.count(fields.len(), self.policy.total_fields)?;
159                    stack
160                        .try_retain(fields.len())
161                        .map_err(|_| CompositionShapeError::Limit)?;
162                    for (name, value) in fields {
163                        self.text(name)?;
164                        if let Some(value) = value {
165                            stack.push((value, depth + 1));
166                        } else {
167                            self.facts.retained_value_nodes = self
168                                .facts
169                                .retained_value_nodes
170                                .checked_add(1)
171                                .ok_or(CompositionShapeError::Limit)?;
172                        }
173                    }
174                    if self.facts.retained_value_nodes
175                        > self.limits.nodes.min(self.policy.value_nodes)
176                    {
177                        return Err(CompositionShapeError::Limit);
178                    }
179                }
180                CompositionValue::Variant { tag, payload } => {
181                    self.text(tag)?;
182                    stack
183                        .try_retain(1)
184                        .map_err(|_| CompositionShapeError::Limit)?;
185                    stack.push((payload, depth + 1));
186                }
187                CompositionValue::Null | CompositionValue::Bool(_) => {}
188            }
189        }
190        Ok(())
191    }
192
193    /// Checks both definition kinds and all defaults, finite choices and restrictions.
194    fn body(&mut self, body: &CompositionBody) -> Result<(), CompositionShapeError> {
195        self.facts.types = self
196            .facts
197            .types
198            .checked_add(1)
199            .ok_or(CompositionShapeError::Limit)?;
200        if self.facts.types > self.policy.total_types {
201            return Err(CompositionShapeError::Limit);
202        }
203        match body {
204            CompositionBody::Record(fields) => {
205                self.count(fields.len(), self.policy.total_fields)?;
206                self.facts.fields = self
207                    .facts
208                    .fields
209                    .checked_add(fields.len() as u64)
210                    .ok_or(CompositionShapeError::Limit)?;
211                if self.facts.fields > self.policy.total_fields {
212                    return Err(CompositionShapeError::Limit);
213                }
214                for field in fields {
215                    self.text(&field.name)?;
216                    self.ty(&field.ty)?;
217                    let restrictions = &field.restrictions;
218                    if let Some(choices) = &restrictions.choices {
219                        self.count(choices.len(), self.policy.choices_per_field)?;
220                        self.facts.choices = self
221                            .facts
222                            .choices
223                            .checked_add(choices.len() as u64)
224                            .ok_or(CompositionShapeError::Limit)?;
225                        if self.facts.choices > self.policy.total_choices {
226                            return Err(CompositionShapeError::Limit);
227                        }
228                        for choice in choices {
229                            self.value(choice, |never| match *never {})?;
230                        }
231                    }
232                    for bound in [&restrictions.minimum, &restrictions.maximum]
233                        .into_iter()
234                        .flatten()
235                    {
236                        self.text(bound.coefficient())?;
237                    }
238                    if let Some(default) = &field.default {
239                        self.value(default, |never| match *never {})?;
240                    }
241                }
242            }
243            CompositionBody::Variant(alternatives) => {
244                self.count(alternatives.len(), self.policy.alternatives_per_type)?;
245                self.facts.alternatives = self
246                    .facts
247                    .alternatives
248                    .checked_add(alternatives.len() as u64)
249                    .ok_or(CompositionShapeError::Limit)?;
250                if self.facts.alternatives > self.policy.total_alternatives {
251                    return Err(CompositionShapeError::Limit);
252                }
253                for alternative in alternatives {
254                    self.text(&alternative.tag)?;
255                    self.ty(&alternative.ty)?;
256                }
257            }
258        }
259        Ok(())
260    }
261}

263/// Preflights every raw retained value/type/companion before consumers clone or recurse.
264///
265/// This is structural resource accounting, not semantic validation or authentication.
266/// All nominal contracts count once; binding/default/choice occurrences count independently.
267///
268/// # Errors
269/// Returns only bounded limit/allocation or cancellation failure, never partial facts.
270pub fn inspect_composition_ir(
271    ir: &CompositionProjectIr,
272    limits: ProjectLimits,
273    policy: CompositionPolicy,
274    cancellation: &CancellationToken,
275) -> Result<CompositionResourceFacts, CompositionShapeError> {
276    let mut inspector = Inspector {
277        limits,
278        policy,
279        cancellation,
280        work: 0,
281        text: 0,
282        facts: CompositionResourceFacts::from_values([0; 7]),
283    };
284    inspector.count(ir.modules.len(), limits.modules)?;
285    inspector.count(ir.declarations.len(), limits.declarations)?;
286    inspector.count(ir.vocabularies.len(), policy.bundles)?;
287    inspector.count(ir.source_maps.len(), limits.declarations)?;
288    inspector.count(ir.provenance.len(), limits.nodes)?;
289    inspector.count(ir.origins.len(), limits.nodes)?;
290    inspector.count(ir.sources.len(), limits.modules)?;
291    inspector.count(ir.vocabulary_sources.len(), policy.bundles)?;
292    let mut imports = 0_u64;
293    for module in &ir.modules {
294        inspector.text(module.identity.module_name())?;
295        inspector.count(module.imports.len(), limits.import_edges)?;
296        imports = imports
297            .checked_add(module.imports.len() as u64)
298            .ok_or(CompositionShapeError::Limit)?;
299        if imports > limits.import_edges {
300            return Err(CompositionShapeError::Limit);
301        }
302        for target in &module.imports {
303            inspector.text(target)?;
304        }
305    }
306    for declaration in &ir.declarations {
307        inspector.symbol(&declaration.identity)?;
308        match &declaration.signature {
309            CompositionSignature::Binding(ty) => inspector.ty(ty)?,
310            CompositionSignature::Definition(body) => inspector.body(body)?,
311        }
312        if let Some(value) = &declaration.value {
313            inspector.value(value, |target| Some(target))?;
314        }
315    }
316    for bundle in &ir.vocabularies {
317        inspector.text(bundle.identity.identity())?;
318        inspector.text(bundle.identity.version())?;
319        inspector.text(bundle.identity.schema_version())?;
320        for feature in bundle.identity.required_features() {
321            inspector.text(feature)?;
322        }
323        inspector.count(bundle.dependencies.len(), policy.dependencies_per_bundle)?;
324        inspector.facts.dependency_edges = inspector
325            .facts
326            .dependency_edges
327            .checked_add(bundle.dependencies.len() as u64)
328            .ok_or(CompositionShapeError::Limit)?;
329        if inspector.facts.dependency_edges > policy.dependency_edges {
330            return Err(CompositionShapeError::Limit);
331        }
332        for dependency in &bundle.dependencies {
333            inspector.text(&dependency.identity)?;
334            inspector.text(&dependency.version)?;
335        }
336        inspector.count(bundle.definitions.len(), policy.total_types)?;
337        for definition in &bundle.definitions {
338            inspector.text(&definition.name)?;
339            inspector.body(&definition.body)?;
340        }
341    }
342    inspect_companions(ir, &mut inspector)?;
343    inspector.facts.bundles = ir.vocabularies.len() as u64;
344    Ok(inspector.facts)
345}

347/// Checks retained companion collection/path text before independent provenance validation.
348fn inspect_companions(
349    ir: &CompositionProjectIr,
350    inspector: &mut Inspector<'_>,
351) -> Result<(), CompositionShapeError> {
352    for source in &ir.sources {
353        inspector.text(&source.module)?;
354        inspector.text(&source.source_id)?;
355    }
356    for source in &ir.vocabulary_sources {
357        inspector.text(&source.identity)?;
358        inspector.text(&source.version)?;
359    }
360    for mapping in &ir.source_maps {
361        inspector.symbol(&mapping.declaration)?;
362    }
363    for provenance in &ir.provenance {
364        inspector.symbol(&provenance.from)?;
365        inspector.symbol(&provenance.to)?;
366    }
367    for origin in &ir.origins {
368        inspector.symbol(&origin.binding)?;
369        inspector.count(
370            origin.path.len(),
371            inspector.policy.value_depth.min(PROJECT_MAX_DEPTH as u64),
372        )?;
373        for segment in &origin.path {
374            if let ValuePathSegment::Field(name) = segment {
375                inspector.text(name)?;
376            }
377        }
378        if let Some(super::CompositionAttribution::Vocabulary {
379            identity,
380            version,
381            type_name,
382            field_name,
383            ..
384        }) = &origin.attribution
385        {
386            inspector.text(identity)?;
387            inspector.text(version)?;
388            inspector.text(type_name)?;
389            inspector.text(field_name)?;
390        }
391    }
392    Ok(())
