17/// Independent composition policy in frozen wire/derivation order, excluding JSON parser controls.
18#[derive(Clone, Copy, Debug, Eq, PartialEq)]
19pub struct CompositionPolicy {
20    /// Bundle count.
21    pub bundles: u64,
22    /// Aggregate captured vocabulary bytes.
23    pub captured_bytes: u64,
24    /// Direct dependencies per bundle.
25    pub dependencies_per_bundle: u64,
26    /// Aggregate dependency edges.
27    pub dependency_edges: u64,
28    /// Longest bundle path, including root and leaf.
29    pub dependency_depth: u64,
30    /// Source and vocabulary nominal definitions.
31    pub total_types: u64,
32    /// Aggregate record fields.
33    pub total_fields: u64,
34    /// Alternatives in one variant.
35    pub alternatives_per_type: u64,
36    /// Aggregate alternatives.
37    pub total_alternatives: u64,
38    /// Choices in one field.
39    pub choices_per_field: u64,
40    /// Aggregate choices.
41    pub total_choices: u64,
42    /// Composed wrapper depth, root zero.
43    pub type_depth: u64,
44    /// Value occurrence depth, root zero.
45    pub value_depth: u64,
46    /// Cumulative materialization visits.
47    pub value_nodes: u64,
48    /// Semantic traversal and comparison work.
49    pub work: u64,
50}
51
52impl CompositionPolicy {
53    /// Projects explicit named controls into the immutable fifteen-position contract.
54    #[must_use]
55    pub const fn values(self) -> [u64; 15] {
56        [
57            self.bundles,
58            self.captured_bytes,
59            self.dependencies_per_bundle,
60            self.dependency_edges,
61            self.dependency_depth,
62            self.total_types,
63            self.total_fields,
64            self.alternatives_per_type,
65            self.total_alternatives,
66            self.choices_per_field,
67            self.total_choices,
68            self.type_depth,
69            self.value_depth,
70            self.value_nodes,
71            self.work,
72        ]
73    }
74
75    /// Reads an already shape-checked contract tuple; positive/hard bounds remain a validator's job.
76    #[must_use]
77    pub const fn from_values(v: [u64; 15]) -> Self {
78        Self {
79            bundles: v[0],
80            captured_bytes: v[1],
81            dependencies_per_bundle: v[2],
82            dependency_edges: v[3],
83            dependency_depth: v[4],
84            total_types: v[5],
85            total_fields: v[6],
86            alternatives_per_type: v[7],
87            total_alternatives: v[8],
88            choices_per_field: v[9],
89            total_choices: v[10],
90            type_depth: v[11],
91            value_depth: v[12],
92            value_nodes: v[13],
93            work: v[14],
94        }
95    }
96
97    /// Intersects producer and independent consumer policies without relaxing either.
98    #[must_use]
99    pub fn intersect(self, other: Self) -> Self {
100        let left = self.values();
101        let right = other.values();
102        Self::from_values(std::array::from_fn(|i| left[i].min(right[i])))
103    }
104}

106/// Complete resolved source signature, including all unused default/restriction facts.

115/// One private or public declaration; type declarations never carry a binding value.
116#[derive(Clone, Debug, Eq, PartialEq)]
117pub struct CompositionDeclaration {
118    /// Exact source module-symbol identity.
119    pub identity: ModuleSymbolIdentity,
120    /// Explicit source visibility.
121    pub public: bool,
122    /// Complete canonical signature.
123    pub signature: CompositionSignature,
124    /// Fully materialized meaning, absent exactly for nominal definitions.
125    pub value: Option<BindingValue>,
126}

128/// Attribution evidence, not authentication or a host acquisition hint.
129#[derive(Clone, Debug, Eq, PartialEq)]
130pub enum CompositionAttribution {
131    /// Checked original source bytes, distinct from a host filesystem path.
132    Source(SourceLocation),
133    /// Exact vocabulary contract owner with an optional checked bundle-byte span.
134    Vocabulary {
135        /// Canonical bundle identity.
136        identity: String,
137        /// Exact semantic revision.
138        version: String,
139        /// Nominal contract name.
140        type_name: String,
141        /// Field whose closed default supplied this occurrence.
142        field_name: String,
143        /// Optional original-byte bundle occurrence; absence is explicit.
144        span: Option<ByteSpan>,
145    },
146}

161/// Independently recomputable retained contract/value facts, never execution-work observations.
162#[derive(Clone, Copy, Debug, Eq, PartialEq)]
163pub struct CompositionResourceFacts {
164    /// Exact canonical bundle count.
165    pub bundles: u64,
166    /// Direct dependency edges after diamond deduplication.
167    pub dependency_edges: u64,

174    /// All finite choices.
175    pub choices: u64,

177    pub retained_value_nodes: u64,

234    pub vocabulary_sources: Vec<ProjectVocabularySource>,

237    /// Frozen independent composition acceptance controls.
