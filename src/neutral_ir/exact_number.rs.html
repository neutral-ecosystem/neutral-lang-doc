3//! Exact decimal values shared by parsing, IR and identity construction.
4//!
5//! Numbers retain a sign, decimal coefficient and scale instead of converting
6//! through floating point. Source parsing checks digit and scale budgets before
7//! normalization; equivalent spellings share a normalized representation.
8//! Display and hash-transcript emission consume that same representation.
9
10use crate::IrError;
11use neutral_core::allocation::RetainCapacity;
12use neutral_core::{CoreError, nht_frame};
13use std::fmt;
14
15/// Exact normalized base-10 rational representation used by Neutral IR.
16#[derive(Clone, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
17pub struct ExactNumber {
18    /// Whether the normalized value is negative.
19    pub(super) negative: bool,
20    /// Magnitude digits without separators or leading zeroes except zero itself.
21    pub(super) coefficient: String,
22    /// Signed base-10 scale applied to the coefficient.
23    pub(super) scale: i64,
24}
25
26impl ExactNumber {
27    /// Reconstructs one validated normalized exact number from external parts.
28    ///
29    /// The value is `sign * coefficient * 10^scale`. Nonzero coefficients have
30    /// neither leading nor trailing zeroes; zero has one spelling, positive with
31    /// scale zero. External decoders must reject noncanonical parts, not silently
32    /// normalize them, so equivalent wire values cannot gain multiple identities.
33    ///
34    /// # Errors
35    ///
36    /// Returns an error when the coefficient is noncanonical or the configured
37    /// digit or absolute-scale limit is exceeded.
38    pub fn from_normalized_parts(
39        negative: bool,
40        coefficient: &str,
41        scale: i64,
42        maximum_digits: u64,
43        maximum_scale: u64,
44    ) -> Result<Self, IrError> {
45        let digit_count = u64::try_from(coefficient.len()).unwrap_or(u64::MAX);
46        if digit_count > maximum_digits || scale.unsigned_abs() > maximum_scale {
47            return Err(IrError::ExactNumberLimitExceeded);
48        }
49        let valid_digits =
50            !coefficient.is_empty() && coefficient.bytes().all(|byte| byte.is_ascii_digit());
51        let valid_zero = coefficient == "0" && !negative && scale == 0;
52        let valid_nonzero = coefficient
53            .starts_with(|character: char| ('1'..='9').contains(&character))
54            && !coefficient.ends_with('0');
55        if !valid_digits || (!valid_zero && !valid_nonzero) {
56            return Err(IrError::InvalidExactNumber);
57        }
58        Ok(Self {
59            negative,
60            coefficient: neutral_core::allocation::text(coefficient)
61                .map_err(|_| IrError::ExactNumberLimitExceeded)?,
62            scale,
63        })
64    }
65
66    /// Normalizes one frozen source number without floating-point conversion.
67    ///
68    /// Unlike the external-parts constructor, this accepts source spelling
69    /// variations. Fractional digits and exponents become a decimal scale, then
70    /// insignificant zeroes are removed without changing the exact value.
71    /// Limits apply during parsing as well as after normalization; spelling a
72    /// tiny value with excessive digits does not bypass the input-work bound.
73    ///
74    /// # Errors
75    ///
76    /// Returns an error for a malformed spelling or an exceeded digit/scale bound.
77    pub fn from_source(
78        spelling: &str,
79        maximum_digits: u64,
80        maximum_scale: u64,
81    ) -> Result<Self, IrError> {
82        let mut parsed = parse_source_number(spelling, maximum_digits, maximum_scale)?;
83        let leading_zeroes =
84            parsed.coefficient.len() - parsed.coefficient.trim_start_matches('0').len();
85        if leading_zeroes == parsed.coefficient.len() {
86            parsed.coefficient.clear();
87            parsed.coefficient.push('0');
88            // All signed/exponent spellings of zero share one canonical value.
89            return Ok(Self {
90                negative: false,
91                coefficient: parsed.coefficient,
92                scale: 0,
93            });
94        }
95        parsed.coefficient.drain(..leading_zeroes);
96        let mut coefficient = parsed.coefficient;
97        let mut scale = parsed.scale;
98        // Removing a coefficient zero multiplies the power of ten by one.
99        while coefficient.ends_with('0') {
100            coefficient.pop();
101            scale = scale
102                .checked_add(1)
103                .ok_or(IrError::ExactNumberLimitExceeded)?;
104        }
105        if scale.unsigned_abs() > maximum_scale {
106            return Err(IrError::ExactNumberLimitExceeded);
107        }
108        Ok(Self {
109            negative: parsed.negative,
110            coefficient,
111            scale,
112        })
113    }
114
115    /// Normalizes one unsigned digits-only integer for minimal compatibility tests.
116    ///
117    /// # Errors
118    ///
119    /// Returns an error when the spelling is empty or contains non-digits.
120    pub fn from_unsigned_integer(spelling: &str) -> Result<Self, IrError> {
121        let maximum_digits = u64::try_from(spelling.len()).unwrap_or(u64::MAX).max(1);
122        Self::from_source(spelling, maximum_digits, maximum_digits)
123    }
124
125    /// Returns whether the normalized number is negative.
126    #[must_use]
127    pub const fn is_negative(&self) -> bool {
128        self.negative
129    }
130
131    /// Returns normalized coefficient digits.
132    #[must_use]
133    pub fn coefficient(&self) -> &str {
134        &self.coefficient
135    }
136
137    /// Returns the signed base-10 scale.
138    #[must_use]
139    pub const fn scale(&self) -> i64 {
140        self.scale
141    }
142
143    /// Builds the frozen NHT logical-definition payload for this number.
144    ///
145    /// # Errors
146    ///
147    /// Returns an error only when fixed transcript lengths cannot be represented.
148    pub fn nht_payload(&self) -> Result<Vec<u8>, CoreError> {
149        let mut payload = Vec::new();
150        payload.extend(nht_frame("sign", &[u8::from(self.negative)])?);
151        payload.extend(nht_frame("coefficient", self.coefficient.as_bytes())?);
152        payload.extend(nht_frame(
153            "scale-sign",
154            &[u8::from(self.scale.is_negative())],
155        )?);
156        payload.extend(nht_frame(
157            "scale",
158            self.scale.unsigned_abs().to_string().as_bytes(),
159        )?);
160        Ok(payload)
161    }
162}

164/// Parsed numeric components retained only until exact normalization completes.
165struct ParsedSourceNumber {
166    /// Whether the source spelling carries a minus sign.
167    negative: bool,
168    /// Coefficient digits without separators.
169    coefficient: String,
170    /// Decimal scale before trailing-zero normalization.
171    scale: i64,
172}
173
174/// Parses the frozen numeric grammar before allocating its normalized coefficient.
175fn parse_source_number(
176    spelling: &str,
177    maximum_digits: u64,
178    maximum_scale: u64,
179) -> Result<ParsedSourceNumber, IrError> {
180    if spelling.is_empty() || maximum_digits == 0 || maximum_scale == 0 {
181        return Err(IrError::InvalidExactNumber);
182    }
183    let bytes = spelling.as_bytes();
184    let mut index = 0;
185    let negative = match bytes.first() {
186        Some(b'+') => {
187            index = 1;
188            false
189        }
190        Some(b'-') => {
191            index = 1;
192            true
193        }
194        _ => false,
195    };
196    let integer = consume_digit_run(bytes, &mut index)?;
197    let fraction = if bytes.get(index) == Some(&b'.') {
198        index += 1;
199        consume_digit_run(bytes, &mut index)?
200    } else {
201        DigitRun::empty(index)
202    };
203    let mut exponent = 0_i64;
204    if matches!(bytes.get(index), Some(b'e' | b'E')) {
205        index += 1;
206        let exponent_negative = if matches!(bytes.get(index), Some(b'+' | b'-')) {
207            let value = bytes[index] == b'-';
208            index += 1;
209            value
210        } else {
211            false
212        };
213        let digits = consume_digit_run(bytes, &mut index)?;
214        exponent = parse_bounded_exponent(
215            &spelling[digits.start..digits.end],
216            maximum_scale,
217            exponent_negative,
218        )?;
219    }
220    if index != bytes.len() {
221        return Err(IrError::InvalidExactNumber);
222    }
223    let digit_count = integer.digit_count.saturating_add(fraction.digit_count);
224    if u64::try_from(digit_count).unwrap_or(u64::MAX) > maximum_digits {
225        return Err(IrError::ExactNumberLimitExceeded);
226    }
227    // Concatenating fractional digits into the integer coefficient shifts the
228    // decimal point right; subtract their count from the source exponent.
229    let fraction_scale =
230        i64::try_from(fraction.digit_count).map_err(|_| IrError::ExactNumberLimitExceeded)?;
231    let scale = exponent
232        .checked_sub(fraction_scale)
233        .ok_or(IrError::ExactNumberLimitExceeded)?;
234    if scale.unsigned_abs() > maximum_scale {
235        return Err(IrError::ExactNumberLimitExceeded);
236    }
237    let mut coefficient = String::new();
238    coefficient
239        .try_retain_exact(digit_count)
240        .map_err(|_| IrError::ExactNumberLimitExceeded)?;
241    coefficient.extend(
242        spelling[integer.start..integer.end]
243            .bytes()
244            .filter(u8::is_ascii_digit)
245            .map(char::from),
246    );
247    coefficient.extend(
248        spelling[fraction.start..fraction.end]
249            .bytes()
250            .filter(u8::is_ascii_digit)
251            .map(char::from),
252    );
253    Ok(ParsedSourceNumber {
254        negative,
255        coefficient,
256        scale,
257    })
258}

260/// One validated source digit-run range without a proportional digit buffer.
261struct DigitRun {
262    /// First included source byte.
263    start: usize,
264    /// First excluded source byte.
265    end: usize,
266    /// Number of decimal digits excluding separators.
267    digit_count: usize,
268}
269
270impl DigitRun {
271    /// Creates an empty omitted fractional run at one source position.
272    const fn empty(index: usize) -> Self {
273        Self {
274            start: index,
275            end: index,
276            digit_count: 0,
277        }
278    }
279}
280
281/// Consumes a nonempty decimal digit run with separators only between digits.
282fn consume_digit_run(bytes: &[u8], index: &mut usize) -> Result<DigitRun, IrError> {
283    if !bytes.get(*index).is_some_and(u8::is_ascii_digit) {
284        return Err(IrError::InvalidExactNumber);
285    }
286    let start = *index;
287    let mut digit_count = 0_usize;
288    while let Some(byte) = bytes.get(*index) {
289        if byte.is_ascii_digit() {
290            digit_count = digit_count.saturating_add(1);
291            *index += 1;
292        } else if *byte == b'_' {
293            if !bytes
294                .get((*index).saturating_sub(1))
295                .is_some_and(u8::is_ascii_digit)
296                || !bytes.get(*index + 1).is_some_and(u8::is_ascii_digit)
297            {
298                return Err(IrError::InvalidExactNumber);
299            }
300            *index += 1;
301        } else {
302            break;
303        }
304    }
305    Ok(DigitRun {
306        start,
307        end: *index,
308        digit_count,
309    })
310}

312/// Parses a bounded decimal exponent without host floating-point conversion.
313fn parse_bounded_exponent(
314    digits: &str,
315    maximum_scale: u64,
316    negative: bool,
317) -> Result<i64, IrError> {
318    let mut magnitude = 0_u64;
319    for byte in digits.bytes().filter(u8::is_ascii_digit) {
320        magnitude = magnitude
321            .checked_mul(10)
322            .and_then(|value| value.checked_add(u64::from(byte - b'0')))
323            .ok_or(IrError::ExactNumberLimitExceeded)?;
324        if magnitude > maximum_scale {
325            return Err(IrError::ExactNumberLimitExceeded);
326        }
327    }
328    let magnitude = i64::try_from(magnitude).map_err(|_| IrError::ExactNumberLimitExceeded)?;
329    Ok(if negative { -magnitude } else { magnitude })
330}

332impl fmt::Display for ExactNumber {
333    /// Formats the normalized exact rational used by reader and probe output.
334    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
335        if self.negative {
336            formatter.write_str("-")?;
337        }
338        if self.scale == 0 {
339            write!(formatter, "{}/1", self.coefficient)
340        } else {
341            write!(formatter, "{}e{}/1", self.coefficient, self.scale)
342        }
343    }
