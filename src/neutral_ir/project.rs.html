3//! Complete project data; constructors never imply independent validation.
4//!
5//! Keep logical declarations, types, and values distinct from source maps,
6//! provenance, captured digests, limits, and resource accounting. The former
7//! express meaning; the latter explain where it came from and how it was built.
8//! Logical comparison and identity deliberately exclude those companions, but
9//! reader validation must still check their consistency with the complete data.
10//! A view is a post-validation public selection, never a pruned compilation input.
11

163impl ProjectLimits {

167        Self {
168            modules: self.modules.min(other.modules),
169            declarations: self.declarations.min(other.declarations),
170            import_edges: self.import_edges.min(other.import_edges),
171            nodes: self.nodes.min(other.nodes),
172            text_bytes: self.text_bytes.min(other.text_bytes),
173            artifact_bytes: self.artifact_bytes.min(other.artifact_bytes),
174        }
175    }

218impl ProjectIr {
219    /// Checks schema work/depth bounds before a consumer clones untrusted types.
220    #[must_use]
221    pub fn within_schema_limits(&self, limits: ProjectLimits) -> bool {
222        if self.modules.len() as u64 > limits.modules
223            || self.declarations.len() as u64 > limits.declarations
224            || self.public_interface.exports().len() as u64 > limits.declarations
225            || self.public_interface.edges().len() as u64 > limits.nodes
226            || self.public_interface.vocabularies().len() as u64 > limits.nodes
227            || self.vocabulary_records.len() as u64 > limits.nodes
228            || !bounded_names(self, limits)
229        {
230            return false;
231        }
232        let mut count = 0_u64;
233        for signature in self.declarations.iter().map(|decl| &decl.signature).chain(
234            self.public_interface
235                .exports()
236                .iter()
237                .map(crate::project_interface::ProjectPublicExport::signature),
238        ) {
239            match signature {
240                ProjectPublicSignature::Binding(ty) => {
241                    if !bounded_type(ty, &mut count, limits) {
242                        return false;
243                    }
244                }
245                ProjectPublicSignature::Record(fields) => {
246                    let Some(next) = count.checked_add(fields.len() as u64) else {
247                        return false;
248                    };
249                    count = next;
250                    if count > limits.nodes {
251                        return false;
252                    }
253                    for field in fields {
254                        if field.name().len() as u64 > limits.text_bytes
255                            || !bounded_type(field.ty(), &mut count, limits)
256                        {
257                            return false;
258                        }
259                    }
260                }
261            }
262        }
263        for record in &self.vocabulary_records {
264            let Some(next) = count.checked_add(record.fields.len() as u64) else {
265                return false;
266            };
267            count = next;
268            if count > limits.nodes {
269                return false;
270            }
271            for (name, ty) in &record.fields {
272                if name.len() as u64 > limits.text_bytes || !bounded_type(ty, &mut count, limits) {
273                    return false;
274                }
275            }
276        }
277        true
278    }

280    /// Derives the exact redacted export index from complete validated content.
281    ///
282    /// # Errors
283    /// Returns an error when public fingerprint framing exceeds representable lengths.
284    pub fn recompute_public_interface(&self) -> Result<ProjectInterface, CoreError> {
285        use crate::project_interface::{ProjectPublicEdge, ProjectPublicExport};
286        let public = self
287            .declarations
288            .iter()
289            .filter(|decl| decl.public)
290            .map(|decl| &decl.identity)
291            .collect::<BTreeSet<_>>();
292        let exports = self
293            .declarations
294            .iter()
295            .filter(|decl| decl.public)
296            .map(|decl| ProjectPublicExport::new(decl.identity.clone(), decl.signature.clone()))
297            .collect();
298        let mut edges = self
299            .provenance
300            .iter()
301            .filter(|edge| public.contains(&edge.from) && public.contains(&edge.to))
302            .map(|edge| (edge.from.clone(), edge.kind, edge.to.clone()))
303            .collect::<BTreeSet<_>>();
304        for decl in self.declarations.iter().filter(|decl| decl.public) {
305            if let Some(value) = &decl.value {
306                for target in value.references() {
307                    edges.insert((
308                        decl.identity.clone(),
309                        ProjectPublicEdgeKind::Reference,
310                        target.clone(),
311                    ));
312                }
313            }
314        }
315        let edges = edges
316            .into_iter()
317            .map(|(from, kind, to)| ProjectPublicEdge::new(from, to, kind))
318            .collect();
319        ProjectInterface::with_vocabularies(
320            self.public_interface.vocabularies().to_vec(),
321            exports,
322            edges,
323        )
324    }
325    /// Compares complete logical content, excluding source and processing evidence.
326    #[must_use]
327    pub fn logical_eq(&self, other: &Self) -> bool {
328        self.schema == other.schema
329            && self.modules == other.modules
330            && self.declarations == other.declarations
331            && self.vocabulary_records == other.vocabulary_records
332            && self.public_interface == other.public_interface
333    }
334}

336/// Bounds one linear nested type without recursion or proportional allocation.
337fn bounded_type(mut ty: &ProjectPublicType, count: &mut u64, limits: ProjectLimits) -> bool {
338    let mut depth = 0;
339    loop {
340        let Some(next) = count.checked_add(1) else {
341            return false;
342        };
343        *count = next;
344        if *count > limits.nodes || depth > PROJECT_MAX_DEPTH {
345            return false;
346        }
347        match ty {
348            ProjectPublicType::List(inner)
349            | ProjectPublicType::Nullable(inner)
350            | ProjectPublicType::Ref(inner) => {
351                ty = inner;
352                depth += 1;
353            }
354            ProjectPublicType::Nominal(identity) => {
355                return bounded_symbol(identity, limits.text_bytes);
356            }
357            ProjectPublicType::VocabularyNominal {
358                identity,
359                version,
360                name,
361            } => {
362                return [identity, version, name]
363                    .iter()
364                    .all(|value| value.len() as u64 <= limits.text_bytes);
365            }
366            _ => return true,
367        }
368    }
369}

371/// Checks each logical identity before name parsing or public-interface copying.
372fn bounded_symbol(identity: &ModuleSymbolIdentity, text_bytes: u64) -> bool {
373    identity.declaration_name().len() as u64 <= text_bytes
374        && identity.module().module_name().len() as u64 <= text_bytes
375        && identity.module().language_behavior_version().len() as u64 <= text_bytes
376}

378/// Bounds schema names and catalogues before any proportional validation work.
379fn bounded_names(ir: &ProjectIr, limits: ProjectLimits) -> bool {
380    for module in &ir.modules {
381        if module.identity.module_name().len() as u64 > limits.text_bytes
382            || module.imports.len() as u64 > limits.import_edges
383            || module
384                .imports
385                .iter()
386                .any(|target| target.len() as u64 > limits.text_bytes)
387        {
388            return false;
389        }
390    }
391    let identities = ir
392        .declarations
393        .iter()
394        .map(|decl| &decl.identity)
395        .chain(
396            ir.public_interface
397                .exports()
398                .iter()
399                .map(crate::project_interface::ProjectPublicExport::identity),
400        )
401        .chain(
402            ir.public_interface
403                .edges()
404                .iter()
405                .flat_map(|edge| [edge.from(), edge.to()]),
406        );
407    if identities
408        .into_iter()
409        .any(|identity| !bounded_symbol(identity, limits.text_bytes))
410    {
411        return false;
412    }
413    let mut names = 0_u64;
414    for vocabulary in ir.public_interface.vocabularies() {
415        let Some(next) = names.checked_add(vocabulary.public_types().len() as u64) else {
416            return false;
417        };
418        names = next;
419        if names > limits.nodes
420            || vocabulary.identity().len() as u64 > limits.text_bytes
421            || vocabulary.version().len() as u64 > limits.text_bytes
422            || vocabulary
423                .public_types()
424                .iter()
425                .any(|name| name.len() as u64 > limits.text_bytes)
426        {
427            return false;
428        }
429    }
430    ir.vocabulary_records.iter().all(|record| {
431        [&record.identity, &record.version, &record.name]
432            .iter()
433            .all(|name| name.len() as u64 <= limits.text_bytes)
434    })
435}

437impl ProjectValue {
438    /// Collects typed identity targets iteratively without following identity edges.
439    #[must_use]
440    pub fn references(&self) -> Vec<&ModuleSymbolIdentity> {
441        let mut targets = Vec::new();
442        let mut pending = vec![self];
443        while let Some(value) = pending.pop() {
444            match value {
445                Self::Reference(target) => targets.push(target),
446                Self::List(values) => pending.extend(values),
447                Self::Record(fields) => pending.extend(fields.iter().map(|(_, value)| value)),
448                _ => {}
449            }
450        }
451        targets
452    }
453}
