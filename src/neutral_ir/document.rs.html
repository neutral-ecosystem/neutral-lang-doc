3//! Document ownership and logical comparison independent of local labels.
4//!
5//! `LogicalDocument` groups module identity, record definitions, declarations
6//! and the captured vocabulary contract. Equivalence matches symbols and builds
7//! a bijection between graph-local element IDs before comparing reference edges.
8//! Source spans, provenance and derivation policy belong to separate companions.
9
10use crate::{
11    Declaration, ElementId, LogicalModuleIdentity, LogicalValue, ModuleSymbolIdentity,
12    RecordTypeDefinition, VocabularyContract,
13};
14use std::collections::BTreeMap;
15
16/// Immutable validated logical Neutral document.
17#[derive(Clone, Debug, Eq, PartialEq)]
18pub struct LogicalDocument {
19    /// Logical module identity.
20    module: LogicalModuleIdentity,
21    /// Exported nominal record declarations in canonical name order.
22    record_types: Vec<RecordTypeDefinition>,
23    /// Optional exact captured vocabulary contract.
24    vocabulary: Option<VocabularyContract>,
25    /// Exported declarations in deterministic source order.
26    pub(super) declarations: Vec<Declaration>,
27}
28
29impl LogicalDocument {
30    /// Creates a validated logical document.
31    #[must_use]
32    pub fn new(module: LogicalModuleIdentity, declarations: Vec<Declaration>) -> Self {
33        Self {
34            module,
35            record_types: Vec::new(),
36            vocabulary: None,
37            declarations,
38        }
39    }
40
41    /// Creates a validated logical document containing nominal records and bindings.
42    #[must_use]
43    pub fn with_record_types(
44        module: LogicalModuleIdentity,
45        record_types: Vec<RecordTypeDefinition>,
46        declarations: Vec<Declaration>,
47    ) -> Self {
48        Self {
49            module,
50            record_types,
51            vocabulary: None,
52            declarations,
53        }
54    }
55
56    /// Attaches the exact validated captured vocabulary contract.
57    #[must_use]
58    pub fn with_vocabulary(mut self, vocabulary: VocabularyContract) -> Self {
59        self.vocabulary = Some(vocabulary);
60        self
61    }
62
63    /// Returns the exact captured vocabulary contract, when used.
64    #[must_use]
65    pub const fn vocabulary(&self) -> Option<&VocabularyContract> {
66        self.vocabulary.as_ref()
67    }
68
69    /// Returns the logical module identity.
70    #[must_use]
71    pub const fn module(&self) -> &LogicalModuleIdentity {
72        &self.module
73    }
74
75    /// Returns exported nominal records in canonical name order.
76    #[must_use]
77    pub fn record_types(&self) -> &[RecordTypeDefinition] {
78        &self.record_types
79    }
80
81    /// Finds one exported nominal record by its validated name.
82    #[must_use]
83    pub fn record_type_by_name(&self, name: &str) -> Option<&RecordTypeDefinition> {
84        self.record_types
85            .iter()
86            .find(|record| record.name() == name)
87    }
88
89    /// Returns exported declarations in deterministic order.
90    #[must_use]
91    pub fn declarations(&self) -> &[Declaration] {
92        &self.declarations
93    }
94
95    /// Compares the complete logical payload under one bijective element-ID mapping.
96    ///
97    /// Companion source maps, provenance, derivation facts, and graph-local ID
98    /// spellings are intentionally outside this comparison. Both documents must
99    /// still contain unique node IDs and references to declared binding nodes.
100    #[must_use]
101    pub fn logically_equivalent(&self, other: &Self) -> bool {
102        let Some(mapping) = ElementIdMapping::between(self, other) else {
103            return false;
104        };
105        self.module == other.module
106            && match (&self.vocabulary, &other.vocabulary) {
107                (Some(left), Some(right)) => left.logically_equivalent(right),
108                (None, None) => true,
109                _ => false,
110            }
111            && logical_record_types_equal(self, other)
112            && logical_declarations_equal(self, other, &mapping)
113    }
114}

116/// One temporary bijection between graph-local IDs in two logical documents.
117struct ElementIdMapping {
118    /// Left-to-right graph-node mapping.
119    forward: BTreeMap<ElementId, ElementId>,
120    /// Right-to-left graph-node mapping proving injectivity.
121    reverse: BTreeMap<ElementId, ElementId>,
122}
123
124impl ElementIdMapping {
125    /// Builds one complete mapping by matching durable module-symbol identities.
126    fn between(left: &LogicalDocument, right: &LogicalDocument) -> Option<Self> {
127        if left.record_types.len() != right.record_types.len()
128            || left.declarations.len() != right.declarations.len()
129        {
130            return None;
131        }
132        let mut mapping = Self {
133            forward: BTreeMap::new(),
134            reverse: BTreeMap::new(),
135        };
136        let right_records = unique_records_by_symbol(right)?;
137        for record in &left.record_types {
138            mapping.insert(
139                record.element_id(),
140                right_records.get(record.symbol_identity())?.element_id(),
141            )?;
142        }
143        let right_declarations = unique_declarations_by_symbol(right)?;
144        for declaration in &left.declarations {
145            mapping.insert(
146                declaration.element_id(),
147                right_declarations
148                    .get(declaration.symbol_identity())?
149                    .element_id(),
150            )?;
151        }
152        (mapping.forward.len() == left.record_types.len() + left.declarations.len())
153            .then_some(mapping)
154    }
155
156    /// Adds one pair only when both graph labels remain one-to-one.
157    fn insert(&mut self, left: ElementId, right: ElementId) -> Option<()> {
158        if self.forward.insert(left, right).is_some() || self.reverse.insert(right, left).is_some()
159        {
160            return None;
161        }
162        Some(())
163    }
164
165    /// Maps one left document label into the right document.
166    fn target(&self, source: ElementId) -> Option<ElementId> {
167        self.forward.get(&source).copied()
168    }
169}

171/// Indexes record declarations by durable identity while rejecting duplicates.
172fn unique_records_by_symbol(
173    document: &LogicalDocument,
174) -> Option<BTreeMap<&ModuleSymbolIdentity, &RecordTypeDefinition>> {
175    let mut records = BTreeMap::new();
176    for record in &document.record_types {
177        if records.insert(record.symbol_identity(), record).is_some() {
178            return None;
179        }
180    }
181    Some(records)
182}
183
184/// Indexes binding declarations by durable identity while rejecting duplicates.
185fn unique_declarations_by_symbol(
186    document: &LogicalDocument,
187) -> Option<BTreeMap<&ModuleSymbolIdentity, &Declaration>> {
188    let mut declarations = BTreeMap::new();
189    for declaration in &document.declarations {
190        if declarations
191            .insert(declaration.symbol_identity(), declaration)
192            .is_some()
193        {
194            return None;
195        }
196    }
197    Some(declarations)
198}

200/// Compares nominal record payloads without graph-local IDs or vector order.
201fn logical_record_types_equal(left: &LogicalDocument, right: &LogicalDocument) -> bool {
202    let (Some(left_records), Some(right_records)) = (
203        unique_records_by_symbol(left),
204        unique_records_by_symbol(right),
205    ) else {
206        return false;
207    };
208    left_records.iter().all(|(identity, left_record)| {
209        right_records.get(identity).is_some_and(|right_record| {
210            left_record.fingerprint() == right_record.fingerprint()
211                && left_record.nominal_identity() == right_record.nominal_identity()
212                && left_record.fields() == right_record.fields()
213        })
214    })
215}

217/// Compares binding payloads and every nested identity edge under one mapping.
218fn logical_declarations_equal(
219    left: &LogicalDocument,
220    right: &LogicalDocument,
221    mapping: &ElementIdMapping,
222) -> bool {
223    let (Some(left_declarations), Some(right_declarations)) = (
224        unique_declarations_by_symbol(left),
225        unique_declarations_by_symbol(right),
226    ) else {
227        return false;
228    };
229    left_declarations
230        .iter()
231        .all(|(identity, left_declaration)| {
232            right_declarations
233                .get(identity)
234                .is_some_and(|right_declaration| {
235                    left_declaration.fingerprint() == right_declaration.fingerprint()
236                        && left_declaration.name() == right_declaration.name()
237                        && left_declaration.resolved_type() == right_declaration.resolved_type()
238                        && logical_values_equal(
239                            left_declaration.value(),
240                            right_declaration.value(),
241                            mapping,
242                            &left_declarations,
243                            &right_declarations,
244                        )
245                })
246        })
247}

249/// Compares recursively nested values while translating identity-edge targets.
250fn logical_values_equal(
251    left: &LogicalValue,
252    right: &LogicalValue,
253    mapping: &ElementIdMapping,
254    left_declarations: &BTreeMap<&ModuleSymbolIdentity, &Declaration>,
255    right_declarations: &BTreeMap<&ModuleSymbolIdentity, &Declaration>,
256) -> bool {
257    match (left, right) {
258        (LogicalValue::Number(left), LogicalValue::Number(right)) => left == right,
259        (LogicalValue::String(left), LogicalValue::String(right)) => left == right,
260        (LogicalValue::Boolean(left), LogicalValue::Boolean(right)) => left == right,
261        (LogicalValue::Null, LogicalValue::Null) => true,
262        (LogicalValue::Record(left), LogicalValue::Record(right)) => {
263            left.nominal_type() == right.nominal_type()
264                && left.fields().len() == right.fields().len()
265                && left
266                    .fields()
267                    .iter()
268                    .zip(right.fields())
269                    .all(|(left_field, right_field)| {
270                        left_field.name() == right_field.name()
271                            && logical_values_equal(
272                                left_field.value(),
273                                right_field.value(),
274                                mapping,
275                                left_declarations,
276                                right_declarations,
277                            )
278                    })
279        }
280        (LogicalValue::VocabularyRecord(left), LogicalValue::VocabularyRecord(right)) => {
281            left.nominal_type() == right.nominal_type()
282                && left.fields().len() == right.fields().len()
283                && left
284                    .fields()
285                    .iter()
286                    .zip(right.fields())
287                    .all(|(left_field, right_field)| {
288                        left_field.name() == right_field.name()
289                            && logical_values_equal(
290                                left_field.value(),
291                                right_field.value(),
292                                mapping,
293                                left_declarations,
294                                right_declarations,
295                            )
296                    })
297        }
298        (LogicalValue::List(left), LogicalValue::List(right)) => {
299            left.len() == right.len()
300                && left.iter().zip(right).all(|(left_item, right_item)| {
301                    logical_values_equal(
302                        left_item,
303                        right_item,
304                        mapping,
305                        left_declarations,
306                        right_declarations,
307                    )
308                })
309        }
310        (LogicalValue::Reference(left), LogicalValue::Reference(right)) => {
311            left_declarations
312                .get(left.target_symbol_identity())
313                .is_some_and(|target| target.element_id() == left.target_element_id())
314                && right_declarations
315                    .get(right.target_symbol_identity())
316                    .is_some_and(|target| target.element_id() == right.target_element_id())
317                && mapping.target(left.target_element_id()) == Some(right.target_element_id())
318                && left.target_symbol_identity() == right.target_symbol_identity()
319                && left.target_type() == right.target_type()
320        }
321        _ => false,
322    }
