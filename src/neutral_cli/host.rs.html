24/// Monotonic process-local suffix for atomic temporary outputs.
25static TEMPORARY_SEQUENCE: AtomicU64 = AtomicU64::new(0);
26
27/// Stable process outcome classes exposed by the command-line boundary.
28#[derive(Clone, Copy, Debug, Eq, PartialEq)]
29pub enum ExitClass {
30    /// Invalid command syntax or option policy.
31    Usage,
32    /// Source or vocabulary input could not be acquired.
33    Input,
34    /// Captured content was rejected.
35    Validation,
36    /// Output could not be safely committed.
37    Output,
38    /// Work was cooperatively cancelled.
39    Cancelled,
40    /// An internal invariant failed.
41    Internal,
42}
43
44impl ExitClass {
45    /// Returns the stable operating-system process exit code.
46    #[must_use]
47    pub const fn code(self) -> u8 {
48        match self {
49            Self::Usage => constants::EXIT_USAGE,
50            Self::Input => constants::EXIT_INPUT,
51            Self::Validation => constants::EXIT_VALIDATION,
52            Self::Output => constants::EXIT_OUTPUT,
53            Self::Cancelled => constants::EXIT_CANCELLED,
54            Self::Internal => constants::EXIT_INTERNAL,
55        }
56    }
57}

59/// Safe categorized command failure with no host path or source disclosure.
60#[derive(Clone, Debug, Eq, PartialEq)]
61pub struct CliFailure {
62    /// Stable exit class.
63    class: ExitClass,
64    /// Safe bounded output lines without category prefixes.
65    messages: Vec<String>,
66}
67
68impl CliFailure {
69    /// Creates one safe single-line command failure.
70    fn one(class: ExitClass, message: impl Into<String>) -> Self {
71        Self {
72            class,
73            messages: vec![message.into()],
74        }
75    }
76
77    /// Returns the stable failure exit class.
78    #[must_use]
79    pub const fn class(&self) -> ExitClass {
80        self.class
81    }
82
83    /// Returns safe bounded failure lines.
84    #[must_use]
85    pub fn messages(&self) -> &[String] {
86        &self.messages
87    }
88}

90/// Parses and executes one command using only explicitly selected host resources.
91///
92/// # Errors
93///
94/// Returns a categorized, path-safe failure when usage is invalid, acquisition
95/// or validation fails, output cannot be committed, work is cancelled, or an
96/// internal invariant is violated.
97pub fn execute(arguments: impl IntoIterator<Item = String>) -> Result<(), CliFailure> {
98    match parse(arguments).map_err(|message| CliFailure::one(ExitClass::Usage, message))? {
99        ParseOutcome::Help(usage) => {
100            eprintln!("{} usage: {usage}", constants::INFO);
101            Ok(())
102        }
103        ParseOutcome::Version => {
104            eprintln!(
105                "{} {} {}",
106                constants::INFO,
107                env!("CARGO_PKG_NAME"),
108                env!("CARGO_PKG_VERSION")
109            );
110            Ok(())
111        }
112        ParseOutcome::Execute(options) => execute_command(&options),
113    }
114}
115
116/// Executes one validated command policy.
117fn execute_command(options: &CommandOptions) -> Result<(), CliFailure> {
118    let request = request_from_options(options)?;
119    match options.kind {
120        CommandKind::Compile => compile_command(request, options),
121        CommandKind::Validate => validate_command(request),
122        CommandKind::Format => format_command(request, options),
123    }
124}
125
126/// Compiles source and atomically publishes one encoded artifact.
127fn compile_command(
128    request: CompilationRequest,
129    options: &CommandOptions,
130) -> Result<(), CliFailure> {
131    let document = successful_document(compile(request).map_err(|error| capture_failure(&error))?)?;
132    let encoded = encode(
133        &document,
134        &ProducerInfo::new(env!("CARGO_PKG_NAME"), env!("CARGO_PKG_VERSION")),
135    )
136    .map_err(|_| CliFailure::one(ExitClass::Validation, "artifact-encoding-limit"))?;
137    write_selected_output(options, encoded.as_bytes())?;
138    eprintln!("{} compile succeeded", constants::INFO);
139    Ok(())
140}
141
142/// Validates source without publishing authoritative output.
143fn validate_command(request: CompilationRequest) -> Result<(), CliFailure> {
144    ensure_success(compile(request).map_err(|error| capture_failure(&error))?)?;
145    eprintln!("{} validation succeeded", constants::INFO);
146    Ok(())
147}
148
149/// Formats validated source and atomically publishes ordinary source bytes.
150fn format_command(request: CompilationRequest, options: &CommandOptions) -> Result<(), CliFailure> {
151    let formatted = format_source(request).map_err(format_failure)?;
152    write_selected_output(options, formatted.as_bytes())?;
153    eprintln!("{} format succeeded", constants::INFO);
154    Ok(())
155}

157/// Builds one captured-input request from explicit source, limits, and vocabulary facts.
158fn request_from_options(options: &CommandOptions) -> Result<CompilationRequest, CliFailure> {
159    let source = read_selected_input(&options.input, options.limits.source_bytes())?;
160    let cancellation = CancellationToken::new();
161    if options.cancel_before_start {
162        cancellation.cancel();
163    }
164    let request = CompilationRequest::new(source, options.limits, cancellation);
165    captured_vocabulary(request, &options.vocabulary, options.limits.source_bytes())
166}

168/// Attaches an exact explicitly acquired vocabulary bundle and immutable lock.
169fn captured_vocabulary(
170    request: CompilationRequest,
171    options: &VocabularyOptions,
172    byte_limit: u64,
173) -> Result<CompilationRequest, CliFailure> {
174    if options.is_empty() {
175        return Ok(request);
176    }
177    let bundle_path = required_option(options.bundle.as_ref())?;
178    let bytes = read_selected_input(bundle_path, byte_limit)?;
179    let digest = VocabularyContentDigest::parse_text(required_option(options.digest.as_ref())?)
180        .map_err(|_| CliFailure::one(ExitClass::Usage, "invalid-vocabulary-digest"))?;
181    let lock = VocabularyLock::new(
182        required_option(options.identity.as_ref())?,
183        required_option(options.version.as_ref())?,
184        required_option(options.encoding_version.as_ref())?,
185        required_option(options.schema_version.as_ref())?,
186        digest,
187        options.features.clone(),
188    )
189    .map_err(|_| CliFailure::one(ExitClass::Usage, "invalid-vocabulary-lock"))?;
190    Ok(request.with_captured_vocabulary(bytes, lock))
191}

193/// Borrows one parser-validated required option or reports an internal defect.
194fn required_option(value: Option<&String>) -> Result<&str, CliFailure> {
195    value
196        .map(String::as_str)
197        .ok_or_else(|| CliFailure::one(ExitClass::Internal, "missing-validated-option"))
198}

200/// Reads a file or standard input without exceeding the selected byte ceiling.
201fn read_selected_input(selection: &str, byte_limit: u64) -> Result<Vec<u8>, CliFailure> {
202    let reader: Box<dyn Read> = if selection == constants::STANDARD_STREAM {
203        Box::new(io::stdin())
204    } else {
205        Box::new(
206            File::open(selection)
207                .map_err(|_| CliFailure::one(ExitClass::Input, "input-read-failed"))?,
208        )
209    };
210    read_bounded(reader, byte_limit)
211}
212
213/// Reads at most one byte beyond a declared limit to classify oversized input.
214fn read_bounded(reader: Box<dyn Read>, byte_limit: u64) -> Result<Vec<u8>, CliFailure> {
215    let maximum_read = byte_limit.saturating_add(1);
216    let mut bytes = Vec::new();
217    reader
218        .take(maximum_read)
219        .read_to_end(&mut bytes)
220        .map_err(|_| CliFailure::one(ExitClass::Input, "input-read-failed"))?;
221    if u64::try_from(bytes.len()).unwrap_or(u64::MAX) > byte_limit {
222        Err(CliFailure::one(
223            ExitClass::Validation,
224            "input-limit-exceeded",
225        ))
226    } else {
227        Ok(bytes)
228    }
229}

231/// Opens authoritative artifacts as a validated document or maps their failure.
232fn successful_document(result: CompilationResult) -> Result<ValidatedDocument, CliFailure> {
233    match result {
234        CompilationResult::Success(artifacts) => ValidatedDocument::from_compiler_output(artifacts)
235            .map_err(|_| CliFailure::one(ExitClass::Internal, "reader-validation-defect")),
236        CompilationResult::Failure(failure) => Err(compilation_failure(&failure)),
237    }
238}
239
240/// Requires successful compilation when a command publishes no artifact.
241fn ensure_success(result: CompilationResult) -> Result<(), CliFailure> {
242    match result {
243        CompilationResult::Success(_) => Ok(()),
244        CompilationResult::Failure(failure) => Err(compilation_failure(&failure)),
245    }
246}

248/// Maps one capture failure to its stable CLI class without path disclosure.
249fn capture_failure(error: &CaptureError) -> CliFailure {
250    match error {
251        CaptureError::Cancelled => CliFailure::one(ExitClass::Cancelled, "cancelled"),
252        CaptureError::MissingSource => CliFailure::one(ExitClass::Validation, "missing-source"),
253        CaptureError::SourceLimitExceeded { .. } => {
254            CliFailure::one(ExitClass::Validation, "source-limit-exceeded")
255        }
256        CaptureError::SourceDigestMismatch { .. } => {
257            CliFailure::one(ExitClass::Validation, "source-digest-mismatch")
258        }
259    }
260}

262/// Maps a compilation failure to safe diagnostic code and byte-span lines.
263fn compilation_failure(failure: &CompilationFailure) -> CliFailure {
264    let class = if failure.class() == ResultClass::Cancellation {
265        ExitClass::Cancelled
266    } else {
267        ExitClass::Validation
268    };
269    let messages = if failure.diagnostics().is_empty() {
270        vec![if class == ExitClass::Cancelled {
271            "cancelled".to_owned()
272        } else {
273            "source-rejected".to_owned()
274        }]
275    } else {
276        failure
277            .diagnostics()
278            .iter()
279            .map(|diagnostic| {
280                let span = diagnostic.primary().span();
281                format!(
282                    "{} bytes {}..{}",
283                    diagnostic.code().as_str(),
284                    span.start(),
285                    span.end()
286                )
287            })
288            .collect()
289    };
290    CliFailure { class, messages }
291}

293/// Maps reference formatter failure classes to stable CLI outcomes.
294fn format_failure(error: FormatError) -> CliFailure {
295    match error {
296        FormatError::Capture(error) => capture_failure(&error),
297        FormatError::Compilation(failure) => compilation_failure(&failure),
298        FormatError::OutputLimitExceeded => {
299            CliFailure::one(ExitClass::Validation, "formatted-output-limit-exceeded")
300        }
301        FormatError::InternalDefect => {
302            CliFailure::one(ExitClass::Internal, "formatter-internal-defect")
303        }
304    }
305}

307/// Publishes bytes to the explicitly selected standard stream or file destination.
308fn write_selected_output(options: &CommandOptions, bytes: &[u8]) -> Result<(), CliFailure> {
309    let destination = options
310        .output
311        .as_deref()
312        .ok_or_else(|| CliFailure::one(ExitClass::Internal, "missing-validated-output"))?;
313    if destination == constants::STANDARD_STREAM {
314        let mut stdout = io::stdout().lock();
315        stdout
316            .write_all(bytes)
317            .and_then(|()| stdout.flush())
318            .map_err(|_| CliFailure::one(ExitClass::Output, "stdout-write-failed"))
319    } else {
320        atomic_write(Path::new(destination), bytes, options.overwrite)
321    }
322}

324/// Commits complete output through a same-directory temporary file.
325fn atomic_write(destination: &Path, bytes: &[u8], overwrite: bool) -> Result<(), CliFailure> {
326    if !overwrite && destination.exists() {
327        return Err(CliFailure::one(ExitClass::Output, "output-exists"));
328    }
329    let parent = destination.parent().unwrap_or_else(|| Path::new("."));
330    let (temporary_path, mut temporary_file) = create_temporary(parent)?;
331    let guard = TemporaryOutput::new(temporary_path.clone());
332    temporary_file
333        .write_all(bytes)
334        .and_then(|()| temporary_file.sync_all())
335        .map_err(|_| CliFailure::one(ExitClass::Output, "output-write-failed"))?;
336    drop(temporary_file);
337    if overwrite {
338        fs::rename(&temporary_path, destination)
339    } else {
340        fs::hard_link(&temporary_path, destination).and_then(|()| fs::remove_file(&temporary_path))
341    }
342    .map_err(|_| CliFailure::one(ExitClass::Output, "output-commit-failed"))?;
343    guard.commit();
344    Ok(())
345}

347/// Creates a collision-resistant temporary output in the destination directory.
348fn create_temporary(parent: &Path) -> Result<(PathBuf, File), CliFailure> {
349    for _ in 0..constants::MAXIMUM_TEMPORARY_ATTEMPTS {
350        let sequence = TEMPORARY_SEQUENCE.fetch_add(1, Ordering::Relaxed);
351        let name = format!(
352            "{}{}-{sequence}",
353            constants::TEMPORARY_MARKER,
354            std::process::id()
355        );
356        let path = parent.join(name);
357        match OpenOptions::new().write(true).create_new(true).open(&path) {
358            Ok(file) => return Ok((path, file)),
359            Err(error) if error.kind() == io::ErrorKind::AlreadyExists => {}
360            Err(_) => return Err(CliFailure::one(ExitClass::Output, "output-create-failed")),
361        }
362    }
363    Err(CliFailure::one(
364        ExitClass::Output,
365        "temporary-name-exhausted",
366    ))
367}

369/// Cleanup guard for an uncommitted atomic temporary output.
370struct TemporaryOutput {
371    /// Exact same-directory temporary path.
372    path: PathBuf,
373    /// Whether ownership was transferred to the final destination.
374    committed: std::cell::Cell<bool>,
375}
376
377impl TemporaryOutput {
378    /// Creates an armed cleanup guard.
379    fn new(path: PathBuf) -> Self {
380        Self {
381            path,
382            committed: std::cell::Cell::new(false),
383        }
384    }
385
386    /// Marks the temporary path as successfully committed.
387    fn commit(&self) {
388        self.committed.set(true);
389    }
390}
391
392impl Drop for TemporaryOutput {
393    /// Removes an uncommitted temporary file on every return path.
394    fn drop(&mut self) {
395        if !self.committed.get() {
396            let _ = fs::remove_file(&self.path);
397        }
398    }
399}
