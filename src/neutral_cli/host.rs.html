28/// Monotonic process-local suffix for atomic temporary outputs.
29static TEMPORARY_SEQUENCE: AtomicU64 = AtomicU64::new(0);

31/// Stable process outcome classes exposed by the command-line boundary.
32#[derive(Clone, Copy, Debug, Eq, PartialEq)]
33pub enum ExitClass {
34    /// Invalid command syntax or option policy.
35    Usage,
36    /// Source or vocabulary input could not be acquired.
37    Input,
38    /// Captured content was rejected.
39    Validation,
40    /// Output could not be safely committed.
41    Output,
42    /// Work was cooperatively cancelled.
43    Cancelled,
44    /// An internal invariant failed.
45    Internal,
46}
47
48impl ExitClass {
49    /// Returns the stable operating-system process exit code.
50    #[must_use]
51    pub const fn code(self) -> u8 {
52        match self {
53            Self::Usage => constants::EXIT_USAGE,
54            Self::Input => constants::EXIT_INPUT,
55            Self::Validation => constants::EXIT_VALIDATION,
56            Self::Output => constants::EXIT_OUTPUT,
57            Self::Cancelled => constants::EXIT_CANCELLED,
58            Self::Internal => constants::EXIT_INTERNAL,
59        }
60    }
61}

63/// Safe categorized command failure with no host path or source disclosure.
64#[derive(Clone, Debug, Eq, PartialEq)]
65pub struct CliFailure {
66    /// Stable exit class.
67    class: ExitClass,
68    /// Safe bounded output lines without category prefixes.
69    messages: Vec<String>,
70}
71
72impl CliFailure {
73    /// Creates one safe single-line command failure.
74    fn one(class: ExitClass, message: impl Into<String>) -> Self {
75        Self {
76            class,
77            messages: vec![message.into()],
78        }
79    }
80
81    /// Returns the stable failure exit class.
82    #[must_use]
83    pub const fn class(&self) -> ExitClass {
84        self.class
85    }
86
87    /// Returns safe bounded failure lines.
88    #[must_use]
89    pub fn messages(&self) -> &[String] {
90        &self.messages
91    }
92}

94/// Parses and executes one command using only explicitly selected host resources.
95///
96/// # Errors
97///
98/// Returns a categorized, path-safe failure when usage is invalid, acquisition
99/// or validation fails, output cannot be committed, work is cancelled, or an
100/// internal invariant is violated.
101pub fn execute(arguments: impl IntoIterator<Item = String>) -> Result<(), CliFailure> {
102    match parse(arguments).map_err(|message| CliFailure::one(ExitClass::Usage, message))? {
103        ParseOutcome::Help(usage) => {
104            eprintln!("{} usage: {usage}", constants::INFO);
105            Ok(())
106        }
107        ParseOutcome::Version => {
108            eprintln!(
109                "{} {} {}",
110                constants::INFO,
111                env!("CARGO_PKG_NAME"),
112                env!("CARGO_PKG_VERSION")
113            );
114            Ok(())
115        }
116        ParseOutcome::Profiles => {
117            report_profiles();
118            Ok(())
119        }
120        ParseOutcome::Execute(options) => execute_command(&options),
121    }
122}
123
124/// Reports the reader-visible profile catalogue in deterministic text form.
125fn report_profiles() {
126    for descriptor in neutral_reader::language_profiles() {
127        let capabilities = descriptor
128            .capabilities()
129            .iter()
130            .map(|capability| capability.as_str())
131            .collect::<Vec<_>>()
132            .join(",");
133        eprintln!(
134            "{} profile={} status={} capabilities={}",
135            constants::INFO,
136            descriptor.profile().source_version(),
137            descriptor.availability().as_str(),
138            if capabilities.is_empty() {
139                "none"
140            } else {
141                capabilities.as_str()
142            }
143        );
144    }
145}

147/// Executes one validated command policy.
148fn execute_command(options: &CommandOptions) -> Result<(), CliFailure> {
149    if options.kind == CommandKind::CaptureProject {
150        return capture_project_command(options);
151    }
152    let request = request_from_options(options)?;
153    match options.kind {
154        CommandKind::Compile => compile_command(request, options),
155        CommandKind::Validate => validate_command(request),
156        CommandKind::Format => format_command(request, options),
157        CommandKind::CaptureProject => unreachable!("project capture returned before compilation"),
158    }
159}

161/// Acquires explicit inputs, constructs the shared request schema, and captures it.
162fn capture_project_command(options: &CommandOptions) -> Result<(), CliFailure> {
163    let source = read_selected_input(&options.input, options.limits.source_bytes())?;
164    let cancellation = CancellationToken::new();
165    if options.cancel_before_start {
166        cancellation.cancel();
167    }
168    let controls = ProjectCaptureControls::new(project_limits(options), cancellation);
169    let mut builder = CapturedProjectRequestBuilder::new(LanguageProfile::V1_0, controls);
170    if let Some(project_key) = &options.project_key {
171        builder = builder.with_project_key(project_key.clone());
172    }
173    builder
174        .add_source(CapturedSourceInput::new(
175            required_option(options.source_id.as_ref())?,
176            required_option(options.module_id.as_ref())?,
177            source,
178        ))
179        .map_err(project_host_failure)?;
180    if let Some(vocabulary) =
181        project_vocabulary(&options.vocabulary, options.limits.source_bytes())?
182    {
183        builder.add_vocabulary(vocabulary);
184    }
185    let captured = capture_project(builder.build()).map_err(project_capture_failure)?;
186    let facts = captured.resource_facts();
187    eprintln!(
188        "{} capture succeeded source-units={} source-bytes={} vocabulary-units={} vocabulary-bytes={}",
189        constants::INFO,
190        facts.source_units(),
191        facts.total_source_bytes(),
192        facts.vocabulary_units(),
193        facts.total_vocabulary_bytes()
194    );
195    Ok(())
196}

198/// Converts the existing explicit CLI limits into the complete project schema.
199fn project_limits(options: &CommandOptions) -> ProjectCaptureLimits {
200    let limits = options.limits;
201    ProjectCaptureLimits::new(ProjectCaptureLimitValues {
202        total_source_bytes: limits.source_bytes(),
203        source_bytes_per_unit: limits.source_bytes(),
204        source_units: 1,
205        source_id_bytes: limits.source_bytes(),
206        module_id_bytes: limits.source_bytes(),
207        vocabulary_units: 1,
208        vocabulary_bytes_per_unit: limits.source_bytes(),
209        total_vocabulary_bytes: limits.source_bytes(),
210        imports_per_module: limits.declarations(),
211        import_edges: limits.declarations(),
212        scc_units: 1,
213        declarations: limits.declarations(),
214        diagnostics: u64::from(limits.diagnostics()),
215        output_bytes: limits.source_bytes(),
216    })
217}

219/// Compiles source and atomically publishes one encoded artifact.
220fn compile_command(
221    request: CompilationRequest,
222    options: &CommandOptions,
223) -> Result<(), CliFailure> {
224    let document = successful_document(compile(request).map_err(|error| capture_failure(&error))?)?;
225    let encoded = encode(
226        &document,
227        &ProducerInfo::new(env!("CARGO_PKG_NAME"), env!("CARGO_PKG_VERSION")),
228    )
229    .map_err(|_| CliFailure::one(ExitClass::Validation, "artifact-encoding-limit"))?;
230    write_selected_output(options, encoded.as_bytes())?;
231    eprintln!("{} compile succeeded", constants::INFO);
232    Ok(())
233}
234
235/// Validates source without publishing authoritative output.
236fn validate_command(request: CompilationRequest) -> Result<(), CliFailure> {
237    ensure_success(compile(request).map_err(|error| capture_failure(&error))?)?;
238    eprintln!("{} validation succeeded", constants::INFO);
239    Ok(())
240}

242/// Formats validated source and atomically publishes ordinary source bytes.
243fn format_command(request: CompilationRequest, options: &CommandOptions) -> Result<(), CliFailure> {
244    let formatted = format_source(request).map_err(format_failure)?;
245    write_selected_output(options, formatted.as_bytes())?;
246    eprintln!("{} format succeeded", constants::INFO);
247    Ok(())
248}

250/// Builds one captured-input request from explicit source, limits, and vocabulary facts.
251fn request_from_options(options: &CommandOptions) -> Result<CompilationRequest, CliFailure> {
252    let source = read_selected_input(&options.input, options.limits.source_bytes())?;
253    let cancellation = CancellationToken::new();
254    if options.cancel_before_start {
255        cancellation.cancel();
256    }
257    let request = CompilationRequest::new(source, options.limits, cancellation);
258    captured_vocabulary(request, &options.vocabulary, options.limits.source_bytes())
259}

261/// Attaches an exact explicitly acquired vocabulary bundle and immutable lock.
262fn captured_vocabulary(
263    request: CompilationRequest,
264    options: &VocabularyOptions,
265    byte_limit: u64,
266) -> Result<CompilationRequest, CliFailure> {
267    if options.is_empty() {
268        return Ok(request);
269    }
270    let bundle_path = required_option(options.bundle.as_ref())?;
271    let bytes = read_selected_input(bundle_path, byte_limit)?;
272    let digest = VocabularyContentDigest::parse_text(required_option(options.digest.as_ref())?)
273        .map_err(|_| CliFailure::one(ExitClass::Usage, "invalid-vocabulary-digest"))?;
274    let lock = VocabularyLock::new(
275        required_option(options.identity.as_ref())?,
276        required_option(options.version.as_ref())?,
277        required_option(options.encoding_version.as_ref())?,
278        required_option(options.schema_version.as_ref())?,
279        digest,
280        options.features.clone(),
281    )
282    .map_err(|_| CliFailure::one(ExitClass::Usage, "invalid-vocabulary-lock"))?;
283    Ok(request.with_captured_vocabulary(bytes, lock))
284}

286/// Constructs one optional vocabulary input for the shared project schema.
287fn project_vocabulary(
288    options: &VocabularyOptions,
289    byte_limit: u64,
290) -> Result<Option<CapturedVocabularyInput>, CliFailure> {
291    if options.is_empty() {
292        return Ok(None);
293    }
294    let bundle_path = required_option(options.bundle.as_ref())?;
295    let bytes = read_selected_input(bundle_path, byte_limit)?;
296    let digest = VocabularyContentDigest::parse_text(required_option(options.digest.as_ref())?)
297        .map_err(|_| CliFailure::one(ExitClass::Usage, "invalid-vocabulary-digest"))?;
298    let lock = VocabularyLock::new(
299        required_option(options.identity.as_ref())?,
300        required_option(options.version.as_ref())?,
301        required_option(options.encoding_version.as_ref())?,
302        required_option(options.schema_version.as_ref())?,
303        digest,
304        options.features.clone(),
305    )
306    .map_err(|_| CliFailure::one(ExitClass::Usage, "invalid-vocabulary-lock"))?;
307    Ok(Some(CapturedVocabularyInput::new(bytes, lock)))
308}

334/// Reads a file or standard input without exceeding the selected byte ceiling.
335fn read_selected_input(selection: &str, byte_limit: u64) -> Result<Vec<u8>, CliFailure> {
336    let reader: Box<dyn Read> = if selection == constants::STANDARD_STREAM {
337        Box::new(io::stdin())
338    } else {
339        Box::new(
340            File::open(selection)
341                .map_err(|_| CliFailure::one(ExitClass::Input, "input-read-failed"))?,
342        )
343    };
344    read_bounded(reader, byte_limit)
345}
346
347/// Reads at most one byte beyond a declared limit to classify oversized input.
348fn read_bounded(reader: Box<dyn Read>, byte_limit: u64) -> Result<Vec<u8>, CliFailure> {
349    let maximum_read = byte_limit.saturating_add(1);
350    let mut bytes = Vec::new();
351    reader
352        .take(maximum_read)
353        .read_to_end(&mut bytes)
354        .map_err(|_| CliFailure::one(ExitClass::Input, "input-read-failed"))?;
355    if u64::try_from(bytes.len()).unwrap_or(u64::MAX) > byte_limit {
356        Err(CliFailure::one(
357            ExitClass::Validation,
358            "input-limit-exceeded",
359        ))
360    } else {
361        Ok(bytes)
362    }
363}

365/// Opens authoritative artifacts as a validated document or maps their failure.
366fn successful_document(result: CompilationResult) -> Result<ValidatedDocument, CliFailure> {
367    match result {
368        CompilationResult::Success(artifacts) => ValidatedDocument::from_compiler_output(artifacts)
369            .map_err(|_| CliFailure::one(ExitClass::Internal, "reader-validation-defect")),
370        CompilationResult::Failure(failure) => Err(compilation_failure(&failure)),
371    }
372}

374/// Requires successful compilation when a command publishes no artifact.
375fn ensure_success(result: CompilationResult) -> Result<(), CliFailure> {
376    match result {
377        CompilationResult::Success(_) => Ok(()),
378        CompilationResult::Failure(failure) => Err(compilation_failure(&failure)),
379    }
380}

382/// Maps one capture failure to its stable CLI class without path disclosure.
383fn capture_failure(error: &CaptureError) -> CliFailure {
384    match error {
385        CaptureError::Cancelled => CliFailure::one(ExitClass::Cancelled, "cancelled"),
386        CaptureError::MissingSource => CliFailure::one(ExitClass::Validation, "missing-source"),
387        CaptureError::SourceLimitExceeded { .. } => {
388            CliFailure::one(ExitClass::Validation, "source-limit-exceeded")
389        }
390        CaptureError::SourceDigestMismatch { .. } => {
391            CliFailure::one(ExitClass::Validation, "source-digest-mismatch")
392        }
393    }
394}

396/// Maps a compilation failure to safe diagnostic code and byte-span lines.
397fn compilation_failure(failure: &CompilationFailure) -> CliFailure {
398    let class = if failure.class() == ResultClass::Cancellation {
399        ExitClass::Cancelled
400    } else {
401        ExitClass::Validation
402    };
403    let messages = if failure.diagnostics().is_empty() {
404        vec![if class == ExitClass::Cancelled {
405            "cancelled".to_owned()
406        } else {
407            "source-rejected".to_owned()
408        }]
409    } else {
410        failure
411            .diagnostics()
412            .iter()
413            .map(|diagnostic| {
414                let span = diagnostic.primary().span();
415                format!(
416                    "{} bytes {}..{}",
417                    diagnostic.code().as_str(),
418                    span.start(),
419                    span.end()
420                )
421            })
422            .collect()
423    };
424    CliFailure { class, messages }
425}

427/// Maps reference formatter failure classes to stable CLI outcomes.
428fn format_failure(error: FormatError) -> CliFailure {
429    match error {
430        FormatError::Capture(error) => capture_failure(&error),
431        FormatError::Compilation(failure) => compilation_failure(&failure),
432        FormatError::OutputLimitExceeded => {
433            CliFailure::one(ExitClass::Validation, "formatted-output-limit-exceeded")
434        }
435        FormatError::InternalDefect => {
436            CliFailure::one(ExitClass::Internal, "formatter-internal-defect")
437        }
438    }
439}

441/// Publishes bytes to the explicitly selected standard stream or file destination.
442fn write_selected_output(options: &CommandOptions, bytes: &[u8]) -> Result<(), CliFailure> {
443    let destination = options
444        .output
445        .as_deref()
446        .ok_or_else(|| CliFailure::one(ExitClass::Internal, "missing-validated-output"))?;
447    if destination == constants::STANDARD_STREAM {
448        let mut stdout = io::stdout().lock();
449        stdout
450            .write_all(bytes)
451            .and_then(|()| stdout.flush())
452            .map_err(|_| CliFailure::one(ExitClass::Output, "stdout-write-failed"))
453    } else {
454        atomic_write(Path::new(destination), bytes, options.overwrite)
455    }
456}

458/// Commits complete output through a same-directory temporary file.
459fn atomic_write(destination: &Path, bytes: &[u8], overwrite: bool) -> Result<(), CliFailure> {
460    if !overwrite && destination.exists() {
461        return Err(CliFailure::one(ExitClass::Output, "output-exists"));
462    }
463    let parent = destination.parent().unwrap_or_else(|| Path::new("."));
464    let (temporary_path, mut temporary_file) = create_temporary(parent)?;
465    let guard = TemporaryOutput::new(temporary_path.clone());
466    temporary_file
467        .write_all(bytes)
468        .and_then(|()| temporary_file.sync_all())
469        .map_err(|_| CliFailure::one(ExitClass::Output, "output-write-failed"))?;
470    drop(temporary_file);
471    if overwrite {
472        fs::rename(&temporary_path, destination)
473    } else {
474        fs::hard_link(&temporary_path, destination).and_then(|()| fs::remove_file(&temporary_path))
475    }
476    .map_err(|_| CliFailure::one(ExitClass::Output, "output-commit-failed"))?;
477    guard.commit();
478    Ok(())
479}

481/// Creates a collision-resistant temporary output in the destination directory.
482fn create_temporary(parent: &Path) -> Result<(PathBuf, File), CliFailure> {
483    for _ in 0..constants::MAXIMUM_TEMPORARY_ATTEMPTS {
484        let sequence = TEMPORARY_SEQUENCE.fetch_add(1, Ordering::Relaxed);
485        let name = format!(
486            "{}{}-{sequence}",
487            constants::TEMPORARY_MARKER,
488            std::process::id()
489        );
490        let path = parent.join(name);
491        match OpenOptions::new().write(true).create_new(true).open(&path) {
492            Ok(file) => return Ok((path, file)),
493            Err(error) if error.kind() == io::ErrorKind::AlreadyExists => {}
494            Err(_) => return Err(CliFailure::one(ExitClass::Output, "output-create-failed")),
495        }
496    }
497    Err(CliFailure::one(
498        ExitClass::Output,
499        "temporary-name-exhausted",
500    ))
501}

503/// Cleanup guard for an uncommitted atomic temporary output.
504struct TemporaryOutput {
505    /// Exact same-directory temporary path.
506    path: PathBuf,
507    /// Whether ownership was transferred to the final destination.
508    committed: std::cell::Cell<bool>,
509}
510
511impl TemporaryOutput {
512    /// Creates an armed cleanup guard.
513    fn new(path: PathBuf) -> Self {
514        Self {
515            path,
516            committed: std::cell::Cell::new(false),
517        }
518    }
519
520    /// Marks the temporary path as successfully committed.
521    fn commit(&self) {
522        self.committed.set(true);
523    }
524}
525
526impl Drop for TemporaryOutput {
527    /// Removes an uncommitted temporary file on every return path.
528    fn drop(&mut self) {
529        if !self.committed.get() {
530            let _ = fs::remove_file(&self.path);
531        }
532    }
533}
