8/// One supported host operation.
9#[derive(Clone, Copy, Debug, Eq, PartialEq)]
10pub enum CommandKind {
11    /// Compile source into an encoded artifact.
12    Compile,
13    /// Validate source without authoritative output.
14    Validate,
15    /// Format valid source into canonical source bytes.
16    Format,
17}

19/// Exact optional vocabulary bundle and immutable lock arguments.
20#[derive(Clone, Debug, Default, Eq, PartialEq)]
21pub struct VocabularyOptions {
22    /// Explicit bundle input path.
23    pub bundle: Option<String>,
24    /// Expected logical identity.
25    pub identity: Option<String>,
26    /// Expected release version.
27    pub version: Option<String>,
28    /// Expected bundle encoding version.
29    pub encoding_version: Option<String>,
30    /// Expected logical schema version.
31    pub schema_version: Option<String>,
32    /// Expected exact content digest text.
33    pub digest: Option<String>,
34    /// Required structural features.
35    pub features: Vec<String>,
36}
37
38impl VocabularyOptions {
39    /// Returns whether no vocabulary capture argument was supplied.
40    #[must_use]
41    pub fn is_empty(&self) -> bool {
42        self.bundle.is_none()
43            && self.identity.is_none()
44            && self.version.is_none()
45            && self.encoding_version.is_none()
46            && self.schema_version.is_none()
47            && self.digest.is_none()
48            && self.features.is_empty()
49    }
50
51    /// Verifies that every required captured vocabulary lock argument exists.
52    pub fn validate(&self) -> Result<(), String> {
53        if self.is_empty()
54            || self.bundle.is_some()
55                && self.identity.is_some()
56                && self.version.is_some()
57                && self.encoding_version.is_some()
58                && self.schema_version.is_some()
59                && self.digest.is_some()
60        {
61            Ok(())
62        } else {
63            Err("vocabulary capture requires bundle, identity, version, encoding, schema, and digest"
64                .to_owned())
65        }
66    }
67}

69/// Fully parsed explicit command policy.
70#[derive(Clone, Debug, Eq, PartialEq)]
71pub struct CommandOptions {
72    /// Selected host operation.
73    pub kind: CommandKind,
74    /// Explicit source path or standard-input sentinel.
75    pub input: String,
76    /// Explicit output path or standard-output sentinel.
77    pub output: Option<String>,
78    /// Whether an existing regular output may be atomically replaced.
79    pub overwrite: bool,
80    /// Whether cancellation is requested before capture starts.
81    pub cancel_before_start: bool,
82    /// Effective deterministic structural limits.
83    pub limits: StructuralLimits,
84    /// Optional exact captured vocabulary policy.
85    pub vocabulary: VocabularyOptions,
86}

88/// A non-executing command parser outcome.
89#[derive(Clone, Debug, Eq, PartialEq)]
90pub enum ParseOutcome {
91    /// Render general or command-specific help.
92    Help(&'static str),
93    /// Render the package version.
94    Version,
95    /// Execute one fully parsed command.
96    Execute(Box<CommandOptions>),
97}

99/// Mutable numeric limit values while parsing command arguments.
100struct LimitValues {
101    /// Maximum source bytes.
102    source_bytes: u64,
103    /// Maximum diagnostics.
104    diagnostics: u32,
105    /// Maximum decoded string bytes.
106    string_bytes: u64,
107    /// Maximum exact-number digits.
108    numeric_digits: u64,
109    /// Maximum exact-number scale.
110    numeric_scale: u64,
111    /// Maximum declarations.
112    declarations: u64,
113    /// Maximum fields.
114    record_fields: u64,
115    /// Maximum nesting depth.
116    nesting_depth: u64,
117    /// Maximum list items.
118    list_items: u64,
119    /// Maximum traversal nodes.
120    traversal_nodes: u64,
121}
122
123impl Default for LimitValues {
124    /// Returns the reviewed bounded CLI defaults.
125    fn default() -> Self {
126        Self {
127            source_bytes: constants::DEFAULT_SOURCE_BYTES,
128            diagnostics: constants::DEFAULT_DIAGNOSTICS,
129            string_bytes: constants::DEFAULT_STRING_BYTES,
130            numeric_digits: constants::DEFAULT_NUMERIC_DIGITS,
131            numeric_scale: constants::DEFAULT_NUMERIC_SCALE,
132            declarations: constants::DEFAULT_DECLARATIONS,
133            record_fields: constants::DEFAULT_RECORD_FIELDS,
134            nesting_depth: constants::DEFAULT_NESTING_DEPTH,
135            list_items: constants::DEFAULT_LIST_ITEMS,
136            traversal_nodes: constants::DEFAULT_TRAVERSAL_NODES,
137        }
138    }
139}
140
141impl LimitValues {
142    /// Constructs the checked shared structural limit contract.
143    fn build(&self) -> Result<StructuralLimits, String> {
144        StructuralLimits::new(self.source_bytes, self.diagnostics)
145            .and_then(|limits| limits.with_string_bytes(self.string_bytes))
146            .and_then(|limits| limits.with_numeric_digits(self.numeric_digits))
147            .and_then(|limits| limits.with_numeric_scale(self.numeric_scale))
148            .and_then(|limits| limits.with_declarations(self.declarations))
149            .and_then(|limits| limits.with_record_fields(self.record_fields))
150            .and_then(|limits| limits.with_nesting_depth(self.nesting_depth))
151            .and_then(|limits| limits.with_list_items(self.list_items))
152            .and_then(|limits| limits.with_traversal_nodes(self.traversal_nodes))
153            .map_err(|_| "every limit must be greater than zero".to_owned())
154    }
155}

157/// Parses command arguments without performing filesystem or process effects.
158pub fn parse(arguments: impl IntoIterator<Item = String>) -> Result<ParseOutcome, String> {
159    let arguments = arguments.into_iter().collect::<Vec<_>>();
160    let Some(first) = arguments.first().map(String::as_str) else {
161        return Err(format!(
162            "a command is required; usage: {}",
163            constants::USAGE
164        ));
165    };
166    if first == constants::HELP {
167        return Ok(ParseOutcome::Help(constants::USAGE));
168    }
169    if first == constants::VERSION {
170        return Ok(ParseOutcome::Version);
171    }
172    let (kind, usage) = command_kind(first)?;
173    if arguments
174        .get(1)
175        .is_some_and(|value| value == constants::HELP)
176    {
177        return Ok(ParseOutcome::Help(usage));
178    }
179    parse_command_arguments(&arguments, kind, usage)
180}

182/// Parses options and positional input for one already selected command.
183fn parse_command_arguments(
184    arguments: &[String],
185    kind: CommandKind,
186    usage: &'static str,
187) -> Result<ParseOutcome, String> {
188    let mut input = None;
189    let mut output = None;
190    let mut overwrite = false;
191    let mut cancel_before_start = false;
192    let mut limits = LimitValues::default();
193    let mut vocabulary = VocabularyOptions::default();
194    let mut index = 1;
195    while index < arguments.len() {
196        let argument = &arguments[index];
197        match argument.as_str() {
198            constants::OVERWRITE => overwrite = true,
199            constants::CANCEL_BEFORE_START => cancel_before_start = true,
200            constants::OUTPUT => set_once(
201                &mut output,
202                next_value(arguments, &mut index, argument)?,
203                argument,
204            )?,
205            constants::VOCABULARY_BUNDLE => set_once(
206                &mut vocabulary.bundle,
207                next_value(arguments, &mut index, argument)?,
208                argument,
209            )?,
210            constants::VOCABULARY_IDENTITY => set_once(
211                &mut vocabulary.identity,
212                next_value(arguments, &mut index, argument)?,
213                argument,
214            )?,
215            constants::VOCABULARY_VERSION => set_once(
216                &mut vocabulary.version,
217                next_value(arguments, &mut index, argument)?,
218                argument,
219            )?,
220            constants::VOCABULARY_ENCODING => set_once(
221                &mut vocabulary.encoding_version,
222                next_value(arguments, &mut index, argument)?,
223                argument,
224            )?,
225            constants::VOCABULARY_SCHEMA => set_once(
226                &mut vocabulary.schema_version,
227                next_value(arguments, &mut index, argument)?,
228                argument,
229            )?,
230            constants::VOCABULARY_DIGEST => set_once(
231                &mut vocabulary.digest,
232                next_value(arguments, &mut index, argument)?,
233                argument,
234            )?,
235            constants::VOCABULARY_FEATURE => vocabulary
236                .features
237                .push(next_value(arguments, &mut index, argument)?),
238            constants::MAX_SOURCE_BYTES => {
239                limits.source_bytes = parse_u64(arguments, &mut index, argument)?;
240            }
241            constants::MAX_DIAGNOSTICS => {
242                limits.diagnostics = parse_u32(arguments, &mut index, argument)?;
243            }
244            constants::MAX_STRING_BYTES => {
245                limits.string_bytes = parse_u64(arguments, &mut index, argument)?;
246            }
247            constants::MAX_NUMERIC_DIGITS => {
248                limits.numeric_digits = parse_u64(arguments, &mut index, argument)?;
249            }
250            constants::MAX_NUMERIC_SCALE => {
251                limits.numeric_scale = parse_u64(arguments, &mut index, argument)?;
252            }
253            constants::MAX_DECLARATIONS => {
254                limits.declarations = parse_u64(arguments, &mut index, argument)?;
255            }
256            constants::MAX_RECORD_FIELDS => {
257                limits.record_fields = parse_u64(arguments, &mut index, argument)?;
258            }
259            constants::MAX_NESTING_DEPTH => {
260                limits.nesting_depth = parse_u64(arguments, &mut index, argument)?;
261            }
262            constants::MAX_LIST_ITEMS => {
263                limits.list_items = parse_u64(arguments, &mut index, argument)?;
264            }
265            constants::MAX_TRAVERSAL_NODES => {
266                limits.traversal_nodes = parse_u64(arguments, &mut index, argument)?;
267            }
268            value if value.starts_with('-') && value != constants::STANDARD_STREAM => {
269                return Err(format!("unknown option: {value}; usage: {usage}"));
270            }
271            value => set_once(&mut input, value.to_owned(), "source")?,
272        }
273        index += 1;
274    }
275    vocabulary.validate()?;
276    validate_output_policy(kind, output.as_deref(), overwrite, usage)?;
277    let input = input.ok_or_else(|| format!("source is required; usage: {usage}"))?;
278    validate_standard_inputs(&input, &vocabulary)?;
279    Ok(ParseOutcome::Execute(Box::new(CommandOptions {
280        kind,
281        input,
282        output,
283        overwrite,
284        cancel_before_start,
285        limits: limits.build()?,
286        vocabulary,
287    })))
288}

290/// Prevents two captured inputs from competing for the same standard stream.
291fn validate_standard_inputs(input: &str, vocabulary: &VocabularyOptions) -> Result<(), String> {
292    if input == constants::STANDARD_STREAM
293        && vocabulary.bundle.as_deref() == Some(constants::STANDARD_STREAM)
294    {
295        Err("source and vocabulary bundle cannot both use standard input".to_owned())
296    } else {
297        Ok(())
298    }
299}

301/// Resolves one stable command spelling and its usage.
302fn command_kind(value: &str) -> Result<(CommandKind, &'static str), String> {
303    match value {
304        constants::COMPILE => Ok((CommandKind::Compile, constants::COMPILE_USAGE)),
305        constants::VALIDATE => Ok((CommandKind::Validate, constants::VALIDATE_USAGE)),
306        constants::FORMAT => Ok((CommandKind::Format, constants::FORMAT_USAGE)),
307        _ => Err(format!(
308            "unknown command: {value}; usage: {}",
309            constants::USAGE
310        )),
311    }
312}

314/// Validates command-specific destination and replacement policy.
315fn validate_output_policy(
316    kind: CommandKind,
317    output: Option<&str>,
318    overwrite: bool,
319    usage: &str,
320) -> Result<(), String> {
321    match kind {
322        CommandKind::Compile | CommandKind::Format if output.is_none() => {
323            Err(format!("output is required; usage: {usage}"))
324        }
325        CommandKind::Validate if output.is_some() || overwrite => {
326            Err(format!("validate produces no output; usage: {usage}"))
327        }
328        CommandKind::Compile | CommandKind::Format
329            if output == Some(constants::STANDARD_STREAM) && overwrite =>
330        {
331            Err(format!(
332                "overwrite is invalid for standard output; usage: {usage}"
333            ))
334        }
335        _ => Ok(()),
336    }
337}

339/// Returns one required option value and advances the parser cursor.
340fn next_value(arguments: &[String], index: &mut usize, option: &str) -> Result<String, String> {
341    *index += 1;
342    arguments
343        .get(*index)
344        .filter(|value| !value.is_empty())
345        .cloned()
346        .ok_or_else(|| format!("{option} requires a value"))
347}
348
349/// Sets one non-repeatable string argument exactly once.
350fn set_once(target: &mut Option<String>, value: String, option: &str) -> Result<(), String> {
351    if target.replace(value).is_some() {
352        Err(format!("{option} may be supplied only once"))
353    } else {
354        Ok(())
355    }
356}

358/// Parses one required positive unsigned 64-bit option value.
359fn parse_u64(arguments: &[String], index: &mut usize, option: &str) -> Result<u64, String> {
360    next_value(arguments, index, option)?
361        .parse::<u64>()
362        .ok()
363        .filter(|value| *value > 0)
364        .ok_or_else(|| format!("{option} requires a positive integer"))
365}
366
367/// Parses one required positive unsigned 32-bit option value.
368fn parse_u32(arguments: &[String], index: &mut usize, option: &str) -> Result<u32, String> {
369    next_value(arguments, index, option)?
370        .parse::<u32>()
371        .ok()
372        .filter(|value| *value > 0)
373        .ok_or_else(|| format!("{option} requires a positive integer"))
374}
