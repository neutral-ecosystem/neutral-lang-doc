18/// Explicit capture-lock and producer facts; no host environment is consulted.
19pub struct ProjectIdentityContext<'a> {
20    /// Exact capture facts, checked against the validated source companions.
21    pub capture: CapturedIdentityInput<'a>,
22    /// Stable producer identity, not a host path or executable discovery result.
23    pub producer: &'a str,
24    /// Explicit producer revision, independent of the transcript profile.
25    pub producer_version: &'a str,
26    /// Acceptance controls in the frozen capture-limit order.
27    pub capture_limits: [u64; CAPTURE_LIMIT_TAGS.len()],
28}
29
30/// Atomic reader identity failures, never a partially published identity set.
31#[derive(Clone, Copy, Debug, Eq, PartialEq)]
32pub enum ProjectIdentityReadError {
33    /// Explicit capture facts do not match the complete project companions.
34    CaptureMismatch,
35    /// Bounded canonical identity construction failed.
36    Identity(IdentityError),
37    /// Selected artifact roots are absent, private, duplicate, or cancelled.
38    View(ProjectReadError),
39}
40
41impl ProjectIdentityReadError {
42    /// Returns the shared project failure envelope, never a package release version.
43    #[must_use]
44    pub const fn schema(self) -> &'static str {
45        crate::PROJECT_RESULT_SCHEMA
46    }
47}

49/// Explicit processing facts associated with a complete validated project.
50#[derive(Clone, Debug, Eq, PartialEq)]
51pub struct ProjectDerivationFacts {
52    /// Exact captured closure identity, not logical meaning.
53    pub captured: CapturedClosureIdentity,
54    /// Complete logical project identity, including private declarations.
55    pub logical: LogicalProjectIdentity,
56    /// Stable producer identifier supplied by the caller.
57    pub producer: String,
58    /// Explicit producer revision supplied by the caller.
59    pub producer_version: String,
60    /// Frozen acceptance controls; limits do not enter logical identity.
61    pub capture_limits: [u64; CAPTURE_LIMIT_TAGS.len()],
62    /// Authoritative project bounds retained in the complete artifact.
63    pub project_limits: ProjectLimits,
64}

66/// Three immutable typed identity layers bound to one validated reader.
67#[derive(Debug)]
68pub struct ProjectIdentities<'a> {
69    /// Complete validated authority used to check artifact selections.
70    project: &'a ValidatedProject,
71    /// Exact captured-closure transcript.
72    captured: IdentityTranscript<CapturedClosureIdentity>,
73    /// Complete logical transcript, never a selected public fingerprint.
74    logical: IdentityTranscript<LogicalProjectIdentity>,
75    /// Explicit processing transcript.
76    derivation: IdentityTranscript<ProjectDerivationIdentity>,
77    /// Caller-supplied facts bound to the validated project companions.
78    facts: ProjectDerivationFacts,
79}
80
81impl ProjectIdentities<'_> {
82    /// Returns the exact captured-closure transcript and typed digest.
83    #[must_use]
84    pub const fn captured(&self) -> &IdentityTranscript<CapturedClosureIdentity> {
85        &self.captured
86    }
87    /// Returns the complete logical transcript, independent of root selection.
88    #[must_use]
89    pub const fn logical(&self) -> &IdentityTranscript<LogicalProjectIdentity> {
90        &self.logical
91    }
92    /// Returns the explicit derivation transcript and typed digest.
93    #[must_use]
94    pub const fn derivation(&self) -> &IdentityTranscript<ProjectDerivationIdentity> {
95        &self.derivation
96    }
97    /// Returns processing facts without inferring authenticity or authority.
98    #[must_use]
99    pub const fn facts(&self) -> &ProjectDerivationFacts {
100        &self.facts
101    }
102    /// Derives an artifact transcript only after checking public selection roots.
103    ///
104    /// # Errors
105    /// Rejects private, unknown, or duplicate roots, malformed options, bounds,
106    /// or cancellation. No complete-project transcript is mutated.
107    pub fn artifact(
108        &self,
109        input: &ArtifactIdentityInput<'_>,
110        limits: IdentityLimits,
111        cancellation: &CancellationToken,
112    ) -> Result<IdentityTranscript<ProjectArtifactIdentity>, ProjectIdentityReadError> {
113        let transcript = artifact_identity(self.derivation.identity(), input, limits, cancellation)
114            .map_err(ProjectIdentityReadError::Identity)?;
115        if input.kind == ArtifactKind::View {
116            self.project
117                .derive_view(
118                    &ViewRequest {
119                        schema: PROJECT_VIEW_SCHEMA.to_owned(),
120                        roots: input.roots.to_vec(),
121                    },
122                    cancellation,
123                )
124                .map_err(ProjectIdentityReadError::View)?;
125        }
126        Ok(transcript)
127    }
128}

130impl ValidatedProject {
131    /// Returns complete logical identity through an independently validated reader.
132    ///
133    /// # Errors
134    /// Rejects explicit work bounds or cancellation, without consulting roots.
135    pub fn logical_identity(
136        &self,
137        limits: IdentityLimits,
138        cancellation: &CancellationToken,
139    ) -> Result<IdentityTranscript<LogicalProjectIdentity>, IdentityError> {
140        canonical_logical_project(self.complete_ir(), limits, cancellation)
141    }
142
143    /// Binds exact capture and explicit producer facts to validated complete IR.
144    ///
145    /// Lock encoding/schema/features and producer claims remain explicit caller
146    /// inputs: this API does not authenticate a producer or recapture source bytes.
147    ///
148    /// # Errors
149    /// Rejects mismatched source/vocabulary companions, work bounds, or cancellation.
150    pub fn identities(
151        &self,
152        context: &ProjectIdentityContext<'_>,
153        limits: IdentityLimits,
154        cancellation: &CancellationToken,
155    ) -> Result<ProjectIdentities<'_>, ProjectIdentityReadError> {
156        if cancellation.is_cancelled() {
157            return Err(ProjectIdentityReadError::Identity(IdentityError::Cancelled));
158        }
159        let ir = self.complete_ir();
160        let capture = &context.capture;
161        if capture.profile != V1_SOURCE_PROFILE
162            || capture.sources.len() != ir.sources.len()
163            || capture.vocabularies.len() != ir.vocabulary_sources.len()
164            || capture
165                .sources
166                .iter()
167                .zip(&ir.sources)
168                .any(|(fact, source)| {
169                    fact.module != source.module
170                        || fact.source_id != source.source_id
171                        || fact.digest != source.digest
172                        || fact.byte_len != source.byte_len
173                })
174            || capture
175                .vocabularies
176                .iter()
177                .zip(&ir.vocabulary_sources)
178                .any(|(fact, source)| {
179                    fact.identity != source.identity
180                        || fact.version != source.version
181                        || fact.digest != source.digest
182                        || fact.byte_len != source.byte_len
183                })
184        {
185            return Err(ProjectIdentityReadError::CaptureMismatch);
186        }
187        let captured = captured_closure(capture, limits, cancellation)
188            .map_err(ProjectIdentityReadError::Identity)?;
189        let logical = self
190            .logical_identity(limits, cancellation)
191            .map_err(ProjectIdentityReadError::Identity)?;
192        let derivation = derivation_identity(
193            logical.identity(),
194            &DerivationContext {
195                captured: captured.identity(),
196                producer: context.producer,
197                producer_version: context.producer_version,
198                capture_limits: context.capture_limits,
199                project_limits: ir.limits,
200            },
201            limits,
202            cancellation,
203        )
204        .map_err(ProjectIdentityReadError::Identity)?;
205        // Own caller text only after the bounded transcript has accepted it.
206        let facts = ProjectDerivationFacts {
207            captured: captured.identity(),
208            logical: logical.identity(),
209            producer: context.producer.to_owned(),
210            producer_version: context.producer_version.to_owned(),
211            capture_limits: context.capture_limits,
212            project_limits: ir.limits,
213        };
214        Ok(ProjectIdentities {
215            project: self,
216            captured,
217            logical,
218            derivation,
219            facts,
220        })
221    }
