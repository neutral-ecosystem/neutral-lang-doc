51/// Safe lookup failures containing no private type names, captured text or host paths.
52#[derive(Clone, Copy, Debug, Eq, PartialEq)]
53pub enum CompositionLookupError {
54    /// Canonical vocabulary is absent from the captured catalogue.
55    UnknownVocabulary,
56    /// Semantic revision does not equal the captured exact lock.
57    RevisionMismatch,
58    /// Type is absent or private; public callers cannot distinguish these cases.
59    UnavailableType,
60}
61
62/// Immutable public inspection surface over a fully validated complete catalogue.
63#[derive(Clone)]
64pub struct CompositionCatalogue {
65    /// Complete contracts retained internally, never mutably exposed.
66    catalogue: Arc<ValidatedComposition>,
67}
68
69impl std::fmt::Debug for CompositionCatalogue {
70    /// Logs only public catalogue counts, never complete private type contracts.
71    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
72        let public_types = self
73            .catalogue
74            .bundles()
75            .iter()
76            .flat_map(|bundle| &bundle.definitions)
77            .filter(|definition| definition.public)
78            .count();
79        formatter
80            .debug_struct("CompositionCatalogue")
81            .field("vocabularies", &self.catalogue.bundles().len())
82            .field("public_types", &public_types)
83            .finish_non_exhaustive()
84    }
85}
86
87impl CompositionCatalogue {
88    /// Opens validated contracts without compiler linkage or source reparsing.
89    ///
90    /// # Errors
91    /// Returns [`CompositionError::Allocation`] if shared ownership cannot be allocated.
92    pub fn new(catalogue: ValidatedComposition) -> Result<Self, CompositionError> {
93        Ok(Self::from_shared(
94            Arc::try_new(catalogue).map_err(|_| CompositionError::Allocation)?,
95        ))
96    }
97
98    /// Opens an immutable validated catalogue without copying contracts or linking the compiler.
99    ///
100    /// Validation cannot be bypassed: raw composition structures cannot construct
101    /// [`ValidatedComposition`]. Sharing does not expose mutable catalogue access.
102    #[must_use]
103    pub fn from_shared(catalogue: Arc<ValidatedComposition>) -> Self {
104        Self { catalogue }
105    }
106
107    /// Enumerates exact canonical identity/revision/schema/feature/content facts in order.
108    pub fn vocabularies(&self) -> impl Iterator<Item = &VocabularyIdentity> {
109        self.catalogue
110            .bundles()
111            .iter()
112            .map(|bundle| &bundle.identity)
113    }
114
115    /// Enumerates every public nominal contract for one exact canonical owner.
116    ///
117    /// # Errors
118    /// Rejects unknown identities or mismatched semantic revisions.
119    pub fn public_types(
120        &self,
121        identity: &str,
122        version: &str,
123    ) -> Result<impl Iterator<Item = &CompositionDefinition>, CompositionLookupError> {
124        Ok(self
125            .bundle(identity, version)?
126            .definitions
127            .iter()
128            .filter(|definition| definition.public))
129    }
130
131    /// Returns a complete public record/variant contract including every alternative and default.
132    ///
133    /// # Errors
134    /// Rejects unknown owners/revisions or an absent/private type.
135    pub fn public_type(
136        &self,
137        identity: &str,
138        version: &str,
139        name: &str,
140    ) -> Result<&CompositionDefinition, CompositionLookupError> {
141        self.public_types(identity, version)?
142            .find(|definition| definition.name == name)
143            .ok_or(CompositionLookupError::UnavailableType)
144    }
145
146    /// Returns exactly locked dependencies, never aliases or acquisition locators.
147    ///
148    /// # Errors
149    /// Rejects unknown identities or mismatched semantic revisions.
150    pub fn dependencies(
151        &self,
152        identity: &str,
153        version: &str,
154    ) -> Result<&[CompositionDependency], CompositionLookupError> {
155        Ok(&self.bundle(identity, version)?.dependencies)
156    }
157
158    /// Checks supplied closed data and returns meaning plus safe origins using the shared validator.
159    ///
160    /// No compiler linkage, source reparsing, reference-target invention or project-profile
161    /// activation occurs. Errors contain no raw private definition, source bytes or host path.
162    ///
163    /// # Errors
164    /// Rejects unavailable public roots, invalid values, exhausted bounds or cancellation.
165    pub fn materialize(
166        &self,
167        owner: (&str, &str, &str),
168        supplied: &ClosedValue,
169        limits: CompositionLimits,
170        cancellation: &CancellationToken,
171    ) -> Result<ValidatedCompositionValue, CompositionError> {
172        if cancellation.is_cancelled() {
173            return Err(CompositionError::Cancelled);
174        }
175        // Keep absent and private lookup indistinguishable at the public reader
176        // boundary, matching the existing public-type lookup contract.
177        self.public_type(owner.0, owner.1, owner.2)
178            .map_err(|_| CompositionError::UnknownType)?;
179        materialize_composition_value(&self.catalogue, owner, supplied, limits, cancellation)
180    }
181
182    /// Enumerates every declared reference type in one public record or variant.
183    ///
184    /// Includes references in unselected alternatives, lists and nullable wrappers.
185    /// Targets are exact nominal contracts, not acquired or executable binding edges.
186    ///
187    /// # Errors
188    /// Rejects unavailable public roots, zero/exhausted budgets, cancellation or allocation failure.
189    pub fn reference_types(
190        &self,
191        owner: (&str, &str, &str),
192        limits: CompositionInspectionLimits,
193        cancellation: &CancellationToken,
194    ) -> Result<Vec<ReferenceTypeDependency<'_>>, CompositionReferenceError> {
195        if cancellation.is_cancelled() {
196            return Err(CompositionReferenceError::Cancelled);
197        }
198        let definition = self
199            .public_type(owner.0, owner.1, owner.2)
200            .map_err(CompositionReferenceError::Lookup)?;
201        references::inspect(definition, limits, cancellation)
202    }
203
204    /// Resolves exact ownership before publishing public traversal state.
205    fn bundle(
206        &self,
207        identity: &str,
208        version: &str,
209    ) -> Result<&CompositionBundle, CompositionLookupError> {
210        let bundle = self
211            .catalogue
212            .bundles()
213            .iter()
214            .find(|bundle| bundle.identity.identity() == identity)
215            .ok_or(CompositionLookupError::UnknownVocabulary)?;
216        if bundle.identity.version() != version {
217            return Err(CompositionLookupError::RevisionMismatch);
218        }
219        Ok(bundle)
220    }
221}
