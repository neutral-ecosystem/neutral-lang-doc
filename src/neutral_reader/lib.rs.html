3//! Validated, immutable access to Neutral artifacts.
4//!
5//! This crate owns validation at the external artifact boundary and the reader
6//! views exposed after validation. It treats encoded data as untrusted and must
7//! not acquire inputs or depend on compiler-private representations.

22/// Immutable typed traversal over already-validated in-process artifacts.
23#[derive(Clone, Debug)]
24pub struct ValidatedDocument {
25    /// Shared immutable compiler artifacts.
26    artifacts: Arc<CompilationArtifacts>,
27}
28
29impl ValidatedDocument {
30    /// Validates cross-artifact indexes before exposing immutable reader views.
31    ///
32    /// # Errors
33    ///
34    /// Returns a fail-closed error for duplicate elements or missing source and
35    /// provenance records.
36    pub fn from_compiler_output(artifacts: Arc<CompilationArtifacts>) -> Result<Self, ReaderError> {
37        validate_artifacts(&artifacts)?;
38        Ok(Self { artifacts })
39    }
40
41    /// Returns the logical module name.
42    #[must_use]
43    pub fn module_name(&self) -> &str {
44        self.artifacts.logical_document().module().module_name()
45    }
46
47    /// Returns immutable declarations in deterministic order.
48    #[must_use]
49    pub fn declarations(&self) -> &[Declaration] {
50        self.artifacts.logical_document().declarations()
51    }
52
53    /// Returns nominal record schemas in canonical name order.
54    #[must_use]
55    pub fn record_types(&self) -> &[RecordTypeDefinition] {
56        self.artifacts.logical_document().record_types()
57    }
58
59    /// Returns the exact captured vocabulary contract, when present.
60    #[must_use]
61    pub fn vocabulary(&self) -> Option<&VocabularyContract> {
62        self.artifacts.logical_document().vocabulary()
63    }
64
65    /// Finds one nominal record schema by its validated name.
66    #[must_use]
67    pub fn record_type_by_name(&self, name: &str) -> Option<&RecordTypeDefinition> {
68        self.artifacts.logical_document().record_type_by_name(name)
69    }
70
71    /// Finds one declaration by its validated source name.
72    #[must_use]
73    pub fn declaration_by_name(&self, name: &str) -> Option<&Declaration> {
74        self.declarations()
75            .iter()
76            .find(|declaration| declaration.name() == name)
77    }
78
79    /// Maps a declaration element to its complete original-byte source location.
80    #[must_use]
81    pub fn source_location(&self, element_id: ElementId) -> Option<SourceLocation> {
82        self.artifacts.source_map().entry(element_id).map(|entry| {
83            SourceLocation::new(
84                self.artifacts.source_map().source_digest(),
85                entry.declaration_span(),
86            )
87        })
88    }
89
90    /// Returns the immutable validated compiler artifacts for advanced readers.
91    #[must_use]
92    pub const fn artifacts(&self) -> &Arc<CompilationArtifacts> {
93        &self.artifacts
94    }
95
96    /// Compares validated logical payloads under document-local ID renaming.
97    ///
98    /// Source locations, provenance, derivation, and numeric ID spellings are
99    /// companion facts and do not participate. Callers must use durable symbol
100    /// identities, never retain an `ElementId` for cross-document continuity.
101    #[must_use]
102    pub fn logically_equivalent(&self, other: &Self) -> bool {
103        self.artifacts.logically_equivalent(&other.artifacts)
104    }
105}

107/// A fail-closed in-process reader validation failure.
108#[derive(Clone, Copy, Debug, Eq, PartialEq)]
109pub enum ReaderError {
110    /// Two declarations reused one graph-local element identifier.
111    DuplicateElementId,
112    /// Two declarations reused one source name.
113    DuplicateDeclarationName,
114    /// A declaration had no matching source-map entry.
115    MissingSourceMapEntry,
116    /// A declaration had no matching value-provenance record.
117    MissingProvenanceRecord,
118    /// A declaration value did not satisfy its resolved type and nullability.
119    TypeValueMismatch,
120    /// A nominal record schema violated ownership, ordering, or recursion rules.
121    InvalidRecordSchema,
122    /// Field provenance was dangling, duplicated, or used an invalid origin.
123    InvalidFieldProvenance,
124    /// Reuse provenance was dangling, duplicated, or inconsistent with final values.
125    InvalidReuseProvenance,
126    /// A typed identity edge or its provenance was dangling or inconsistent.
127    InvalidReferenceEdge,
128    /// A declaration fingerprint did not match its complete logical definition.
129    InvalidDeclarationFingerprint,
130    /// Logical and derivation vocabulary contracts were missing or inconsistent.
131    InvalidVocabularyContract,
132}

134/// Validates relationships among logical declarations and companion artifacts.
135fn validate_artifacts(artifacts: &CompilationArtifacts) -> Result<(), ReaderError> {
136    let mut element_ids = BTreeSet::new();
137    let mut names = BTreeSet::new();
138    let records = artifacts
139        .logical_document()
140        .record_types()
141        .iter()
142        .map(|record| (record.name(), record))
143        .collect::<BTreeMap<_, _>>();
144    validate_record_schemas(artifacts, &records)?;
145    let logical_vocabulary = artifacts.logical_document().vocabulary();
146    let derivation_vocabulary = artifacts.derivation().vocabulary();
147    if logical_vocabulary.map(VocabularyContract::identity) != derivation_vocabulary {
148        return Err(ReaderError::InvalidVocabularyContract);
149    }
150    validate_vocabulary_contract(logical_vocabulary)?;
151    for record in artifacts.logical_document().record_types() {
152        if DeclarationFingerprint::for_record(record.fields())
153            .map_err(|_| ReaderError::InvalidDeclarationFingerprint)?
154            != record.fingerprint()
155        {
156            return Err(ReaderError::InvalidDeclarationFingerprint);
157        }
158        if !element_ids.insert(record.element_id()) {
159            return Err(ReaderError::DuplicateElementId);
160        }
161        if !names.insert(record.name()) {
162            return Err(ReaderError::DuplicateDeclarationName);
163        }
164        if artifacts.source_map().entry(record.element_id()).is_none() {
165            return Err(ReaderError::MissingSourceMapEntry);
166        }
167    }
168    for declaration in artifacts.logical_document().declarations() {
169        if DeclarationFingerprint::for_binding(declaration.resolved_type(), declaration.value())
170            .map_err(|_| ReaderError::InvalidDeclarationFingerprint)?
171            != declaration.fingerprint()
172        {
173            return Err(ReaderError::InvalidDeclarationFingerprint);
174        }
175        if !validate_value(declaration.resolved_type(), declaration.value(), &records) {
176            return Err(ReaderError::TypeValueMismatch);
177        }
178        if !validate_vocabulary_value(
179            declaration.resolved_type(),
180            declaration.value(),
181            artifacts.logical_document().vocabulary(),
182        ) {
183            return Err(ReaderError::TypeValueMismatch);
184        }
185        if !element_ids.insert(declaration.element_id()) {
186            return Err(ReaderError::DuplicateElementId);
187        }
188        if !names.insert(declaration.name()) {
189            return Err(ReaderError::DuplicateDeclarationName);
190        }
191        if artifacts
192            .source_map()
193            .entry(declaration.element_id())
194            .is_none()
195        {
196            return Err(ReaderError::MissingSourceMapEntry);
197        }
198        if !artifacts
199            .provenance()
200            .iter()
201            .any(|record| record.element_id() == declaration.element_id())
202        {
203            return Err(ReaderError::MissingProvenanceRecord);
204        }
205    }
206    validate_field_provenance(artifacts, &records)?;
207    validate_reuse_provenance(artifacts, &records)?;
208    validate_reference_provenance(artifacts, &records)?;
209    Ok(())
210}

212/// Validates canonical shape and closed defaults in an embedded vocabulary contract.
213fn validate_vocabulary_contract(contract: Option<&VocabularyContract>) -> Result<(), ReaderError> {
214    let Some(contract) = contract else {
215        return Ok(());
216    };
217    let mut previous_type = None;
218    for definition in contract.types() {
219        if definition.identity().vocabulary() != contract.identity().identity()
220            || previous_type.is_some_and(|name| name >= definition.identity().name())
221        {
222            return Err(ReaderError::InvalidRecordSchema);
223        }
224        previous_type = Some(definition.identity().name());
225        let mut previous_field = None;
226        for field in definition.fields() {
227            if previous_field.is_some_and(|name| name >= field.name())
228                || field.default_value().is_some_and(|value| {
229                    !is_closed_default(value)
230                        || !validate_vocabulary_value(field.resolved_type(), value, Some(contract))
231                })
232            {
233                return Err(ReaderError::InvalidRecordSchema);
234            }
235            previous_field = Some(field.name());
236        }
237    }
238    Ok(())
239}

241/// Validates qualified values solely against the embedded exact contract.
242fn validate_vocabulary_value(
243    expected: &ResolvedType,
244    value: &LogicalValue,
245    contract: Option<&VocabularyContract>,
246) -> bool {
247    match (expected, value) {
248        (ResolvedType::Nullable(_), LogicalValue::Null) => true,
249        (ResolvedType::Nullable(inner), value) => validate_vocabulary_value(inner, value, contract),
250        (ResolvedType::List(inner), LogicalValue::List(items)) => items
251            .iter()
252            .all(|item| validate_vocabulary_value(inner, item, contract)),
253        (ResolvedType::VocabularyRecord(identity), LogicalValue::VocabularyRecord(value)) => {
254            let Some(contract) = contract else {
255                return false;
256            };
257            let Some(schema) = contract.type_by_name(identity.name()) else {
258                return false;
259            };
260            identity == value.nominal_type()
261                && identity.vocabulary() == contract.identity().identity()
262                && schema.fields().len() == value.fields().len()
263                && schema
264                    .fields()
265                    .iter()
266                    .zip(value.fields())
267                    .all(|(field, actual)| {
268                        field.name() == actual.name()
269                            && validate_value(
270                                field.resolved_type(),
271                                actual.value(),
272                                &BTreeMap::new(),
273                            )
274                            && validate_vocabulary_value(
275                                field.resolved_type(),
276                                actual.value(),
277                                Some(contract),
278                            )
279                    })
280        }
281        (ResolvedType::VocabularyRecord(_), _) => false,
282        _ => true,
283    }
284}

286/// Validates every typed identity edge and requires exact provenance coverage.
287fn validate_reference_provenance(
288    artifacts: &CompilationArtifacts,
289    records: &BTreeMap<&str, &RecordTypeDefinition>,
290) -> Result<(), ReaderError> {
291    let declarations = artifacts
292        .logical_document()
293        .declarations()
294        .iter()
295        .map(|declaration| (declaration.element_id(), declaration))
296        .collect::<BTreeMap<_, _>>();
297    let root_reuses = artifacts
298        .reuse_provenance()
299        .iter()
300        .filter(|record| record.value_path().is_empty())
301        .map(neutral_ir::ReuseProvenanceRecord::element_id)
302        .collect::<BTreeSet<_>>();
303    let mut observed = BTreeMap::new();
304    for provenance in artifacts.reference_provenance() {
305        let Some(owner) = declarations.get(&provenance.element_id()) else {
306            return Err(ReaderError::InvalidReferenceEdge);
307        };
308        let Some(target) = declarations.get(&provenance.target_element_id()) else {
309            return Err(ReaderError::InvalidReferenceEdge);
310        };
311        let Some((expected, value)) = value_at_path(
312            owner.resolved_type(),
313            owner.value(),
314            provenance.value_path(),
315            records,
316            artifacts.logical_document().vocabulary(),
317        ) else {
318            return Err(ReaderError::InvalidReferenceEdge);
319        };
320        let expected = match expected {
321            ResolvedType::Nullable(inner) => inner.as_ref(),
322            expected => expected,
323        };
324        let (ResolvedType::Ref(expected_target), LogicalValue::Reference(reference)) =
325            (expected, value)
326        else {
327            return Err(ReaderError::InvalidReferenceEdge);
328        };
329        if reference.target_element_id() != provenance.target_element_id()
330            || reference.target_symbol_identity() != target.symbol_identity()
331            || reference.target_type() != target.resolved_type()
332            || expected_target.as_ref() != target.resolved_type()
333            || observed
334                .insert(
335                    (provenance.element_id(), provenance.value_path().to_vec()),
336                    provenance.target_element_id(),
337                )
338                .is_some()
339        {
340            return Err(ReaderError::InvalidReferenceEdge);
341        }
342    }
343    for declaration in declarations.values() {
344        if !reference_provenance_is_complete(
345            declaration.element_id(),
346            declaration.value(),
347            &mut Vec::new(),
348            &observed,
349        ) {
350            return Err(ReaderError::InvalidReferenceEdge);
351        }
352    }
353    for provenance in artifacts.provenance() {
354        let has_root_reference = observed.contains_key(&(provenance.element_id(), Vec::new()));
355        let expected_direct = has_root_reference && !root_reuses.contains(&provenance.element_id());
356        if expected_direct
357            != matches!(
358                (provenance.origin(), provenance.normalization()),
359                (
360                    ValueOrigin::IdentityReference,
361                    neutral_ir::Normalization::IdentityReferenceResolution
362                )
363            )
364        {
365            return Err(ReaderError::InvalidReferenceEdge);
366        }
367    }
368    Ok(())
369}

371/// Returns whether every final logical identity edge has one provenance entry.
372fn reference_provenance_is_complete(
373    element_id: ElementId,
374    value: &LogicalValue,
375    path: &mut Vec<String>,
376    observed: &BTreeMap<(ElementId, Vec<String>), ElementId>,
377) -> bool {
378    match value {
379        LogicalValue::Reference(reference) => observed
380            .get(&(element_id, path.clone()))
381            .is_some_and(|target| *target == reference.target_element_id()),
382        LogicalValue::Record(record) => record.fields().iter().all(|field| {
383            path.push(field.name().to_owned());
384            let complete =
385                reference_provenance_is_complete(element_id, field.value(), path, observed);
386            path.pop();
387            complete
388        }),
389        LogicalValue::VocabularyRecord(record) => record.fields().iter().all(|field| {
390            path.push(field.name().to_owned());
391            let complete =
392                reference_provenance_is_complete(element_id, field.value(), path, observed);
393            path.pop();
394            complete
395        }),
396        LogicalValue::List(items) => items.iter().enumerate().all(|(index, item)| {
397            path.push(index.to_string());
398            let complete = reference_provenance_is_complete(element_id, item, path, observed);
399            path.pop();
400            complete
401        }),
402        LogicalValue::Number(_)
403        | LogicalValue::String(_)
404        | LogicalValue::Boolean(_)
405        | LogicalValue::Null => true,
406    }
407}

409/// Validates immutable-value reuse ownership, paths, targets, types, and values.
410fn validate_reuse_provenance(
411    artifacts: &CompilationArtifacts,
412    records: &BTreeMap<&str, &RecordTypeDefinition>,
413) -> Result<(), ReaderError> {
414    let declarations = artifacts
415        .logical_document()
416        .declarations()
417        .iter()
418        .map(|declaration| (declaration.element_id(), declaration))
419        .collect::<BTreeMap<_, _>>();
420    let mut observed = BTreeSet::new();
421    for provenance in artifacts.reuse_provenance() {
422        let Some(owner) = declarations.get(&provenance.element_id()) else {
423            return Err(ReaderError::InvalidReuseProvenance);
424        };
425        let Some(source) = declarations.get(&provenance.source_element_id()) else {
426            return Err(ReaderError::InvalidReuseProvenance);
427        };
428        let Some((expected, value)) = value_at_path(
429            owner.resolved_type(),
430            owner.value(),
431            provenance.value_path(),
432            records,
433            artifacts.logical_document().vocabulary(),
434        ) else {
435            return Err(ReaderError::InvalidReuseProvenance);
436        };
437        if !observed.insert((provenance.element_id(), provenance.value_path().to_vec()))
438            || !reuse_type_is_compatible(expected, source.resolved_type())
439            || value != source.value()
440        {
441            return Err(ReaderError::InvalidReuseProvenance);
442        }
443    }
444    for provenance in artifacts.provenance() {
445        let root_reuse = observed.contains(&(provenance.element_id(), Vec::new()));
446        if matches!(
447            (provenance.origin(), provenance.normalization(), root_reuse),
448            (
449                ValueOrigin::OrdinaryReuse,
450                neutral_ir::Normalization::ImmutableValueReuse,
451                true
452            )
453        ) {
454            continue;
455        }
456        if provenance.origin() == ValueOrigin::OrdinaryReuse
457            || provenance.normalization() == neutral_ir::Normalization::ImmutableValueReuse
458            || root_reuse
459        {
460            return Err(ReaderError::InvalidReuseProvenance);
461        }
462    }
463    Ok(())
464}

466/// Resolves one canonical value path to its contextual type and final value.
467fn value_at_path<'a>(
468    expected: &'a ResolvedType,
469    value: &'a LogicalValue,
470    path: &[String],
471    records: &BTreeMap<&str, &'a RecordTypeDefinition>,
472    vocabulary: Option<&'a VocabularyContract>,
473) -> Option<(&'a ResolvedType, &'a LogicalValue)> {
474    if path.is_empty() {
475        return Some((expected, value));
476    }
477    let expected = match expected {
478        ResolvedType::Nullable(inner) => inner.as_ref(),
479        expected => expected,
480    };
481    let (head, tail) = path.split_first()?;
482    if let (ResolvedType::List(inner), LogicalValue::List(items)) = (expected, value) {
483        let index = head.parse::<usize>().ok()?;
484        return value_at_path(inner, items.get(index)?, tail, records, vocabulary);
485    }
486    if let (
487        ResolvedType::VocabularyRecord(identity),
488        LogicalValue::VocabularyRecord(record_value),
489    ) = (expected, value)
490    {
491        let schema = vocabulary?.type_by_name(identity.name())?;
492        let index = schema
493            .fields()
494            .iter()
495            .position(|field| field.name() == head)?;
496        return value_at_path(
497            schema.fields()[index].resolved_type(),
498            record_value.fields()[index].value(),
499            tail,
500            records,
501            vocabulary,
502        );
503    }
504    let (ResolvedType::Record(identity), LogicalValue::Record(record_value)) = (expected, value)
505    else {
506        return None;
507    };
508    let schema = records.get(identity.name())?;
509    let index = schema
510        .fields()
511        .iter()
512        .position(|field| field.name() == head)?;
513    value_at_path(
514        schema.fields()[index].resolved_type(),
515        record_value.fields()[index].value(),
516        tail,
517        records,
518        vocabulary,
519    )
520}

522/// Returns whether a source binding type may satisfy one reuse occurrence.
523fn reuse_type_is_compatible(expected: &ResolvedType, source: &ResolvedType) -> bool {
524    expected == source
525        || matches!(expected, ResolvedType::Nullable(inner) if inner.as_ref() == source)
526}

528/// Validates field provenance ownership, paths, uniqueness, and origin kinds.
529fn validate_field_provenance(
530    artifacts: &CompilationArtifacts,
531    records: &BTreeMap<&str, &RecordTypeDefinition>,
532) -> Result<(), ReaderError> {
533    let declarations = artifacts
534        .logical_document()
535        .declarations()
536        .iter()
537        .map(|declaration| (declaration.element_id(), declaration))
538        .collect::<BTreeMap<_, _>>();
539    let mut observed = BTreeMap::new();
540    for provenance in artifacts.field_provenance() {
541        let Some(declaration) = declarations.get(&provenance.element_id()) else {
542            return Err(ReaderError::InvalidFieldProvenance);
543        };
544        if provenance.field_path().is_empty()
545            || !matches!(
546                provenance.origin(),
547                ValueOrigin::ExplicitRecordField
548                    | ValueOrigin::UserRecordDefault
549                    | ValueOrigin::VocabularyDefault
550            )
551            || observed
552                .insert(
553                    (provenance.element_id(), provenance.field_path().to_vec()),
554                    provenance.origin(),
555                )
556                .is_some()
557            || !field_path_exists(
558                declaration.resolved_type(),
559                declaration.value(),
560                provenance.field_path(),
561                records,
562                artifacts.logical_document().vocabulary(),
563            )
564        {
565            return Err(ReaderError::InvalidFieldProvenance);
566        }
567    }
568    for declaration in declarations.values() {
569        if !field_provenance_is_complete(
570            declaration.element_id(),
571            declaration.resolved_type(),
572            declaration.value(),
573            &mut Vec::new(),
574            records,
575            artifacts.logical_document().vocabulary(),
576            &observed,
577        ) {
578            return Err(ReaderError::InvalidFieldProvenance);
579        }
580    }
581    Ok(())
582}

584/// Returns whether every final record field has exact explicit/default evidence.
585fn field_provenance_is_complete(
586    element_id: ElementId,
587    expected: &ResolvedType,
588    value: &LogicalValue,
589    path: &mut Vec<String>,
590    records: &BTreeMap<&str, &RecordTypeDefinition>,
591    vocabulary: Option<&VocabularyContract>,
592    observed: &BTreeMap<(ElementId, Vec<String>), ValueOrigin>,
593) -> bool {
594    let expected = match expected {
595        ResolvedType::Nullable(inner) => inner.as_ref(),
596        expected => expected,
597    };
598    if let (ResolvedType::List(inner), LogicalValue::List(items)) = (expected, value) {
599        for (index, item) in items.iter().enumerate() {
600            path.push(index.to_string());
601            if !field_provenance_is_complete(
602                element_id, inner, item, path, records, vocabulary, observed,
603            ) {
604                return false;
605            }
606            path.pop();
607        }
608        return true;
609    }
610    if let (ResolvedType::VocabularyRecord(identity), LogicalValue::VocabularyRecord(value)) =
611        (expected, value)
612    {
613        let Some(schema) = vocabulary.and_then(|contract| contract.type_by_name(identity.name()))
614        else {
615            return false;
616        };
617        for (schema_field, value_field) in schema.fields().iter().zip(value.fields()) {
618            path.push(schema_field.name().to_owned());
619            let Some(origin) = observed.get(&(element_id, path.clone())) else {
620                return false;
621            };
622            if *origin == ValueOrigin::ExplicitRecordField
623                && !field_provenance_is_complete(
624                    element_id,
625                    schema_field.resolved_type(),
626                    value_field.value(),
627                    path,
628                    records,
629                    vocabulary,
630                    observed,
631                )
632            {
633                return false;
634            }
635            path.pop();
636        }
637        return true;
638    }
639    let (ResolvedType::Record(identity), LogicalValue::Record(value)) = (expected, value) else {
640        return true;
641    };
642    let Some(schema) = records.get(identity.name()) else {
643        return false;
644    };
645    for (schema_field, value_field) in schema.fields().iter().zip(value.fields()) {
646        path.push(schema_field.name().to_owned());
647        let key = (element_id, path.clone());
648        let Some(origin) = observed.get(&key) else {
649            return false;
650        };
651        if *origin == ValueOrigin::ExplicitRecordField
652            && !field_provenance_is_complete(
653                element_id,
654                schema_field.resolved_type(),
655                value_field.value(),
656                path,
657                records,
658                vocabulary,
659                observed,
660            )
661        {
662            return false;
663        }
664        path.pop();
665    }
666    true
667}

669/// Returns whether a path identifies a final field in a typed record value.
670fn field_path_exists(
671    expected: &ResolvedType,
672    value: &LogicalValue,
673    path: &[String],
674    records: &BTreeMap<&str, &RecordTypeDefinition>,
675    vocabulary: Option<&VocabularyContract>,
676) -> bool {
677    let expected = match expected {
678        ResolvedType::Nullable(inner) => inner.as_ref(),
679        expected => expected,
680    };
681    let Some((head, tail)) = path.split_first() else {
682        return false;
683    };
684    if let (ResolvedType::List(inner), LogicalValue::List(items)) = (expected, value) {
685        let Ok(index) = head.parse::<usize>() else {
686            return false;
687        };
688        let Some(item) = items.get(index) else {
689            return false;
690        };
691        return field_path_exists(inner, item, tail, records, vocabulary);
692    }
693    if let (ResolvedType::VocabularyRecord(identity), LogicalValue::VocabularyRecord(value)) =
694        (expected, value)
695    {
696        let Some(schema) = vocabulary.and_then(|contract| contract.type_by_name(identity.name()))
697        else {
698            return false;
699        };
700        let Some(index) = schema
701            .fields()
702            .iter()
703            .position(|field| field.name() == head)
704        else {
705            return false;
706        };
707        if tail.is_empty() {
708            return true;
709        }
710        return field_path_exists(
711            schema.fields()[index].resolved_type(),
712            value.fields()[index].value(),
713            tail,
714            records,
715            vocabulary,
716        );
717    }
718    let (ResolvedType::Record(identity), LogicalValue::Record(value)) = (expected, value) else {
719        return false;
720    };
721    let Some(schema) = records.get(identity.name()) else {
722        return false;
723    };
724    let Some(index) = schema
725        .fields()
726        .iter()
727        .position(|field| field.name() == head)
728    else {
729        return false;
730    };
731    if tail.is_empty() {
732        return true;
733    }
734    field_path_exists(
735        schema.fields()[index].resolved_type(),
736        value.fields()[index].value(),
737        tail,
738        records,
739        vocabulary,
740    )
741}

743/// Validates one recursively typed value against public record schemas.
744fn validate_value(
745    expected: &ResolvedType,
746    value: &LogicalValue,
747    records: &BTreeMap<&str, &RecordTypeDefinition>,
748) -> bool {
749    match (expected, value) {
750        (ResolvedType::Nullable(_), LogicalValue::Null) => true,
751        (ResolvedType::Nullable(inner), value) => validate_value(inner, value, records),
752        (ResolvedType::Record(identity), LogicalValue::Record(value)) => {
753            if identity != value.nominal_type() {
754                return false;
755            }
756            let Some(schema) = records.get(identity.name()) else {
757                return false;
758            };
759            if identity != schema.nominal_identity() {
760                return false;
761            }
762            schema.fields().len() == value.fields().len()
763                && schema
764                    .fields()
765                    .iter()
766                    .zip(value.fields())
767                    .all(|(schema_field, value_field)| {
768                        schema_field.name() == value_field.name()
769                            && validate_value(
770                                schema_field.resolved_type(),
771                                value_field.value(),
772                                records,
773                            )
774                    })
775        }
776        (ResolvedType::List(inner), LogicalValue::List(items)) => items
777            .iter()
778            .all(|item| validate_value(inner, item, records)),
779        _ => expected.accepts_value(value),
780    }
781}

783/// Validates nominal ownership, canonical fields, targets, and acyclic embedding.
784fn validate_record_schemas(
785    artifacts: &CompilationArtifacts,
786    records: &BTreeMap<&str, &RecordTypeDefinition>,
787) -> Result<(), ReaderError> {
788    for record in records.values() {
789        if record.nominal_identity().module() != artifacts.logical_document().module() {
790            return Err(ReaderError::InvalidRecordSchema);
791        }
792        let mut previous = None;
793        for field in record.fields() {
794            if previous.is_some_and(|name| name >= field.name())
795                || !resolved_record_targets_exist(field.resolved_type(), records)
796                || field.default_value().is_some_and(|value| {
797                    !is_closed_default(value)
798                        || !validate_value(field.resolved_type(), value, records)
799                })
800            {
801                return Err(ReaderError::InvalidRecordSchema);
802            }
803            previous = Some(field.name());
804        }
805    }
806    let mut visiting = BTreeSet::new();
807    let mut visited = BTreeSet::new();
808    for name in records.keys() {
809        if record_schema_has_cycle(name, records, &mut visiting, &mut visited) {
810            return Err(ReaderError::InvalidRecordSchema);
811        }
812    }
813    Ok(())
814}

816/// Returns whether one schema default is recursively closed and reference-free.
817fn is_closed_default(value: &LogicalValue) -> bool {
818    match value {
819        LogicalValue::Record(record) => record
820            .fields()
821            .iter()
822            .all(|field| is_closed_default(field.value())),
823        LogicalValue::VocabularyRecord(record) => record
824            .fields()
825            .iter()
826            .all(|field| is_closed_default(field.value())),
827        LogicalValue::List(items) => items.iter().all(is_closed_default),
828        LogicalValue::Reference(_) => false,
829        LogicalValue::Number(_)
830        | LogicalValue::String(_)
831        | LogicalValue::Boolean(_)
832        | LogicalValue::Null => true,
833    }
834}

836/// Returns whether every nominal target in one resolved type has an exact schema.
837fn resolved_record_targets_exist(
838    resolved_type: &ResolvedType,
839    records: &BTreeMap<&str, &RecordTypeDefinition>,
840) -> bool {
841    match resolved_type {
842        ResolvedType::Record(identity) => records
843            .get(identity.name())
844            .is_some_and(|record| record.nominal_identity() == identity),
845        ResolvedType::Nullable(inner) | ResolvedType::List(inner) | ResolvedType::Ref(inner) => {
846            resolved_record_targets_exist(inner, records)
847        }
848        ResolvedType::VocabularyRecord(_)
849        | ResolvedType::Num
850        | ResolvedType::String
851        | ResolvedType::Bool => true,
852    }
853}
854
855/// Detects an embedded cycle from one public nominal record schema.
856fn record_schema_has_cycle(
857    name: &str,
858    records: &BTreeMap<&str, &RecordTypeDefinition>,
859    visiting: &mut BTreeSet<String>,
860    visited: &mut BTreeSet<String>,
861) -> bool {
862    if visited.contains(name) {
863        return false;
864    }
865    visiting.insert(name.to_owned());
866    for field in records[name].fields() {
867        let Some(target) = resolved_record_target(field.resolved_type()) else {
868            continue;
869        };
870        if visiting.contains(target) || record_schema_has_cycle(target, records, visiting, visited)
871        {
872            return true;
873        }
874    }
875    visiting.remove(name);
876    visited.insert(name.to_owned());
877    false
878}
879
880/// Returns the nominal target embedded by an active public resolved type.
881fn resolved_record_target(resolved_type: &ResolvedType) -> Option<&str> {
882    match resolved_type {
883        ResolvedType::Record(identity) => Some(identity.name()),
884        ResolvedType::VocabularyRecord(_) => None,
885        ResolvedType::Nullable(inner) | ResolvedType::List(inner) => resolved_record_target(inner),
886        ResolvedType::Ref(_) | ResolvedType::Num | ResolvedType::String | ResolvedType::Bool => {
887            None
888        }
889    }
