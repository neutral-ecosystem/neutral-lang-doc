53/// Immutable typed traversal over already-validated in-process artifacts.
54#[derive(Clone, Debug)]
55pub struct ValidatedDocument {
56    /// Shared immutable compiler artifacts.
57    artifacts: Arc<CompilationArtifacts>,
58}
59
60impl ValidatedDocument {
61    /// Validates cross-artifact indexes before exposing immutable reader views.
62    ///
63    /// # Errors
64    ///
65    /// Returns a fail-closed error for invalid semantics, duplicate elements or
66    /// inconsistent derivation, source-map and provenance companions.
67    pub fn from_compiler_output(artifacts: Arc<CompilationArtifacts>) -> Result<Self, ReaderError> {
68        validate_artifacts(&artifacts)?;
69        Ok(Self { artifacts })
70    }
71
72    /// Returns the logical module name.
73    #[must_use]
74    pub fn module_name(&self) -> &str {
75        self.artifacts.logical_document().module().module_name()
76    }
77
78    /// Returns immutable declarations in deterministic order.
79    #[must_use]
80    pub fn declarations(&self) -> &[Declaration] {
81        self.artifacts.logical_document().declarations()
82    }
83
84    /// Returns nominal record schemas in canonical name order.
85    #[must_use]
86    pub fn record_types(&self) -> &[RecordTypeDefinition] {
87        self.artifacts.logical_document().record_types()
88    }
89
90    /// Returns the exact captured vocabulary contract, when present.
91    #[must_use]
92    pub fn vocabulary(&self) -> Option<&VocabularyContract> {
93        self.artifacts.logical_document().vocabulary()
94    }
95
96    /// Finds one nominal record schema by its validated name.
97    #[must_use]
98    pub fn record_type_by_name(&self, name: &str) -> Option<&RecordTypeDefinition> {
99        self.artifacts.logical_document().record_type_by_name(name)
100    }
101
102    /// Finds one declaration by its validated source name.
103    #[must_use]
104    pub fn declaration_by_name(&self, name: &str) -> Option<&Declaration> {
105        self.declarations()
106            .iter()
107            .find(|declaration| declaration.name() == name)
108    }
109
110    /// Maps a declaration element to its complete original-byte source location.
111    #[must_use]
112    pub fn source_location(&self, element_id: ElementId) -> Option<SourceLocation> {
113        self.artifacts.source_map().entry(element_id).map(|entry| {
114            SourceLocation::new(
115                self.artifacts.source_map().source_digest(),
116                entry.declaration_span(),
117            )
118        })
119    }
120
121    /// Returns the immutable validated compiler artifacts for advanced readers.
122    #[must_use]
123    pub const fn artifacts(&self) -> &Arc<CompilationArtifacts> {
124        &self.artifacts
125    }
126
127    /// Compares validated logical payloads under document-local ID renaming.
128    ///
129    /// Source locations, provenance, derivation, and numeric ID spellings are
130    /// companion facts and do not participate. Callers must use durable symbol
131    /// identities, never retain an `ElementId` for cross-document continuity.
132    #[must_use]
133    pub fn logically_equivalent(&self, other: &Self) -> bool {
134        self.artifacts.logically_equivalent(&other.artifacts)
135    }
136}

138/// A fail-closed in-process reader validation failure.
139#[derive(Clone, Copy, Debug, Eq, PartialEq)]
140pub enum ReaderError {
141    /// Two declarations reused one graph-local element identifier.
142    DuplicateElementId,
143    /// Two declarations reused one source name.
144    DuplicateDeclarationName,
145    /// A declaration had no matching source-map entry.
146    MissingSourceMapEntry,
147    /// A declaration had no matching value-provenance record.
148    MissingProvenanceRecord,
149    /// A declaration value did not satisfy its resolved type and nullability.
150    TypeValueMismatch,
151    /// A nominal record schema violated ownership, ordering, or recursion rules.
152    InvalidRecordSchema,
153    /// Field provenance was dangling, duplicated, or used an invalid origin.
154    InvalidFieldProvenance,
155    /// Reuse provenance was dangling, duplicated, or inconsistent with final values.
156    InvalidReuseProvenance,
157    /// A typed identity edge or its provenance was dangling or inconsistent.
158    InvalidReferenceEdge,
159    /// A declaration fingerprint did not match its complete logical definition.
160    InvalidDeclarationFingerprint,
161    /// Logical and derivation vocabulary contracts were missing or inconsistent.
162    InvalidVocabularyContract,
163    /// Source spans, coverage or canonical ordering were inconsistent.
164    InvalidSourceMap,
165    /// Companion provenance coverage or ordering was inconsistent.
166    InvalidProvenance,
167    /// Resource and derivation facts disagreed with the document or source.
168    InvalidDerivation,
169    /// Symbol/module ownership or declaration ordering was inconsistent.
170    InvalidLogicalIdentity,
171}
