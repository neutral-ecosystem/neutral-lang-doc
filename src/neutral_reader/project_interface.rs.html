19/// A reader rejection of an invalid or non-public project interface.
20#[derive(Clone, Copy, Debug, Eq, PartialEq)]
21pub enum ProjectInterfaceError {
22    /// An export identity was malformed, duplicated, or out of order.
23    InvalidExport,
24    /// A public signature had malformed fields, inaccessible types, or excess depth.
25    InvalidSignature,
26    /// A public edge was dangling, duplicated, incorrectly typed, or out of order.
27    InvalidEdge,
28    /// The declared public signature fingerprint did not match its content.
29    InvalidFingerprint,
30}
31
32/// Immutable, independently validated public-only project interface.
33#[derive(Clone, Debug)]
34pub struct ValidatedProjectInterface {
35    /// Complete public-only snapshot after structural and digest validation.
36    interface: Arc<ProjectInterface>,
37}
38
39impl ValidatedProjectInterface {
40    /// Validates a public interface without compiler linkage or source parsing.
41    ///
42    /// # Errors
43    ///
44    /// Rejects invalid export order, types, edges, or fingerprint before a view
45    /// is published. No source IDs, private roots, or raw provenance are exposed.
46    pub fn from_interface(interface: Arc<ProjectInterface>) -> Result<Self, ProjectInterfaceError> {
47        let exports = interface.exports();
48        let mut index = BTreeMap::new();
49        let mut previous = None;
50        for (position, export) in exports.iter().enumerate() {
51            if previous.is_some_and(|identity| identity >= export.identity())
52                || !valid_identity(export.identity(), export.signature())
53            {
54                return Err(ProjectInterfaceError::InvalidExport);
55            }
56            previous = Some(export.identity());
57            index.insert(export.identity(), position);
58        }
59        for export in exports {
60            if !valid_signature(export.signature(), &index, exports) {
61                return Err(ProjectInterfaceError::InvalidSignature);
62            }
63        }
64        let mut previous_edge = None;
65        for edge in interface.edges() {
66            let key = (edge.from(), edge.kind(), edge.to());
67            if previous_edge.is_some_and(|previous| previous >= key)
68                || !valid_edge(edge, &index, exports)
69            {
70                return Err(ProjectInterfaceError::InvalidEdge);
71            }
72            previous_edge = Some(key);
73        }
74        let mut expected_types = BTreeSet::new();
75        for export in exports {
76            match export.signature() {
77                ProjectPublicSignature::Binding(ty) => {
78                    collect_type_edges(export.identity(), ty, false, &mut expected_types);
79                }
80                ProjectPublicSignature::Record(fields) => {
81                    for field in fields {
82                        collect_type_edges(
83                            export.identity(),
84                            field.ty(),
85                            false,
86                            &mut expected_types,
87                        );
88                    }
89                }
90            }
91        }
92        let actual_types = interface
93            .edges()
94            .iter()
95            .filter(|edge| {
96                matches!(
97                    edge.kind(),
98                    ProjectPublicEdgeKind::Type | ProjectPublicEdgeKind::ReferenceType
99                )
100            })
101            .map(|edge| (edge.from().clone(), edge.kind(), edge.to().clone()))
102            .collect::<BTreeSet<_>>();
103        if actual_types != expected_types {
104            return Err(ProjectInterfaceError::InvalidEdge);
105        }
106        if interface.recompute_fingerprint().ok() != Some(interface.fingerprint()) {
107            return Err(ProjectInterfaceError::InvalidFingerprint);
108        }
109        Ok(Self { interface })
110    }
111
112    /// Returns public exports in canonical module-symbol order.
113    #[must_use]
114    pub fn exports(&self) -> &[ProjectPublicExport] {
115        self.interface.exports()
116    }
117
118    /// Finds one public export by exact full module-symbol identity.
119    #[must_use]
120    pub fn export_by_identity(
121        &self,
122        identity: &ModuleSymbolIdentity,
123    ) -> Option<&ProjectPublicExport> {
124        self.interface
125            .exports()
126            .binary_search_by(|export| export.identity().cmp(identity))
127            .ok()
128            .map(|index| &self.interface.exports()[index])
129    }
130
131    /// Returns only public-to-public dependency edges with source details redacted.
132    #[must_use]
133    pub fn public_edges(&self) -> &[ProjectPublicEdge] {
134        self.interface.edges()
135    }
136
137    /// Enumerates cross-module immutable reuse without a compiler dependency.
138    pub fn cross_module_values(&self) -> impl Iterator<Item = &ProjectPublicEdge> {
139        self.public_edges().iter().filter(|edge| {
140            edge.kind() == ProjectPublicEdgeKind::Value
141                && edge.from().module() != edge.to().module()
142        })
143    }
144
145    /// Enumerates cross-module identity references without a compiler dependency.
146    pub fn cross_module_references(&self) -> impl Iterator<Item = &ProjectPublicEdge> {
147        self.public_edges().iter().filter(|edge| {
148            edge.kind() == ProjectPublicEdgeKind::Reference
149                && edge.from().module() != edge.to().module()
150        })
151    }
152
153    /// Returns the checked public-signature fingerprint.
154    #[must_use]
155    pub fn fingerprint(&self) -> neutral_core::SemanticDigest {
156        self.interface.fingerprint()
157    }
158}

160/// Collects the exact nominal dependencies implied by one validated public type.
161fn collect_type_edges(
162    from: &ModuleSymbolIdentity,
163    ty: &ProjectPublicType,
164    under_ref: bool,
165    edges: &mut BTreeSet<(
166        ModuleSymbolIdentity,
167        ProjectPublicEdgeKind,
168        ModuleSymbolIdentity,
169    )>,
170) {
171    match ty {
172        ProjectPublicType::Nominal(to) => {
173            let kind = if under_ref {
174                ProjectPublicEdgeKind::ReferenceType
175            } else {
176                ProjectPublicEdgeKind::Type
177            };
178            edges.insert((from.clone(), kind, to.clone()));
179        }
180        ProjectPublicType::List(inner) | ProjectPublicType::Nullable(inner) => {
181            collect_type_edges(from, inner, under_ref, edges);
182        }
183        ProjectPublicType::Ref(inner) => collect_type_edges(from, inner, true, edges),
184        ProjectPublicType::Num | ProjectPublicType::String | ProjectPublicType::Bool => {}
185    }
186}

188/// Checks exact profile, qualified module grammar, and root-name category.
189fn valid_identity(identity: &ModuleSymbolIdentity, signature: &ProjectPublicSignature) -> bool {
190    if identity.module().language_behavior_version() != V1_SOURCE_PROFILE
191        || !identity
192            .module()
193            .module_name()
194            .split("::")
195            .all(is_snake_name)
196    {
197        return false;
198    }
199    match signature {
200        ProjectPublicSignature::Binding(_) => is_snake_name(identity.declaration_name()),
201        ProjectPublicSignature::Record(_) => is_upper_name(identity.declaration_name()),
202    }
203}
204
205/// Checks a complete public signature and canonical field ordering.
206fn valid_signature(
207    signature: &ProjectPublicSignature,
208    index: &BTreeMap<&ModuleSymbolIdentity, usize>,
209    exports: &[ProjectPublicExport],
210) -> bool {
211    match signature {
212        ProjectPublicSignature::Binding(ty) => valid_type(ty, 0, index, exports),
213        ProjectPublicSignature::Record(fields) => {
214            let mut previous = None;
215            for field in fields {
216                if !is_snake_name(field.name())
217                    || previous.is_some_and(|name| name >= field.name())
218                    || !valid_type(field.ty(), 0, index, exports)
219                {
220                    return false;
221                }
222                previous = Some(field.name());
223            }
224            true
225        }
226    }
227}

229/// Checks nested public nominal closure before recursive fingerprinting.
230fn valid_type(
231    ty: &ProjectPublicType,
232    depth: usize,
233    index: &BTreeMap<&ModuleSymbolIdentity, usize>,
234    exports: &[ProjectPublicExport],
235) -> bool {
236    if depth > MAX_PROJECT_INTERFACE_TYPE_DEPTH {
237        return false;
238    }
239    match ty {
240        ProjectPublicType::Num | ProjectPublicType::String | ProjectPublicType::Bool => true,
241        ProjectPublicType::Nominal(identity) => index.get(identity).is_some_and(|position| {
242            matches!(
243                exports[*position].signature(),
244                ProjectPublicSignature::Record(_)
245            )
246        }),
247        ProjectPublicType::List(inner) | ProjectPublicType::Ref(inner) => {
248            valid_type(inner, depth + 1, index, exports)
249        }
250        ProjectPublicType::Nullable(inner) => {
251            !matches!(inner.as_ref(), ProjectPublicType::Nullable(_))
252                && valid_type(inner, depth + 1, index, exports)
253        }
254    }
255}

257/// Checks public endpoints and edge category against their export kinds.
258fn valid_edge(
259    edge: &ProjectPublicEdge,
260    index: &BTreeMap<&ModuleSymbolIdentity, usize>,
261    exports: &[ProjectPublicExport],
262) -> bool {
263    let Some(&from) = index.get(edge.from()) else {
264        return false;
265    };
266    let Some(&to) = index.get(edge.to()) else {
267        return false;
268    };
269    match edge.kind() {
270        ProjectPublicEdgeKind::Type | ProjectPublicEdgeKind::ReferenceType => {
271            matches!(exports[to].signature(), ProjectPublicSignature::Record(_))
272        }
273        ProjectPublicEdgeKind::Value | ProjectPublicEdgeKind::Reference => {
274            matches!(
275                exports[from].signature(),
276                ProjectPublicSignature::Binding(_)
277            ) && matches!(exports[to].signature(), ProjectPublicSignature::Binding(_))
278        }
279    }
