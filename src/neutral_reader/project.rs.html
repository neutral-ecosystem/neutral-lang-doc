22/// Fail-closed complete-project reader and view classifications.
23#[derive(Clone, Copy, Debug, Eq, PartialEq)]
24pub enum ProjectReadError {
25    /// Unknown schema or profile.
26    Schema,
27    /// An explicit caller bound was exceeded.
28    Limit,
29    /// Invalid, duplicate, unordered, or dangling module/import.
30    Module,
31    /// Invalid declaration, type, value, or semantic cycle.
32    Declaration,
33    /// Missing, mismatched, private-leaking, or malformed export index.
34    PublicInterface,
35    /// Invalid original-byte source or provenance companion.
36    Companion,
37    /// Declared resource facts differ from actual content.
38    Resources,
39    /// Cancellation prevented publication.
40    Cancelled,
41    /// Invalid, duplicate, absent, or private selected root.
42    View,
43}
44
45impl ProjectReadError {
46    /// Returns the typed outcome schema, independent of package releases.
47    #[must_use]
48    pub const fn schema(self) -> &'static str {
49        PROJECT_RESULT_SCHEMA
50    }
51}

62/// Public-only selected dependency closure, without source or private provenance.
63#[derive(Clone, Debug, Eq, PartialEq)]
64pub struct ProjectView {
65    /// Canonical selected roots, separate from the complete project.
66    roots: Vec<ModuleSymbolIdentity>,
67    /// Full public interpretive declaration closure.
68    exports: Vec<ProjectPublicExport>,
69    /// Fully materialized public binding values in canonical order.
70    values: Vec<(ModuleSymbolIdentity, ProjectValue)>,
71    /// Public locked schemas needed to interpret the selected nominal values.
72    vocabulary_records: Vec<ProjectVocabularyRecord>,
73}
74
75impl ProjectView {
76    /// Returns the post-compilation view schema, never a complete IR schema.
77    #[must_use]
78    pub const fn schema(&self) -> &'static str {
79        PROJECT_VIEW_SCHEMA
80    }
81    /// Returns the normalized selection, never a capture input.
82    #[must_use]
83    pub fn roots(&self) -> &[ModuleSymbolIdentity] {
84        &self.roots
85    }
86    /// Returns required public declaration signatures, without private defaults.
87    #[must_use]
88    pub fn exports(&self) -> &[ProjectPublicExport] {
89        &self.exports
90    }
91    /// Returns public materialized binding values, with no source accounting.
92    #[must_use]
93    pub fn values(&self) -> &[(ModuleSymbolIdentity, ProjectValue)] {
94        &self.values
95    }
96    /// Returns exact public vocabulary interpretation dependencies, never metadata.
97    #[must_use]
98    pub fn vocabulary_records(&self) -> &[ProjectVocabularyRecord] {
99        &self.vocabulary_records
100    }

103impl ValidatedProject {
104    /// Validates complete data without source parsing or compiler-private linkage.
105    ///
106    /// # Errors
107    /// Rejects malformed, incomplete, over-budget, or cancelled projects atomically.
108    pub fn from_ir(
109        ir: Arc<ProjectIr>,
110        limits: ProjectLimits,
111        cancellation: &CancellationToken,
112    ) -> Result<Self, ProjectReadError> {
113        check_cancel(cancellation)?;
114        if ir.schema != PROJECT_IR_SCHEMA {
115            return Err(ProjectReadError::Schema);
116        }
117        let limits = limits.intersect(ir.limits);
118        check_schema_bounds(&ir, limits, cancellation)?;
119        check_modules(&ir, limits)?;
120        let index = declaration_index(&ir)?;
121        check_vocabularies(&ir, limits)?;
122        check_vocabulary_cycles(&ir, cancellation)?;
123        let mut budget = ValueBudget {
124            nodes: 0,
125            bytes: 0,
126            limits,
127            cancellation,
128        };
129        check_declarations(&ir, &index, &mut budget)?;
130        check_companions(&ir, &index, limits)?;
131        check_semantic_graph(&ir, &index, cancellation)?;
132        let interface =
133            ValidatedProjectInterface::from_interface(Arc::new(ir.public_interface.clone()))
134                .map_err(|_| ProjectReadError::PublicInterface)?;
135        let expected = ir
136            .declarations
137            .iter()
138            .filter(|decl| decl.public)
139            .map(|decl| ProjectPublicExport::new(decl.identity.clone(), decl.signature.clone()))
140            .collect::<Vec<_>>();
141        if interface.exports() != expected {
142            return Err(ProjectReadError::PublicInterface);
143        }
144        if ir.recompute_public_interface().ok().as_ref() != Some(&ir.public_interface) {
145            return Err(ProjectReadError::PublicInterface);
146        }
147        let actual = ProjectResourceFacts {
148            source_units: ir.sources.len() as u64,
149            source_bytes: ir
150                .sources
151                .iter()
152                .try_fold(0_u64, |sum, source| sum.checked_add(source.byte_len))
153                .ok_or(ProjectReadError::Limit)?,
154            vocabulary_units: ir.vocabulary_sources.len() as u64,
155            vocabulary_bytes: ir
156                .vocabulary_sources
157                .iter()
158                .try_fold(0_u64, |sum, source| sum.checked_add(source.byte_len))
159                .ok_or(ProjectReadError::Limit)?,
160            declarations: ir.declarations.len() as u64,
161            import_edges: ir
162                .modules
163                .iter()
164                .map(|module| module.imports.len() as u64)
165                .sum(),
166            value_nodes: budget.nodes,
167        };
168        if actual != ir.resources {
169            return Err(ProjectReadError::Resources);
170        }
171        check_cancel(cancellation)?;
172        Ok(Self { ir, interface })
173    }

175    /// Returns complete validated data for trusted inspection, not a redacted view.
176    #[must_use]
177    pub const fn complete_ir(&self) -> &Arc<ProjectIr> {
178        &self.ir
179    }

181    /// Derives a canonical public dependency closure without changing complete IR.
182    ///
183    /// # Errors
184    /// Rejects unknown/private/duplicate roots, schema mismatch, limits, or cancellation.
185    pub fn derive_view(
186        &self,
187        request: &ViewRequest,
188        cancellation: &CancellationToken,
189    ) -> Result<ProjectView, ProjectReadError> {
190        check_cancel(cancellation)?;
191        if request.schema != PROJECT_VIEW_SCHEMA {
192            return Err(ProjectReadError::Schema);
193        }
194        if request.roots.len() > self.ir.declarations.len() {
195            return Err(ProjectReadError::Limit);
196        }
197        let roots = request.roots.iter().cloned().collect::<BTreeSet<_>>();
198        if roots.len() != request.roots.len()
199            || roots
200                .iter()
201                .any(|root| self.interface.export_by_identity(root).is_none())
202        {
203            return Err(ProjectReadError::View);
204        }
205        let mut closure = roots.clone();
206        let mut pending = roots.iter().cloned().collect::<Vec<_>>();
207        let mut adjacency = BTreeMap::<_, Vec<_>>::new();
208        for edge in self.interface.public_edges() {
209            adjacency.entry(edge.from()).or_default().push(edge.to());
210        }
211        while let Some(from) = pending.pop() {
212            check_cancel(cancellation)?;
213            for to in adjacency.get(&from).into_iter().flatten() {
214                if closure.insert((*to).clone()) {
215                    pending.push((*to).clone());
216                }
217            }
218        }
219        let exports: Vec<_> = self
220            .interface
221            .exports()
222            .iter()
223            .filter(|export| closure.contains(export.identity()))
224            .cloned()
225            .collect();
226        let values = self
227            .ir
228            .declarations
229            .iter()
230            .filter(|decl| closure.contains(&decl.identity))
231            .filter_map(|decl| {
232                decl.value
233                    .as_ref()
234                    .map(|value| (decl.identity.clone(), value.clone()))
235            })
236            .collect();
237        let vocabulary_records = view_vocabularies(&exports, &self.ir, cancellation)?;
238        check_cancel(cancellation)?;
239        Ok(ProjectView {
240            roots: roots.into_iter().collect(),
241            exports,
242            values,
243            vocabulary_records,
244        })
245    }
246}

248/// Derives the transitive public vocabulary schema closure of exported types.
249fn view_vocabularies(
250    exports: &[ProjectPublicExport],
251    ir: &ProjectIr,
252    cancellation: &CancellationToken,
253) -> Result<Vec<ProjectVocabularyRecord>, ProjectReadError> {
254    let mut types = Vec::new();
255    for export in exports {
256        match export.signature() {
257            ProjectPublicSignature::Binding(ty) => types.push(ty),
258            ProjectPublicSignature::Record(fields) => types.extend(
259                fields
260                    .iter()
261                    .map(neutral_ir::project_interface::ProjectPublicField::ty),
262            ),
263        }
264    }
265    let mut selected = BTreeSet::new();
266    while let Some(ty) = types.pop() {
267        check_cancel(cancellation)?;
268        match ty {
269            ProjectPublicType::List(inner)
270            | ProjectPublicType::Ref(inner)
271            | ProjectPublicType::Nullable(inner) => types.push(inner),
272            ProjectPublicType::VocabularyNominal {
273                identity,
274                version,
275                name,
276            } if selected.insert((identity, version, name)) => {
277                let record = ir
278                    .vocabulary_records
279                    .iter()
280                    .find(|record| {
281                        record.identity == *identity
282                            && record.version == *version
283                            && record.name == *name
284                            && record.public
285                    })
286                    .ok_or(ProjectReadError::PublicInterface)?;
287                types.extend(record.fields.iter().map(|(_, ty)| ty));
288            }
289            _ => {}
290        }
291    }
292    Ok(ir
293        .vocabulary_records
294        .iter()
295        .filter(|record| selected.contains(&(&record.identity, &record.version, &record.name)))
296        .cloned()
297        .collect())
298}

300/// Checks cooperative cancellation at bounded work boundaries.
301fn check_cancel(cancellation: &CancellationToken) -> Result<(), ProjectReadError> {
302    if cancellation.is_cancelled() {
303        Err(ProjectReadError::Cancelled)
304    } else {
305        Ok(())
306    }

309/// Rejects malformed module identities, imports, and caller limit violations.
310fn check_modules(ir: &ProjectIr, limits: ProjectLimits) -> Result<(), ProjectReadError> {
311    if ir.modules.is_empty()
312        || ir.modules.len() as u64 > limits.modules
313        || ir.declarations.len() as u64 > limits.declarations
314        || ir.provenance.len() as u64 > limits.nodes
315        || ir.sources.len() != ir.modules.len()
316    {
317        return Err(ProjectReadError::Limit);
318    }
319    let mut previous = None;
320    let modules = ir
321        .modules
322        .iter()
323        .map(|module| module.identity.module_name())
324        .collect::<BTreeSet<_>>();
325    let mut edges = 0_u64;
326    for module in &ir.modules {
327        let name = module.identity.module_name();
328        if module.identity.language_behavior_version() != V1_SOURCE_PROFILE
329            || !name.split("::").all(is_snake_name)
330            || previous.is_some_and(|prior| prior >= name)
331        {
332            return Err(ProjectReadError::Module);
333        }
334        previous = Some(name);
335        let mut prior_import = None;
336        for target in &module.imports {
337            if target == name
338                || !modules.contains(target.as_str())
339                || prior_import.is_some_and(|prior| prior >= target)
340            {
341                return Err(ProjectReadError::Module);
342            }
343            prior_import = Some(target);
344            edges = edges.checked_add(1).ok_or(ProjectReadError::Limit)?;
345        }
346    }
347    if edges > limits.import_edges {
348        return Err(ProjectReadError::Limit);
349    }
350    Ok(())
351}

353/// Builds a checked canonical declaration index with valid name categories.
354fn declaration_index(
355    ir: &ProjectIr,
356) -> Result<BTreeMap<&ModuleSymbolIdentity, &ProjectDeclaration>, ProjectReadError> {
357    let mut index = BTreeMap::new();
358    let mut previous = None;
359    for decl in &ir.declarations {
360        let valid_name = match decl.signature {
361            ProjectPublicSignature::Binding(_) => is_snake_name(decl.identity.declaration_name()),
362            ProjectPublicSignature::Record(_) => is_upper_name(decl.identity.declaration_name()),
363        };
364        if !valid_name
365            || is_protected_name(decl.identity.declaration_name())
366            || previous.is_some_and(|prior| prior >= &decl.identity)
367            || !ir
368                .modules
369                .iter()
370                .any(|module| module.identity == *decl.identity.module())
371        {
372            return Err(ProjectReadError::Declaration);
373        }
374        previous = Some(&decl.identity);
375        index.insert(&decl.identity, decl);
376    }
377    Ok(index)
378}

380/// Bounds complete and exported schemas before cloning or graph construction.
381fn check_schema_bounds(
382    ir: &ProjectIr,
383    limits: ProjectLimits,
384    cancellation: &CancellationToken,
385) -> Result<(), ProjectReadError> {
386    check_cancel(cancellation)?;
387    if !ir.within_schema_limits(limits) {
388        return Err(ProjectReadError::Limit);
389    }
390    check_cancel(cancellation)
391}

393/// Checks complete canonical vocabulary schemas against the public catalogue.
394fn check_vocabularies(ir: &ProjectIr, limits: ProjectLimits) -> Result<(), ProjectReadError> {
395    if ir.vocabulary_records.len() as u64 > limits.nodes {
396        return Err(ProjectReadError::Limit);
397    }
398    let mut previous = None;
399    for record in &ir.vocabulary_records {
400        let key = (&record.identity, &record.version, &record.name);
401        if !is_upper_name(&record.identity)
402            || !is_exact_release_version(&record.version)
403            || !is_upper_name(&record.name)
404            || previous.is_some_and(|prior| prior >= key)
405        {
406            return Err(ProjectReadError::Declaration);
407        }
408        previous = Some(key);
409        let vocabulary = ir
410            .public_interface
411            .vocabularies()
412            .iter()
413            .find(|vocabulary| {
414                vocabulary.identity() == record.identity && vocabulary.version() == record.version
415            })
416            .ok_or(ProjectReadError::PublicInterface)?;
417        if vocabulary
418            .public_types()
419            .binary_search(&record.name)
420            .is_ok()
421            != record.public
422        {
423            return Err(ProjectReadError::PublicInterface);
424        }
425        let mut prior = None;
426        if record.fields.len() as u64 > limits.nodes {
427            return Err(ProjectReadError::Limit);
428        }
429        for (name, ty) in &record.fields {
430            if !is_snake_name(name)
431                || is_protected_name(name)
432                || prior.is_some_and(|old| old >= name)
433            {
434                return Err(ProjectReadError::Declaration);
435            }
436            prior = Some(name);
437            match ty {
438                ProjectPublicType::Num
439                | ProjectPublicType::String
440                | ProjectPublicType::Bool
441                | ProjectPublicType::Url
442                | ProjectPublicType::Path => {}
443                ProjectPublicType::VocabularyNominal {
444                    identity,
445                    version,
446                    name,
447                } if identity == &record.identity
448                    && version == &record.version
449                    && ir.vocabulary_records.iter().any(|target| {
450                        target.identity == *identity
451                            && target.version == *version
452                            && target.name == *name
453                            && (!record.public || target.public)
454                    }) => {}
455                _ => return Err(ProjectReadError::Declaration),
456            }
457        }
458    }
459    for vocabulary in ir.public_interface.vocabularies() {
460        for name in vocabulary.public_types() {
461            if !ir.vocabulary_records.iter().any(|record| {
462                record.identity == vocabulary.identity()
463                    && record.version == vocabulary.version()
464                    && record.name == *name
465                    && record.public
466            }) {
467                return Err(ProjectReadError::PublicInterface);
468            }
469        }
470    }
471    Ok(())
472}

474/// Rejects embedded vocabulary type cycles without recursive graph traversal.
475fn check_vocabulary_cycles(
476    ir: &ProjectIr,
477    cancellation: &CancellationToken,
478) -> Result<(), ProjectReadError> {
479    let index = ir
480        .vocabulary_records
481        .iter()
482        .enumerate()
483        .map(|(position, record)| {
484            (
485                (
486                    record.identity.as_str(),
487                    record.version.as_str(),
488                    record.name.as_str(),
489                ),
490                position,
491            )
492        })
493        .collect::<BTreeMap<_, _>>();
494    let mut outgoing = vec![BTreeSet::new(); index.len()];
495    let mut incoming = vec![0_usize; index.len()];
496    for (position, record) in ir.vocabulary_records.iter().enumerate() {
497        check_cancel(cancellation)?;
498        for (_, ty) in &record.fields {
499            if let ProjectPublicType::VocabularyNominal {
500                identity,
501                version,
502                name,
503            } = ty
504            {
505                let target = *index
506                    .get(&(identity.as_str(), version.as_str(), name.as_str()))
507                    .ok_or(ProjectReadError::Declaration)?;
508                if outgoing[position].insert(target) {
509                    incoming[target] += 1;
510                }
511            }
512        }
513    }
514    let mut pending = incoming
515        .iter()
516        .enumerate()
517        .filter_map(|(position, count)| (*count == 0).then_some(position))
518        .collect::<Vec<_>>();
519    let mut visited = 0;
520    while let Some(position) = pending.pop() {
521        check_cancel(cancellation)?;
522        visited += 1;
523        for target in &outgoing[position] {
524            incoming[*target] -= 1;
525            if incoming[*target] == 0 {
526                pending.push(*target);
527            }
528        }
529    }
530    if visited != index.len() {
531        return Err(ProjectReadError::Declaration);
532    }
533    Ok(())
534}

536/// Verifies complete type/provenance edges, imported visibility, and non-reference cycles.
537fn check_semantic_graph(
538    ir: &ProjectIr,
539    index: &BTreeMap<&ModuleSymbolIdentity, &ProjectDeclaration>,
540    cancellation: &CancellationToken,
541) -> Result<(), ProjectReadError> {
542    use neutral_ir::project_interface::ProjectPublicEdgeKind;
543    let expected = expected_type_edges(ir, cancellation)?;
544    let actual = ir
545        .provenance
546        .iter()
547        .filter(|edge| {
548            matches!(
549                edge.kind,
550                ProjectPublicEdgeKind::Type | ProjectPublicEdgeKind::ReferenceType
551            )
552        })
553        .map(|edge| (&edge.from, edge.kind, &edge.to))
554        .collect::<BTreeSet<_>>();
555    if actual != expected {
556        return Err(ProjectReadError::Companion);
557    }
558    let mut adjacency = BTreeMap::<_, BTreeSet<_>>::new();
559    let mut indegrees = index
560        .keys()
561        .map(|identity| (*identity, 0_usize))
562        .collect::<BTreeMap<_, _>>();
563    for edge in &ir.provenance {
564        let from = index[&edge.from];
565        let to = index[&edge.to];
566        if edge.from.module() != edge.to.module()
567            && (!to.public
568                || !ir.modules.iter().any(|module| {
569                    module.identity == *edge.from.module()
570                        && module
571                            .imports
572                            .iter()
573                            .any(|target| target == edge.to.module().module_name())
574                }))
575        {
576            return Err(ProjectReadError::Declaration);
577        }
578        if matches!(
579            edge.kind,
580            ProjectPublicEdgeKind::Reference | ProjectPublicEdgeKind::Value
581        ) && (!matches!(from.signature, ProjectPublicSignature::Binding(_))
582            || !matches!(to.signature, ProjectPublicSignature::Binding(_)))
583        {
584            return Err(ProjectReadError::Declaration);
585        }
586        if matches!(
587            edge.kind,
588            ProjectPublicEdgeKind::Type | ProjectPublicEdgeKind::Value
589        ) && adjacency.entry(&edge.from).or_default().insert(&edge.to)
590        {
591            *indegrees
592                .get_mut(&edge.to)
593                .ok_or(ProjectReadError::Declaration)? += 1;
594        }
595    }
596    let mut ready = indegrees
597        .iter()
598        .filter(|(_, degree)| **degree == 0)
599        .map(|(identity, _)| *identity)
600        .collect::<Vec<_>>();
601    let mut seen = 0;
602    while let Some(from) = ready.pop() {
603        check_cancel(cancellation)?;
604        seen += 1;
605        for to in adjacency.get(from).into_iter().flatten() {
606            let degree = indegrees.get_mut(to).ok_or(ProjectReadError::Declaration)?;
607            *degree -= 1;
608            if *degree == 0 {
609                ready.push(to);
610            }
611        }
612    }
613    if seen != index.len() {
614        return Err(ProjectReadError::Declaration);
615    }
616    Ok(())
617}

619/// Canonical nominal type-edge keys, independent of source occurrence multiplicity.
620type TypeEdges<'a> = BTreeSet<(
621    &'a ModuleSymbolIdentity,
622    neutral_ir::project_interface::ProjectPublicEdgeKind,
623    &'a ModuleSymbolIdentity,
624)>;
625
626/// Derives the required nominal type edges from all complete signatures.
627fn expected_type_edges<'a>(
628    ir: &'a ProjectIr,
629    cancellation: &CancellationToken,
630) -> Result<TypeEdges<'a>, ProjectReadError> {
631    use neutral_ir::project_interface::ProjectPublicEdgeKind;
632    let mut expected = BTreeSet::new();
633    for decl in &ir.declarations {
634        check_cancel(cancellation)?;
635        let types = match &decl.signature {
636            ProjectPublicSignature::Binding(ty) => vec![ty],
637            ProjectPublicSignature::Record(fields) => fields
638                .iter()
639                .map(neutral_ir::project_interface::ProjectPublicField::ty)
640                .collect(),
641        };
642        let mut pending = types.into_iter().map(|ty| (ty, false)).collect::<Vec<_>>();
643        while let Some((ty, under_ref)) = pending.pop() {
644            match ty {
645                ProjectPublicType::Nominal(target) => {
646                    expected.insert((
647                        &decl.identity,
648                        if under_ref {
649                            ProjectPublicEdgeKind::ReferenceType
650                        } else {
651                            ProjectPublicEdgeKind::Type
652                        },
653                        target,
654                    ));
655                }
656                ProjectPublicType::Ref(inner) => pending.push((inner, true)),
657                ProjectPublicType::List(inner) | ProjectPublicType::Nullable(inner) => {
658                    pending.push((inner, under_ref));
659                }
660                _ => {}
661            }
662        }
663    }
664    Ok(expected)
665}

667/// Shared aggregate value/text work accounting across all declarations and defaults.
668struct ValueBudget<'a> {
669    /// Nodes processed so far.
670    nodes: u64,
671    /// Retained text bytes processed so far.
672    bytes: u64,
673    /// Caller-provided maximums.
674    limits: ProjectLimits,
675    /// Cooperative cancellation input.
676    cancellation: &'a CancellationToken,
677}
678
679impl ValueBudget<'_> {
680    /// Charges a node and its retained textual payload before further work.
681    fn charge(&mut self, bytes: usize) -> Result<(), ProjectReadError> {
682        check_cancel(self.cancellation)?;
683        self.nodes = self.nodes.checked_add(1).ok_or(ProjectReadError::Limit)?;
684        self.bytes = self
685            .bytes
686            .checked_add(bytes as u64)
687            .ok_or(ProjectReadError::Limit)?;
688        if self.nodes > self.limits.nodes || self.bytes > self.limits.text_bytes {
689            return Err(ProjectReadError::Limit);
690        }
691        Ok(())
692    }
693}

695/// Validates all typed roots and closed defaults before publication.
696fn check_declarations(
697    ir: &ProjectIr,
698    index: &BTreeMap<&ModuleSymbolIdentity, &ProjectDeclaration>,
699    budget: &mut ValueBudget<'_>,
700) -> Result<(), ProjectReadError> {
701    for decl in &ir.declarations {
702        match (&decl.signature, &decl.value) {
703            (ProjectPublicSignature::Binding(ty), Some(value)) if decl.defaults.is_empty() => {
704                check_type(ty, ir, index, 0)?;
705                check_value(value, ty, decl.public, false, ir, index, budget, 0)?;
706            }
707            (ProjectPublicSignature::Record(fields), None) => {
708                let mut previous = None;
709                for field in fields {
710                    if !is_snake_name(field.name())
711                        || is_protected_name(field.name())
712                        || previous.is_some_and(|old| old >= field.name())
713                    {
714                        return Err(ProjectReadError::Declaration);
715                    }
716                    previous = Some(field.name());
717                    check_type(field.ty(), ir, index, 0)?;
718                }
719                let mut prior = None;
720                for (name, value) in &decl.defaults {
721                    if prior.is_some_and(|old| old >= name) {
722                        return Err(ProjectReadError::Declaration);
723                    }
724                    prior = Some(name);
725                    let field = fields
726                        .iter()
727                        .find(|field| field.name() == name)
728                        .ok_or(ProjectReadError::Declaration)?;
729                    check_value(value, field.ty(), decl.public, true, ir, index, budget, 0)?;
730                }
731            }
732            _ => return Err(ProjectReadError::Declaration),
733        }
734    }
735    Ok(())
736}

738/// Validates bounded canonical type closure against complete nominal schemas.
739fn check_type(
740    ty: &ProjectPublicType,
741    ir: &ProjectIr,
742    index: &BTreeMap<&ModuleSymbolIdentity, &ProjectDeclaration>,
743    depth: usize,
744) -> Result<(), ProjectReadError> {
745    if depth > PROJECT_MAX_DEPTH {
746        return Err(ProjectReadError::Limit);
747    }
748    match ty {
749        ProjectPublicType::Nominal(identity) => {
750            if !index
751                .get(identity)
752                .is_some_and(|decl| matches!(decl.signature, ProjectPublicSignature::Record(_)))
753            {
754                return Err(ProjectReadError::Declaration);
755            }
756        }
757        ProjectPublicType::VocabularyNominal {
758            identity,
759            version,
760            name,
761        } => {
762            if !ir.vocabulary_records.iter().any(|record| {
763                record.identity == *identity
764                    && record.version == *version
765                    && record.name == *name
766                    && record.public
767            }) {
768                return Err(ProjectReadError::Declaration);
769            }
770        }
771        ProjectPublicType::List(inner) | ProjectPublicType::Ref(inner) => {
772            check_type(inner, ir, index, depth + 1)?;
773        }
774        ProjectPublicType::Nullable(inner) => {
775            if matches!(inner.as_ref(), ProjectPublicType::Nullable(_)) {
776                return Err(ProjectReadError::Declaration);
777            }
778            check_type(inner, ir, index, depth + 1)?;
779        }
780        _ => {}
781    }
782    Ok(())
783}

785/// Validates materialized contextual values and public reference visibility.
786#[expect(
787    clippy::too_many_arguments,
788    reason = "explicit validation context and budget prevent ambient or unbounded work"
789)]
790fn check_value(
791    value: &ProjectValue,
792    ty: &ProjectPublicType,
793    public: bool,
794    closed: bool,
795    ir: &ProjectIr,
796    index: &BTreeMap<&ModuleSymbolIdentity, &ProjectDeclaration>,
797    budget: &mut ValueBudget<'_>,
798    depth: usize,
799) -> Result<(), ProjectReadError> {
800    if depth > PROJECT_MAX_DEPTH {
801        return Err(ProjectReadError::Limit);
802    }
803    let ty = if let ProjectPublicType::Nullable(inner) = ty {
804        if matches!(value, ProjectValue::Null) {
805            budget.charge(0)?;
806            return Ok(());
807        }
808        inner.as_ref()
809    } else {
810        ty
811    };
812    budget.charge(match value {
813        ProjectValue::Number(number) => number.coefficient().len(),
814        ProjectValue::String(text) | ProjectValue::Url(text) | ProjectValue::Path(text) => {
815            text.len()
816        }
817        _ => 0,
818    })?;
819    match (value, ty) {
820        (ProjectValue::Number(number), ProjectPublicType::Num) => {
821            if number.scale().unsigned_abs() > budget.limits.text_bytes {
822                return Err(ProjectReadError::Limit);
823            }
824        }
825        (ProjectValue::String(_), ProjectPublicType::String)
826        | (ProjectValue::Bool(_), ProjectPublicType::Bool)
827        | (ProjectValue::Url(_), ProjectPublicType::Url)
828        | (ProjectValue::Path(_), ProjectPublicType::Path) => {}
829        (ProjectValue::List(values), ProjectPublicType::List(inner)) => {
830            for value in values {
831                check_value(value, inner, public, closed, ir, index, budget, depth + 1)?;
832            }
833        }
834        (ProjectValue::Reference(target), ProjectPublicType::Ref(inner)) if !closed => {
835            let decl = index.get(target).ok_or(ProjectReadError::Declaration)?;
836            if (public && !decl.public)
837                || decl.signature != ProjectPublicSignature::Binding(inner.as_ref().clone())
838            {
839                return Err(ProjectReadError::Declaration);
840            }
841        }
842        (ProjectValue::Record(values), ProjectPublicType::Nominal(identity)) => {
843            let ProjectPublicSignature::Record(fields) = &index[identity].signature else {
844                return Err(ProjectReadError::Declaration);
845            };
846            let fields = fields
847                .iter()
848                .map(|field| (field.name(), field.ty()))
849                .collect::<Vec<_>>();
850            check_record(
851                values,
852                &fields,
853                public,
854                closed,
855                ir,
856                index,
857                budget,
858                depth + 1,
859            )?;
860        }
861        (
862            ProjectValue::Record(values),
863            ProjectPublicType::VocabularyNominal {
864                identity,
865                version,
866                name,
867            },
868        ) => {
869            let record = ir
870                .vocabulary_records
871                .iter()
872                .find(|record| {
873                    record.identity == *identity
874                        && record.version == *version
875                        && record.name == *name
876                })
877                .ok_or(ProjectReadError::Declaration)?;
878            let fields = record
879                .fields
880                .iter()
881                .map(|(name, ty)| (name.as_str(), ty))
882                .collect::<Vec<_>>();
883            check_record(
884                values,
885                &fields,
886                public,
887                closed,
888                ir,
889                index,
890                budget,
891                depth + 1,
892            )?;
893        }
894        _ => return Err(ProjectReadError::Declaration),
895    }
896    Ok(())
897}

899/// Checks exact contextual field coverage, order, and compatibility.
900#[expect(
901    clippy::too_many_arguments,
902    reason = "shares the explicit bounded value-validation context"
903)]
904fn check_record(
905    values: &[(String, ProjectValue)],
906    fields: &[(&str, &ProjectPublicType)],
907    public: bool,
908    closed: bool,
909    ir: &ProjectIr,
910    index: &BTreeMap<&ModuleSymbolIdentity, &ProjectDeclaration>,
911    budget: &mut ValueBudget<'_>,
912    depth: usize,
913) -> Result<(), ProjectReadError> {
914    if values.len() != fields.len() {
915        return Err(ProjectReadError::Declaration);
916    }
917    for ((name, value), (field, ty)) in values.iter().zip(fields) {
918        if name != field {
919            return Err(ProjectReadError::Declaration);
920        }
921        check_value(value, ty, public, closed, ir, index, budget, depth)?;
922    }
923    Ok(())
924}

926/// Checks complete source-map coverage and bounded, non-dangling provenance.
927fn check_companions(
928    ir: &ProjectIr,
929    index: &BTreeMap<&ModuleSymbolIdentity, &ProjectDeclaration>,
930    limits: ProjectLimits,
931) -> Result<(), ProjectReadError> {
932    if ir.source_maps.len() != ir.declarations.len() {
933        return Err(ProjectReadError::Companion);
934    }
935    if ir.vocabulary_sources.len() != ir.public_interface.vocabularies().len() {
936        return Err(ProjectReadError::Companion);
937    }
938    for (source, vocabulary) in ir
939        .vocabulary_sources
940        .iter()
941        .zip(ir.public_interface.vocabularies())
942    {
943        if source.identity != vocabulary.identity()
944            || source.version != vocabulary.version()
945            || source.byte_len == 0
946            || source.byte_len > limits.text_bytes
947        {
948            return Err(ProjectReadError::Companion);
949        }
950    }
951    let mut ids = BTreeSet::new();
952    let mut lengths = BTreeMap::new();
953    for (source, module) in ir.sources.iter().zip(&ir.modules) {
954        if source.module != module.identity.module_name()
955            || source.source_id.is_empty()
956            || source.source_id.len() as u64 > limits.text_bytes
957            || !ids.insert(&source.source_id)
958        {
959            return Err(ProjectReadError::Companion);
960        }
961        lengths.insert(source.digest, source.byte_len);
962    }
963    for (map, decl) in ir.source_maps.iter().zip(&ir.declarations) {
964        let source = ir
965            .sources
966            .iter()
967            .find(|source| source.module == decl.identity.module().module_name())
968            .ok_or(ProjectReadError::Companion)?;
969        if map.declaration != decl.identity
970            || map.location.source() != source.digest
971            || map.location.span().end() > source.byte_len
972            || map.location.span().is_empty()
973        {
974            return Err(ProjectReadError::Companion);
975        }
976    }
977    let maps = ir
978        .source_maps
979        .iter()
980        .map(|map| (&map.declaration, map.location))
981        .collect::<BTreeMap<_, _>>();
982    let mut previous = None;
983    for edge in &ir.provenance {
984        let key = (&edge.from, edge.kind, &edge.to, edge.location);
985        let owner = maps.get(&edge.from).ok_or(ProjectReadError::Companion)?;
986        if !index.contains_key(&edge.from)
987            || !index.contains_key(&edge.to)
988            || previous.is_some_and(|prior| prior >= key)
989            || lengths
990                .get(&edge.location.source())
991                .is_none_or(|length| edge.location.span().end() > *length)
992            || edge.location.span().is_empty()
993            || owner.source() != edge.location.source()
994            || edge.location.span().start() < owner.span().start()
995            || edge.location.span().end() > owner.span().end()
996        {
997            return Err(ProjectReadError::Companion);
998        }
999        previous = Some(key);
1000    }
1001    if ir.resources.source_bytes > limits.text_bytes {
1002        return Err(ProjectReadError::Limit);
1003    }
1004    Ok(())
1005}
