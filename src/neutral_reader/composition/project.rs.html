30/// Safe whole-project failures; no supplied text, private names or host paths are retained.
31#[derive(Clone, Copy, Debug, Eq, PartialEq)]
32pub enum CompositionReadError {
33    /// Unknown successor selector.
34    Schema,
35    /// Independent work, count, byte, allocation or depth bound.
36    Limit,
37    /// Invalid module, declaration, type, value or semantic graph.
38    Semantic,
39    /// Mismatched public interpretive digest.
40    Interface,
41    /// Missing, malformed, dangling or inconsistent companions.
42    Companion,
43    /// Claimed retained counts differ from independent inspection.
44    Resources,
45    /// Cancellation prevented atomic publication.
46    Cancelled,
47}
48impl CompositionReadError {
49    /// Returns the successor result envelope, not a package release number.
50    #[must_use]
51    pub const fn schema(self) -> &'static str {
52        profile::PROJECT_RESULT_SCHEMA
53    }
54}
55/// Short complete-reader failure name used by private checks.
56type E = CompositionReadError;

58/// Immutable complete authority; construction requires independent revalidation of producer data.
59#[derive(Clone)]
60pub struct ValidatedCompositionProject {
61    /// Complete data; never exposed by public-only catalogue projections.
62    ir: Arc<CompositionProjectIr>,
63    /// Independently validated interpretation and actual reference paths.
64    bindings: Arc<ValidatedCompositionBindings>,
65    /// Intersected consumer acceptance policy, retained for later bounded projections.
66    effective_composition: CompositionLimits,
67}
68impl std::fmt::Debug for ValidatedCompositionProject {
69    /// Logs counts only, avoiding private values or identity keys.
70    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
71        f.debug_struct("ValidatedCompositionProject")
72            .field("modules", &self.ir.modules.len())
73            .field("declarations", &self.ir.declarations.len())
74            .finish_non_exhaustive()
75    }

77impl ValidatedCompositionProject {
78    /// Rechecks complete source/vocabulary semantics and companions without compiler linkage.
79    ///
80    /// # Errors
81    /// Rejects unknown profiles, noncanonical data, invalid defaults/references,
82    /// stale facts, companion defects, exhausted bounds or cancellation atomically.
83    pub fn from_ir(
84        ir: Arc<CompositionProjectIr>,
85        caller: ProjectLimits,
86        composition: CompositionLimits,
87        cancellation: &CancellationToken,
88    ) -> Result<Self, E> {
89        if cancellation.is_cancelled() {
90            return Err(E::Cancelled);
91        }
92        if ir.schema != profile::PROJECT_IR_SCHEMA {
93            return Err(E::Schema);
94        }
95        let limits = caller.intersect(ir.limits);
96        let composition = CompositionLimits::from_policy(
97            composition.json,
98            composition.policy().intersect(ir.composition_limits),
99        );
100        composition.validate().map_err(|e| error(&e))?;
101        if ir.resources.vocabulary_bytes > composition.captured_bytes {
102            return Err(E::Limit);
103        }
104        let facts = inspect_composition_ir(&ir, limits, composition.policy(), cancellation)
105            .map_err(|e| match e {
106                neutral_ir::composition::project::CompositionShapeError::Cancelled => E::Cancelled,
107                neutral_ir::composition::project::CompositionShapeError::Limit => E::Limit,
108            })?;
109        if facts != ir.composition_resources {
110            return Err(E::Resources);
111        }
112        check_comparison_work(&ir, composition.policy().work)?;
113        modules(&ir)?;
114        canonical_lookup_order(&ir)?;
115        let catalogue = Arc::try_new(
116            validate_composition_model(&ir.vocabularies, composition, cancellation)
117                .map_err(|e| error(&e))?,
118        )
119        .map_err(|_| E::Limit)?;
120        let (sources, mut raw) = declarations(&ir, cancellation)?;
121        let expected_sources = sources.try_clone().map_err(|_| E::Limit)?;
122        let scope = Arc::try_new(
123            validate_composition_scope(catalogue, sources, composition, cancellation)
124                .map_err(|e| error(&e))?,
125        )
126        .map_err(|_| E::Limit)?;
127        if scope.sources() != expected_sources {
128            return Err(E::Semantic);
129        }
130        let origin_index = index_origins(&ir, composition.work, cancellation)?;
131        for binding in &mut raw {
132            binding.value = supplied(
133                &binding.owner,
134                &binding.value,
135                &mut Vec::new(),
136                &origin_index,
137                cancellation,
138            )?;
139        }
140        let bindings = Arc::try_new(
141            validate_composition_bindings(scope, &raw, composition, cancellation)
142                .map_err(|e| error(&e))?,
143        )
144        .map_err(|_| E::Limit)?;
145        let mut origins = ir.origins.iter();
146        let mut attribution_work = composition.work;
147        for binding in bindings.bindings() {
148            let d = declaration(&ir, &binding.binding().owner)?;
149            if d.value.as_ref() != Some(&binding.binding().value) {
150                return Err(E::Semantic);
151            }
152            for expected in binding.origins() {
153                let actual = origins.next().ok_or(E::Companion)?;
154                if actual.binding != d.identity
155                    || actual.path != expected.path
156                    || actual.kind != expected.kind
157                {
158                    return Err(E::Companion);
159                }
160                attribution(actual, &ir, &mut attribution_work, cancellation)?;
161            }
162        }
163        if origins.next().is_some() {
164            return Err(E::Companion);
165        }
166        companions(&ir, &bindings, limits, cancellation)?;
167        check_interface(&ir, composition.work, cancellation)?;
168        if cancellation.is_cancelled() {
169            return Err(E::Cancelled);
170        }
171        Ok(Self {
172            ir,
173            bindings,
174            effective_composition: composition,
175        })
176    }
177    /// Returns immutable complete data for trusted consumers, not a public redacted view.
178    #[must_use]
179    pub const fn complete_ir(&self) -> &Arc<CompositionProjectIr> {
180        &self.ir
181    }
182    /// Returns independently checked contracts and binding-reference paths without source reparsing.
183    #[must_use]
184    pub const fn bindings(&self) -> &Arc<ValidatedCompositionBindings> {
185        &self.bindings
186    }
187    /// Keeps post-validation work within the independent consumer's policy.
188    pub(super) const fn effective_composition_limits(&self) -> CompositionLimits {
189        self.effective_composition
190    }
191}

193/// Checks producer ordering before any binary search can confer authority.
194fn canonical_lookup_order(ir: &CompositionProjectIr) -> Result<(), E> {
195    if ir
196        .declarations
197        .windows(2)
198        .any(|p| p[0].identity >= p[1].identity)
199        || ir
200            .source_maps
201            .windows(2)
202            .any(|p| p[0].declaration >= p[1].declaration)
203        || ir
204            .vocabulary_sources
205            .windows(2)
206            .any(|p| (&p[0].identity, &p[0].version) >= (&p[1].identity, &p[1].version))
207    {
208        return Err(E::Semantic);
209    }
210    Ok(())
211}

213/// Independently recomputes the interface identity under the intersected reader budget.
214fn check_interface(
215    ir: &CompositionProjectIr,
216    work: u64,
217    cancellation: &CancellationToken,
218) -> Result<(), E> {
219    let digest = composition_interface(
220        ir,
221        IdentityLimits {
222            // Transcript framing is not encoded artifact size; the two independent bounds cannot be conflated.
223            bytes: neutral_ir::project_identity::MAX_TRANSCRIPT_BYTES,
224            nodes: work,
225        },
226        cancellation,
227    )
228    .map_err(|e| match e {
229        neutral_ir::project_identity::IdentityError::Cancelled => E::Cancelled,
230        neutral_ir::project_identity::IdentityError::Limit => E::Limit,
231        neutral_ir::project_identity::IdentityError::InvalidInput => E::Interface,
232    })?
233    .identity();
234    if ir.interface_digest != digest {
235        return Err(E::Interface);
236    }
237    Ok(())
238}

240/// Checks canonical complete declaration categories before retaining any semantic clones.
241fn declarations(
242    ir: &CompositionProjectIr,
243    cancellation: &CancellationToken,
244) -> Result<(Vec<SourceCompositionDefinition>, Vec<CompositionBinding>), E> {
245    let mut sources = Vec::new();
246    let mut raw = Vec::new();
247    sources
248        .try_retain_exact(ir.declarations.len())
249        .map_err(|_| E::Limit)?;
250    raw.try_retain_exact(ir.declarations.len())
251        .map_err(|_| E::Limit)?;
252    for (i, d) in ir.declarations.iter().enumerate() {
253        if cancellation.is_cancelled() {
254            return Err(E::Cancelled);
255        }
256        if i > 0 && ir.declarations[i - 1].identity >= d.identity {
257            return Err(E::Semantic);
258        }
259        if ir
260            .modules
261            .binary_search_by(|m| m.identity.cmp(d.identity.module()))
262            .is_err()
263        {
264            return Err(E::Semantic);
265        }
266        match (&d.signature, &d.value) {
267            (S::Definition(body), None) => {
268                if let B::Record(fields) = body
269                    && fields
270                        .iter()
271                        .any(|f| f.restrictions != FieldRestrictions::default())
272                {
273                    return Err(E::Semantic);
274                }
275                sources.push(SourceCompositionDefinition {
276                    owner: d.identity.try_clone().map_err(|_| E::Limit)?,
277                    definition: CompositionDefinition {
278                        name: text(d.identity.declaration_name()).map_err(|_| E::Limit)?,
279                        public: d.public,
280                        body: body.try_clone().map_err(|_| E::Limit)?,
281                    },
282                });
283            }
284            (S::Binding(ty), Some(value)) => raw.push(CompositionBinding {
285                owner: d.identity.try_clone().map_err(|_| E::Limit)?,
286                public: d.public,
287                ty: ty.try_clone().map_err(|_| E::Limit)?,
288                value: value.try_clone().map_err(|_| E::Limit)?,
289            }),
290            _ => return Err(E::Semantic),
291        }
292    }
293    Ok((sources, raw))
294}

296/// Borrow canonical occurrences directly, avoiding a second allocated, shift-heavy index.
297type OriginIndex<'a> = &'a [neutral_ir::composition::project::CompositionOrigin];
298
299/// Validates ordered membership once before binary searches of untrusted occurrences.
300fn index_origins<'a>(
301    ir: &'a CompositionProjectIr,
302    work: u64,
303    cancellation: &CancellationToken,
304) -> Result<OriginIndex<'a>, E> {
305    if ir.origins.len() as u64 > work {
306        return Err(E::Limit);
307    }
308    let mut previous = None;
309    for origin in &ir.origins {
310        if cancellation.is_cancelled() {
311            return Err(E::Cancelled);
312        }
313        let key = (&origin.binding, origin.path.as_slice());
314        if previous.is_some_and(|p| p >= key) {
315            return Err(E::Companion);
316        }
317        previous = Some(key);
318    }
319    Ok(&ir.origins)
320}

322/// Looks up a checked canonical declaration without retaining another identity copy.
323fn declaration<'a>(
324    ir: &'a CompositionProjectIr,
325    owner: &ModuleSymbolIdentity,
326) -> Result<&'a neutral_ir::composition::project::CompositionDeclaration, E> {
327    ir.declarations
328        .binary_search_by(|d| d.identity.cmp(owner))
329        .map(|i| &ir.declarations[i])
330        .map_err(|_| E::Semantic)

344/// Converts shared contract failures into a source-independent, non-disclosing envelope.
345fn error(e: &CompositionError) -> E {
346    match e {
347        CompositionError::Cancelled => E::Cancelled,
348        CompositionError::Limit
349        | CompositionError::Allocation
350        | CompositionError::InvalidLimits => E::Limit,
351        _ => E::Semantic,
352    }
353}

355/// Checks canonical complete module topology without requiring source text or acquisition.
356fn modules(ir: &CompositionProjectIr) -> Result<(), E> {
357    if ir.modules.is_empty()
358        || ir.sources.len() != ir.modules.len()
359        || ir
360            .modules
361            .windows(2)
362            .any(|p| p[0].identity >= p[1].identity)
363    {
364        return Err(E::Semantic);
365    }
366    for (i, m) in ir.modules.iter().enumerate() {
367        if m.identity.language_behavior_version() != V1_SOURCE_PROFILE
368            || !m.identity.module_name().split("::").all(is_snake_name)
369            || i > 0 && ir.modules[i - 1].identity >= m.identity
370        {
371            return Err(E::Semantic);
372        }
373        for (j, target) in m.imports.iter().enumerate() {
374            if target == m.identity.module_name()
375                || j > 0 && m.imports[j - 1] >= *target
376                || ir
377                    .modules
378                    .binary_search_by(|m| m.identity.module_name().cmp(target))
379                    .is_err()
380            {
381                return Err(E::Semantic);
382            }
383        }
384    }
385    Ok(())
386}

388/// Reconstructs explicit input by pruning only claimed default/optional occurrences.
389/// Re-materialization below must reproduce both meaning and every classification.
390fn supplied(
391    owner: &ModuleSymbolIdentity,
392    value: &V,
393    path: &mut Vec<P>,
394    origins: &OriginIndex<'_>,
395    cancellation: &CancellationToken,
396) -> Result<V, E> {
397    if cancellation.is_cancelled() {
398        return Err(E::Cancelled);
399    }
400    if origins
401        .binary_search_by(|o| (&o.binding, o.path.as_slice()).cmp(&(owner, path.as_slice())))
402        .is_err()
403    {
404        return Err(E::Companion);
405    }
406    Ok(match value {
407        V::Record(fields) => {
408            if fields.windows(2).any(|f| f[0].0 >= f[1].0) {
409                return Err(E::Semantic);
410            }
411            let mut result = Vec::new();
412            result
413                .try_retain_exact(fields.len())
414                .map_err(|_| E::Limit)?;
415            for (name, value) in fields {
416                if cancellation.is_cancelled() {
417                    return Err(E::Cancelled);
418                }
419                path.try_retain(1).map_err(|_| E::Limit)?;
420                path.push(P::Field(text(name).map_err(|_| E::Limit)?));
421                let origin = &origins[origins
422                    .binary_search_by(|o| {
423                        (&o.binding, o.path.as_slice()).cmp(&(owner, path.as_slice()))
424                    })
425                    .map_err(|_| E::Companion)?];
426                if !matches!(origin.kind, K::Defaulted | K::OmittedOptional) {
427                    result.push((
428                        text(name).map_err(|_| E::Limit)?,
429                        Some(supplied(
430                            owner,
431                            value.as_ref().ok_or(E::Companion)?,
432                            path,
433                            origins,
434                            cancellation,
435                        )?),
436                    ));
437                }
438                path.pop();
439            }
440            V::Record(result)
441        }
442        V::List(values) => {
443            let mut result = Vec::new();
444            result
445                .try_retain_exact(values.len())
446                .map_err(|_| E::Limit)?;
447            for (i, v) in values.iter().enumerate() {
448                path.try_retain(1).map_err(|_| E::Limit)?;
449                path.push(P::Element(i as u64));
450                result.push(supplied(owner, v, path, origins, cancellation)?);
451                path.pop();
452            }
453            V::List(result)
454        }
455        V::Variant { tag, payload } => {
456            path.try_retain(1).map_err(|_| E::Limit)?;
457            path.push(P::Payload);
458            let payload = supplied(owner, payload, path, origins, cancellation)?;
459            path.pop();
460            V::Variant {
461                tag: text(tag).map_err(|_| E::Limit)?,
462                payload: boxed(payload).map_err(|_| E::Limit)?,
463            }
464        }
465        _ => value.try_clone().map_err(|_| E::Limit)?,
466    })
467}

469/// Checks supplied source evidence or exact closed-default vocabulary attribution.
470fn attribution(
471    origin: &neutral_ir::composition::project::CompositionOrigin,
472    ir: &CompositionProjectIr,
473    work: &mut u64,
474    cancel: &CancellationToken,
475) -> Result<(), E> {
476    match &origin.attribution {
477        None => Ok(()),
478        Some(A::Source(location)) => source_attribution(ir, origin, *location, work, cancel),
479        Some(A::Vocabulary {
480            identity,
481            version,
482            type_name,
483            field_name,
484            span,
485        }) => {
486            if origin.kind != K::Defaulted {
487                return Err(E::Companion);
488            }
489            let bundle = ir
490                .vocabularies
491                .binary_search_by(|b| {
492                    (b.identity.identity(), b.identity.version())
493                        .cmp(&(identity.as_str(), version.as_str()))
494                })
495                .ok()
496                .map(|i| &ir.vocabularies[i])
497                .ok_or(E::Companion)?;
498            let definition = bundle
499                .definitions
500                .binary_search_by(|d| d.name.cmp(type_name))
501                .ok()
502                .map(|i| &bundle.definitions[i])
503                .ok_or(E::Companion)?;
504            let B::Record(fields) = &definition.body else {
505                return Err(E::Companion);
506            };
507            if !fields
508                .iter()
509                .any(|f| f.name == *field_name && f.default.is_some())
510            {
511                return Err(E::Companion);
512            }
513            let source = ir
514                .vocabulary_sources
515                .binary_search_by(|s| (&s.identity, &s.version).cmp(&(identity, version)))
516                .ok()
517                .map(|i| &ir.vocabulary_sources[i])
518                .ok_or(E::Companion)?;
519            if span.is_some_and(|s| s.is_empty() || s.end() > source.byte_len) {
520                return Err(E::Companion);
521            }
522            // The owner must be on this binding's actual typed path, not merely exist elsewhere.
523            if !default_path_matches(ir, origin, (identity, version, type_name, field_name))? {
524                return Err(E::Companion);
525            }
526            Ok(())
527        }
528    }
529}

531/// Resolves a nominal body from complete interpretation facts, without aliases.
532fn body<'a>(ir: &'a CompositionProjectIr, ty: &T) -> Result<&'a B, E> {
533    match ty {
534        T::Nominal(owner) => match &declaration(ir, owner)?.signature {
535            S::Definition(body) => Ok(body),
536            S::Binding(_) => Err(E::Semantic),
537        },
538        T::VocabularyNominal {
539            identity,
540            version,
541            name,
542        } => ir
543            .vocabularies
544            .binary_search_by(|b| {
545                (b.identity.identity(), b.identity.version())
546                    .cmp(&(identity.as_str(), version.as_str()))
547            })
548            .ok()
549            .map(|i| &ir.vocabularies[i])
550            .and_then(|b| {
551                b.definitions
552                    .binary_search_by(|d| d.name.cmp(name))
553                    .ok()
554                    .map(|i| &b.definitions[i])
555            })
556            .map(|d| &d.body)
557            .ok_or(E::Semantic),
558        _ => Err(E::Semantic),
559    }
560}

562/// Binds a vocabulary default annotation to an actual traversed field prefix.
563fn default_path_matches(
564    ir: &CompositionProjectIr,
565    origin: &neutral_ir::composition::project::CompositionOrigin,
566    owner: (&str, &str, &str, &str),
567) -> Result<bool, E> {
568    default_path_check(
569        ir,
570        origin,
571        |ty, field| matches!(ty,T::VocabularyNominal {identity,version,name} if (identity.as_str(),version.as_str(),name.as_str(),field.name.as_str())==owner),
572    )
573}

575/// Checks actual typed field prefixes for source or vocabulary default ownership.
576fn default_path_check(
577    ir: &CompositionProjectIr,
578    origin: &neutral_ir::composition::project::CompositionOrigin,
579    matches: impl Fn(&T, &neutral_ir::composition::CompositionField) -> bool,
580) -> Result<bool, E> {
581    let declaration = declaration(ir, &origin.binding)?;
582    let S::Binding(initial) = &declaration.signature else {
583        return Err(E::Semantic);
584    };
585    let mut ty = initial;
586    let mut value = declaration.value.as_ref();
587    for segment in &origin.path {
588        while let T::Nullable(inner) = ty {
589            ty = inner;
590        }
591        match segment {
592            P::Field(name) => {
593                let B::Record(fields) = body(ir, ty)? else {
594                    return Err(E::Companion);
595                };
596                let field = fields
597                    .iter()
598                    .find(|f| f.name == *name)
599                    .ok_or(E::Companion)?;
600                if matches(ty, field) && field.default.is_some() {
601                    return Ok(true);
602                }
603                ty = &field.ty;
604                value = match value {
605                    Some(V::Record(fields)) => fields
606                        .iter()
607                        .find(|(n, _)| n == name)
608                        .and_then(|(_, v)| v.as_ref()),
609                    _ => None,
610                };
611            }
612            P::Element(index) => {
613                let T::List(inner) = ty else {
614                    return Err(E::Companion);
615                };
616                ty = inner;
617                value = match value {
618                    Some(V::List(values)) => {
619                        usize::try_from(*index).ok().and_then(|i| values.get(i))
620                    }
621                    _ => None,
622                };
623            }
624            P::Payload => {
625                let B::Variant(alternatives) = body(ir, ty)? else {
626                    return Err(E::Companion);
627                };
628                let Some(V::Variant { tag, payload }) = value else {
629                    return Err(E::Companion);
630                };
631                ty = &alternatives
632                    .iter()
633                    .find(|a| a.tag == *tag)
634                    .ok_or(E::Companion)?
635                    .ty;
636                value = Some(payload);
637            }
638        }
639    }
640    Ok(false)
641}

643/// Checks a source occurrence against its consuming declaration, exact default owner or real reuse closure.
644fn source_attribution(
645    ir: &CompositionProjectIr,
646    origin: &neutral_ir::composition::project::CompositionOrigin,
647    location: neutral_core::SourceLocation,
648    work: &mut u64,
649    cancel: &CancellationToken,
650) -> Result<(), E> {
651    if location.span().is_empty() {
652        return Err(E::Companion);
653    }
654    let contains = |map: &neutral_ir::project::ProjectSourceMap| {
655        map.location.source() == location.source()
656            && location.span().start() >= map.location.span().start()
657            && location.span().end() <= map.location.span().end()
658    };
659    if source_map(ir, &origin.binding).is_some_and(contains) {
660        return Ok(());
661    }
662    if default_path_check(
663        ir,
664        origin,
665        |ty, _| matches!(ty, T::Nominal(owner) if source_map(ir, owner).is_some_and(contains)),
666    )? {
667        return Ok(());
668    }
669    let mut seen = BTreeSet::new();
670    let mut pending = Vec::new();
671    pending.try_retain(1).map_err(|_| E::Limit)?;
672    pending.push(&origin.binding);
673    while let Some(owner) = pending.pop() {
674        if cancel.is_cancelled() {
675            return Err(E::Cancelled);
676        }
677        *work = work
678            .checked_sub(ir.provenance.len() as u64 + ir.declarations.len() as u64 + 1)
679            .ok_or(E::Limit)?;
680        charge_index(work, seen.len(), cancel)?;
681        if !seen.insert(owner).map_err(|_| E::Limit)? {
682            continue;
683        }
684        if source_map(ir, owner).is_some_and(contains) {
685            return Ok(());
686        }
687        for edge in ir
688            .provenance
689            .iter()
690            .filter(|e| e.from == *owner && e.kind == Edge::Value)
691        {
692            pending.try_retain(1).map_err(|_| E::Limit)?;
693            pending.push(&edge.to);
694        }
695    }
696    Err(E::Companion)
697}

699/// Validates source coverage, dependency provenance and independently recomputed base facts.
700fn companions(
701    ir: &CompositionProjectIr,
702    bindings: &ValidatedCompositionBindings,
703    limits: ProjectLimits,
704    cancel: &CancellationToken,
705) -> Result<(), E> {
706    let mut work = ir.composition_limits.work;
707    source_companions(ir, &mut work, cancel)?;
708    let mut actual = BTreeSet::new();
709    let mut previous = None;
710    for edge in &ir.provenance {
711        if cancel.is_cancelled() {
712            return Err(E::Cancelled);
713        }
714        let key = (&edge.from, edge.kind, &edge.to, edge.location);
715        let map = source_map(ir, &edge.from).ok_or(E::Companion)?;
716        if previous.is_some_and(|p| p >= key)
717            || declaration(ir, &edge.to).is_err()
718            || edge.location.source() != map.location.source()
719            || edge.location.span().is_empty()
720            || edge.location.span().start() < map.location.span().start()
721            || edge.location.span().end() > map.location.span().end()
722        {
723            return Err(E::Companion);
724        }
725        previous = Some(key);
726        charge_index(&mut work, actual.len(), cancel)?;
727        actual
728            .insert((&edge.from, edge.kind, &edge.to))
729            .map_err(|_| E::Limit)?;
730    }
731    let mut expected = BTreeSet::new();
732    for d in &ir.declarations {
733        match &d.signature {
734            S::Binding(ty) => type_edges(&d.identity, ty, false, &mut expected, &mut work, cancel)?,
735            S::Definition(B::Record(fields)) => {
736                for f in fields {
737                    type_edges(&d.identity, &f.ty, false, &mut expected, &mut work, cancel)?;
738                }
739            }
740            S::Definition(B::Variant(alts)) => {
741                for a in alts {
742                    type_edges(&d.identity, &a.ty, false, &mut expected, &mut work, cancel)?;
743                }
744            }
745        }
746    }
747    for b in bindings.bindings() {
748        for r in b.references() {
749            charge_index(&mut work, expected.len(), cancel)?;
750            expected
751                .insert((&b.binding().owner, Edge::Reference, &r.target))
752                .map_err(|_| E::Limit)?;
753        }
754    }
755    if expected
756        .iter()
757        .any(|(from, kind, to)| !actual.contains(&(from, *kind, to)))
758    {
759        return Err(E::Companion);
760    }
761    for (from, kind, to) in &actual {
762        if *kind != Edge::Value && !expected.contains(&(from, *kind, to)) {
763            return Err(E::Companion);
764        }
765        if from.module() != to.module()
766            && (!declaration(ir, to)?.public
767                || !reachable(
768                    ir,
769                    from.module().module_name(),
770                    to.module().module_name(),
771                    &mut work,
772                    cancel,
773                )?)
774        {
775            return Err(E::Semantic);
776        }
777        if *kind == Edge::Value {
778            let left = declaration(ir, from)?;
779            let right = declaration(ir, to)?;
780            let (S::Binding(a), S::Binding(b), Some(value), Some(target)) =
781                (&left.signature, &right.signature, &left.value, &right.value)
782            else {
783                return Err(E::Semantic);
784            };
785            if !contains_reuse(ir, value, a, target, b)? {
786                return Err(E::Semantic);
787            }
788        }
789    }
790    // Ordinary reuse evaluates; identity references deliberately do not enter this cycle check.
791    check_reuse_cycles(&actual, &mut work, cancel)?;
792    resource_facts(ir, limits)
793}

795/// Recomputes aggregate captured and retained counts instead of trusting advertised facts.
796fn resource_facts(ir: &CompositionProjectIr, limits: ProjectLimits) -> Result<(), E> {
797    let facts = ProjectResourceFacts {
798        source_units: ir.sources.len() as u64,
799        source_bytes: ir
800            .sources
801            .iter()
802            .map(|s| s.byte_len)
803            .try_fold(0_u64, u64::checked_add)
804            .ok_or(E::Limit)?,
805        vocabulary_units: ir.vocabulary_sources.len() as u64,
806        vocabulary_bytes: ir
807            .vocabulary_sources
808            .iter()
809            .map(|s| s.byte_len)
810            .try_fold(0_u64, u64::checked_add)
811            .ok_or(E::Limit)?,
812        declarations: ir.declarations.len() as u64,
813        import_edges: ir
814            .modules
815            .iter()
816            .map(|m| m.imports.len() as u64)
817            .try_fold(0_u64, u64::checked_add)
818            .ok_or(E::Limit)?,
819        value_nodes: ir.composition_resources.retained_value_nodes,
820    };
821    if facts != ir.resources {
822        return Err(E::Resources);
823    }
824    if facts.source_bytes > limits.text_bytes
825        || facts.vocabulary_bytes > ir.composition_limits.captured_bytes
826    {
827        return Err(E::Limit);
828    }
829    Ok(())
830}

832/// Checks source accounting, original-byte coverage and exact vocabulary evidence ownership.
833fn source_companions(
834    ir: &CompositionProjectIr,
835    work: &mut u64,
836    cancel: &CancellationToken,
837) -> Result<(), E> {
838    if ir.source_maps.len() != ir.declarations.len()
839        || ir.vocabulary_sources.len() != ir.vocabularies.len()
840    {
841        return Err(E::Companion);
842    }
843    let mut ids = BTreeSet::new();
844    for (s, m) in ir.sources.iter().zip(&ir.modules) {
845        charge_index(work, ids.len(), cancel)?;
846        if s.module != m.identity.module_name()
847            || s.source_id.is_empty()
848            || !ids.insert(&s.source_id).map_err(|_| E::Limit)?
849            || s.byte_len == 0
850        {
851            return Err(E::Companion);
852        }
853    }
854    for (s, b) in ir.vocabulary_sources.iter().zip(&ir.vocabularies) {
855        if s.identity != b.identity.identity()
856            || s.version != b.identity.version()
857            || s.digest != b.identity.content_digest()
858            || s.byte_len == 0
859        {
860            return Err(E::Companion);
861        }
862    }
863    for (map, d) in ir.source_maps.iter().zip(&ir.declarations) {
864        let s = &ir.sources[ir
865            .sources
866            .binary_search_by(|s| s.module.as_str().cmp(d.identity.module().module_name()))
867            .map_err(|_| E::Companion)?];
868        if map.declaration != d.identity
869            || map.location.source() != s.digest
870            || map.location.span().is_empty()
871            || map.location.span().end() > s.byte_len
872        {
873            return Err(E::Companion);
874        }
875    }
876    Ok(())
877}

879/// Bounds binary lookups and remaining type/reuse scans before processing producer-owned data.
880fn check_comparison_work(ir: &CompositionProjectIr, limit: u64) -> Result<(), E> {
881    let declarations = ir.declarations.len() as u64;
882    let modules = ir.modules.len() as u64;
883    let facts = ir.composition_resources;
884    let lookup_depth = u64::from(usize::BITS - ir.declarations.len().leading_zeros())
885        + u64::from(usize::BITS - ir.origins.len().leading_zeros());
886    let size = lookup_depth
887        .checked_add(modules)
888        .and_then(|n| n.checked_add(facts.types))
889        .and_then(|n| n.checked_add(facts.fields))
890        .and_then(|n| n.checked_add(facts.alternatives))
891        .and_then(|n| n.checked_add(1))
892        .ok_or(E::Limit)?;
893    let paths = ir
894        .origins
895        .iter()
896        .try_fold(0_u64, |n, o| n.checked_add(o.path.len() as u64 + 1))
897        .ok_or(E::Limit)?;
898    let scans = paths
899        .checked_add(declarations)
900        .and_then(|n| n.checked_mul(size))
901        .and_then(|n| {
902            (ir.provenance.len() as u64)
903                .checked_mul(facts.retained_value_nodes.checked_add(size)?)
904                .and_then(|reuse| n.checked_add(reuse))
905        })
906        .ok_or(E::Limit)?;
907    if scans > limit {
908        return Err(E::Limit);
909    }
910    Ok(())
911}

913/// Proves a reuse edge points to an invariant typed materialized occurrence, including nested fields/lists.
914fn contains_reuse(
915    ir: &CompositionProjectIr,
916    value: &V,
917    ty: &T,
918    target: &V,
919    target_ty: &T,
920) -> Result<bool, E> {
921    if ty == target_ty && value == target {
922        return Ok(true);
923    }
924    let ty = if let T::Nullable(inner) = ty {
925        inner.as_ref()
926    } else {
927        ty
928    };
929    match (ty, value) {
930        (T::List(inner), V::List(values)) => {
931            for v in values {
932                if contains_reuse(ir, v, inner, target, target_ty)? {
933                    return Ok(true);
934                }
935            }
936        }
937        (T::Nominal(_) | T::VocabularyNominal { .. }, V::Record(values)) => {
938            let B::Record(fields) = body(ir, ty)? else {
939                return Err(E::Semantic);
940            };
941            for ((name, value), field) in values.iter().zip(fields) {
942                if name != &field.name {
943                    return Err(E::Semantic);
944                }
945                if let Some(v) = value
946                    && contains_reuse(ir, v, &field.ty, target, target_ty)?
947                {
948                    return Ok(true);
949                }
950            }
951        }
952        (T::Nominal(_) | T::VocabularyNominal { .. }, V::Variant { tag, payload }) => {
953            let B::Variant(alternatives) = body(ir, ty)? else {
954                return Err(E::Semantic);
955            };
956            let alternative = alternatives
957                .iter()
958                .find(|a| a.tag == *tag)
959                .ok_or(E::Semantic)?;
960            return contains_reuse(ir, payload, &alternative.ty, target, target_ty);
961        }
962        _ => {}
963    }
964    Ok(false)
965}

967/// Checks ordinary value dependencies in linear graph work; identity-only references may cycle.
968fn check_reuse_cycles(
969    edges: &BTreeSet<(&ModuleSymbolIdentity, Edge, &ModuleSymbolIdentity)>,
970    work: &mut u64,
971    cancel: &CancellationToken,
972) -> Result<(), E> {
973    let mut incoming = BTreeMap::new();
974    let mut outgoing: BTreeMap<_, Vec<_>> = BTreeMap::new();
975    for (from, kind, to) in edges {
976        if *kind == Edge::Value {
977            charge_index(work, incoming.len() * 2 + outgoing.len(), cancel)?;
978            incoming
979                .entry(*from)
980                .map_err(|_| E::Limit)?
981                .or_insert(0_usize);
982            *incoming.entry(*to).map_err(|_| E::Limit)?.or_insert(0) += 1;
983            let targets = outgoing.entry(*from).map_err(|_| E::Limit)?.or_default();
984            targets.try_retain(1).map_err(|_| E::Limit)?;
985            targets.push(*to);
986        }
987    }
988    let mut ready = Vec::new();
989    ready.try_retain(incoming.len()).map_err(|_| E::Limit)?;
990    for (key, count) in &incoming {
991        if *count == 0 {
992            ready.push(*key);
993        }
994    }
995    let mut count = 0;
996    while let Some(node) = ready.pop() {
997        if cancel.is_cancelled() {
998            return Err(E::Cancelled);
999        }
1000        count += 1;
1001        if let Some(targets) = outgoing.get(node) {
1002            for target in targets {
1003                let n = incoming.get_mut(target).ok_or(E::Semantic)?;
1004                *n -= 1;
1005                if *n == 0 {
1006                    ready.push(*target);
1007                }
1008            }
1009        }
1010    }
1011    if count != incoming.len() {
1012        return Err(E::Semantic);
1013    }
1014    Ok(())
1015}

1017/// Enumerates every source nominal dependency, including unselected alternatives.
1018fn type_edges<'a>(
1019    from: &'a ModuleSymbolIdentity,
1020    ty: &'a T,
1021    reference: bool,
1022    edges: &mut BTreeSet<(&'a ModuleSymbolIdentity, Edge, &'a ModuleSymbolIdentity)>,
1023    work: &mut u64,
1024    cancel: &CancellationToken,
1025) -> Result<(), E> {
1026    charge_index(work, edges.len(), cancel)?;
1027    match ty {
1028        T::Nominal(to) => {
1029            edges
1030                .insert((
1031                    from,
1032                    if reference {
1033                        Edge::ReferenceType
1034                    } else {
1035                        Edge::Type
1036                    },
1037                    to,
1038                ))
1039                .map_err(|_| E::Limit)?;
1040        }
1041        T::Ref(inner) => type_edges(from, inner, true, edges, work, cancel)?,
1042        T::List(inner) | T::Nullable(inner) => {
1043            type_edges(from, inner, reference, edges, work, cancel)?;
1044        }
1045        _ => {}
1046    }
1047    Ok(())
1048}

1050/// Checks logical reachability without acquiring imports or interpreting host locations.
1051fn reachable(
1052    ir: &CompositionProjectIr,
1053    from: &str,
1054    to: &str,
1055    work: &mut u64,
1056    cancel: &CancellationToken,
1057) -> Result<bool, E> {
1058    let mut stack = Vec::new();
1059    stack.try_retain(1).map_err(|_| E::Limit)?;
1060    stack.push(from);
1061    let mut seen = BTreeSet::new();
1062    while let Some(name) = stack.pop() {
1063        if cancel.is_cancelled() {
1064            return Err(E::Cancelled);
1065        }
1066        if name == to {
1067            return Ok(true);
1068        }
1069        charge_index(work, seen.len() + ir.modules.len(), cancel)?;
1070        if !seen.insert(name).map_err(|_| E::Limit)? {
1071            continue;
1072        }
1073        let module = &ir.modules[ir
1074            .modules
1075            .binary_search_by(|m| m.identity.module_name().cmp(name))
1076            .map_err(|_| E::Semantic)?];
1077        stack
1078            .try_retain(module.imports.len())
1079            .map_err(|_| E::Limit)?;
1080        stack.extend(module.imports.iter().map(String::as_str));
1081    }
1082    Ok(false)
1083}
1084/// Charges sorted-index movement before reservation or expansion and checks cancellation.
1085fn charge_index(work: &mut u64, count: usize, cancel: &CancellationToken) -> Result<(), E> {
1086    if cancel.is_cancelled() {
1087        return Err(E::Cancelled);
1088    }
1089    *work = work.checked_sub(count as u64 + 1).ok_or(E::Limit)?;
1090    Ok(())
