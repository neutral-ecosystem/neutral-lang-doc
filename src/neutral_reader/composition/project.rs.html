30/// Safe whole-project failures; no supplied text, private names or host paths are retained.
31#[derive(Clone, Copy, Debug, Eq, PartialEq)]
32pub enum CompositionReadError {
33    /// Unknown successor selector.
34    Schema,
35    /// Independent work, count, byte, allocation or depth bound.
36    Limit,
37    /// Invalid module, declaration, type, value or semantic graph.
38    Semantic,
39    /// Mismatched public interpretive digest.
40    Interface,
41    /// Missing, malformed, dangling or inconsistent companions.
42    Companion,
43    /// Claimed retained counts differ from independent inspection.
44    Resources,
45    /// Cancellation prevented atomic publication.
46    Cancelled,
47}
48impl CompositionReadError {
49    /// Returns the successor result envelope, not a package release number.
50    #[must_use]
51    pub const fn schema(self) -> &'static str {
52        profile::PROJECT_RESULT_SCHEMA
53    }
54}
55/// Short complete-reader failure name used by private checks.
56type E = CompositionReadError;

58/// Immutable complete authority; construction requires independent revalidation of producer data.
59#[derive(Clone)]
60pub struct ValidatedCompositionProject {
61    /// Complete data; never exposed by public-only catalogue projections.
62    ir: Arc<CompositionProjectIr>,
63    /// Independently validated interpretation and actual reference paths.
64    bindings: Arc<ValidatedCompositionBindings>,
65    /// Intersected consumer acceptance policy, retained for later bounded projections.
66    effective_composition: CompositionLimits,
67}
68impl std::fmt::Debug for ValidatedCompositionProject {
69    /// Logs counts only, avoiding private values or identity keys.
70    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
71        f.debug_struct("ValidatedCompositionProject")
72            .field("modules", &self.ir.modules.len())
73            .field("declarations", &self.ir.declarations.len())
74            .finish_non_exhaustive()
75    }
76}
77impl ValidatedCompositionProject {
78    /// Rechecks complete source/vocabulary semantics and companions without compiler linkage.
79    ///
80    /// # Errors
81    /// Rejects unknown profiles, noncanonical data, invalid defaults/references,
82    /// stale facts, companion defects, exhausted bounds or cancellation atomically.
83    pub fn from_ir(
84        ir: Arc<CompositionProjectIr>,
85        caller: ProjectLimits,
86        composition: CompositionLimits,
87        cancellation: &CancellationToken,
88    ) -> Result<Self, E> {
89        if cancellation.is_cancelled() {
90            return Err(E::Cancelled);
91        }
92        if ir.schema != profile::PROJECT_IR_SCHEMA {
93            return Err(E::Schema);
94        }
95        let limits = caller.intersect(ir.limits);
96        let composition = CompositionLimits::from_policy(
97            composition.json,
98            composition.policy().intersect(ir.composition_limits),
99        );
100        composition.validate().map_err(|e| error(&e))?;
101        if ir.resources.vocabulary_bytes > composition.captured_bytes {
102            return Err(E::Limit);
103        }
104        let facts = inspect_composition_ir(&ir, limits, composition.policy(), cancellation)
105            .map_err(|e| match e {
106                neutral_ir::composition::project::CompositionShapeError::Cancelled => E::Cancelled,
107                neutral_ir::composition::project::CompositionShapeError::Limit => E::Limit,
108            })?;
109        if facts != ir.composition_resources {
110            return Err(E::Resources);
111        }
112        check_comparison_work(&ir, composition.policy().work)?;
113        modules(&ir)?;
114        let catalogue = Arc::try_new(
115            validate_composition_model(&ir.vocabularies, composition, cancellation)
116                .map_err(|e| error(&e))?,
117        )
118        .map_err(|_| E::Limit)?;
119        let (sources, mut raw) = declarations(&ir, cancellation)?;
120        let expected_sources = sources.try_clone().map_err(|_| E::Limit)?;
121        let scope = Arc::try_new(
122            validate_composition_scope(catalogue, sources, composition, cancellation)
123                .map_err(|e| error(&e))?,
124        )
125        .map_err(|_| E::Limit)?;
126        if scope.sources() != expected_sources {
127            return Err(E::Semantic);
128        }
129        let origin_index = index_origins(&ir, composition.work, cancellation)?;
130        for binding in &mut raw {
131            binding.value = supplied(
132                &binding.owner,
133                &binding.value,
134                &mut Vec::new(),
135                &origin_index,
136                cancellation,
137            )?;
138        }
139        let bindings = Arc::try_new(
140            validate_composition_bindings(scope, &raw, composition, cancellation)
141                .map_err(|e| error(&e))?,
142        )
143        .map_err(|_| E::Limit)?;
144        let mut origins = ir.origins.iter();
145        let mut attribution_work = composition.work;
146        for binding in bindings.bindings() {
147            let d = ir
148                .declarations
149                .iter()
150                .find(|d| d.identity == binding.binding().owner)
151                .ok_or(E::Semantic)?;
152            if d.value.as_ref() != Some(&binding.binding().value) {
153                return Err(E::Semantic);
154            }
155            for expected in binding.origins() {
156                let actual = origins.next().ok_or(E::Companion)?;
157                if actual.binding != d.identity
158                    || actual.path != expected.path
159                    || actual.kind != expected.kind
160                {
161                    return Err(E::Companion);
162                }
163                attribution(actual, &ir, &mut attribution_work, cancellation)?;
164            }
165        }
166        if origins.next().is_some() {
167            return Err(E::Companion);
168        }
169        companions(&ir, &bindings, limits, cancellation)?;
170        check_interface(&ir, composition.work, cancellation)?;
171        if cancellation.is_cancelled() {
172            return Err(E::Cancelled);
173        }
174        Ok(Self {
175            ir,
176            bindings,
177            effective_composition: composition,
178        })
179    }
180    /// Returns immutable complete data for trusted consumers, not a public redacted view.
181    #[must_use]
182    pub const fn complete_ir(&self) -> &Arc<CompositionProjectIr> {
183        &self.ir
184    }
185    /// Returns independently checked contracts and binding-reference paths without source reparsing.
186    #[must_use]
187    pub const fn bindings(&self) -> &Arc<ValidatedCompositionBindings> {
188        &self.bindings
189    }
190    /// Keeps post-validation work within the independent consumer's policy.
191    pub(super) const fn effective_composition_limits(&self) -> CompositionLimits {
192        self.effective_composition
193    }
194}

196/// Independently recomputes the interface identity under the intersected reader budget.
197fn check_interface(
198    ir: &CompositionProjectIr,
199    work: u64,
200    cancellation: &CancellationToken,
201) -> Result<(), E> {
202    let digest = composition_interface(
203        ir,
204        IdentityLimits {
205            // Transcript framing is not encoded artifact size; the two independent bounds cannot be conflated.
206            bytes: neutral_ir::project_identity::MAX_TRANSCRIPT_BYTES,
207            nodes: work,
208        },
209        cancellation,
210    )
211    .map_err(|e| match e {
212        neutral_ir::project_identity::IdentityError::Cancelled => E::Cancelled,
213        neutral_ir::project_identity::IdentityError::Limit => E::Limit,
214        neutral_ir::project_identity::IdentityError::InvalidInput => E::Interface,
215    })?
216    .identity();
217    if ir.interface_digest != digest {
218        return Err(E::Interface);
219    }
220    Ok(())
221}

223/// Checks canonical complete declaration categories before retaining any semantic clones.
224fn declarations(
225    ir: &CompositionProjectIr,
226    cancellation: &CancellationToken,
227) -> Result<(Vec<SourceCompositionDefinition>, Vec<CompositionBinding>), E> {
228    let mut sources = Vec::new();
229    let mut raw = Vec::new();
230    sources
231        .try_retain_exact(ir.declarations.len())
232        .map_err(|_| E::Limit)?;
233    raw.try_retain_exact(ir.declarations.len())
234        .map_err(|_| E::Limit)?;
235    for (i, d) in ir.declarations.iter().enumerate() {
236        if cancellation.is_cancelled() {
237            return Err(E::Cancelled);
238        }
239        if i > 0 && ir.declarations[i - 1].identity >= d.identity {
240            return Err(E::Semantic);
241        }
242        if !ir
243            .modules
244            .iter()
245            .any(|m| m.identity == *d.identity.module())
246        {
247            return Err(E::Semantic);
248        }
249        match (&d.signature, &d.value) {
250            (S::Definition(body), None) => {
251                if let B::Record(fields) = body
252                    && fields
253                        .iter()
254                        .any(|f| f.restrictions != FieldRestrictions::default())
255                {
256                    return Err(E::Semantic);
257                }
258                sources.push(SourceCompositionDefinition {
259                    owner: d.identity.try_clone().map_err(|_| E::Limit)?,
260                    definition: CompositionDefinition {
261                        name: text(d.identity.declaration_name()).map_err(|_| E::Limit)?,
262                        public: d.public,
263                        body: body.try_clone().map_err(|_| E::Limit)?,
264                    },
265                });
266            }
267            (S::Binding(ty), Some(value)) => raw.push(CompositionBinding {
268                owner: d.identity.try_clone().map_err(|_| E::Limit)?,
269                public: d.public,
270                ty: ty.try_clone().map_err(|_| E::Limit)?,
271                value: value.try_clone().map_err(|_| E::Limit)?,
272            }),
273            _ => return Err(E::Semantic),
274        }
275    }
276    Ok((sources, raw))
277}

279/// Exact borrowed origin occurrences with fallibly reserved ordered membership.
280type OriginIndex<'a> = neutral_core::ordered::OrderedMap<
281    (&'a ModuleSymbolIdentity, &'a [P]),
282    &'a neutral_ir::composition::project::CompositionOrigin,
283>;
284
285/// Charges index shifts and observes cancellation before retaining companion membership.
286fn index_origins<'a>(
287    ir: &'a CompositionProjectIr,
288    mut work: u64,
289    cancellation: &CancellationToken,
290) -> Result<OriginIndex<'a>, E> {
291    let mut index = OriginIndex::new();
292    for origin in &ir.origins {
293        if cancellation.is_cancelled() {
294            return Err(E::Cancelled);
295        }
296        work = work.checked_sub(index.len() as u64).ok_or(E::Limit)?;
297        if index
298            .insert((&origin.binding, origin.path.as_slice()), origin)
299            .map_err(|_| E::Limit)?
300            .is_some()
301        {
302            return Err(E::Companion);
303        }
304    }
305    Ok(index)
306}

308/// Converts shared contract failures into a source-independent, non-disclosing envelope.
309fn error(e: &CompositionError) -> E {
310    match e {
311        CompositionError::Cancelled => E::Cancelled,
312        CompositionError::Limit
313        | CompositionError::Allocation
314        | CompositionError::InvalidLimits => E::Limit,
315        _ => E::Semantic,
316    }
317}

319/// Checks canonical complete module topology without requiring source text or acquisition.
320fn modules(ir: &CompositionProjectIr) -> Result<(), E> {
321    if ir.modules.is_empty() || ir.sources.len() != ir.modules.len() {
322        return Err(E::Semantic);
323    }
324    for (i, m) in ir.modules.iter().enumerate() {
325        if m.identity.language_behavior_version() != V1_SOURCE_PROFILE
326            || !m.identity.module_name().split("::").all(is_snake_name)
327            || i > 0 && ir.modules[i - 1].identity >= m.identity
328        {
329            return Err(E::Semantic);
330        }
331        for (j, target) in m.imports.iter().enumerate() {
332            if target == m.identity.module_name()
333                || j > 0 && m.imports[j - 1] >= *target
334                || !ir
335                    .modules
336                    .iter()
337                    .any(|m| m.identity.module_name() == target)
338            {
339                return Err(E::Semantic);
340            }
341        }
342    }
343    Ok(())
344}

346/// Reconstructs explicit input by pruning only claimed default/optional occurrences.
347/// Re-materialization below must reproduce both meaning and every classification.
348fn supplied(
349    owner: &ModuleSymbolIdentity,
350    value: &V,
351    path: &mut Vec<P>,
352    origins: &OriginIndex<'_>,
353    cancellation: &CancellationToken,
354) -> Result<V, E> {
355    if cancellation.is_cancelled() {
356        return Err(E::Cancelled);
357    }
358    if !origins.contains_key(&(owner, path.as_slice())) {
359        return Err(E::Companion);
360    }
361    Ok(match value {
362        V::Record(fields) => {
363            if fields.windows(2).any(|f| f[0].0 >= f[1].0) {
364                return Err(E::Semantic);
365            }
366            let mut result = Vec::new();
367            result
368                .try_retain_exact(fields.len())
369                .map_err(|_| E::Limit)?;
370            for (name, value) in fields {
371                if cancellation.is_cancelled() {
372                    return Err(E::Cancelled);
373                }
374                path.try_retain(1).map_err(|_| E::Limit)?;
375                path.push(P::Field(text(name).map_err(|_| E::Limit)?));
376                let origin = origins.get(&(owner, path.as_slice())).ok_or(E::Companion)?;
377                if !matches!(origin.kind, K::Defaulted | K::OmittedOptional) {
378                    result.push((
379                        text(name).map_err(|_| E::Limit)?,
380                        Some(supplied(
381                            owner,
382                            value.as_ref().ok_or(E::Companion)?,
383                            path,
384                            origins,
385                            cancellation,
386                        )?),
387                    ));
388                }
389                path.pop();
390            }
391            V::Record(result)
392        }
393        V::List(values) => {
394            let mut result = Vec::new();
395            result
396                .try_retain_exact(values.len())
397                .map_err(|_| E::Limit)?;
398            for (i, v) in values.iter().enumerate() {
399                path.try_retain(1).map_err(|_| E::Limit)?;
400                path.push(P::Element(i as u64));
401                result.push(supplied(owner, v, path, origins, cancellation)?);
402                path.pop();
403            }
404            V::List(result)
405        }
406        V::Variant { tag, payload } => {
407            path.try_retain(1).map_err(|_| E::Limit)?;
408            path.push(P::Payload);
409            let payload = supplied(owner, payload, path, origins, cancellation)?;
410            path.pop();
411            V::Variant {
412                tag: text(tag).map_err(|_| E::Limit)?,
413                payload: boxed(payload).map_err(|_| E::Limit)?,
414            }
415        }
416        _ => value.try_clone().map_err(|_| E::Limit)?,
417    })
418}

420/// Checks supplied source evidence or exact closed-default vocabulary attribution.
421fn attribution(
422    origin: &neutral_ir::composition::project::CompositionOrigin,
423    ir: &CompositionProjectIr,
424    work: &mut u64,
425    cancel: &CancellationToken,
426) -> Result<(), E> {
427    match &origin.attribution {
428        None => Ok(()),
429        Some(A::Source(location)) => source_attribution(ir, origin, *location, work, cancel),
430        Some(A::Vocabulary {
431            identity,
432            version,
433            type_name,
434            field_name,
435            span,
436        }) => {
437            if origin.kind != K::Defaulted {
438                return Err(E::Companion);
439            }
440            let bundle = ir
441                .vocabularies
442                .iter()
443                .find(|b| b.identity.identity() == identity && b.identity.version() == version)
444                .ok_or(E::Companion)?;
445            let definition = bundle
446                .definitions
447                .iter()
448                .find(|d| d.name == *type_name)
449                .ok_or(E::Companion)?;
450            let B::Record(fields) = &definition.body else {
451                return Err(E::Companion);
452            };
453            if !fields
454                .iter()
455                .any(|f| f.name == *field_name && f.default.is_some())
456            {
457                return Err(E::Companion);
458            }
459            let source = ir
460                .vocabulary_sources
461                .iter()
462                .find(|s| s.identity == *identity && s.version == *version)
463                .ok_or(E::Companion)?;
464            if span.is_some_and(|s| s.is_empty() || s.end() > source.byte_len) {
465                return Err(E::Companion);
466            }
467            // The owner must be on this binding's actual typed path, not merely exist elsewhere.
468            if !default_path_matches(ir, origin, (identity, version, type_name, field_name))? {
469                return Err(E::Companion);
470            }
471            Ok(())
472        }
473    }
474}

476/// Resolves a nominal body from complete interpretation facts, without aliases.
477fn body<'a>(ir: &'a CompositionProjectIr, ty: &T) -> Result<&'a B, E> {
478    match ty {
479        T::Nominal(owner) => match &ir
480            .declarations
481            .iter()
482            .find(|d| d.identity == *owner)
483            .ok_or(E::Semantic)?
484            .signature
485        {
486            S::Definition(body) => Ok(body),
487            S::Binding(_) => Err(E::Semantic),
488        },
489        T::VocabularyNominal {
490            identity,
491            version,
492            name,
493        } => ir
494            .vocabularies
495            .iter()
496            .find(|b| b.identity.identity() == identity && b.identity.version() == version)
497            .and_then(|b| b.definitions.iter().find(|d| d.name == *name))
498            .map(|d| &d.body)
499            .ok_or(E::Semantic),
500        _ => Err(E::Semantic),
501    }
502}

504/// Binds a vocabulary default annotation to an actual traversed field prefix.
505fn default_path_matches(
506    ir: &CompositionProjectIr,
507    origin: &neutral_ir::composition::project::CompositionOrigin,
508    owner: (&str, &str, &str, &str),
509) -> Result<bool, E> {
510    default_path_check(
511        ir,
512        origin,
513        |ty, field| matches!(ty,T::VocabularyNominal {identity,version,name} if (identity.as_str(),version.as_str(),name.as_str(),field.name.as_str())==owner),
514    )
515}

517/// Checks actual typed field prefixes for source or vocabulary default ownership.
518fn default_path_check(
519    ir: &CompositionProjectIr,
520    origin: &neutral_ir::composition::project::CompositionOrigin,
521    matches: impl Fn(&T, &neutral_ir::composition::CompositionField) -> bool,
522) -> Result<bool, E> {
523    let declaration = ir
524        .declarations
525        .iter()
526        .find(|d| d.identity == origin.binding)
527        .ok_or(E::Semantic)?;
528    let S::Binding(initial) = &declaration.signature else {
529        return Err(E::Semantic);
530    };
531    let mut ty = initial;
532    let mut value = declaration.value.as_ref();
533    for segment in &origin.path {
534        while let T::Nullable(inner) = ty {
535            ty = inner;
536        }
537        match segment {
538            P::Field(name) => {
539                let B::Record(fields) = body(ir, ty)? else {
540                    return Err(E::Companion);
541                };
542                let field = fields
543                    .iter()
544                    .find(|f| f.name == *name)
545                    .ok_or(E::Companion)?;
546                if matches(ty, field) && field.default.is_some() {
547                    return Ok(true);
548                }
549                ty = &field.ty;
550                value = match value {
551                    Some(V::Record(fields)) => fields
552                        .iter()
553                        .find(|(n, _)| n == name)
554                        .and_then(|(_, v)| v.as_ref()),
555                    _ => None,
556                };
557            }
558            P::Element(index) => {
559                let T::List(inner) = ty else {
560                    return Err(E::Companion);
561                };
562                ty = inner;
563                value = match value {
564                    Some(V::List(values)) => {
565                        usize::try_from(*index).ok().and_then(|i| values.get(i))
566                    }
567                    _ => None,
568                };
569            }
570            P::Payload => {
571                let B::Variant(alternatives) = body(ir, ty)? else {
572                    return Err(E::Companion);
573                };
574                let Some(V::Variant { tag, payload }) = value else {
575                    return Err(E::Companion);
576                };
577                ty = &alternatives
578                    .iter()
579                    .find(|a| a.tag == *tag)
580                    .ok_or(E::Companion)?
581                    .ty;
582                value = Some(payload);
583            }
584        }
585    }
586    Ok(false)
587}

589/// Checks a source occurrence against its consuming declaration, exact default owner or real reuse closure.
590fn source_attribution(
591    ir: &CompositionProjectIr,
592    origin: &neutral_ir::composition::project::CompositionOrigin,
593    location: neutral_core::SourceLocation,
594    work: &mut u64,
595    cancel: &CancellationToken,
596) -> Result<(), E> {
597    if location.span().is_empty() {
598        return Err(E::Companion);
599    }
600    let contains = |map: &neutral_ir::project::ProjectSourceMap| {
601        map.location.source() == location.source()
602            && location.span().start() >= map.location.span().start()
603            && location.span().end() <= map.location.span().end()
604    };
605    if ir
606        .source_maps
607        .iter()
608        .any(|m| m.declaration == origin.binding && contains(m))
609    {
610        return Ok(());
611    }
612    if default_path_check(
613        ir,
614        origin,
615        |ty, _| matches!(ty, T::Nominal(owner) if ir.source_maps.iter().any(|m| m.declaration == *owner && contains(m))),
616    )? {
617        return Ok(());
618    }
619    let mut seen = BTreeSet::new();
620    let mut pending = Vec::new();
621    pending.try_retain(1).map_err(|_| E::Limit)?;
622    pending.push(&origin.binding);
623    while let Some(owner) = pending.pop() {
624        if cancel.is_cancelled() {
625            return Err(E::Cancelled);
626        }
627        *work = work
628            .checked_sub(ir.provenance.len() as u64 + ir.declarations.len() as u64 + 1)
629            .ok_or(E::Limit)?;
630        charge_index(work, seen.len(), cancel)?;
631        if !seen.insert(owner).map_err(|_| E::Limit)? {
632            continue;
633        }
634        if ir
635            .source_maps
636            .iter()
637            .any(|m| m.declaration == *owner && contains(m))
638        {
639            return Ok(());
640        }
641        for edge in ir
642            .provenance
643            .iter()
644            .filter(|e| e.from == *owner && e.kind == Edge::Value)
645        {
646            pending.try_retain(1).map_err(|_| E::Limit)?;
647            pending.push(&edge.to);
648        }
649    }
650    Err(E::Companion)
651}

653/// Validates source coverage, dependency provenance and independently recomputed base facts.
654fn companions(
655    ir: &CompositionProjectIr,
656    bindings: &ValidatedCompositionBindings,
657    limits: ProjectLimits,
658    cancel: &CancellationToken,
659) -> Result<(), E> {
660    let mut work = ir.composition_limits.work;
661    source_companions(ir, &mut work, cancel)?;
662    let mut index = BTreeMap::new();
663    for d in &ir.declarations {
664        charge_index(&mut work, index.len(), cancel)?;
665        index.insert(&d.identity, d).map_err(|_| E::Limit)?;
666    }
667    let mut actual = BTreeSet::new();
668    let mut previous = None;
669    for edge in &ir.provenance {
670        if cancel.is_cancelled() {
671            return Err(E::Cancelled);
672        }
673        let key = (&edge.from, edge.kind, &edge.to, edge.location);
674        let map = ir
675            .source_maps
676            .iter()
677            .find(|m| m.declaration == edge.from)
678            .ok_or(E::Companion)?;
679        if previous.is_some_and(|p| p >= key)
680            || !index.contains_key(&edge.to)
681            || edge.location.source() != map.location.source()
682            || edge.location.span().is_empty()
683            || edge.location.span().start() < map.location.span().start()
684            || edge.location.span().end() > map.location.span().end()
685        {
686            return Err(E::Companion);
687        }
688        previous = Some(key);
689        charge_index(&mut work, actual.len(), cancel)?;
690        actual
691            .insert((&edge.from, edge.kind, &edge.to))
692            .map_err(|_| E::Limit)?;
693    }
694    let mut expected = BTreeSet::new();
695    for d in &ir.declarations {
696        match &d.signature {
697            S::Binding(ty) => type_edges(&d.identity, ty, false, &mut expected, &mut work, cancel)?,
698            S::Definition(B::Record(fields)) => {
699                for f in fields {
700                    type_edges(&d.identity, &f.ty, false, &mut expected, &mut work, cancel)?;
701                }
702            }
703            S::Definition(B::Variant(alts)) => {
704                for a in alts {
705                    type_edges(&d.identity, &a.ty, false, &mut expected, &mut work, cancel)?;
706                }
707            }
708        }
709    }
710    for b in bindings.bindings() {
711        for r in b.references() {
712            charge_index(&mut work, expected.len(), cancel)?;
713            expected
714                .insert((&b.binding().owner, Edge::Reference, &r.target))
715                .map_err(|_| E::Limit)?;
716        }
717    }
718    if expected
719        .iter()
720        .any(|(from, kind, to)| !actual.contains(&(from, *kind, to)))
721    {
722        return Err(E::Companion);
723    }
724    for (from, kind, to) in &actual {
725        if *kind != Edge::Value && !expected.contains(&(from, *kind, to)) {
726            return Err(E::Companion);
727        }
728        if from.module() != to.module()
729            && (!index.get(to).ok_or(E::Companion)?.public
730                || !reachable(
731                    ir,
732                    from.module().module_name(),
733                    to.module().module_name(),
734                    &mut work,
735                    cancel,
736                )?)
737        {
738            return Err(E::Semantic);
739        }
740        if *kind == Edge::Value {
741            let left = index.get(from).ok_or(E::Companion)?;
742            let right = index.get(to).ok_or(E::Companion)?;
743            let (S::Binding(a), S::Binding(b), Some(value), Some(target)) =
744                (&left.signature, &right.signature, &left.value, &right.value)
745            else {
746                return Err(E::Semantic);
747            };
748            if !contains_reuse(ir, value, a, target, b)? {
749                return Err(E::Semantic);
750            }
751        }
752    }
753    // Ordinary reuse evaluates; identity references deliberately do not enter this cycle check.
754    check_reuse_cycles(&actual, &mut work, cancel)?;
755    resource_facts(ir, limits)
756}

758/// Recomputes aggregate captured and retained counts instead of trusting advertised facts.
759fn resource_facts(ir: &CompositionProjectIr, limits: ProjectLimits) -> Result<(), E> {
760    let facts = ProjectResourceFacts {
761        source_units: ir.sources.len() as u64,
762        source_bytes: ir
763            .sources
764            .iter()
765            .map(|s| s.byte_len)
766            .try_fold(0_u64, u64::checked_add)
767            .ok_or(E::Limit)?,
768        vocabulary_units: ir.vocabulary_sources.len() as u64,
769        vocabulary_bytes: ir
770            .vocabulary_sources
771            .iter()
772            .map(|s| s.byte_len)
773            .try_fold(0_u64, u64::checked_add)
774            .ok_or(E::Limit)?,
775        declarations: ir.declarations.len() as u64,
776        import_edges: ir
777            .modules
778            .iter()
779            .map(|m| m.imports.len() as u64)
780            .try_fold(0_u64, u64::checked_add)
781            .ok_or(E::Limit)?,
782        value_nodes: ir.composition_resources.retained_value_nodes,
783    };
784    if facts != ir.resources {
785        return Err(E::Resources);
786    }
787    if facts.source_bytes > limits.text_bytes
788        || facts.vocabulary_bytes > ir.composition_limits.captured_bytes
789    {
790        return Err(E::Limit);
791    }
792    Ok(())
793}

795/// Checks source accounting, original-byte coverage and exact vocabulary evidence ownership.
796fn source_companions(
797    ir: &CompositionProjectIr,
798    work: &mut u64,
799    cancel: &CancellationToken,
800) -> Result<(), E> {
801    if ir.source_maps.len() != ir.declarations.len()
802        || ir.vocabulary_sources.len() != ir.vocabularies.len()
803    {
804        return Err(E::Companion);
805    }
806    let mut ids = BTreeSet::new();
807    for (s, m) in ir.sources.iter().zip(&ir.modules) {
808        charge_index(work, ids.len(), cancel)?;
809        if s.module != m.identity.module_name()
810            || s.source_id.is_empty()
811            || !ids.insert(&s.source_id).map_err(|_| E::Limit)?
812            || s.byte_len == 0
813        {
814            return Err(E::Companion);
815        }
816    }
817    for (s, b) in ir.vocabulary_sources.iter().zip(&ir.vocabularies) {
818        if s.identity != b.identity.identity()
819            || s.version != b.identity.version()
820            || s.digest != b.identity.content_digest()
821            || s.byte_len == 0
822        {
823            return Err(E::Companion);
824        }
825    }
826    for (map, d) in ir.source_maps.iter().zip(&ir.declarations) {
827        let s = ir
828            .sources
829            .iter()
830            .find(|s| s.module == d.identity.module().module_name())
831            .ok_or(E::Companion)?;
832        if map.declaration != d.identity
833            || map.location.source() != s.digest
834            || map.location.span().is_empty()
835            || map.location.span().end() > s.byte_len
836        {
837            return Err(E::Companion);
838        }
839    }
840    Ok(())
841}

843/// Bounds conservative comparison/lookup work before repeated scans of producer-owned data.
844fn check_comparison_work(ir: &CompositionProjectIr, limit: u64) -> Result<(), E> {
845    let declarations = ir.declarations.len() as u64;
846    let modules = ir.modules.len() as u64;
847    let facts = ir.composition_resources;
848    let size = declarations
849        .checked_add(modules)
850        .and_then(|n| n.checked_add(facts.types))
851        .and_then(|n| n.checked_add(facts.fields))
852        .and_then(|n| n.checked_add(facts.alternatives))
853        .and_then(|n| n.checked_add(facts.retained_value_nodes))
854        .and_then(|n| n.checked_add(1))
855        .ok_or(E::Limit)?;
856    let paths = ir
857        .origins
858        .iter()
859        .try_fold(0_u64, |n, o| n.checked_add(o.path.len() as u64 + 1))
860        .ok_or(E::Limit)?;
861    let scans = paths
862        .checked_add(ir.provenance.len() as u64)
863        .and_then(|n| n.checked_add(declarations))
864        .and_then(|n| n.checked_mul(size))
865        .ok_or(E::Limit)?;
866    if scans > limit {
867        return Err(E::Limit);
868    }
869    Ok(())
870}

872/// Proves a reuse edge points to an invariant typed materialized occurrence, including nested fields/lists.
873fn contains_reuse(
874    ir: &CompositionProjectIr,
875    value: &V,
876    ty: &T,
877    target: &V,
878    target_ty: &T,
879) -> Result<bool, E> {
880    if ty == target_ty && value == target {
881        return Ok(true);
882    }
883    let ty = if let T::Nullable(inner) = ty {
884        inner.as_ref()
885    } else {
886        ty
887    };
888    match (ty, value) {
889        (T::List(inner), V::List(values)) => {
890            for v in values {
891                if contains_reuse(ir, v, inner, target, target_ty)? {
892                    return Ok(true);
893                }
894            }
895        }
896        (T::Nominal(_) | T::VocabularyNominal { .. }, V::Record(values)) => {
897            let B::Record(fields) = body(ir, ty)? else {
898                return Err(E::Semantic);
899            };
900            for ((name, value), field) in values.iter().zip(fields) {
901                if name != &field.name {
902                    return Err(E::Semantic);
903                }
904                if let Some(v) = value
905                    && contains_reuse(ir, v, &field.ty, target, target_ty)?
906                {
907                    return Ok(true);
908                }
909            }
910        }
911        (T::Nominal(_) | T::VocabularyNominal { .. }, V::Variant { tag, payload }) => {
912            let B::Variant(alternatives) = body(ir, ty)? else {
913                return Err(E::Semantic);
914            };
915            let alternative = alternatives
916                .iter()
917                .find(|a| a.tag == *tag)
918                .ok_or(E::Semantic)?;
919            return contains_reuse(ir, payload, &alternative.ty, target, target_ty);
920        }
921        _ => {}
922    }
923    Ok(false)
924}

926/// Checks ordinary value dependencies in linear graph work; identity-only references may cycle.
927fn check_reuse_cycles(
928    edges: &BTreeSet<(&ModuleSymbolIdentity, Edge, &ModuleSymbolIdentity)>,
929    work: &mut u64,
930    cancel: &CancellationToken,
931) -> Result<(), E> {
932    let mut incoming = BTreeMap::new();
933    let mut outgoing: BTreeMap<_, Vec<_>> = BTreeMap::new();
934    for (from, kind, to) in edges {
935        if *kind == Edge::Value {
936            charge_index(work, incoming.len() * 2 + outgoing.len(), cancel)?;
937            incoming
938                .entry(*from)
939                .map_err(|_| E::Limit)?
940                .or_insert(0_usize);
941            *incoming.entry(*to).map_err(|_| E::Limit)?.or_insert(0) += 1;
942            let targets = outgoing.entry(*from).map_err(|_| E::Limit)?.or_default();
943            targets.try_retain(1).map_err(|_| E::Limit)?;
944            targets.push(*to);
945        }
946    }
947    let mut ready = Vec::new();
948    ready.try_retain(incoming.len()).map_err(|_| E::Limit)?;
949    for (key, count) in &incoming {
950        if *count == 0 {
951            ready.push(*key);
952        }
953    }
954    let mut count = 0;
955    while let Some(node) = ready.pop() {
956        if cancel.is_cancelled() {
957            return Err(E::Cancelled);
958        }
959        count += 1;
960        if let Some(targets) = outgoing.get(node) {
961            for target in targets {
962                let n = incoming.get_mut(target).ok_or(E::Semantic)?;
963                *n -= 1;
964                if *n == 0 {
965                    ready.push(*target);
966                }
967            }
968        }
969    }
970    if count != incoming.len() {
971        return Err(E::Semantic);
972    }
973    Ok(())
974}

976/// Enumerates every source nominal dependency, including unselected alternatives.
977fn type_edges<'a>(
978    from: &'a ModuleSymbolIdentity,
979    ty: &'a T,
980    reference: bool,
981    edges: &mut BTreeSet<(&'a ModuleSymbolIdentity, Edge, &'a ModuleSymbolIdentity)>,
982    work: &mut u64,
983    cancel: &CancellationToken,
984) -> Result<(), E> {
985    charge_index(work, edges.len(), cancel)?;
986    match ty {
987        T::Nominal(to) => {
988            edges
989                .insert((
990                    from,
991                    if reference {
992                        Edge::ReferenceType
993                    } else {
994                        Edge::Type
995                    },
996                    to,
997                ))
998                .map_err(|_| E::Limit)?;
999        }
1000        T::Ref(inner) => type_edges(from, inner, true, edges, work, cancel)?,
1001        T::List(inner) | T::Nullable(inner) => {
1002            type_edges(from, inner, reference, edges, work, cancel)?;
1003        }
1004        _ => {}
1005    }
1006    Ok(())
1007}

1009/// Checks logical reachability without acquiring imports or interpreting host locations.
1010fn reachable(
1011    ir: &CompositionProjectIr,
1012    from: &str,
1013    to: &str,
1014    work: &mut u64,
1015    cancel: &CancellationToken,
1016) -> Result<bool, E> {
1017    let mut stack = Vec::new();
1018    stack.try_retain(1).map_err(|_| E::Limit)?;
1019    stack.push(from);
1020    let mut seen = BTreeSet::new();
1021    while let Some(name) = stack.pop() {
1022        if cancel.is_cancelled() {
1023            return Err(E::Cancelled);
1024        }
1025        if name == to {
1026            return Ok(true);
1027        }
1028        charge_index(work, seen.len() + ir.modules.len(), cancel)?;
1029        if !seen.insert(name).map_err(|_| E::Limit)? {
1030            continue;
1031        }
1032        let module = ir
1033            .modules
1034            .iter()
1035            .find(|m| m.identity.module_name() == name)
1036            .ok_or(E::Semantic)?;
1037        stack
1038            .try_retain(module.imports.len())
1039            .map_err(|_| E::Limit)?;
1040        stack.extend(module.imports.iter().map(String::as_str));
1041    }
1042    Ok(false)
1043}
1044/// Charges sorted-index movement before reservation or expansion and checks cancellation.
1045fn charge_index(work: &mut u64, count: usize, cancel: &CancellationToken) -> Result<(), E> {
1046    if cancel.is_cancelled() {
1047        return Err(E::Cancelled);
1048    }
1049    *work = work.checked_sub(count as u64 + 1).ok_or(E::Limit)?;
1050    Ok(())
