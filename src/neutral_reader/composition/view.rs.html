31/// Immutable public interpretive closure without private source IDs, spans or implementation edges.
32#[derive(Clone, Debug, Eq, PartialEq)]
33pub struct CompositionView {
34    /// Normalized explicit selection.
35    roots: Vec<ModuleSymbolIdentity>,
36    /// All public source signatures/values needed to interpret selected roots.
37    declarations: Vec<CompositionDeclaration>,
38    /// Public contract definitions and exact transitive dependency metadata.
39    vocabularies: Vec<CompositionBundle>,
40    /// Redacted occurrence facts; source attribution is explicitly unavailable.
41    origins: Vec<CompositionOrigin>,
42    /// Actual public binding references, keyed by their consuming owner.
43    references: Vec<(ModuleSymbolIdentity, CompositionBindingReference)>,
44}
45impl CompositionView {
46    /// Returns the successor selection schema, never complete-project authority.
47    #[must_use]
48    pub const fn schema(&self) -> &'static str {
49        profile::PROJECT_VIEW_SCHEMA
50    }
51    /// Returns canonical selected roots.
52    #[must_use]
53    pub fn roots(&self) -> &[ModuleSymbolIdentity] {
54        &self.roots
55    }
56    /// Returns complete public source interpretation contracts and materialized values.
57    #[must_use]
58    pub fn declarations(&self) -> &[CompositionDeclaration] {
59        &self.declarations
60    }
61    /// Returns public vocabulary facts, including all alternative/reference types and restrictions/defaults.
62    #[must_use]
63    pub fn vocabularies(&self) -> &[CompositionBundle] {
64        &self.vocabularies
65    }
66    /// Returns safe classifications and canonical vocabulary attribution without private source evidence.
67    #[must_use]
68    pub fn origins(&self) -> &[CompositionOrigin] {
69        &self.origins
70    }
71    /// Returns every actual exposed reference occurrence and target in canonical order.
72    #[must_use]
73    pub fn references(&self) -> &[(ModuleSymbolIdentity, CompositionBindingReference)] {
74        &self.references
75    }
76}

78/// Bounded closure state over independently validated complete data.
79struct Closure<'a> {
80    /// Immutable complete reader authority.
81    project: &'a ValidatedCompositionProject,
82    /// Cooperative cancellation.
83    cancel: &'a CancellationToken,
84    /// Remaining independently intersected work budget.
85    work: u64,
86    /// Required source owners, including reference targets.
87    sources: BTreeSet<ModuleSymbolIdentity>,
88    /// Iterative binding/type worklist; reference chains do not consume Rust stack.
89    pending: Vec<ModuleSymbolIdentity>,
90    /// Required public vocabulary definitions.
91    types: BTreeSet<(String, String, String)>,
92    /// Iterative nominal worklist; reference-type chains do not consume Rust stack.
93    pending_types: Vec<(String, String, String)>,
94    /// Exact transitive bundle closure.
95    bundles: BTreeSet<(String, String)>,
96}
97impl Closure<'_> {
98    /// Charges traversal before following another interpretation dependency.
99    fn step(&mut self) -> Result<(), E> {
100        if self.cancel.is_cancelled() {
101            return Err(E::Cancelled);
102        }
103        let ir = self.project.complete_ir();
104        let scan = 1_u64
105            + ir.declarations.len() as u64
106            + ir.vocabularies
107                .iter()
108                .map(|b| b.definitions.len() as u64)
109                .sum::<u64>();
110        self.work = self.work.checked_sub(scan).ok_or(E::Limit)?;
111        Ok(())
112    }
113    /// Includes nominal definitions behind every wrapper, not just selected variant payloads.
114    fn ty(&mut self, ty: &T) -> Result<(), E> {
115        self.step()?;
116        match ty {
117            T::List(inner) | T::Ref(inner) | T::Nullable(inner) => self.ty(inner),
118            T::Nominal(owner) => self.source(owner),
119            T::VocabularyNominal {
120                identity,
121                version,
122                name,
123            } => {
124                self.shift(self.types.len())?;
125                let key = (owned(identity)?, owned(version)?, owned(name)?);
126                if self.types.insert(copy(&key)?).map_err(|_| E::Limit)? {
127                    self.pending_types.try_retain(1).map_err(|_| E::Limit)?;
128                    self.pending_types.push(key);
129                }
130                Ok(())
131            }
132            _ => Ok(()),
133        }
134    }
135    /// Expands one vocabulary contract; nominal reference targets are queued instead of recursively expanded.
136    fn expand_type(&mut self, identity: &str, version: &str, name: &str) -> Result<(), E> {
137        self.step()?;
138        let bundle = self
139            .project
140            .complete_ir()
141            .vocabularies
142            .iter()
143            .find(|b| b.identity.identity() == identity && b.identity.version() == version)
144            .ok_or(E::Semantic)?;
145        let definition = bundle
146            .definitions
147            .iter()
148            .find(|d| d.name == name && d.public)
149            .ok_or(E::Semantic)?;
150        self.body(&definition.body)?;
151        self.bundle(identity, version)
152    }
153    /// Includes every field and alternative; closed defaults cannot introduce binding references.
154    fn body(&mut self, body: &B) -> Result<(), E> {
155        match body {
156            B::Record(fields) => {
157                for field in fields {
158                    self.ty(&field.ty)?;
159                }
160            }
161            B::Variant(alternatives) => {
162                for alternative in alternatives {
163                    self.ty(&alternative.ty)?;
164                }
165            }
166        }
167        Ok(())
168    }
169    /// Includes public source definitions and actual binding-reference targets, never ordinary reuse provenance.
170    fn source(&mut self, owner: &ModuleSymbolIdentity) -> Result<(), E> {
171        self.step()?;
172        self.shift(self.sources.len())?;
173        if !self.sources.insert(copy(owner)?).map_err(|_| E::Limit)? {
174            return Ok(());
175        }
176        self.pending.try_retain(1).map_err(|_| E::Limit)?;
177        self.pending.push(copy(owner)?);
178        Ok(())
179    }
180    /// Expands one queued public declaration without recursively following binding references.
181    fn expand(&mut self, owner: &ModuleSymbolIdentity) -> Result<(), E> {
182        let declaration = self
183            .project
184            .complete_ir()
185            .declarations
186            .iter()
187            .find(|d| d.identity == *owner && d.public)
188            .ok_or(E::Semantic)?;
189        match &declaration.signature {
190            S::Definition(body) => self.body(body)?,
191            S::Binding(ty) => {
192                self.ty(ty)?;
193                let binding = self
194                    .project
195                    .bindings()
196                    .bindings()
197                    .iter()
198                    .find(|b| b.binding().owner == *owner)
199                    .ok_or(E::Semantic)?;
200                for edge in binding.references() {
201                    self.source(&edge.target)?;
202                }
203            }
204        }
205        Ok(())
206    }
207    /// Includes exact transitive bundle locks even when a dependency has no selected type.
208    fn bundle(&mut self, identity: &str, version: &str) -> Result<(), E> {
209        self.step()?;
210        self.shift(self.bundles.len())?;
211        if !self
212            .bundles
213            .insert((owned(identity)?, owned(version)?))
214            .map_err(|_| E::Limit)?
215        {
216            return Ok(());
217        }
218        let bundle = self
219            .project
220            .complete_ir()
221            .vocabularies
222            .iter()
223            .find(|b| b.identity.identity() == identity && b.identity.version() == version)
224            .ok_or(E::Semantic)?;
225        for dependency in &bundle.dependencies {
226            self.bundle(&dependency.identity, &dependency.version)?;
227        }
228        Ok(())
229    }
230    /// Charges worst-case sorted-index shifts before fallible insertion.
231    fn shift(&mut self, count: usize) -> Result<(), E> {
232        if self.cancel.is_cancelled() {
233            return Err(E::Cancelled);
234        }
235        self.work = self.work.checked_sub(count as u64).ok_or(E::Limit)?;
236        Ok(())
237    }
238}

240impl ValidatedCompositionProject {
241    /// Derives a public dependency-closed view only after complete validation, without changing identities.
242    ///
243    /// # Errors
244    /// Rejects selector mismatch, private/unknown/duplicate roots, bounds or cancellation atomically.
245    pub fn derive_view(
246        &self,
247        request: &CompositionViewRequest,
248        cancel: &CancellationToken,
249    ) -> Result<CompositionView, E> {
250        if cancel.is_cancelled() {
251            return Err(E::Cancelled);
252        }
253        if request.schema != profile::PROJECT_VIEW_SCHEMA {
254            return Err(E::Schema);
255        }
256        let ir = self.complete_ir();
257        if request.roots.len() > ir.declarations.len() {
258            return Err(E::Limit);
259        }
260        let key_bytes = request
261            .roots
262            .iter()
263            .try_fold(0_u64, |sum, owner| {
264                sum.checked_add(
265                    (owner.module().language_behavior_version().len()
266                        + owner.module().module_name().len()
267                        + owner.declaration_name().len()) as u64,
268                )
269            })
270            .ok_or(E::Limit)?;
271        if key_bytes > self.effective_composition_limits().work {
272            return Err(E::Limit);
273        }
274        let mut roots = copy_slice(&request.roots).map_err(|_| E::Limit)?;
275        roots.sort_unstable();
276        if roots.windows(2).any(|pair| pair[0] == pair[1])
277            || roots.iter().any(|r| {
278                !ir.declarations
279                    .binary_search_by(|d| d.identity.cmp(r))
280                    .is_ok_and(|i| ir.declarations[i].public)
281            })
282        {
283            return Err(E::Semantic);
284        }
285        let mut closure = Closure {
286            project: self,
287            cancel,
288            work: self
289                .effective_composition_limits()
290                .work
291                .saturating_sub(key_bytes)
292                .min(profile::MAX_ITEMS),
293            sources: BTreeSet::new(),
294            pending: Vec::new(),
295            types: BTreeSet::new(),
296            pending_types: Vec::new(),
297            bundles: BTreeSet::new(),
298        };
299        for root in &roots {
300            closure.source(root)?;
301        }
302        while !closure.pending.is_empty() || !closure.pending_types.is_empty() {
303            if let Some(owner) = closure.pending.pop() {
304                closure.expand(&owner)?;
305            }
306            if let Some((identity, version, name)) = closure.pending_types.pop() {
307                closure.expand_type(&identity, &version, &name)?;
308            }
309        }
310        project_view(&mut closure, roots)
311    }
312}

314/// Copies only interpretation closure and redacts every source/private implementation companion.
315fn project_view(
316    closure: &mut Closure<'_>,
317    roots: Vec<ModuleSymbolIdentity>,
318) -> Result<CompositionView, E> {
319    let ir = closure.project.complete_ir();
320    let mut declarations = Vec::new();
321    for d in &ir.declarations {
322        if !closure.sources.contains(&d.identity) {
323            continue;
324        }
325        closure.step()?;
326        push(&mut declarations, copy(d)?)?;
327    }
328    let mut vocabularies = Vec::new();
329    for b in &ir.vocabularies {
330        closure.step()?;
331        if !closure.bundles.iter().any(|(identity, version)| {
332            identity == b.identity.identity() && version == b.identity.version()
333        }) {
334            continue;
335        }
336        let mut definitions = Vec::new();
337        for d in &b.definitions {
338            closure.step()?;
339            if d.public
340                && has_type(
341                    closure,
342                    b.identity.identity(),
343                    b.identity.version(),
344                    &d.name,
345                )
346            {
347                push(&mut definitions, copy(d)?)?;
348            }
349        }
350        push(
351            &mut vocabularies,
352            CompositionBundle {
353                identity: copy(&b.identity)?,
354                dependencies: copy(&b.dependencies)?,
355                definitions,
356            },
357        )?;
358    }
359    let mut origins = Vec::new();
360    for o in &ir.origins {
361        if !closure.sources.contains(&o.binding) {
362            continue;
363        }
364        closure.step()?;
365        let attribution = match &o.attribution {
366            Some(A::Vocabulary {
367                identity,
368                version,
369                type_name,
370                field_name,
371                ..
372            }) if has_type(closure, identity, version, type_name) => Some(A::Vocabulary {
373                identity: owned(identity)?,
374                version: owned(version)?,
375                type_name: owned(type_name)?,
376                field_name: owned(field_name)?,
377                span: None,
378            }),
379            _ => None,
380        };
381        push(
382            &mut origins,
383            CompositionOrigin {
384                binding: copy(&o.binding)?,
385                path: copy(&o.path)?,
386                kind: o.kind,
387                attribution,
388            },
389        )?;
390    }
391    let mut references = Vec::new();
392    for b in closure.project.bindings().bindings() {
393        if !closure.sources.contains(&b.binding().owner) {
394            continue;
395        }
396        for r in b.references() {
397            closure.step()?;
398            push(&mut references, (copy(&b.binding().owner)?, copy(r)?))?;
399        }
400    }
401    closure.step()?;
402    Ok(CompositionView {
403        roots,
404        declarations,
405        vocabularies,
406        origins,
407        references,
408    })
409}
410/// Checks nominal membership using borrowed spellings without allocating temporary tuple keys.
411fn has_type(closure: &Closure<'_>, identity: &str, version: &str, name: &str) -> bool {
412    closure
413        .types
414        .iter()
415        .any(|(i, v, n)| i == identity && v == version && n == name)
416}
417/// Copies preflighted retained data without permitting allocator-abort ownership constructors.
418fn copy<T: TryClone>(value: &T) -> Result<T, E> {
419    value.try_clone().map_err(|_| E::Limit)
420}
421/// Retains one bounded logical spelling through a fallible string reservation.
422fn owned(value: &str) -> Result<String, E> {
423    text(value).map_err(|_| E::Limit)
424}
425/// Reserves unpublished view retention before appending, never exposing a partial view.
426fn push<T>(values: &mut Vec<T>, value: T) -> Result<(), E> {
427    values.try_retain(1).map_err(|_| E::Limit)?;
428    values.push(value);
429    Ok(())
