14/// Independent per-definition bounds, not semantic field or list-length restrictions.
15#[derive(Clone, Copy, Debug)]
16pub struct CompositionInspectionLimits {
17    /// Maximum inspected type nodes and path-copy/key bytes.
18    pub visits: u64,
19    /// Maximum emitted reference-type occurrences.
20    pub references: u64,
21    /// Maximum field/alternative/composed-type path depth.
22    pub depth: u64,
23}
24
25/// Safe public inspection failures with no captured text or private definition details.
26#[derive(Clone, Copy, Debug, Eq, PartialEq)]
27pub enum CompositionReferenceError {
28    /// Existing public owner/type lookup classification.
29    Lookup(CompositionLookupError),
30    /// Zero independent caller budget.
31    InvalidLimits,
32    /// Caller or hard traversal/reference/depth ceiling exceeded.
33    Limit,
34    /// Cooperative cancellation before publishing any partial enumeration.
35    Cancelled,
36    /// Bounded fallible path/result reservation failed.
37    Allocation,
38}
39
40/// A canonical type-contract path, distinct from source/value/host paths.
41#[derive(Clone, Copy, Debug, Eq, PartialEq)]
42pub enum ReferenceTypeSegment<'a> {
43    /// Record field, including optional/defaulted fields.
44    Field(&'a str),
45    /// Closed variant alternative, including unselected branches.
46    Alternative(&'a str),
47    /// Nested ordered-list element type.
48    ListElement,
49    /// Nested nullable inner type.
50    NullableInner,
51}
52
53/// One reference type occurrence with an exact public nominal target.
54#[derive(Clone, Debug, Eq, PartialEq)]
55pub struct ReferenceTypeDependency<'a> {
56    /// Canonical field/tag/wrapper path within the selected public definition.
57    pub path: Vec<ReferenceTypeSegment<'a>>,
58    /// Complete alias-independent target type, not a binding/value reference edge.
59    pub target: &'a T,
60}
61
62/// Request-local counters shared by every branch of one definition traversal.
63struct Inspector<'a> {
64    /// Complete independent caller policy.
65    limits: CompositionInspectionLimits,
66    /// Explicit cooperative cancellation signal.
67    cancellation: &'a CancellationToken,
68    /// Charged traversal/path-copy work.
69    visits: u64,
70}
71
72impl Inspector<'_> {
73    /// Observes cancellation and charges work before traversal or proportional allocation.
74    fn step(&mut self, amount: u64) -> Result<(), CompositionReferenceError> {
75        if self.cancellation.is_cancelled() {
76            return Err(CompositionReferenceError::Cancelled);
77        }
78        self.visits = self
79            .visits
80            .checked_add(amount)
81            .ok_or(CompositionReferenceError::Limit)?;
82        if self.visits > self.limits.visits.min(MAX_WORK) {
83            return Err(CompositionReferenceError::Limit);
84        }
85        Ok(())
86    }
87
88    /// Walks list/nullable wrappers but emits rather than expands a nominal reference target.
89    fn walk<'a>(
90        &mut self,
91        ty: &'a T,
92        path: &mut Vec<ReferenceTypeSegment<'a>>,
93        result: &mut Vec<ReferenceTypeDependency<'a>>,
94    ) -> Result<(), CompositionReferenceError> {
95        self.step(1)?;
96        if path.len() as u64 > self.limits.depth.min(MAX_DEPTH) {
97            return Err(CompositionReferenceError::Limit);
98        }
99        match ty {
100            T::Ref(target) => {
101                if result.len() as u64 >= self.limits.references.min(MAX_WORK) {
102                    return Err(CompositionReferenceError::Limit);
103                }
104                let bytes = path.iter().try_fold(1_u64, |total, segment| {
105                    total
106                        .checked_add(match segment {
107                            ReferenceTypeSegment::Field(name)
108                            | ReferenceTypeSegment::Alternative(name) => name.len() as u64,
109                            _ => 1,
110                        })
111                        .ok_or(CompositionReferenceError::Limit)
112                })?;
113                self.step(bytes)?;
114                let mut copied = Vec::new();
115                copied
116                    .try_retain_exact(path.len())
117                    .map_err(|_| CompositionReferenceError::Allocation)?;
118                copied.extend_from_slice(path);
119                result
120                    .try_retain(1)
121                    .map_err(|_| CompositionReferenceError::Allocation)?;
122                result.push(ReferenceTypeDependency {
123                    path: copied,
124                    target,
125                });
126            }
127            T::List(inner) | T::Nullable(inner) => {
128                path.try_retain(1)
129                    .map_err(|_| CompositionReferenceError::Allocation)?;
130                path.push(if matches!(ty, T::List(_)) {
131                    ReferenceTypeSegment::ListElement
132                } else {
133                    ReferenceTypeSegment::NullableInner
134                });
135                let outcome = self.walk(inner, path, result);
136                path.pop();
137                outcome?;
138            }
139            _ => {}
140        }
141        Ok(())
142    }
143}

145/// Enumerates direct declared reference contracts atomically in validated canonical field/tag order.
146pub(super) fn inspect<'a>(
147    definition: &'a CompositionDefinition,
148    limits: CompositionInspectionLimits,
149    cancellation: &CancellationToken,
150) -> Result<Vec<ReferenceTypeDependency<'a>>, CompositionReferenceError> {
151    if [limits.visits, limits.references, limits.depth].contains(&0) {
152        return Err(CompositionReferenceError::InvalidLimits);
153    }
154    let mut inspector = Inspector {
155        limits,
156        cancellation,
157        visits: 0,
158    };
159    let mut result = Vec::new();
160    let mut path = Vec::new();
161    path.try_retain(1)
162        .map_err(|_| CompositionReferenceError::Allocation)?;
163    inspector.step(1)?;
164    match &definition.body {
165        CompositionBody::Record(fields) => {
166            for field in fields {
167                path.push(ReferenceTypeSegment::Field(&field.name));
168                inspector.walk(&field.ty, &mut path, &mut result)?;
169                path.pop();
170            }
171        }
172        CompositionBody::Variant(alternatives) => {
173            for alternative in alternatives {
174                path.push(ReferenceTypeSegment::Alternative(&alternative.tag));
175                inspector.walk(&alternative.ty, &mut path, &mut result)?;
176                path.pop();
177            }
178        }
179    }
180    inspector.step(1)?;
181    Ok(result)
182}
