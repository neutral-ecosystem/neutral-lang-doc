3//! Check that provenance paths and edges describe the actual logical values.
4//!
5//! Reference evidence must identify the typed target; reuse evidence must point
6//! to a compatible source value; field evidence must cover materialized fields.
7//! Path traversal and completeness checks reject missing, extra or contradictory
8//! claims instead of inventing source history on behalf of a producer.
9
10use super::{
11    BTreeMap, BTreeSet, CompilationArtifacts, ElementId, LogicalValue, ReaderError,
12    RecordTypeDefinition, ResolvedType, ValueOrigin, VocabularyContract,
13};

15/// Validates every typed identity edge and requires exact provenance coverage.
16pub(super) fn validate_reference_provenance(
17    artifacts: &CompilationArtifacts,
18    records: &BTreeMap<&str, &RecordTypeDefinition>,
19) -> Result<(), ReaderError> {
20    let declarations = artifacts
21        .logical_document()
22        .declarations()
23        .iter()
24        .map(|declaration| (declaration.element_id(), declaration))
25        .collect::<BTreeMap<_, _>>();
26    let root_reuses = artifacts
27        .reuse_provenance()
28        .iter()
29        .filter(|record| record.value_path().is_empty())
30        .map(neutral_ir::ReuseProvenanceRecord::element_id)
31        .collect::<BTreeSet<_>>();
32    let mut observed = BTreeMap::new();
33    for provenance in artifacts.reference_provenance() {
34        let Some(owner) = declarations.get(&provenance.element_id()) else {
35            return Err(ReaderError::InvalidReferenceEdge);
36        };
37        let Some(target) = declarations.get(&provenance.target_element_id()) else {
38            return Err(ReaderError::InvalidReferenceEdge);
39        };
40        let Some((expected, value)) = value_at_path(
41            owner.resolved_type(),
42            owner.value(),
43            provenance.value_path(),
44            records,
45            artifacts.logical_document().vocabulary(),
46        ) else {
47            return Err(ReaderError::InvalidReferenceEdge);
48        };
49        let expected = match expected {
50            ResolvedType::Nullable(inner) => inner.as_ref(),
51            expected => expected,
52        };
53        let (ResolvedType::Ref(expected_target), LogicalValue::Reference(reference)) =
54            (expected, value)
55        else {
56            return Err(ReaderError::InvalidReferenceEdge);
57        };
58        if reference.target_element_id() != provenance.target_element_id()
59            || reference.target_symbol_identity() != target.symbol_identity()
60            || reference.target_type() != target.resolved_type()
61            || expected_target.as_ref() != target.resolved_type()
62            || observed
63                .insert(
64                    (provenance.element_id(), provenance.value_path().to_vec()),
65                    provenance.target_element_id(),
66                )
67                .is_some()
68        {
69            return Err(ReaderError::InvalidReferenceEdge);
70        }
71    }
72    for declaration in declarations.values() {
73        if !reference_provenance_is_complete(
74            declaration.element_id(),
75            declaration.value(),
76            &mut Vec::new(),
77            &observed,
78        ) {
79            return Err(ReaderError::InvalidReferenceEdge);
80        }
81    }
82    for provenance in artifacts.provenance() {
83        let has_root_reference = observed.contains_key(&(provenance.element_id(), Vec::new()));
84        let expected_direct = has_root_reference && !root_reuses.contains(&provenance.element_id());
85        if expected_direct
86            != matches!(
87                (provenance.origin(), provenance.normalization()),
88                (
89                    ValueOrigin::IdentityReference,
90                    neutral_ir::Normalization::IdentityReferenceResolution
91                )
92            )
93        {
94            return Err(ReaderError::InvalidReferenceEdge);
95        }
96    }
97    Ok(())
98}

100/// Returns whether every final logical identity edge has one provenance entry.
101fn reference_provenance_is_complete(
102    element_id: ElementId,
103    value: &LogicalValue,
104    path: &mut Vec<String>,
105    observed: &BTreeMap<(ElementId, Vec<String>), ElementId>,
106) -> bool {
107    match value {
108        LogicalValue::Reference(reference) => observed
109            .get(&(element_id, path.clone()))
110            .is_some_and(|target| *target == reference.target_element_id()),
111        LogicalValue::Record(record) => record.fields().iter().all(|field| {
112            path.push(field.name().to_owned());
113            let complete =
114                reference_provenance_is_complete(element_id, field.value(), path, observed);
115            path.pop();
116            complete
117        }),
118        LogicalValue::VocabularyRecord(record) => record.fields().iter().all(|field| {
119            path.push(field.name().to_owned());
120            let complete =
121                reference_provenance_is_complete(element_id, field.value(), path, observed);
122            path.pop();
123            complete
124        }),
125        LogicalValue::List(items) => items.iter().enumerate().all(|(index, item)| {
126            path.push(index.to_string());
127            let complete = reference_provenance_is_complete(element_id, item, path, observed);
128            path.pop();
129            complete
130        }),
131        LogicalValue::Number(_)
132        | LogicalValue::String(_)
133        | LogicalValue::Boolean(_)
134        | LogicalValue::Null => true,
135    }
136}

138/// Validates immutable-value reuse ownership, paths, targets, types, and values.
139pub(super) fn validate_reuse_provenance(
140    artifacts: &CompilationArtifacts,
141    records: &BTreeMap<&str, &RecordTypeDefinition>,
142) -> Result<(), ReaderError> {
143    let declarations = artifacts
144        .logical_document()
145        .declarations()
146        .iter()
147        .map(|declaration| (declaration.element_id(), declaration))
148        .collect::<BTreeMap<_, _>>();
149    let mut observed = BTreeSet::new();
150    for provenance in artifacts.reuse_provenance() {
151        let Some(owner) = declarations.get(&provenance.element_id()) else {
152            return Err(ReaderError::InvalidReuseProvenance);
153        };
154        let Some(source) = declarations.get(&provenance.source_element_id()) else {
155            return Err(ReaderError::InvalidReuseProvenance);
156        };
157        let Some((expected, value)) = value_at_path(
158            owner.resolved_type(),
159            owner.value(),
160            provenance.value_path(),
161            records,
162            artifacts.logical_document().vocabulary(),
163        ) else {
164            return Err(ReaderError::InvalidReuseProvenance);
165        };
166        if !observed.insert((provenance.element_id(), provenance.value_path().to_vec()))
167            || !reuse_type_is_compatible(expected, source.resolved_type())
168            || value != source.value()
169        {
170            return Err(ReaderError::InvalidReuseProvenance);
171        }
172    }
173    for provenance in artifacts.provenance() {
174        let root_reuse = observed.contains(&(provenance.element_id(), Vec::new()));
175        if matches!(
176            (provenance.origin(), provenance.normalization(), root_reuse),
177            (
178                ValueOrigin::OrdinaryReuse,
179                neutral_ir::Normalization::ImmutableValueReuse,
180                true
181            )
182        ) {
183            continue;
184        }
185        if provenance.origin() == ValueOrigin::OrdinaryReuse
186            || provenance.normalization() == neutral_ir::Normalization::ImmutableValueReuse
187            || root_reuse
188        {
189            return Err(ReaderError::InvalidReuseProvenance);
190        }
191    }
192    Ok(())
193}

195/// Resolves one canonical value path to its contextual type and final value.
196fn value_at_path<'a>(
197    expected: &'a ResolvedType,
198    value: &'a LogicalValue,
199    path: &[String],
200    records: &BTreeMap<&str, &'a RecordTypeDefinition>,
201    vocabulary: Option<&'a VocabularyContract>,
202) -> Option<(&'a ResolvedType, &'a LogicalValue)> {
203    if path.is_empty() {
204        return Some((expected, value));
205    }
206    let expected = match expected {
207        ResolvedType::Nullable(inner) => inner.as_ref(),
208        expected => expected,
209    };
210    let (head, tail) = path.split_first()?;
211    if let (ResolvedType::List(inner), LogicalValue::List(items)) = (expected, value) {
212        let index = head.parse::<usize>().ok()?;
213        return value_at_path(inner, items.get(index)?, tail, records, vocabulary);
214    }
215    if let (
216        ResolvedType::VocabularyRecord(identity),
217        LogicalValue::VocabularyRecord(record_value),
218    ) = (expected, value)
219    {
220        let schema = vocabulary?.type_by_name(identity.name())?;
221        let index = schema
222            .fields()
223            .iter()
224            .position(|field| field.name() == head)?;
225        return value_at_path(
226            schema.fields()[index].resolved_type(),
227            record_value.fields()[index].value(),
228            tail,
229            records,
230            vocabulary,
231        );
232    }
233    let (ResolvedType::Record(identity), LogicalValue::Record(record_value)) = (expected, value)
234    else {
235        return None;
236    };
237    let schema = records.get(identity.name())?;
238    let index = schema
239        .fields()
240        .iter()
241        .position(|field| field.name() == head)?;
242    value_at_path(
243        schema.fields()[index].resolved_type(),
244        record_value.fields()[index].value(),
245        tail,
246        records,
247        vocabulary,
248    )
249}

251/// Returns whether a source binding type may satisfy one reuse occurrence.
252fn reuse_type_is_compatible(expected: &ResolvedType, source: &ResolvedType) -> bool {
253    expected == source
254        || matches!(expected, ResolvedType::Nullable(inner) if inner.as_ref() == source)
255}

257/// Validates field provenance ownership, paths, uniqueness, and origin kinds.
258pub(super) fn validate_field_provenance(
259    artifacts: &CompilationArtifacts,
260    records: &BTreeMap<&str, &RecordTypeDefinition>,
261) -> Result<(), ReaderError> {
262    let declarations = artifacts
263        .logical_document()
264        .declarations()
265        .iter()
266        .map(|declaration| (declaration.element_id(), declaration))
267        .collect::<BTreeMap<_, _>>();
268    let mut observed = BTreeMap::new();
269    for provenance in artifacts.field_provenance() {
270        let Some(declaration) = declarations.get(&provenance.element_id()) else {
271            return Err(ReaderError::InvalidFieldProvenance);
272        };
273        if provenance.field_path().is_empty()
274            || !matches!(
275                provenance.origin(),
276                ValueOrigin::ExplicitRecordField
277                    | ValueOrigin::UserRecordDefault
278                    | ValueOrigin::VocabularyDefault
279            )
280            || observed
281                .insert(
282                    (provenance.element_id(), provenance.field_path().to_vec()),
283                    provenance.origin(),
284                )
285                .is_some()
286            || !field_path_exists(
287                declaration.resolved_type(),
288                declaration.value(),
289                provenance.field_path(),
290                records,
291                artifacts.logical_document().vocabulary(),
292            )
293        {
294            return Err(ReaderError::InvalidFieldProvenance);
295        }
296    }
297    for declaration in declarations.values() {
298        if !field_provenance_is_complete(
299            declaration.element_id(),
300            declaration.resolved_type(),
301            declaration.value(),
302            &mut Vec::new(),
303            records,
304            artifacts.logical_document().vocabulary(),
305            &observed,
306        ) {
307            return Err(ReaderError::InvalidFieldProvenance);
308        }
309    }
310    Ok(())
311}

313/// Returns whether every final record field has exact explicit/default evidence.
314fn field_provenance_is_complete(
315    element_id: ElementId,
316    expected: &ResolvedType,
317    value: &LogicalValue,
318    path: &mut Vec<String>,
319    records: &BTreeMap<&str, &RecordTypeDefinition>,
320    vocabulary: Option<&VocabularyContract>,
321    observed: &BTreeMap<(ElementId, Vec<String>), ValueOrigin>,
322) -> bool {
323    let expected = match expected {
324        ResolvedType::Nullable(inner) => inner.as_ref(),
325        expected => expected,
326    };
327    if let (ResolvedType::List(inner), LogicalValue::List(items)) = (expected, value) {
328        for (index, item) in items.iter().enumerate() {
329            path.push(index.to_string());
330            if !field_provenance_is_complete(
331                element_id, inner, item, path, records, vocabulary, observed,
332            ) {
333                return false;
334            }
335            path.pop();
336        }
337        return true;
338    }
339    if let (ResolvedType::VocabularyRecord(identity), LogicalValue::VocabularyRecord(value)) =
340        (expected, value)
341    {
342        let Some(schema) = vocabulary.and_then(|contract| contract.type_by_name(identity.name()))
343        else {
344            return false;
345        };
346        for (schema_field, value_field) in schema.fields().iter().zip(value.fields()) {
347            path.push(schema_field.name().to_owned());
348            let Some(origin) = observed.get(&(element_id, path.clone())) else {
349                return false;
350            };
351            if *origin == ValueOrigin::ExplicitRecordField
352                && !field_provenance_is_complete(
353                    element_id,
354                    schema_field.resolved_type(),
355                    value_field.value(),
356                    path,
357                    records,
358                    vocabulary,
359                    observed,
360                )
361            {
362                return false;
363            }
364            path.pop();
365        }
366        return true;
367    }
368    let (ResolvedType::Record(identity), LogicalValue::Record(value)) = (expected, value) else {
369        return true;
370    };
371    let Some(schema) = records.get(identity.name()) else {
372        return false;
373    };
374    for (schema_field, value_field) in schema.fields().iter().zip(value.fields()) {
375        path.push(schema_field.name().to_owned());
376        let key = (element_id, path.clone());
377        let Some(origin) = observed.get(&key) else {
378            return false;
379        };
380        if *origin == ValueOrigin::ExplicitRecordField
381            && !field_provenance_is_complete(
382                element_id,
383                schema_field.resolved_type(),
384                value_field.value(),
385                path,
386                records,
387                vocabulary,
388                observed,
389            )
390        {
391            return false;
392        }
393        path.pop();
394    }
395    true
396}

398/// Returns whether a path identifies a final field in a typed record value.
399fn field_path_exists(
400    expected: &ResolvedType,
401    value: &LogicalValue,
402    path: &[String],
403    records: &BTreeMap<&str, &RecordTypeDefinition>,
404    vocabulary: Option<&VocabularyContract>,
405) -> bool {
406    let expected = match expected {
407        ResolvedType::Nullable(inner) => inner.as_ref(),
408        expected => expected,
409    };
410    let Some((head, tail)) = path.split_first() else {
411        return false;
412    };
413    if let (ResolvedType::List(inner), LogicalValue::List(items)) = (expected, value) {
414        let Ok(index) = head.parse::<usize>() else {
415            return false;
416        };
417        let Some(item) = items.get(index) else {
418            return false;
419        };
420        return field_path_exists(inner, item, tail, records, vocabulary);
421    }
422    if let (ResolvedType::VocabularyRecord(identity), LogicalValue::VocabularyRecord(value)) =
423        (expected, value)
424    {
425        let Some(schema) = vocabulary.and_then(|contract| contract.type_by_name(identity.name()))
426        else {
427            return false;
428        };
429        let Some(index) = schema
430            .fields()
431            .iter()
432            .position(|field| field.name() == head)
433        else {
434            return false;
435        };
436        if tail.is_empty() {
437            return true;
438        }
439        return field_path_exists(
440            schema.fields()[index].resolved_type(),
441            value.fields()[index].value(),
442            tail,
443            records,
444            vocabulary,
445        );
446    }
447    let (ResolvedType::Record(identity), LogicalValue::Record(value)) = (expected, value) else {
448        return false;
449    };
450    let Some(schema) = records.get(identity.name()) else {
451        return false;
452    };
453    let Some(index) = schema
454        .fields()
455        .iter()
456        .position(|field| field.name() == head)
457    else {
458        return false;
459    };
460    if tail.is_empty() {
461        return true;
462    }
463    field_path_exists(
464        schema.fields()[index].resolved_type(),
465        value.fields()[index].value(),
466        tail,
467        records,
468        vocabulary,
469    )
