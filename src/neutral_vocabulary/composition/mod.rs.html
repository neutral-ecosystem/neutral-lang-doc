42/// Caller policy for independent composition and dependency resource budgets.
43#[derive(Clone, Copy, Debug, Eq, PartialEq)]
44pub struct CompositionLimits {
45    /// Existing strict per-bundle JSON, scalar, type and field budgets.
46    pub json: VocabularyLimits,
47    /// Maximum supplied bundles.
48    pub bundles: u64,
49    /// Maximum aggregate captured bytes.
50    pub captured_bytes: u64,
51    /// Maximum direct dependencies per bundle.

53    /// Maximum total dependency edges.
54    pub dependency_edges: u64,
55    /// Maximum number of bundles along one dependency path.
56    pub dependency_depth: u64,
57    /// Maximum aggregate nominal definitions.
58    pub total_types: u64,
59    /// Maximum aggregate record fields.
60    pub total_fields: u64,
61    /// Maximum alternatives in one variant.
62    pub alternatives_per_type: u64,
63    /// Maximum aggregate variant alternatives.
64    pub total_alternatives: u64,
65    /// Maximum finite choices in one field.
66    pub choices_per_field: u64,

150    /// Checks policy before processing input, rather than silently disabling a resource gate.
151    ///
152    /// # Errors
153    /// Returns [`CompositionError::InvalidLimits`] if any composition budget is zero.
154    pub fn validate(self) -> Result<(), CompositionError> {
155        if [
156            self.bundles,
157            self.captured_bytes,
158            self.dependencies_per_bundle,
159            self.dependency_edges,
160            self.dependency_depth,
161            self.total_types,
162            self.total_fields,
163            self.alternatives_per_type,
164            self.total_alternatives,
165            self.choices_per_field,
166            self.total_choices,
167            self.type_depth,
168            self.value_depth,
169            self.value_nodes,
170            self.work,
171        ]
172        .contains(&0)
173        {
174            Err(CompositionError::InvalidLimits)
175        } else {
176            Ok(())
177        }
178    }

190/// Bounded composition failure; errors never contain captured text or host paths.
191#[derive(Clone, Debug, Eq, PartialEq)]
192pub enum CompositionError {
193    /// Existing strict JSON/envelope/scalar failure.
194    Vocabulary(VocabularyError),
195    /// A caller supplied a zero independent limit.
196    InvalidLimits,
197    /// An independent caller or hard resource bound was exceeded.
198    Limit,
199    /// Cooperative cancellation, before any partial catalogue publication.
200    Cancelled,
201    /// Duplicate canonical identity or conflicting revision.
202    DuplicateBundle,
203    /// Missing root/dependency bundle or revision mismatch.
204    MissingDependency,
205    /// Supplied bundle is outside the required transitive closure.
206    ExtraBundle,
207    /// Dependency is duplicate, self-referential, or unused by any type.
208    InvalidDependency,
209    /// Direct bundle dependency cycle.
210    DependencyCycle,
211    /// Public signature or external target exposes a private type.
212    PrivateType,
213    /// Missing nominal owner/type or unsupported reference target.
214    UnknownType,
215    /// Embedded nominal types form a cycle, including unselected alternatives.
216    EmbeddedCycle,
217    /// Invalid tag/name/type/field/presence shape.
218    InvalidContract,
219    /// Invalid/incompatible/contradictory declarative restrictions.
220    InvalidRestrictions,
221    /// Duplicate finite scalar choice after normalization.
222    DuplicateChoice,
223    /// Closed default is incomplete, incompatible, or violates restrictions.
224    InvalidDefault,
225    /// Supplied value violates its selected public type or declarative restrictions.
226    InvalidValue,
227    /// A bounded fallible result/origin reservation failed.
228    Allocation,
229}

284/// Validates exact transitive lock cover, closed schemas, typing, defaults and public closure.
285///
286/// Roots are canonical identities/revisions required by source, not post-compilation
287/// view selections. Existing schema bundles are permitted only as dependency leaves.
288/// No partially validated catalogue is returned when any bundle fails.
289///
290/// # Errors
291/// Returns a classified schema/lock/closure/semantic/limit/cancellation failure.
292pub fn validate_composition_closure(
293    inputs: &[CapturedCompositionBundle<'_>],
294    roots: &[(&str, &str)],
295    limits: CompositionLimits,
296    cancellation: &CancellationToken,
297) -> Result<ValidatedComposition, CompositionError> {
298    limits.validate()?;
299    // A caller cannot remove the JSON stage's hard node/depth/object ceilings
300    // by making its semantic budgets generous. Every legitimate object has at
301    // most eight members; duplicate detection is therefore bounded per key.
302    let mut limits = limits;
303    limits.json.object_members = limits.json.object_members.min(MAX_OBJECT_MEMBERS);
304    limits.json.total_nodes = limits.json.total_nodes.min(MAX_WORK);
305    limits.json.array_items = limits.json.array_items.min(MAX_WORK);
306    limits.json.nesting_depth = limits.json.nesting_depth.min(MAX_DEPTH);
307    let mut budget = Budget::new(limits, cancellation);
308    budget.step(1)?;
309    check_count(inputs.len(), limits.bundles)?;
310    check_count(roots.len(), limits.bundles)?;
311    let mut bytes = 0_u64;
312    let mut owners = neutral_core::ordered::OrderedMap::new();
313    for input in inputs {
314        budget.step(input.bytes.len() as u64 + 1)?;
315        bytes = bytes
316            .checked_add(input.bytes.len() as u64)
317            .ok_or(CompositionError::Limit)?;
318        if bytes > limits.captured_bytes.min(MAX_CAPTURED_BYTES) {
319            return Err(CompositionError::Limit);
320        }
321        budget.step(owners.len() as u64)?;
322        budget.allocation()?;
323        if owners
324            .insert(input.lock.identity(), input.lock.version())
325            .map_err(|_| CompositionError::Allocation)?
326            .is_some()
327        {
328            return Err(CompositionError::DuplicateBundle);
329        }
330        if !VocabularyContentDigest::from_bytes(input.bytes)
331            .securely_matches(input.lock.content_digest())
332        {
333            return Err(VocabularyError::DigestMismatch.into());
334        }
335    }
336    let mut bundles = budget.storage(inputs.len())?;
337    for input in inputs {
338        bundles.push(decode::bundle(*input, &mut budget)?);
339    }
340    bundles.sort_unstable_by(|left, right| left.identity.identity().cmp(right.identity.identity()));
341    closure::validate(&bundles, roots, &mut budget)?;
342    values::validate_defaults(&mut bundles, &mut budget)?;
343    budget.step(1)?;
344    Ok(ValidatedComposition { bundles })
345}

347/// Per-request semantic budget; no process-global state can leak across requests.
348struct Budget<'a> {
349    /// Request-local reservation checkpoint count, never process-global.
350    allocations: usize,
351    /// Deterministic allocation failure or cancellation at one private checkpoint.
352    allocation_fault: Option<(usize, bool)>,
353    /// Complete caller policy.
354    limits: CompositionLimits,
355    /// Cancellation signal shared by all phases.
356    cancellation: &'a CancellationToken,
357    /// Total charged semantic/comparison/default work.
358    work: u64,
359    /// Total record fields decoded.
360    fields: u64,
361    /// Total nominal types decoded.
362    types: u64,
363    /// Total alternatives decoded.
364    alternatives: u64,
365    /// Total choices decoded.
366    choices: u64,
367    /// Total dependency edges decoded.
368    edges: u64,
369    /// Cumulative materialization visits, independent of aggregate work.
370    value_nodes: u64,
371    /// Standalone origins start at one; explicitly selected project /2 starts at zero.
372    origin_root_depth: u64,
373}
374
375impl<'a> Budget<'a> {
376    /// Starts isolated counters for one captured closure.
377    fn new(limits: CompositionLimits, cancellation: &'a CancellationToken) -> Self {
378        Self {
379            allocations: 0,
380            allocation_fault: None,
381            limits,
382            cancellation,
383            work: 0,
384            fields: 0,
385            types: 0,
386            alternatives: 0,
387            choices: 0,
388            edges: 0,
389            value_nodes: 0,
390            origin_root_depth: 1,
391        }
392    }
393    /// Charges work and observes cancellation before proportional semantic operations.
394    fn step(&mut self, amount: u64) -> Result<(), CompositionError> {
395        if self.cancellation.is_cancelled() {
396            return Err(CompositionError::Cancelled);
397        }
398        charge(&mut self.work, amount, self.limits.work)
399    }
400
401    /// Checks cancellation/work before an allocation and supports isolated private fault tests.
402    fn allocation(&mut self) -> Result<(), CompositionError> {
403        self.step(1)?;
404        {
405            let current = self.allocations;
406            self.allocations += 1;
407            if let Some((index, cancel)) = self.allocation_fault
408                && index == current
409            {
410                if cancel {
411                    self.cancellation.cancel();
412                    return Err(CompositionError::Cancelled);
413                }
414                return Err(CompositionError::Allocation);
415            }
416        }
417        Ok(())
418    }
419
420    /// Reserves complete outer storage after the caller has checked its independent count.
421    fn storage<T>(&mut self, count: usize) -> Result<Vec<T>, CompositionError> {
422        self.allocation()?;
423        let mut result = Vec::new();
424        result
425            .try_retain_exact(count)
426            .map_err(|_| CompositionError::Allocation)?;
427        self.step(0)?;
428        Ok(result)
429    }
430
431    /// Copies bounded exact UTF-8 bytes without infallible string growth.
432    fn owned(&mut self, value: &str) -> Result<String, CompositionError> {
433        self.allocation()?;
434        let result =
435            neutral_core::allocation::text(value).map_err(|_| CompositionError::Allocation)?;
436        self.step(0)?;
437        Ok(result)
438    }
439
440    /// Reserves a recursive wrapper after the caller has checked depth and shape.
441    fn boxed<T>(&mut self, value: T) -> Result<Box<T>, CompositionError> {
442        self.allocation()?;
443        let result =
444            neutral_core::allocation::boxed(value).map_err(|_| CompositionError::Allocation)?;
445        self.step(0)?;
446        Ok(result)
447    }
448
449    /// Charges ordered insertion shifts before reserving and retaining unique membership.
450    fn unique<K: Ord>(
451        &mut self,
452        set: &mut neutral_core::ordered::OrderedSet<K>,
453        key: K,
454    ) -> Result<bool, CompositionError> {
455        self.step(set.len() as u64)?;
456        self.allocation()?;
457        set.insert(key).map_err(|_| CompositionError::Allocation)
458    }
459    /// Intersects recursion policy with the stack-safe hard ceiling.
460    fn depth(&mut self, depth: u64, limit: u64) -> Result<(), CompositionError> {
461        self.step(1)?;
462        if depth > limit.min(MAX_DEPTH) {
463            Err(CompositionError::Limit)
464        } else {
465            Ok(())
466        }
467    }
468
469    /// Charges an independent output-value visit before reserving materialized storage.
470    fn value_node(&mut self) -> Result<(), CompositionError> {
471        self.step(1)?;
472        charge(&mut self.value_nodes, 1, self.limits.value_nodes)
473    }
474
475    /// Reserves conservative string-key inspection work before ordered insertion and sorting.
476    fn key(&mut self, bytes: usize, items: usize) -> Result<(), CompositionError> {
477        let factor = u64::from(items.max(1).ilog2()) + 1;
478        self.step(
479            (bytes as u64)
480                .checked_mul(factor)
481                .and_then(|n| n.checked_mul(4))
482                .ok_or(CompositionError::Limit)?,
483        )
484    }
485}
486
487/// Checks independent counts before reserving/decoding their collections.
488fn check_count(count: usize, limit: u64) -> Result<(), CompositionError> {
489    if count as u64 > limit.min(MAX_WORK) {
490        Err(CompositionError::Limit)
491    } else {
492        Ok(())
493    }
494}
495
496/// Increments a cumulative counter with checked arithmetic and a fixed hard ceiling.
497fn charge(counter: &mut u64, amount: u64, limit: u64) -> Result<(), CompositionError> {
498    *counter = counter.checked_add(amount).ok_or(CompositionError::Limit)?;
499    if *counter > limit.min(MAX_WORK) {
500        Err(CompositionError::Limit)
501    } else {
502        Ok(())
503    }
