17/// Complete immutable value plus origin classifications, never a compiled project artifact.
18#[derive(Clone, Debug, Eq, PartialEq)]
19pub struct ValidatedCompositionValue {
20    /// Canonical materialized meaning, separate from occurrence evidence.
21    pub(super) value: ClosedValue,
22    /// Ordered safe paths; no invented source span or host information.
23    pub(super) origins: Vec<ValueOrigin>,
24}
25
26impl ValidatedCompositionValue {
27    /// Returns immutable materialized meaning; compare this rather than origins for value equality.
28    #[must_use]
29    pub fn value(&self) -> &ClosedValue {
30        &self.value
31    }
32
33    /// Returns canonical root/field/list/payload occurrence facts without source attribution claims.
34    #[must_use]
35    pub fn origins(&self) -> &[ValueOrigin] {
36        &self.origins
37    }
38}

40/// Materializes a closed supplied value against one exact public vocabulary nominal contract.
41///
42/// Uses the same validator as bundle defaults, including every nested restriction,
43/// selected variant payload and omission rule. References cannot be supplied as
44/// closed values; nullable-reference null never creates a binding edge. This API
45/// does not acquire input, parse source, or select a project/wire/identity profile.
46///
47/// # Errors
48/// Returns a safe type/value/limit/cancellation/allocation classification and no partial result.
49pub fn materialize_composition_value(
50    catalogue: &ValidatedComposition,
51    owner: (&str, &str, &str),
52    supplied: &ClosedValue,
53    limits: CompositionLimits,
54    cancellation: &CancellationToken,
55) -> Result<ValidatedCompositionValue, E> {
56    limits.validate()?;
57    let mut budget = Budget::new(limits, cancellation);
58    budget.step(1)?;
59    // Public callers cannot use this boundary to materialize a private root.
60    // All its reachable public type closure was validated before publication.
61    let mut lookup = values::Catalogue::new();
62    for bundle in &catalogue.bundles {
63        for definition in &bundle.definitions {
64            budget.key(
65                bundle.identity.identity().len()
66                    + bundle.identity.version().len()
67                    + definition.name.len(),
68                lookup.len() + 1,
69            )?;
70            budget.step(lookup.len() as u64)?;
71            lookup.insert(
72                (
73                    bundle.identity.identity(),
74                    bundle.identity.version(),
75                    definition.name.as_str(),
76                ),
77                definition,
78            )?;
79        }
80    }
81    budget.key(owner.0.len() + owner.1.len() + owner.2.len(), lookup.len())?;
82    let root = lookup.get(&owner).ok_or(E::UnknownType)?;
83    if !root.public {
84        return Err(E::PrivateType);
85    }
86    let ty = T::VocabularyNominal {
87        identity: super::copy::text(owner.0)?,
88        version: super::copy::text(owner.1)?,
89        name: super::copy::text(owner.2)?,
90    };
91    let value = values::materialize(supplied, &ty, &lookup, &mut budget, 1).map_err(|error| {
92        if error == E::InvalidDefault {
93            E::InvalidValue
94        } else {
95            error
96        }
97    })?;
98    let mut origins = Vec::new();
99    classify(
100        Some(supplied),
101        Some(&value),
102        false,
103        &mut Vec::new(),
104        &mut origins,
105        &mut budget,
106    )?;
107    budget.step(1)?;
108    Ok(ValidatedCompositionValue { value, origins })
109}

111/// Classifies canonical final occurrences without equating optional absence with explicit null.
112pub(super) fn classify<R>(
113    raw: Option<&V<R>>,
114    value: Option<&V<R>>,
115    inherited_default: bool,
116    path: &mut Vec<P>,
117    origins: &mut Vec<ValueOrigin>,
118    budget: &mut Budget<'_>,
119) -> Result<(), E> {
120    budget.depth(
121        path.len() as u64 + budget.origin_root_depth,
122        budget.limits.value_depth,
123    )?;
124    super::check_count(origins.len() + 1, budget.limits.value_nodes)?;
125    let defaulted = inherited_default || (raw.is_none() && value.is_some());
126    let kind = if value.is_none() {
127        K::OmittedOptional
128    } else if defaulted {
129        K::Defaulted
130    } else if matches!(value, Some(V::Null)) {
131        K::ExplicitNull
132    } else {
133        K::Supplied
134    };
135    let path_bytes = path.iter().try_fold(1_u64, |total, segment| {
136        total
137            .checked_add(match segment {
138                P::Field(name) => name.len() as u64,
139                _ => 1,
140            })
141            .ok_or(E::Limit)
142    })?;
143    budget.step(path_bytes)?;
144    origins.try_retain(1).map_err(|_| E::Allocation)?;
145    origins.push(ValueOrigin {
146        path: super::copy::path(path)?,
147        kind,
148    });
149    match value {
150        Some(V::Record(fields)) => {
151            let raw_fields = if let Some(V::Record(fields)) = raw {
152                fields.as_slice()
153            } else {
154                &[]
155            };
156            let mut provided = BTreeMap::new();
157            for (name, value) in raw_fields {
158                budget.key(name.len(), raw_fields.len())?;
159                budget.step(provided.len() as u64)?;
160                provided
161                    .insert(name.as_str(), value.as_ref())
162                    .map_err(|_| E::Allocation)?;
163            }
164            for (name, value) in fields {
165                budget.key(name.len(), raw_fields.len())?;
166                descend(
167                    P::Field(super::copy::text(name)?),
168                    provided.get(name.as_str()).copied().flatten(),
169                    value.as_ref(),
170                    defaulted,
171                    path,
172                    origins,
173                    budget,
174                )?;
175            }
176        }
177        Some(V::List(items)) => {
178            let raw_items = if let Some(V::List(items)) = raw {
179                items.as_slice()
180            } else {
181                &[]
182            };
183            for (index, value) in items.iter().enumerate() {
184                descend(
185                    P::Element(index as u64),
186                    raw_items.get(index),
187                    Some(value),
188                    defaulted,
189                    path,
190                    origins,
191                    budget,
192                )?;
193            }
194        }
195        Some(V::Variant { payload, .. }) => {
196            let raw_payload = if let Some(V::Variant { payload, .. }) = raw {
197                Some(payload.as_ref())
198            } else {
199                None
200            };
201            descend(
202                P::Payload,
203                raw_payload,
204                Some(payload),
205                defaulted,
206                path,
207                origins,
208                budget,
209            )?;
210        }
211        _ => {}
212    }
213    Ok(())
214}

216/// Pushes one safe path segment and restores the caller path even when nested classification fails.
217fn descend<R>(
218    segment: P,
219    raw: Option<&V<R>>,
220    value: Option<&V<R>>,
221    defaulted: bool,
222    path: &mut Vec<P>,
223    origins: &mut Vec<ValueOrigin>,
224    budget: &mut Budget<'_>,
225) -> Result<(), E> {
226    path.try_retain(1).map_err(|_| E::Allocation)?;
227    path.push(segment);
228    let result = classify(raw, value, defaulted, path, origins, budget);
229    path.pop();
230    result
231}
