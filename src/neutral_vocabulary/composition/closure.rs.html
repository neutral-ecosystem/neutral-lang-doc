10/// Canonical vocabulary/revision/type identity, never a source alias.
11type TypeKey<'a> = (&'a str, &'a str, &'a str);
12
13/// Checks exact supplied closure and every type edge, including private/unselected definitions.
14pub(super) fn validate(
15    bundles: &[CompositionBundle],
16    roots: &[(&str, &str)],
17    budget: &mut Budget<'_>,
18) -> Result<(), E> {
19    let mut owners = BTreeMap::new();
20    for bundle in bundles {
21        budget.step(owners.len() as u64 + 1)?;
22        owners
23            .insert(bundle.identity.identity(), bundle)
24            .map_err(|_| E::Allocation)?;
25    }
26    // Check complete dependency availability/revisions/cycles before resolving
27    // type targets, so a missing locked owner is not misclassified as a type typo.
28    let mut heights = BTreeMap::new();
29    for bundle in bundles {
30        dependency_height(bundle.identity.identity(), &owners, &mut heights, budget)?;
31    }
32    let mut types = BTreeMap::new();
33    for bundle in bundles {
34        for definition in &bundle.definitions {
35            budget.step(1)?;
36            budget.step(types.len() as u64)?;
37            types
38                .insert(
39                    (
40                        bundle.identity.identity(),
41                        bundle.identity.version(),
42                        definition.name.as_str(),
43                    ),
44                    definition,
45                )
46                .map_err(|_| E::Allocation)?;
47        }
48    }
49    let mut embedded = BTreeMap::<TypeKey<'_>, BTreeSet<TypeKey<'_>>>::new();
50    for bundle in bundles {
51        let mut dependencies = BTreeMap::new();
52        for d in &bundle.dependencies {
53            budget.step(dependencies.len() as u64 + 1)?;
54            dependencies
55                .insert(d.identity.as_str(), d.version.as_str())
56                .map_err(|_| E::Allocation)?;
57        }
58        let mut used = BTreeSet::new();
59        for definition in &bundle.definitions {
60            let key = (
61                bundle.identity.identity(),
62                bundle.identity.version(),
63                definition.name.as_str(),
64            );
65            budget.step(embedded.len() as u64)?;
66            let targets = embedded.entry(key).map_err(|_| E::Allocation)?.or_default();
67            for ty in definition_types(definition) {
68                let mut stack = initial_stack((ty, false))?;
69                while let Some((ty, reference)) = stack.pop() {
70                    budget.step(1)?;
71                    match ty {
72                        T::List(inner) | T::Nullable(inner) => stack.push((inner, reference)),
73                        T::Ref(inner) => stack.push((inner, true)),
74                        T::VocabularyNominal {
75                            identity,
76                            version,
77                            name,
78                        } => {
79                            let target = (identity.as_str(), version.as_str(), name.as_str());
80                            let target_definition = types.get(&target).ok_or(E::UnknownType)?;
81                            if identity != bundle.identity.identity() {
82                                if dependencies.get(identity.as_str()) != Some(&version.as_str()) {
83                                    return Err(E::InvalidDependency);
84                                }
85                                budget.step(used.len() as u64)?;
86                                used.insert(identity.as_str()).map_err(|_| E::Allocation)?;
87                                if !target_definition.public {
88                                    return Err(E::PrivateType);
89                                }
90                            }
91                            if definition.public && !target_definition.public {
92                                return Err(E::PrivateType);
93                            }
94                            if !reference {
95                                budget.step(targets.len() as u64)?;
96                                targets.insert(target).map_err(|_| E::Allocation)?;
97                            }
98                        }
99                        T::Nominal(_) => return Err(E::UnknownType),
100                        _ => {}
101                    }
102                }
103            }
104        }
105        if dependencies.keys().any(|name| !used.contains(name)) {
106            return Err(E::InvalidDependency);
107        }
108    }
109    reject_embedded_cycles(&embedded, budget)?;
110    // Traverse *all* supplied bundle dependency paths, not just the first path
111    // reaching a shared leaf. The longest-path memo enforces diamond depth too.
112    validate_root_cover(&owners, roots, budget)
113}

115/// Verifies supplied bundles equal the union of direct source requirements and their dependencies.
116fn validate_root_cover(
117    owners: &BTreeMap<&str, &CompositionBundle>,
118    roots: &[(&str, &str)],
119    budget: &mut Budget<'_>,
120) -> Result<(), E> {
121    let mut required = BTreeSet::new();
122    let mut root_versions = BTreeMap::new();
123    let mut stack = Vec::new();
124    stack
125        .try_retain_exact(roots.len())
126        .map_err(|_| E::Allocation)?;
127    stack.extend_from_slice(roots);
128    while let Some((identity, version)) = stack.pop() {
129        budget.step(1)?;
130        budget.step(root_versions.len() as u64)?;
131        if root_versions
132            .insert(identity, version)
133            .map_err(|_| E::Allocation)?
134            .is_some_and(|previous| previous != version)
135        {
136            return Err(E::DuplicateBundle);
137        }
138        let bundle = owners.get(identity).ok_or(E::MissingDependency)?;
139        if bundle.identity.version() != version {
140            return Err(E::MissingDependency);
141        }
142        budget.step(required.len() as u64)?;
143        if required.insert(identity).map_err(|_| E::Allocation)? {
144            stack
145                .try_retain(bundle.dependencies.len())
146                .map_err(|_| E::Allocation)?;
147            stack.extend(
148                bundle
149                    .dependencies
150                    .iter()
151                    .map(|d| (d.identity.as_str(), d.version.as_str())),
152            );
153        }
154    }
155    if required.len() != owners.len() {
156        return Err(E::ExtraBundle);
157    }
158    Ok(())
159}

161/// Rejects embedded recursion with iterative enter/leave states rather than call-stack traversal.
162fn reject_embedded_cycles<'a>(
163    graph: &BTreeMap<TypeKey<'a>, BTreeSet<TypeKey<'a>>>,
164    budget: &mut Budget<'_>,
165) -> Result<(), E> {
166    let mut done = BTreeSet::new();
167    for root in graph.keys() {
168        let mut visiting = BTreeSet::new();
169        let mut stack = initial_stack((*root, false))?;
170        while let Some((key, leaving)) = stack.pop() {
171            budget.step(1)?;
172            if leaving {
173                budget.step(visiting.len() as u64 + done.len() as u64)?;
174                visiting.remove(&key);
175                done.insert(key).map_err(|_| E::Allocation)?;
176                continue;
177            }
178            if done.contains(&key) {
179                continue;
180            }
181            budget.step(visiting.len() as u64)?;
182            if !visiting.insert(key).map_err(|_| E::Allocation)? {
183                return Err(E::EmbeddedCycle);
184            }
185            stack
186                .try_retain(1 + graph.get(&key).map_or(0, BTreeSet::len))
187                .map_err(|_| E::Allocation)?;
188            stack.push((key, true));
189            if let Some(targets) = graph.get(&key) {
190                stack.extend(targets.iter().map(|target| (*target, false)));
191            }
192        }
193    }
194    Ok(())
195}

197/// Computes longest dependency paths with iterative memoization and exact revision checking.
198fn dependency_height<'a>(
199    root: &'a str,
200    owners: &BTreeMap<&'a str, &'a CompositionBundle>,
201    heights: &mut BTreeMap<&'a str, u64>,
202    budget: &mut Budget<'_>,
203) -> Result<(), E> {
204    let mut visiting = BTreeSet::new();
205    let mut stack = initial_stack((root, false))?;
206    while let Some((name, leaving)) = stack.pop() {
207        budget.step(1)?;
208        let bundle = owners.get(name).ok_or(E::MissingDependency)?;
209        if leaving {
210            let mut height = 1;
211            for dependency in &bundle.dependencies {
212                let child = heights
213                    .get(dependency.identity.as_str())
214                    .ok_or(E::DependencyCycle)?;
215                height = height.max(child.checked_add(1).ok_or(E::Limit)?);
216            }
217            if height > budget.limits.dependency_depth.min(super::MAX_DEPTH) {
218                return Err(E::Limit);
219            }
220            budget.step(heights.len() as u64 + visiting.len() as u64)?;
221            heights.insert(name, height).map_err(|_| E::Allocation)?;
222            visiting.remove(name);
223            continue;
224        }
225        if heights.contains_key(name) {
226            continue;
227        }
228        budget.step(visiting.len() as u64)?;
229        if !visiting.insert(name).map_err(|_| E::Allocation)? {
230            return Err(E::DependencyCycle);
231        }
232        stack
233            .try_retain(1 + bundle.dependencies.len())
234            .map_err(|_| E::Allocation)?;
235        stack.push((name, true));
236        for dependency in &bundle.dependencies {
237            let child = owners
238                .get(dependency.identity.as_str())
239                .ok_or(E::MissingDependency)?;
240            if child.identity.version() != dependency.version {
241                return Err(E::MissingDependency);
242            }
243            stack.push((child.identity.identity(), false));
244        }
245    }
246    Ok(())
247}

249/// Starts an iterative traversal with one fallible allocation rather than an infallible vector literal.
250fn initial_stack<T>(value: T) -> Result<Vec<T>, E> {
251    let mut stack = Vec::new();
252    stack.try_retain_exact(1).map_err(|_| E::Allocation)?;
253    stack.push(value);
254    Ok(stack)
255}
