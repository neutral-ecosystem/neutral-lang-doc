25/// Immutable materialized binding and its separate, occurrence-sensitive safe facts.
26#[derive(Clone, Eq, PartialEq)]
27pub struct ValidatedCompositionBinding {
28    /// Complete canonical meaning, never mutably exposed.
29    binding: CompositionBinding,
30    /// Supplied/null/default/absence classification, not fabricated source coordinates.
31    origins: Vec<ValueOrigin>,
32    /// Actual identity-only reference occurrences, in canonical value traversal order.
33    references: Vec<CompositionBindingReference>,
34}

104/// Identity-only target validation against the whole request, including later/cyclic bindings.
105struct References<'a> {
106    /// Every exact binding owner and its already checked signature/visibility.
107    index: &'a BTreeMap<&'a ModuleSymbolIdentity, &'a CompositionBinding>,
108    /// Consumer context for public and cross-module visibility checks.
109    consumer: &'a CompositionBinding,
110}
111
112impl values::ReferencePolicy<ModuleSymbolIdentity> for References<'_> {
113    /// Rejects dangling, differently typed or inaccessible targets without inspecting target values.
114    fn check(
115        &self,
116        target: &ModuleSymbolIdentity,
117        expected: &T,
118        budget: &mut Budget<'_>,
119    ) -> Result<(), E> {
120        budget.key(
121            target.module().module_name().len() + target.declaration_name().len(),
122            self.index.len(),
123        )?;
124        let target = self.index.get(target).ok_or(E::InvalidValue)?;
125        // Repeated references must pay for nominal-key/type comparison too;
126        // a long type name cannot amplify work behind a short binding name.
127        scope::preflight_type(expected, budget)?;
128        if &target.ty != expected {
129            return Err(E::InvalidValue);
130        }
131        if !target.public
132            && (self.consumer.public || target.owner.module() != self.consumer.owner.module())
133        {
134            return Err(E::PrivateType);
135        }
136        Ok(())
137    }
138}

140/// Validates and materializes every resolved binding atomically under cumulative request budgets.
141///
142/// Public roots have public nominal closure and public binding reference targets.
143/// Lists, nullable wrappers and selected variant payloads use the default validator;
144/// references are invariant identity edges and may form cycles without evaluation.
145/// Defaults remain closed by construction and optional absence is not null.
146/// Input order cannot affect typing, reference availability or final ordering.
147/// A caller must separately establish source imports and complete project companions.
148///
149/// # Errors
150/// Rejects malformed/duplicate owners, inaccessible/dangling/mismatched types or
151/// references, invalid defaults/constraints, limits, allocation or cancellation;
152/// never publishes the passing prefix of an invalid request.
153pub fn validate_composition_bindings(
154    scope: Arc<ValidatedCompositionScope>,
155    bindings: &[CompositionBinding],
156    limits: CompositionLimits,
157    cancellation: &CancellationToken,
158) -> Result<ValidatedCompositionBindings, E> {
159    limits.validate()?;
160    let mut budget = Budget::new(limits, cancellation);
161    budget.origin_root_depth = 0;
162    budget.step(1)?;
163    check_count(bindings.len(), limits.value_nodes)?;
164    let catalogue = recheck_scope(&scope, &mut budget)?;
165    let mut index = BTreeMap::new();
166    for binding in bindings {
167        let owner = &binding.owner;
168        budget.key(
169            owner.module().module_name().len() + owner.declaration_name().len(),
170            bindings.len(),
171        )?;
172        if owner.module().language_behavior_version() != V1_SOURCE_PROFILE
173            || !owner.module().module_name().split("::").all(is_snake_name)
174            || !is_snake_name(owner.declaration_name())
175            || is_protected_name(owner.declaration_name())
176        {
177            return Err(E::InvalidContract);
178        }
179        if owner.module().module_name().len() as u64 > limits.json.string_bytes()
180            || owner.declaration_name().len() as u64 > limits.json.string_bytes()
181        {
182            return Err(E::Limit);
183        }
184        scope::preflight_type(&binding.ty, &mut budget)?;
185        check_type_visibility(&binding.ty, binding, &catalogue, &mut budget)?;
186        budget.step(index.len() as u64)?;
187        if index
188            .insert(owner, binding)
189            .map_err(|_| E::Allocation)?
190            .is_some()
191        {
192            return Err(E::InvalidContract);
193        }
194    }
195    let mut result = Vec::new();
196    result
197        .try_retain(bindings.len())
198        .map_err(|_| E::Allocation)?;
199    for binding in index.values() {
200        let value = values::materialize_with(
201            &binding.value,
202            &binding.ty,
203            &catalogue,
204            &mut budget,
205            0,
206            &References {
207                index: &index,
208                consumer: binding,
209            },
210        )
211        .map_err(|error| {
212            if error == E::InvalidDefault {
213                E::InvalidValue
214            } else {
215                error
216            }
217        })?;
218        let mut origins = Vec::new();
219        supplied::classify(
220            Some(&binding.value),
221            Some(&value),
222            false,
223            &mut Vec::new(),
224            &mut origins,
225            &mut budget,
226        )?;
227        let mut references = Vec::new();
228        for origin in &origins {
229            if let Some(V::Reference(target)) = occurrence(&value, &origin.path, &mut budget)? {
230                budget.step(
231                    target.module().module_name().len() as u64
232                        + target.declaration_name().len() as u64,
233                )?;
234                references.try_retain(1).map_err(|_| E::Allocation)?;
235                references.push(CompositionBindingReference {
236                    path: origin.path.try_clone().map_err(|_| E::Allocation)?,
237                    target: target.try_clone().map_err(|_| E::Allocation)?,
238                });
239            }
240        }
241        result.push(ValidatedCompositionBinding {
242            binding: CompositionBinding {
243                owner: binding.owner.try_clone().map_err(|_| E::Allocation)?,
244                public: binding.public,
245                ty: binding.ty.try_clone().map_err(|_| E::Allocation)?,
246                value,
247            },
248            origins,
249            references,
250        });
251    }
252    budget.step(1)?;
253    Ok(ValidatedCompositionBindings {
254        scope,
255        bindings: result,
256    })
257}

259/// Applies stricter semantic policy to dormant contracts before materializing any supplied binding.
260///
261/// Captured-byte/digest policy belongs to capture, not this resolved model;
262/// original bundle bytes are deliberately not reconstructed from contracts.
263fn recheck_scope<'a>(
264    scope: &'a ValidatedCompositionScope,
265    budget: &mut Budget<'_>,
266) -> Result<values::Catalogue<'a>, E> {
267    let limits = budget.limits;
268    check_count(scope.catalogue().bundles().len(), limits.bundles)?;
269    for bundle in scope.catalogue().bundles() {
270        check_count(bundle.definitions.len(), limits.json.types())?;
271        check_count(bundle.dependencies.len(), limits.dependencies_per_bundle)?;
272        super::charge(
273            &mut budget.edges,
274            bundle.dependencies.len() as u64,
275            limits.dependency_edges,
276        )?;
277        for definition in &bundle.definitions {
278            scope::preflight(definition, false, budget)?;
279        }
280    }
281    let mut roots = Vec::new();
282    roots
283        .try_retain(scope.catalogue().bundles().len())
284        .map_err(|_| E::Allocation)?;
285    roots.extend(
286        scope
287            .catalogue()
288            .bundles()
289            .iter()
290            .map(|bundle| (bundle.identity.identity(), bundle.identity.version())),
291    );
292    closure::validate(scope.catalogue().bundles(), &roots, budget)?;
293    let mut module_types = BTreeMap::new();
294    for source in scope.sources() {
295        budget.step(module_types.len() as u64)?;
296        super::charge(
297            module_types
298                .entry(source.owner.module())
299                .map_err(|_| E::Allocation)?
300                .or_insert(0),
301            1,
302            limits.json.types(),
303        )?;
304        scope::preflight(&source.definition, true, budget)?;
305    }
306    let catalogue = scope::lookup(scope.catalogue(), scope.sources(), budget)?;
307    // Unused defaults remain subject to stricter bounds too; checking only
308    // supplied bindings would allow dormant invalid data through this boundary.
309    for definition in scope
310        .sources()
311        .iter()
312        .map(|source| &source.definition)
313        .chain(
314            scope
315                .catalogue()
316                .bundles()
317                .iter()
318                .flat_map(|bundle| &bundle.definitions),
319        )
320    {
321        if let neutral_ir::composition::CompositionBody::Record(fields) = &definition.body {
322            for field in fields {
323                let restrictions = values::validate_restrictions(field, budget)?;
324                if let Some(default) = &field.default {
325                    let value = values::materialize(default, &field.ty, &catalogue, budget, 0)?;
326                    values::check_restrictions(&restrictions, &value, budget)?;
327                }
328            }
329        }
330    }
331    Ok(catalogue)
332}

334/// Checks every type wrapper before public/external access to an exact nominal owner.
335fn check_type_visibility(
336    ty: &T,
337    binding: &CompositionBinding,
338    catalogue: &values::Catalogue<'_>,
339    budget: &mut Budget<'_>,
340) -> Result<(), E> {
341    let mut current = ty;
342    loop {
343        budget.step(1)?;
344        match current {
345            T::List(inner) | T::Nullable(inner) | T::Ref(inner) => current = inner,
346            T::Nominal(owner) => {
347                if !catalogue.resolve(current, budget)?.public
348                    && (binding.public || owner.module() != binding.owner.module())
349                {
350                    return Err(E::PrivateType);
351                }
352                return Ok(());
353            }
354            T::VocabularyNominal { .. } => {
355                if !catalogue.resolve(current, budget)?.public {
356                    return Err(E::PrivateType);
357                }
358                return Ok(());
359            }
360            _ => return Ok(()),
361        }
362    }
363}

365/// Resolves one already classified value occurrence without expanding any reference target.
366fn occurrence<'a>(
367    mut value: &'a V,
368    path: &[P],
369    budget: &mut Budget<'_>,
370) -> Result<Option<&'a V>, E> {
371    for segment in path {
372        budget.step(1)?;
373        value = match (segment, value) {
374            (P::Field(name), V::Record(fields)) => {
375                budget.key(name.len(), fields.len())?;
376                let index = fields
377                    .binary_search_by(|(field, _)| field.cmp(name))
378                    .map_err(|_| E::InvalidValue)?;
379                let Some(value) = &fields[index].1 else {
380                    return Ok(None);
381                };
382                value
383            }
384            (P::Element(index), V::List(items)) => items
385                .get(usize::try_from(*index).map_err(|_| E::Limit)?)
386                .ok_or(E::InvalidValue)?,
387            (P::Payload, V::Variant { payload, .. }) => payload,
388            _ => return Err(E::InvalidValue),
389        };
390    }
391    Ok(Some(value))
