18/// Validates canonical complete contracts without captured JSON, source parsing or compiler linkage.
19///
20/// All metadata, dependencies, public/embedded closure and unused defaults are
21/// checked independently. Unknown schemas and noncanonical materialization reject,
22/// rather than silently sorting or rewriting an external producer's data. Digests
23/// are evidence claims; complete project companion checks bind them separately.
24///
25/// # Errors
26/// Rejects malformed/noncanonical contracts, limits, allocation or cancellation
27/// before exposing any immutable catalogue or recursively cloning unbounded data.
28pub fn validate_composition_model(
29    bundles: &[CompositionBundle],
30    limits: CompositionLimits,
31    cancellation: &CancellationToken,
32) -> Result<ValidatedComposition, E> {
33    limits.validate()?;
34    let mut budget = Budget::new(limits, cancellation);
35    budget.step(1)?;
36    check_count(bundles.len(), limits.bundles)?;
37    let mut previous = None;
38    for bundle in bundles {
39        let identity = &bundle.identity;
40        budget.key(
41            identity.identity().len() + identity.version().len(),
42            bundles.len(),
43        )?;
44        if previous.is_some_and(|owner: (&str, &str)| {
45            owner >= (identity.identity(), identity.version()) || owner.0 == identity.identity()
46        }) {
47            return Err(E::InvalidContract);
48        }
49        previous = Some((identity.identity(), identity.version()));
50        if !is_upper_name(identity.identity())
51            || !is_exact_release_version(identity.version())
52            || identity.encoding_version() != ENCODING_VERSION
53        {
54            return Err(E::InvalidContract);
55        }
56        check_count(identity.required_features().len(), limits.json.features())?;
57        for text in [
58            identity.identity(),
59            identity.version(),
60            identity.schema_version(),
61            identity.encoding_version(),
62        ]
63        .into_iter()
64        .chain(identity.required_features().iter().map(String::as_str))
65        {
66            budget.step(text.len() as u64)?;
67            if text.len() as u64 > limits.json.string_bytes() {
68                return Err(E::Limit);
69            }
70        }
71        let legacy = identity.schema_version() == crate::PROJECT_VOCABULARY_SCHEMA_VERSION;
72        if !(legacy && identity.required_features().is_empty()
73            || identity.schema_version() == SCHEMA_VERSION
74                && identity.required_features() == [REQUIRED_FEATURE])
75        {
76            return Err(E::InvalidContract);
77        }
78        dependencies(bundle, legacy, &mut budget)?;
79        check_count(bundle.definitions.len(), limits.json.types())?;
80        let mut name = None;
81        for definition in &bundle.definitions {
82            if name.is_some_and(|previous| previous >= definition.name.as_str()) {
83                return Err(E::InvalidContract);
84            }
85            name = Some(definition.name.as_str());
86            scope::preflight(definition, false, &mut budget)?;
87            canonical_body(
88                &definition.body,
89                legacy,
90                identity.identity(),
91                identity.version(),
92            )?;
93        }
94    }
95    let mut roots = Vec::new();
96    roots.try_retain(bundles.len()).map_err(|_| E::Allocation)?;
97    roots.extend(
98        bundles
99            .iter()
100            .map(|bundle| (bundle.identity.identity(), bundle.identity.version())),
101    );
102    closure::validate(bundles, &roots, &mut budget)?;
103    let mut normalized = Vec::new();
104    normalized
105        .try_retain(bundles.len())
106        .map_err(|_| E::Allocation)?;
107    for bundle in bundles {
108        budget.step(1)?;
109        normalized.push(bundle.try_clone().map_err(|_| E::Allocation)?);
110    }
111    values::validate_defaults_policy(&mut normalized, &mut budget, true)?;
112    if normalized != bundles {
113        return Err(E::InvalidContract);
114    }
115    budget.step(1)?;
116    Ok(ValidatedComposition {
117        bundles: normalized,
118    })
119}

121/// Validates canonical exact dependency metadata before any graph traversal.
122fn dependencies(
123    bundle: &CompositionBundle,
124    legacy: bool,
125    budget: &mut Budget<'_>,
126) -> Result<(), E> {
127    check_count(
128        bundle.dependencies.len(),
129        budget.limits.dependencies_per_bundle,
130    )?;
131    super::charge(
132        &mut budget.edges,
133        bundle.dependencies.len() as u64,
134        budget.limits.dependency_edges,
135    )?;
136    if legacy && !bundle.dependencies.is_empty() {
137        return Err(E::InvalidContract);
138    }
139    let mut previous = None;
140    for dependency in &bundle.dependencies {
141        budget.key(
142            dependency.identity.len() + dependency.version.len(),
143            bundle.dependencies.len(),
144        )?;
145        if dependency.identity.len() as u64 > budget.limits.json.string_bytes()
146            || dependency.version.len() as u64 > budget.limits.json.string_bytes()
147        {
148            return Err(E::Limit);
149        }
150        if !is_upper_name(&dependency.identity)
151            || !is_exact_release_version(&dependency.version)
152            || dependency.identity == bundle.identity.identity()
153            || previous.is_some_and(|owner| owner >= (&dependency.identity, &dependency.version))
154        {
155            return Err(E::InvalidDependency);
156        }
157        previous = Some((&dependency.identity, &dependency.version));
158    }
159    Ok(())
160}

162/// Checks canonical member order and the intentionally narrow explicit legacy-leaf adapter.
163fn canonical_body(
164    body: &CompositionBody,
165    legacy: bool,
166    identity: &str,
167    version: &str,
168) -> Result<(), E> {
169    match body {
170        CompositionBody::Record(fields) => {
171            if fields.windows(2).any(|pair| pair[0].name >= pair[1].name) {
172                return Err(E::InvalidContract);
173            }
174            for field in fields {
175                if legacy
176                    && (field.presence != FieldPresence::Required
177                        || field.default.is_some()
178                        || field.restrictions != FieldRestrictions::default()
179                        || !matches!(&field.ty, T::Num | T::String | T::Bool | T::Url | T::Path)
180                            && !matches!(&field.ty, T::VocabularyNominal { identity: i, version: v, .. } if i == identity && v == version))
181                {
182                    return Err(E::InvalidContract);
183                }
184            }
185        }
186        CompositionBody::Variant(alternatives) => {
187            if legacy
188                || alternatives
189                    .windows(2)
190                    .any(|pair| pair[0].tag >= pair[1].tag)
191            {
192                return Err(E::InvalidContract);
193            }
194        }
195    }
196    Ok(())
197}
