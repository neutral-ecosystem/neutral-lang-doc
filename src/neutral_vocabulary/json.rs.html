3//! Minimal strict bounded JSON decoder for untrusted vocabulary bundles.
4
5use crate::{VocabularyError, VocabularyLimits};
6
7/// Untrusted JSON tree retained only until closed-schema validation succeeds.
8#[derive(Clone, Debug, Eq, PartialEq)]
9pub(crate) enum JsonValue {
10    /// JSON null.
11    Null,
12    /// JSON Boolean.
13    Bool(bool),
14    /// Decoded Unicode string.
15    String(String),
16    /// Ordered JSON array.
17    Array(Vec<JsonValue>),
18    /// Ordered object members retained without map collapse.
19    Object(Vec<(String, JsonValue)>),
20}
21
22/// Parses one complete strict JSON value under explicit allocation limits.
23pub(crate) fn parse(text: &str, limits: VocabularyLimits) -> Result<JsonValue, VocabularyError> {
24    let mut parser = Parser {
25        text,
26        index: 0,
27        limits,
28        nodes: 0,
29    };
30    parser.skip_whitespace();
31    let value = parser.parse_value(1)?;
32    parser.skip_whitespace();
33    if parser.index == text.len() {
34        Ok(value)
35    } else {
36        Err(VocabularyError::MalformedJson)
37    }
38}

40/// Stateful byte-offset parser over one already validated UTF-8 string.
41struct Parser<'a> {
42    /// Complete untrusted JSON text.
43    text: &'a str,
44    /// Current UTF-8 byte offset.
45    index: usize,
46    /// Frozen resource limits.
47    limits: VocabularyLimits,
48    /// Total parsed JSON value nodes.
49    nodes: u64,
50}
51
52impl Parser<'_> {
53    /// Parses one JSON value and rejects raw numeric tokens.
54    fn parse_value(&mut self, depth: u64) -> Result<JsonValue, VocabularyError> {
55        if depth > self.limits.nesting_depth() {
56            return Err(VocabularyError::JsonLimitExceeded);
57        }
58        self.nodes = self
59            .nodes
60            .checked_add(1)
61            .ok_or(VocabularyError::JsonLimitExceeded)?;
62        if self.nodes > self.limits.total_nodes() {
63            return Err(VocabularyError::JsonLimitExceeded);
64        }
65        match self.peek_byte() {
66            Some(b'n') => self.parse_literal(b"null", JsonValue::Null),
67            Some(b't') => self.parse_literal(b"true", JsonValue::Bool(true)),
68            Some(b'f') => self.parse_literal(b"false", JsonValue::Bool(false)),
69            Some(b'"') => self.parse_string().map(JsonValue::String),
70            Some(b'[') => self.parse_array(depth),
71            Some(b'{') => self.parse_object(depth),
72            Some(b'-' | b'0'..=b'9') => Err(VocabularyError::RawJsonNumberForbidden),
73            _ => Err(VocabularyError::MalformedJson),
74        }
75    }
76
77    /// Parses one fixed ASCII literal.
78    fn parse_literal(
79        &mut self,
80        expected: &[u8],
81        value: JsonValue,
82    ) -> Result<JsonValue, VocabularyError> {
83        let remaining = &self.text.as_bytes()[self.index..];
84        if !remaining.starts_with(expected) {
85            return Err(VocabularyError::MalformedJson);
86        }
87        self.index += expected.len();
88        Ok(value)
89    }
90
91    /// Parses one bounded array while preserving logical order.
92    fn parse_array(&mut self, depth: u64) -> Result<JsonValue, VocabularyError> {
93        self.index += 1;
94        self.skip_whitespace();
95        let mut items = Vec::new();
96        if self.consume_byte(b']') {
97            return Ok(JsonValue::Array(items));
98        }
99        loop {
100            if u64::try_from(items.len()).unwrap_or(u64::MAX) >= self.limits.array_items() {
101                return Err(VocabularyError::JsonLimitExceeded);
102            }
103            items.push(self.parse_value(depth.saturating_add(1))?);
104            self.skip_whitespace();
105            if self.consume_byte(b']') {
106                return Ok(JsonValue::Array(items));
107            }
108            if !self.consume_byte(b',') {
109                return Err(VocabularyError::MalformedJson);
110            }
111            self.skip_whitespace();
112        }
113    }
114
115    /// Parses one bounded object and rejects duplicate keys before map collapse.
116    fn parse_object(&mut self, depth: u64) -> Result<JsonValue, VocabularyError> {
117        self.index += 1;
118        self.skip_whitespace();
119        let mut members: Vec<(String, JsonValue)> = Vec::new();
120        if self.consume_byte(b'}') {
121            return Ok(JsonValue::Object(members));
122        }
123        loop {
124            if u64::try_from(members.len()).unwrap_or(u64::MAX) >= self.limits.object_members() {
125                return Err(VocabularyError::JsonLimitExceeded);
126            }
127            if self.peek_byte() != Some(b'"') {
128                return Err(VocabularyError::MalformedJson);
129            }
130            let name = self.parse_string()?;
131            if members.iter().any(|(existing, _)| existing == &name) {
132                return Err(VocabularyError::DuplicateJsonMember);
133            }
134            self.skip_whitespace();
135            if !self.consume_byte(b':') {
136                return Err(VocabularyError::MalformedJson);
137            }
138            self.skip_whitespace();
139            let value = self.parse_value(depth.saturating_add(1))?;
140            members.push((name, value));
141            self.skip_whitespace();
142            if self.consume_byte(b'}') {
143                return Ok(JsonValue::Object(members));
144            }
145            if !self.consume_byte(b',') {
146                return Err(VocabularyError::MalformedJson);
147            }
148            self.skip_whitespace();
149        }
150    }
151
152    /// Parses and decodes one bounded JSON string including surrogate pairs.
153    fn parse_string(&mut self) -> Result<String, VocabularyError> {
154        if !self.consume_byte(b'"') {
155            return Err(VocabularyError::MalformedJson);
156        }
157        let mut output = String::new();
158        loop {
159            let byte = self.peek_byte().ok_or(VocabularyError::MalformedJson)?;
160            match byte {
161                b'"' => {
162                    self.index += 1;
163                    return Ok(output);
164                }
165                b'\\' => {
166                    self.index += 1;
167                    self.parse_escape(&mut output)?;
168                }
169                0x00..=0x1f => return Err(VocabularyError::MalformedJson),
170                0x20..=0x7f => {
171                    self.index += 1;
172                    output.push(char::from(byte));
173                }
174                _ => {
175                    let character = self.text[self.index..]
176                        .chars()
177                        .next()
178                        .ok_or(VocabularyError::MalformedJson)?;
179                    self.index += character.len_utf8();
180                    output.push(character);
181                }
182            }
183            if u64::try_from(output.len()).unwrap_or(u64::MAX) > self.limits.string_bytes() {
184                return Err(VocabularyError::JsonLimitExceeded);
185            }
186        }
187    }
188
189    /// Decodes one JSON escape into a bounded output string.
190    fn parse_escape(&mut self, output: &mut String) -> Result<(), VocabularyError> {
191        let escaped = self.take_byte().ok_or(VocabularyError::MalformedJson)?;
192        match escaped {
193            b'"' => output.push('"'),
194            b'\\' => output.push('\\'),
195            b'/' => output.push('/'),
196            b'b' => output.push('\u{0008}'),
197            b'f' => output.push('\u{000c}'),
198            b'n' => output.push('\n'),
199            b'r' => output.push('\r'),
200            b't' => output.push('\t'),
201            b'u' => {
202                let first = self.parse_hex_quad()?;
203                let scalar = if (0xd800..=0xdbff).contains(&first) {
204                    if self.take_byte() != Some(b'\\') || self.take_byte() != Some(b'u') {
205                        return Err(VocabularyError::MalformedJson);
206                    }
207                    let second = self.parse_hex_quad()?;
208                    if !(0xdc00..=0xdfff).contains(&second) {
209                        return Err(VocabularyError::MalformedJson);
210                    }
211                    0x1_0000 + ((u32::from(first) - 0xd800) << 10) + (u32::from(second) - 0xdc00)
212                } else if (0xdc00..=0xdfff).contains(&first) {
213                    return Err(VocabularyError::MalformedJson);
214                } else {
215                    u32::from(first)
216                };
217                output.push(char::from_u32(scalar).ok_or(VocabularyError::MalformedJson)?);
218            }
219            _ => return Err(VocabularyError::MalformedJson),
220        }
221        Ok(())
222    }
223
224    /// Parses exactly four lowercase or uppercase JSON hexadecimal digits.
225    fn parse_hex_quad(&mut self) -> Result<u16, VocabularyError> {
226        let mut value = 0_u16;
227        for _ in 0..4 {
228            let digit = self.take_byte().ok_or(VocabularyError::MalformedJson)?;
229            let digit = match digit {
230                b'0'..=b'9' => u16::from(digit - b'0'),
231                b'a'..=b'f' => u16::from(digit - b'a' + 10),
232                b'A'..=b'F' => u16::from(digit - b'A' + 10),
233                _ => return Err(VocabularyError::MalformedJson),
234            };
235            value = (value << 4) | digit;
236        }
237        Ok(value)
238    }
239
240    /// Skips only RFC JSON insignificant whitespace bytes.
241    fn skip_whitespace(&mut self) {
242        while matches!(self.peek_byte(), Some(b' ' | b'\t' | b'\r' | b'\n')) {
243            self.index += 1;
244        }
245    }
246
247    /// Consumes one exact byte when present.
248    fn consume_byte(&mut self, expected: u8) -> bool {
249        if self.peek_byte() == Some(expected) {
250            self.index += 1;
251            true
252        } else {
253            false
254        }
255    }
256
257    /// Returns and advances past one byte.
258    fn take_byte(&mut self) -> Option<u8> {
259        let byte = self.peek_byte()?;
260        self.index += 1;
261        Some(byte)
262    }
263
264    /// Returns the next byte without advancing.
265    fn peek_byte(&self) -> Option<u8> {
266        self.text.as_bytes().get(self.index).copied()
267    }
