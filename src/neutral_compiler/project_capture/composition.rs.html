5//! Sources are captured and their headers checked, not semantically compiled.

32/// Closed successor request with explicit feature selection and independent policy.
33///
34/// The inner envelope must use [`profile::CAPTURE_REQUEST_VERSION`]. Keeping
35/// this separate type prevents passing composition capture to `compile_project`
36/// or treating successful header/catalogue validation as successful compilation.
37#[derive(Clone)]
38pub struct CapturedCompositionProjectRequest {
39    /// Exact bytes, locks and existing capture policy, never acquisition instructions.
40    envelope: CapturedProjectRequest,
41    /// Caller-supplied feature set; validated exactly, never sorted or inferred.
42    required_features: Vec<String>,
43    /// Independent catalogue, default and dependency resource policy.
44    composition_limits: CompositionLimits,
45}
46
47impl CapturedCompositionProjectRequest {
48    /// Creates an explicit data-only request; validation happens at capture, not construction.
49    #[must_use]
50    pub fn new(
51        envelope: CapturedProjectRequest,
52        required_features: Vec<String>,
53        composition_limits: CompositionLimits,
54    ) -> Self {
55        Self {
56            envelope,
57            required_features,
58            composition_limits,
59        }
60    }
61}

75/// Complete immutable capture plus validated catalogue, **not** a compiled project /2 artifact.
76///
77/// The old compiler cannot silently reinterpret this successor capture:
78///
79/// ```compile_fail,E0308
80/// # let captured: &neutral_compiler::CapturedCompositionProject = todo!();
81/// neutral_compiler::compile_project(captured, &neutral_core::CancellationToken::new());
82/// ```
83#[derive(Clone)]
84pub struct CapturedCompositionProject {
85    /// Reuses exact byte ownership only; never exposes a downcast to the /1 compiler.
86    captured: CapturedProject,
87    /// Fully checked transitive vocabulary contracts shared with independent consumers.
88    catalogue: Arc<ValidatedComposition>,
89    /// Independent policy preserved for replay and later compilation.
90    composition_limits: CompositionLimits,
91    /// Module-local alias -> canonical vocabulary identity; no alias enters a type owner.
92    aliases: Arc<OrderedMap<String, OrderedMap<String, String>>>,
93    /// Exact selected feature set, immutable and retained independently of source syntax.
94    required_features: Arc<Vec<String>>,
95}

98    /// Excludes source bytes, private contracts, aliases and host paths from debug output.

100        formatter
101            .debug_struct("CapturedCompositionProject")
102            .field("resources", &self.captured.resource_facts())
103            .finish_non_exhaustive()
104    }
105}

118    }
119
120    /// Returns existing independent capture policy without mutable cancellation state.
121    #[must_use]
122    pub const fn limits(&self) -> ProjectCaptureLimits {
123        self.captured.limits()
124    }
125
126    /// Returns explicit capture controls in the immutable derivation transcript order.
127    #[must_use]
128    pub fn identity_capture_limits(
129        &self,
130    ) -> [u64; neutral_ir::project_identity::CAPTURE_LIMIT_TAGS.len()] {
131        self.captured.identity_capture_limits()
132    }
133
134    /// Returns every independent composition budget, preserved exactly for replay.
135    #[must_use]
136    pub const fn composition_limits(&self) -> CompositionLimits {
137        self.composition_limits
138    }
139
140    /// Returns the exact validated capability set; a source header never infers it.
141    #[must_use]
142    pub fn required_features(&self) -> &[String] {
143        &self.required_features
144    }
145
146    /// Shares the already retained exact feature set without copying host strings.
147    pub(crate) fn shared_features(&self) -> Arc<Vec<String>> {
148        self.required_features.clone()
149    }
150
151    /// Projects verified input facts into the frozen captured identity /2 transcript.
152    ///
153    /// # Errors
154    /// Rejects independent identity bounds, allocation failure or cancellation.
155    /// This is not logical/project/artifact identity or semantic compilation evidence.
156    pub fn identity_transcript(
157        &self,
158        limits: IdentityLimits,
159        cancellation: &CancellationToken,
160    ) -> Result<IdentityTranscript<CompositionCapturedClosureIdentity>, IdentityError> {
161        let (sources, vocabularies) = crate::project_identity::capture_facts(
162            self.sources(),
163            self.vocabularies(),
164            limits,
165            cancellation,
166        )?;
167        captured_composition_closure(
168            &CapturedIdentityInput {
169                profile: self.captured.profile().source_version(),
170                sources: &sources,
171                vocabularies: &vocabularies,
172            },
173            self.required_features(),
174            limits,
175            cancellation,
176        )
177    }
178
179    /// Returns exact accepted source and complete-closure byte/count facts.
180    #[must_use]
181    pub const fn resource_facts(&self) -> ProjectCaptureResourceFacts {
182        self.captured.resource_facts()
183    }
184
185    /// Shares validated contracts without reparsing input, copying defaults or linking a reader.
186    #[must_use]
187    pub fn catalogue(&self) -> Arc<ValidatedComposition> {
188        Arc::clone(&self.catalogue)
189    }
190
191    /// Resolves a module-local source alias to its canonical exact vocabulary owner.
192    ///
193    /// Returns no contract when the module or alias is absent. Transitive dependencies
194    /// are not implicit source aliases; source must explicitly declare a `use`.
195    #[must_use]
196    pub fn vocabulary_for_alias(&self, module: &str, alias: &str) -> Option<&VocabularyIdentity> {
197        let identity = self.aliases.get(module)?.get(alias)?;
198        let bundles = self.catalogue.bundles();
199        let index = bundles
200            .binary_search_by(|bundle| bundle.identity.identity().cmp(identity))
201            .ok()?;
202        Some(&bundles[index].identity)
203    }
204
205    /// Builds the bounded import graph only; this does not validate source variant/value semantics.
206    ///
207    /// # Errors
208    /// Rejects invalid imports, graph bounds or cancellation without publishing a partial graph.
209    pub fn module_graph(
210        &self,
211        cancellation: &CancellationToken,
212    ) -> Result<ModuleGraph, ModuleGraphFailure> {
213        self.captured.module_graph(cancellation)
214    }
215
216    /// Reconstructs exact bytes, locks, features and policies under the explicit successor envelope.
217    ///
218    /// # Errors
219    /// Rejects allocation failure or cancellation without publishing a partial request.
220    pub fn replay_request(
221        &self,
222        cancellation: CancellationToken,
223    ) -> Result<CapturedCompositionProjectRequest, ProjectCaptureError> {
224        capture_cancelled(Some(&cancellation))?;
225        let mut sources = Vec::new();
226        sources
227            .try_retain_exact(self.sources().len())
228            .map_err(|_| ProjectCaptureError::LimitExceeded)?;
229        for source in self.sources() {
230            capture_cancelled(Some(&cancellation))?;
231            sources.push(
232                super::CapturedSourceInput::new(
233                    text(source.source_id()).map_err(|_| ProjectCaptureError::LimitExceeded)?,
234                    text(source.module_id()).map_err(|_| ProjectCaptureError::LimitExceeded)?,
235                    replay_bytes(source.bytes(), &cancellation)?,
236                )
237                .requiring_digest(source.digest()),
238            );
239        }
240        let mut vocabularies = Vec::new();
241        vocabularies
242            .try_retain_exact(self.vocabularies().len())
243            .map_err(|_| ProjectCaptureError::LimitExceeded)?;
244        for vocabulary in self.vocabularies() {
245            capture_cancelled(Some(&cancellation))?;
246            vocabularies.push(super::CapturedVocabularyInput::new(
247                replay_bytes(vocabulary.bytes(), &cancellation)?,
248                vocabulary
249                    .lock()
250                    .try_clone()
251                    .map_err(|_| ProjectCaptureError::LimitExceeded)?,
252            ));
253        }
254        let version = text(profile::CAPTURE_REQUEST_VERSION)
255            .map_err(|_| ProjectCaptureError::LimitExceeded)?;
256        let features = self
257            .required_features
258            .try_clone()
259            .map_err(|_| ProjectCaptureError::LimitExceeded)?;
260        capture_cancelled(Some(&cancellation))?;
261        let envelope = CapturedProjectRequest::new(
262            version,
263            self.captured.profile,
264            sources,
265            vocabularies,
266            super::ProjectCaptureControls::new(self.limits(), cancellation),
267        );
268        Ok(CapturedCompositionProjectRequest::new(
269            envelope,
270            features,
271            self.composition_limits,
272        ))
273    }
274}

276/// Copies already bounded exact captured bytes, checking cancellation before retention and copying.
277fn replay_bytes(bytes: &[u8], cancel: &CancellationToken) -> Result<Vec<u8>, ProjectCaptureError> {
278    capture_cancelled(Some(cancel))?;
279    let mut copied = Vec::new();
280    copied
281        .try_retain_exact(bytes.len())
282        .map_err(|_| ProjectCaptureError::LimitExceeded)?;
283    capture_cancelled(Some(cancel))?;
284    copied.extend_from_slice(bytes);
285    Ok(copied)
286}

288/// Whole-request rejection with phase-specific codes and no raw input or host information.
289#[derive(Clone, Debug, Eq, PartialEq)]
290pub enum CompositionCaptureFailure {
291    /// Existing capture envelope/header/integrity/resource classification.
292    Capture(ProjectCaptureError),
293    /// Strict vocabulary schema, closure, defaults, restrictions or public-type failure.
294    Catalogue(CompositionError),
295}
296
297impl CompositionCaptureFailure {
298    /// Returns the stable diagnostic identifier of the phase that rejected the complete request.
299    #[must_use]
300    pub fn code(&self) -> &'static str {
301        match self {
302            Self::Capture(error) => error.code(),
303            Self::Catalogue(error) => error.diagnostic_code(),
304        }
305    }
306}
307
308impl From<ProjectCaptureError> for CompositionCaptureFailure {
309    /// Preserves capture failure precedence and its safe diagnostic classification.
310    fn from(error: ProjectCaptureError) -> Self {
311        Self::Capture(error)
312    }
313}

315impl From<CompositionError> for CompositionCaptureFailure {
316    /// Preserves catalogue failure classification without introducing invented source attribution.
317    fn from(error: CompositionError) -> Self {
318        Self::Catalogue(error)
319    }
320}

322/// Captures an explicitly selected /2 request and validates its complete vocabulary closure.
323///
324/// This is the no-I/O capture boundary, not source compilation or a binary decoder.
325/// The existing /1 API remains strict and cannot accept this output type.
326///
327/// # Errors
328/// Rejects unsupported features/envelopes, invalid budgets, cancellation, integrity,
329/// malformed headers or any invalid/extra/missing transitive catalogue input.
330pub fn capture_composition_project(
331    request: CapturedCompositionProjectRequest,
332) -> Result<CapturedCompositionProject, CompositionCaptureFailure> {
333    capture_with_checkpoints(request, |_| {})
334}

336/// Validates ordered phases with private deterministic cancellation handoffs for tests.
337fn capture_with_checkpoints(
338    request: CapturedCompositionProjectRequest,
339    mut checkpoint: impl FnMut(ProjectCaptureCheckpoint),
340) -> Result<CapturedCompositionProject, CompositionCaptureFailure> {
341    let envelope = request.envelope;
342    // Unsupported selection wins over zero controls or an already-cancelled
343    // request. Never normalize features, infer them from source or retry /1.
344    if envelope.request_version != profile::CAPTURE_REQUEST_VERSION
345        || !request
346            .required_features
347            .iter()
348            .map(String::as_str)
349            .eq(profile::REQUIRED_FEATURES.iter().copied())
350    {
351        return Err(ProjectCaptureError::InvalidRequest.into());
352    }
353    if envelope.profile != LanguageProfile::V1_0 {
354        return Err(ProjectCaptureError::ProfileMismatch.into());
355    }
356    if !envelope.controls.limits.all_nonzero() || request.composition_limits.validate().is_err() {
357        return Err(ProjectCaptureError::LimitExceeded.into());
358    }
359    let cancellation = &envelope.controls.cancellation;
360    checkpoint(ProjectCaptureCheckpoint::Start);
361    capture_cancelled(Some(cancellation))?;
362    validate_envelope_version(&envelope, profile::CAPTURE_REQUEST_VERSION)?;
363    // Intersect successor aggregate ceilings before reserving per-bundle storage
364    // or hashing/parsing oversized inputs; generous capture policy is not a bypass.
365    validate_catalogue_capture_bounds(&envelope, request.composition_limits)?;
366    let values = envelope.controls.limits.values();
367    let mut sources = validate_sources_cancellable(envelope.sources, values, Some(cancellation))?;
368    validate_vocabularies_cancellable(&envelope.vocabularies, values, Some(cancellation))?;
369    checkpoint(ProjectCaptureCheckpoint::Integrity);
370    capture_cancelled(Some(cancellation))?;
371    let requirements =
372        validate_headers_policy(&sources, envelope.profile, true, Some(cancellation))?;
373    checkpoint(ProjectCaptureCheckpoint::Headers);
374    capture_cancelled(Some(cancellation))?;
375
376    let mut roots = Vec::new();
377    roots
378        .try_retain(requirements.len())
379        .map_err(|_| CompositionError::Allocation)?;
380    for identity in &requirements {
381        capture_cancelled(Some(cancellation))?;
382        let lock = envelope
383            .vocabularies
384            .iter()
385            .map(|input| &input.lock)
386            .find(|lock| lock.identity() == identity)
387            .ok_or(ProjectCaptureError::MissingVocabulary)?;
388        roots.push((lock.identity(), lock.version()));
389    }
390    let mut inputs = Vec::new();
391    inputs
392        .try_retain(envelope.vocabularies.len())
393        .map_err(|_| CompositionError::Allocation)?;
394    inputs.extend(
395        envelope
396            .vocabularies
397            .iter()
398            .map(|input| CapturedCompositionBundle {
399                bytes: &input.bytes,
400                lock: &input.lock,
401            }),
402    );
403    let catalogue =
404        validate_composition_closure(&inputs, &roots, request.composition_limits, cancellation)?;
405    let aliases = capture_aliases(
406        sources.iter().map(|(source, _)| source),
407        request.composition_limits.work,
408        cancellation,
409    )?;
410    checkpoint(ProjectCaptureCheckpoint::Publish);
411    capture_cancelled(Some(cancellation))?;
412    let captured = freeze_project(
413        cancellation,
414        envelope.profile,
415        envelope.controls.limits,
416        &mut sources,
417        envelope.vocabularies,
418    )?;
419    let result = CapturedCompositionProject {
420        captured,
421        catalogue: Arc::try_new(catalogue).map_err(|_| ProjectCaptureError::LimitExceeded)?,
422        composition_limits: request.composition_limits,
423        aliases: Arc::try_new(aliases).map_err(|_| ProjectCaptureError::LimitExceeded)?,
424        required_features: Arc::try_new(request.required_features)
425            .map_err(|_| ProjectCaptureError::LimitExceeded)?,
426    };
427    capture_cancelled(Some(cancellation))?;
428    Ok(result)
429}

431/// Retains module-local aliases fallibly, charging sorted-index shifts before insertion.
432fn capture_aliases<'a>(
433    sources: impl Iterator<Item = &'a super::CapturedSourceInput>,
434    mut work: u64,
435    cancellation: &CancellationToken,
436) -> Result<OrderedMap<String, OrderedMap<String, String>>, ProjectCaptureError> {
437    let mut aliases = OrderedMap::new();
438    for source in sources {
439        capture_cancelled(Some(cancellation))?;
440        let text =
441            std::str::from_utf8(&source.bytes).map_err(|_| ProjectCaptureError::InvalidHeader)?;
442        let mut local = OrderedMap::new();
443        for (identity, alias) in
444            scan_vocabulary_requirements_policy(text, true, Some(cancellation))?
445        {
446            capture_cancelled(Some(cancellation))?;
447            work = work
448                .checked_sub(local.len() as u64 + 1)
449                .ok_or(ProjectCaptureError::LimitExceeded)?;
450            local
451                .insert(alias, identity)
452                .map_err(|_| ProjectCaptureError::LimitExceeded)?;
453        }
454        work = work
455            .checked_sub(aliases.len() as u64 + 1)
456            .ok_or(ProjectCaptureError::LimitExceeded)?;
457        aliases
458            .insert(
459                neutral_core::allocation::text(&source.module_id)
460                    .map_err(|_| ProjectCaptureError::LimitExceeded)?,
461                local,
462            )
463            .map_err(|_| ProjectCaptureError::LimitExceeded)?;
464    }
465    Ok(aliases)
466}

468/// Enforces catalogue aggregate count/byte ceilings before proportional capture work.
469fn validate_catalogue_capture_bounds(
470    envelope: &CapturedProjectRequest,
471    limits: CompositionLimits,
472) -> Result<(), CompositionCaptureFailure> {
473    if super::exceeds(envelope.vocabularies.len(), limits.bundles.min(MAX_WORK)) {
474        return Err(CompositionError::Limit.into());
475    }
476    let mut total = 0_u64;
477    for input in &envelope.vocabularies {
478        capture_cancelled(Some(&envelope.controls.cancellation))?;
479        total = total
480            .checked_add(super::length(input.bytes.len()))
481            .ok_or(CompositionError::Limit)?;
482        if total > limits.captured_bytes.min(MAX_CAPTURED_BYTES) {
483            return Err(CompositionError::Limit.into());
484        }
485    }
486    Ok(())
487}
