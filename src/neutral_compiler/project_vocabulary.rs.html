3//! Canonical project vocabulary validation over an already captured closure.
4
5use crate::{CapturedProject, project_capture::scan_vocabulary_requirements};
6use neutral_core::StructuralLimits;
7use neutral_vocabulary::{
8    ProjectVocabulary, VocabularyError, VocabularyLimits, validate_project_bundle,
9};
10use std::collections::BTreeMap;
11
12/// The complete validated vocabulary set; aliases remain source-local only.
13#[derive(Clone, Debug, Eq, PartialEq)]
14pub struct ProjectVocabularySet {
15    /// Validated contracts in canonical identity order.
16    vocabularies: BTreeMap<String, ProjectVocabulary>,
17    /// Module-local aliases kept separate from semantic contracts.
18    aliases: BTreeMap<String, BTreeMap<String, String>>,
19}
20
21impl ProjectVocabularySet {
22    /// Returns canonical identities and locked revisions in identity order.
23    #[must_use]
24    pub fn vocabularies(&self) -> &BTreeMap<String, ProjectVocabulary> {
25        &self.vocabularies
26    }
27
28    /// Resolves a module-local alias to its canonical locked contract.
29    #[must_use]
30    pub fn resolve(&self, module_id: &str, alias: &str) -> Option<&ProjectVocabulary> {
31        let identity = self.aliases.get(module_id)?.get(alias)?;
32        self.vocabularies.get(identity)
33    }
34}
35
36/// Validation failure before any partially validated vocabulary set is published.
37#[derive(Clone, Copy, Debug, Eq, PartialEq)]
38pub enum ProjectVocabularyValidationError {
39    /// A captured source requirement was malformed despite capture validation.
40    InvalidRequirement,
41    /// A captured bundle violated the exact lock or closed v1 schema.
42    InvalidBundle(VocabularyError),
43}
44
45/// Validates every captured v1 bundle and source-local alias against exact locks.
46///
47/// # Errors
48///
49/// Rejects a malformed requirement or any bundle with no partial result.
50pub fn validate_project_vocabularies(
51    captured: &CapturedProject,
52) -> Result<ProjectVocabularySet, ProjectVocabularyValidationError> {
53    let values = captured.limits().values();
54    let structural = StructuralLimits::new(values.vocabulary_bytes_per_unit, 1)
55        .map_err(|_| ProjectVocabularyValidationError::InvalidRequirement)?
56        .with_declarations(values.declarations)
57        .map_err(|_| ProjectVocabularyValidationError::InvalidRequirement)?;
58    let limits = VocabularyLimits::from_structural(structural);
59    let mut vocabularies = BTreeMap::new();
60    for input in captured.vocabularies() {
61        let validated = validate_project_bundle(input.bytes(), input.lock(), limits)
62            .map_err(ProjectVocabularyValidationError::InvalidBundle)?;
63        vocabularies.insert(validated.identity().to_owned(), validated);
64    }
65    let mut aliases = BTreeMap::new();
66    for source in captured.sources() {
67        let text = std::str::from_utf8(source.bytes())
68            .map_err(|_| ProjectVocabularyValidationError::InvalidRequirement)?;
69        let mut local = BTreeMap::new();
70        let requirements = scan_vocabulary_requirements(text)
71            .map_err(|_| ProjectVocabularyValidationError::InvalidRequirement)?;
72        for (identity, alias) in requirements {
73            if !vocabularies.contains_key(&identity) || local.insert(alias, identity).is_some() {
74                return Err(ProjectVocabularyValidationError::InvalidRequirement);
75            }
76        }
77        aliases.insert(source.module_id().to_owned(), local);
78    }
79    Ok(ProjectVocabularySet {
80        vocabularies,
81        aliases,
82    })
83}
