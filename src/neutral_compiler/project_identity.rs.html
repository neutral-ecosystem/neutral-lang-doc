3//! Borrowed identity projections from successfully frozen capture, without recapture or I/O.
4
5use crate::{CapturedProject, CapturedProjectSource, CapturedProjectVocabulary};
6use neutral_core::CancellationToken;
7use neutral_core::allocation::RetainCapacity;
8use neutral_ir::project_identity::{
9    CapturedClosureIdentity, CapturedIdentityInput, CapturedIdentitySource,
10    CapturedIdentityVocabulary, IdentityError, IdentityLimits, IdentityTranscript,
11    MAX_TRANSCRIPT_NODES, captured_closure,
12};
13
14impl CapturedProject {
15    /// Returns the exact captured-closure transcript using caller-supplied independent bounds.
16    ///
17    /// # Errors
18    /// Rejects identity bounds, allocation failures, or cancellation; capture meaning is unchanged.
19    pub fn identity_transcript(
20        &self,
21        limits: IdentityLimits,
22        cancellation: &CancellationToken,
23    ) -> Result<IdentityTranscript<CapturedClosureIdentity>, IdentityError> {
24        let (sources, vocabularies) =
25            capture_facts(self.sources(), self.vocabularies(), limits, cancellation)?;
26        captured_closure(
27            &CapturedIdentityInput {
28                profile: self.profile().source_version(),
29                sources: &sources,
30                vocabularies: &vocabularies,
31            },
32            limits,
33            cancellation,
34        )
35    }
36
37    /// Projects all explicit acceptance controls into the frozen derivation transcript order.
38    #[must_use]
39    pub fn identity_capture_limits(
40        &self,
41    ) -> [u64; neutral_ir::project_identity::CAPTURE_LIMIT_TAGS.len()] {
42        let value = self.limits().values();
43        [
44            value.total_source_bytes,
45            value.source_bytes_per_unit,
46            value.source_units,
47            value.source_id_bytes,
48            value.module_id_bytes,
49            value.vocabulary_units,
50            value.vocabulary_bytes_per_unit,
51            value.total_vocabulary_bytes,
52            value.imports_per_module,
53            value.import_edges,
54            value.scc_units,
55            value.declarations,
56            value.diagnostics,
57            value.output_bytes,
58        ]
59    }
60}

62/// Bounded borrowed projection buffers shared by old and successor capture identities.
63type CaptureFacts<'a> = (
64    Vec<CapturedIdentitySource<'a>>,
65    Vec<CapturedIdentityVocabulary<'a>>,
66);
67
68/// Projects only verified immutable capture facts, enforcing independent bounds before reservation.
69pub(crate) fn capture_facts<'a>(
70    captured_sources: &'a [CapturedProjectSource],
71    captured_vocabularies: &'a [CapturedProjectVocabulary],
72    limits: IdentityLimits,
73    cancellation: &CancellationToken,
74) -> Result<CaptureFacts<'a>, IdentityError> {
75    if cancellation.is_cancelled() {
76        return Err(IdentityError::Cancelled);
77    }
78    if limits.bytes == 0
79        || limits.nodes == 0
80        || captured_sources.len() as u64 > limits.nodes.min(MAX_TRANSCRIPT_NODES)
81        || captured_vocabularies.len() as u64 > limits.nodes.min(MAX_TRANSCRIPT_NODES)
82    {
83        return Err(IdentityError::Limit);
84    }
85    let mut sources = Vec::new();
86    sources
87        .try_retain(captured_sources.len())
88        .map_err(|_| IdentityError::Limit)?;
89    for source in captured_sources {
90        if cancellation.is_cancelled() {
91            return Err(IdentityError::Cancelled);
92        }
93        sources.push(CapturedIdentitySource {
94            module: source.module_id(),
95            source_id: source.source_id(),
96            digest: source.digest(),
97            byte_len: source.bytes().len() as u64,
98        });
99    }
100    let mut vocabularies = Vec::new();
101    vocabularies
102        .try_retain(captured_vocabularies.len())
103        .map_err(|_| IdentityError::Limit)?;
104    for vocabulary in captured_vocabularies {
105        if cancellation.is_cancelled() {
106            return Err(IdentityError::Cancelled);
107        }
108        let lock = vocabulary.lock();
109        vocabularies.push(CapturedIdentityVocabulary {
110            identity: lock.identity(),
111            version: lock.version(),
112            encoding_version: lock.encoding_version(),
113            schema_version: lock.schema_version(),
114            digest: lock.content_digest(),
115            byte_len: vocabulary.bytes().len() as u64,
116            required_features: lock.required_features(),
117        });
118    }
119    Ok((sources, vocabularies))
120}
