3//! Explicit bounded in-process syntax reuse; graph, semantics, and companions stay fresh.
4//!
5//! A generation contains only private parsed source units from a successful run.
6//! Entries cannot be supplied by callers or loaded from disk. Reuse is safe only
7//! for exact bytes under the same module/profile; source IDs and current controls
8//! are deliberately not taken from cached entries. Retention budgets describe
9//! retained units and original bytes, not a promise about total heap usage.
10
11use super::{Root, lowering::compile_project_with_parser, parse_roots};
12use crate::{CapturedProject, ProjectCompileFailure};
13use neutral_core::{CancellationToken, SourceContentDigest};
14use neutral_ir::project::ProjectIr;
15use std::{collections::BTreeMap, sync::Arc};
16
17/// Independent retained-unit and exact-byte budgets for one caller-owned cache.
18#[derive(Clone, Copy, Debug, Eq, PartialEq)]
19pub struct ProjectCacheLimits {
20    /// Maximum retained parsed source units.
21    pub source_units: u64,
22    /// Maximum aggregate retained original source bytes.
23    pub source_bytes: u64,
24}
25
26/// Observed unit parser execution, not captured replay or inferred equivalence.
27#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
28pub struct ProjectCacheStats {
29    /// Units actually sent through the lexer and declaration parser this run.
30    pub parsed_units: u64,
31    /// Units reused from the previous successful cache generation.
32    pub reused_units: u64,
33    /// Existing module entries rejected for changed bytes or identity context.
34    pub rejected_entries: u64,
35}
36
37/// Immutable parse entry; no API accepts externally supplied cached syntax.
38#[derive(Clone)]
39struct Entry {
40    /// Logical module context; a misplaced entry cannot reuse another module's AST.
41    module: String,
42    /// Exact language profile used to parse the source.
43    profile: String,
44    /// Exact source-byte identity, never a public-interface or logical digest.
45    digest: SourceContentDigest,
46    /// Original bytes also compared to reject stale or digest-collision entries.
47    bytes: Arc<[u8]>,
48    /// Private source-sensitive roots, including exact original spans.
49    roots: Vec<Root>,
50}
51
52/// Caller-owned syntax cache; no process-global state, I/O, clock, or host mapping.
53pub struct ProjectCompilationCache {
54    /// Explicit retention budgets, separate from per-request compilation controls.
55    limits: ProjectCacheLimits,
56    /// Only the previous successful generation, keyed by exact logical module.
57    entries: BTreeMap<String, Entry>,
58    /// Sum of retained exact source bytes.
59    source_bytes: u64,
60}
61
62impl ProjectCompilationCache {
63    /// Creates an empty bounded cache, returning `None` for zero retention budgets.
64    #[must_use]
65    pub fn new(limits: ProjectCacheLimits) -> Option<Self> {
66        (limits.source_units != 0 && limits.source_bytes != 0).then(|| Self {
67            limits,
68            entries: BTreeMap::new(),
69            source_bytes: 0,
70        })
71    }
72
73    /// Returns retained source units, never the number of compilation requests.
74    #[must_use]
75    pub fn retained_units(&self) -> usize {
76        self.entries.len()
77    }
78
79    /// Returns the exact retained source-byte total under the independent budget.
80    #[must_use]
81    pub const fn retained_source_bytes(&self) -> u64 {
82        self.source_bytes
83    }
84
85    /// Compiles incrementally using exact source/profile/module-sensitive parse keys.
86    ///
87    /// Every request rebuilds and validates the graph, vocabulary locks, complete
88    /// semantics, values, source maps, provenance, and resource facts. Source IDs,
89    /// aliases, controls, private bodies, and roots are never substituted through
90    /// a public fingerprint. Only successfully compiled generations are retained;
91    /// over-budget entries are skipped deterministically in module order.
92    ///
93    /// # Errors
94    /// Returns the clean compiler's failure and leaves the prior cache generation
95    /// unchanged on syntax, graph, semantic, lowering, or cancellation failure.
96    pub fn compile(
97        &mut self,
98        captured: &CapturedProject,
99        cancellation: &CancellationToken,
100    ) -> Result<(Arc<ProjectIr>, ProjectCacheStats), ProjectCompileFailure> {
101        let mut pending = BTreeMap::<String, Entry>::new();
102        let mut stats = ProjectCacheStats::default();
103        let ir = compile_project_with_parser(captured, cancellation, &mut |source| {
104            // Semantics passes these parsed roots directly to lowering. This
105            // pending generation is retained only after the entire run succeeds.
106            let previous = self.entries.get(source.module_id());
107            // A logical hash or public fingerprint loses spelling/private facts.
108            // Even a matching source digest needs exact bytes to reject collisions.
109            let reusable = previous.filter(|entry| {
110                entry.module == source.module_id()
111                    && entry.profile == captured.profile().source_version()
112                    && entry.digest == source.digest()
113                    && entry.bytes.as_ref() == source.bytes()
114            });
115            let roots = if let Some(entry) = reusable {
116                stats.reused_units += 1;
117                entry.roots.clone()
118            } else {
119                stats.rejected_entries += u64::from(previous.is_some());
120                stats.parsed_units += 1;
121                parse_roots(source.module_id(), source.digest(), source.bytes())?
122            };
123            pending.insert(
124                source.module_id().to_owned(),
125                Entry {
126                    module: source.module_id().to_owned(),
127                    profile: captured.profile().source_version().to_owned(),
128                    digest: source.digest(),
129                    bytes: Arc::from(source.bytes()),
130                    roots: roots.clone(),
131                },
132            );
133            Ok(roots)
134        })?;
135        // Publication follows the same final cancellation check as clean lowering.
136        if cancellation.is_cancelled() {
137            return Err(ProjectCompileFailure::Lowering {
138                code: super::lowering::codes::CANCELLED,
139                location: None,
140            });
141        }
142        let mut retained = BTreeMap::new();
143        let mut retained_bytes = 0_u64;
144        // Replace, rather than accumulate, generations: removed modules cannot
145        // remain resident forever. Canonical map order makes budget eviction stable.
146        for (module, entry) in pending {
147            let length = entry.bytes.len() as u64;
148            if (retained.len() as u64) < self.limits.source_units
149                && retained_bytes
150                    .checked_add(length)
151                    .is_some_and(|sum| sum <= self.limits.source_bytes)
152            {
153                retained_bytes += length;
154                retained.insert(module, entry);
155            }
156        }
157        self.entries = retained;
158        self.source_bytes = retained_bytes;
159        Ok((ir, stats))
160    }
161}
