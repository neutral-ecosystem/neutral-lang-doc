11/// UTF-8 byte-order mark accepted only once at byte offset zero.
12const UTF8_BOM: &[u8; 3] = b"\xef\xbb\xbf";
13
14/// Lexes exact captured bytes while retaining physical newline tokens and spans.
15pub(super) fn lex(source: &[u8]) -> Result<LexedSource, FrontendError> {
16    validate_source_text(source)?;
17    let mut tokens = Vec::new();
18    let mut trivia = Vec::new();
19    let mut index = usize::from(source.starts_with(UTF8_BOM)) * UTF8_BOM.len();
20
21    while index < source.len() {
22        match source[index] {
23            b' ' | b'\t' => {
24                let end = consume_while(source, index, |value| matches!(value, b' ' | b'\t'));
25                trivia.push(Trivia {
26                    kind: TriviaKind::HorizontalWhitespace,
27                    span: span(index, end),
28                });
29                index = end;
30            }
31            b'\n' => {
32                tokens.push(token(
33                    TokenKind::PhysicalLineEnd(PhysicalLineEnd::LineFeed),
34                    index,
35                    index + 1,
36                ));
37                index += 1;
38            }
39            b'\r' => {
40                let end = if source.get(index + 1) == Some(&b'\n') {
41                    index + 2
42                } else {
43                    index + 1
44                };
45                let kind = if end == index + 2 {
46                    PhysicalLineEnd::CarriageReturnLineFeed
47                } else {
48                    PhysicalLineEnd::CarriageReturn
49                };
50                tokens.push(token(TokenKind::PhysicalLineEnd(kind), index, end));
51                index = end;
52            }
53            b':' if source.get(index + 1) == Some(&b':') => {
54                tokens.push(token(TokenKind::DoubleColon, index, index + 2));
55                index += 2;
56            }
57            byte if punctuation_token(byte).is_some() => {
58                let kind = punctuation_token(byte).expect("matched punctuation must classify");
59                tokens.push(token(kind, index, index + 1));
60                index += 1;
61            }
62            b'/' if source.get(index + 1) == Some(&b'/') => {
63                let end = consume_while(source, index + 2, |value| !matches!(value, b'\n' | b'\r'));
64                trivia.push(Trivia {
65                    kind: TriviaKind::LineComment,
66                    span: span(index, end),
67                });
68                index = end;
69            }
70            b'/' if source.get(index + 1) == Some(&b'*') => {
71                let (end, line_ends) = block_comment(source, index)?;
72                trivia.push(Trivia {
73                    kind: TriviaKind::BlockComment,
74                    span: span(index, end),
75                });
76                tokens.extend(line_ends);
77                index = end;
78            }
79            b'"' => {
80                let (next, value) = string_literal(source, index)?;
81                tokens.push(token(TokenKind::StringLiteral(value), index, next));
82                index = next;
83            }
84            byte if byte.is_ascii_alphabetic() || byte == b'_' => {
85                let end = consume_while(source, index, |value| {
86                    value.is_ascii_alphanumeric() || *value == b'_'
87                });
88                let text = ascii_text(&source[index..end], index)?;
89                tokens.push(token(word_token(text), index, end));
90                index = end;
91            }
92            byte if starts_number(source, index, byte) => {
93                let end = consume_number_candidate(source, index);
94                let value = ascii_text(&source[index..end], index)?;
95                tokens.push(token(TokenKind::Number(value), index, end));
96                index = end;
97            }
98            _ => return Err(FrontendError::unsupported_symbol(span(index, index + 1))),
99        }
100    }
101
102    tokens.push(token(TokenKind::EndOfFile, source.len(), source.len()));
103    Ok(LexedSource { tokens, trivia })
104}

106/// Maps one active punctuation byte to its compiler-private token category.
107fn punctuation_token(byte: u8) -> Option<TokenKind> {
108    match byte {
109        b'=' => Some(TokenKind::Equals),
110        b'?' => Some(TokenKind::Question),
111        b'{' => Some(TokenKind::OpenBrace),
112        b'}' => Some(TokenKind::CloseBrace),
113        b':' => Some(TokenKind::Colon),
114        b',' => Some(TokenKind::Comma),
115        b'<' => Some(TokenKind::Less),
116        b'>' => Some(TokenKind::Greater),
117        b'[' => Some(TokenKind::OpenBracket),
118        b']' => Some(TokenKind::CloseBracket),
119        b'(' => Some(TokenKind::OpenParen),
120        b')' => Some(TokenKind::CloseParen),
121        _ => None,
122    }
123}

125/// Returns whether `byte` begins a candidate frozen numeric literal.
126fn starts_number(source: &[u8], index: usize, byte: u8) -> bool {
127    byte.is_ascii_digit()
128        || matches!(byte, b'+' | b'-') && source.get(index + 1).is_some_and(u8::is_ascii_digit)
129}

131/// Consumes a numeric candidate for later exact semantic validation.
132fn consume_number_candidate(source: &[u8], start: usize) -> usize {
133    consume_while(source, start, |byte| {
134        byte.is_ascii_alphanumeric() || matches!(*byte, b'_' | b'.' | b'+' | b'-')
135    })
136}

138/// Classifies one ASCII word through the centralized language-name namespace.
139fn word_token(text: String) -> TokenKind {
140    if text == names::NEU {
141        TokenKind::Neu
142    } else if text == names::MODULE {
143        TokenKind::Module
144    } else if text == names::USE {
145        TokenKind::Use
146    } else if text == names::RECORD {
147        TokenKind::Record
148    } else if text == names::LIST {
149        TokenKind::List
150    } else if text == names::REF_TYPE {
151        TokenKind::RefType
152    } else if text == names::REF {
153        TokenKind::RefValue
154    } else if text == names::NUM {
155        TokenKind::Num
156    } else if text == names::STRING {
157        TokenKind::StringType
158    } else if text == names::BOOL {
159        TokenKind::BoolType
160    } else if text == names::TRUE {
161        TokenKind::True
162    } else if text == names::FALSE {
163        TokenKind::False
164    } else if text == names::NULL {
165        TokenKind::Null
166    } else if names::is_protected_name(&text) {
167        TokenKind::ProtectedName(text)
168    } else {
169        TokenKind::Identifier(text)
170    }
171}

173/// Reads one non-nesting block comment and retains any physical line endings.
174fn block_comment(source: &[u8], start: usize) -> Result<(usize, Vec<Token>), FrontendError> {
175    let mut index = start + 2;
176    let mut line_ends = Vec::new();
177    while index < source.len() {
178        if source.get(index) == Some(&b'*') && source.get(index + 1) == Some(&b'/') {
179            return Ok((index + 2, line_ends));
180        }
181        match source[index] {
182            b'\n' => {
183                line_ends.push(token(
184                    TokenKind::PhysicalLineEnd(PhysicalLineEnd::LineFeed),
185                    index,
186                    index + 1,
187                ));
188                index += 1;
189            }
190            b'\r' => {
191                let end = if source.get(index + 1) == Some(&b'\n') {
192                    index + 2
193                } else {
194                    index + 1
195                };
196                let kind = if end == index + 2 {
197                    PhysicalLineEnd::CarriageReturnLineFeed
198                } else {
199                    PhysicalLineEnd::CarriageReturn
200                };
201                line_ends.push(token(TokenKind::PhysicalLineEnd(kind), index, end));
202                index = end;
203            }
204            _ => index += 1,
205        }
206    }
207    Err(FrontendError::unterminated_block_comment(span(
208        start,
209        source.len(),
210    )))
211}

213/// Rejects malformed UTF-8, unescaped NUL, and a BOM outside byte offset zero.
214fn validate_source_text(source: &[u8]) -> Result<(), FrontendError> {
215    if let Some(index) = source.iter().position(|byte| *byte == 0) {
216        return Err(FrontendError::other(span(index, index + 1)));
217    }
218    if let Err(error) = std::str::from_utf8(source) {
219        let start = error.valid_up_to();
220        let end = start
221            .saturating_add(error.error_len().unwrap_or(1))
222            .min(source.len());
223        return Err(FrontendError::other(span(start, end)));
224    }
225    let search_start = usize::from(source.starts_with(UTF8_BOM)) * UTF8_BOM.len();
226    if let Some(relative) = source[search_start..]
227        .windows(UTF8_BOM.len())
228        .position(|window| window == UTF8_BOM)
229    {
230        let start = search_start + relative;
231        return Err(FrontendError::other(span(start, start + UTF8_BOM.len())));
232    }
233    Ok(())
234}

236/// Reads one simple, unescaped string literal without crossing a physical line.
237fn string_literal(source: &[u8], start: usize) -> Result<(usize, DecodedString), FrontendError> {
238    let mut index = start + 1;
239    let mut value = String::new();
240    let mut had_escape = false;
241    while let Some(byte) = source.get(index).copied() {
242        match byte {
243            b'"' => {
244                return Ok((index + 1, DecodedString { value, had_escape }));
245            }
246            b'\\' => {
247                had_escape = true;
248                index = decode_escape(source, index, &mut value)?;
249            }
250            b'\n' | b'\r' => {
251                return Err(FrontendError::unterminated_string_literal(span(
252                    start, index,
253                )));
254            }
255            byte if byte.is_ascii_control() => {
256                return Err(FrontendError::invalid_string_literal(span(
257                    index,
258                    index + 1,
259                )));
260            }
261            byte if byte.is_ascii() => {
262                value.push(char::from(byte));
263                index += 1;
264            }
265            _ => {
266                let remaining = std::str::from_utf8(&source[index..])
267                    .expect("source UTF-8 was validated before string decoding");
268                let character = remaining
269                    .chars()
270                    .next()
271                    .expect("non-ASCII byte must start one character");
272                let end = index + character.len_utf8();
273                if character.is_control() {
274                    return Err(FrontendError::invalid_string_literal(span(index, end)));
275                }
276                value.push(character);
277                index = end;
278            }
279        }
280    }
281    Err(FrontendError::unterminated_string_literal(span(
282        start,
283        source.len(),
284    )))
285}

287/// Decodes one frozen string escape and returns the next unread byte offset.
288fn decode_escape(source: &[u8], start: usize, value: &mut String) -> Result<usize, FrontendError> {
289    let Some(escaped) = source.get(start + 1).copied() else {
290        return Err(FrontendError::unterminated_string_literal(span(
291            start,
292            source.len(),
293        )));
294    };
295    let character = match escaped {
296        b'"' => '"',
297        b'\\' => '\\',
298        b'n' => '\n',
299        b'r' => '\r',
300        b't' => '\t',
301        b'0' => '\0',
302        b'u' => return decode_unicode_escape(source, start, value),
303        _ => {
304            return Err(FrontendError::invalid_string_literal(span(
305                start,
306                (start + 2).min(source.len()),
307            )));
308        }
309    };
310    value.push(character);
311    Ok(start + 2)
312}

314/// Decodes one `\u{HEX}` escape containing one to six hexadecimal digits.
315fn decode_unicode_escape(
316    source: &[u8],
317    start: usize,
318    value: &mut String,
319) -> Result<usize, FrontendError> {
320    if source.get(start + 2) != Some(&b'{') {
321        return Err(FrontendError::invalid_string_literal(span(
322            start,
323            (start + 3).min(source.len()),
324        )));
325    }
326    let digits_start = start + 3;
327    let mut end = digits_start;
328    while source.get(end).is_some_and(u8::is_ascii_hexdigit) && end - digits_start < 6 {
329        end += 1;
330    }
331    if end == digits_start || source.get(end) != Some(&b'}') {
332        return Err(FrontendError::invalid_string_literal(span(
333            start,
334            (end + 1).min(source.len()),
335        )));
336    }
337    let digits = ascii_text(&source[digits_start..end], digits_start)?;
338    let scalar = u32::from_str_radix(&digits, 16)
339        .ok()
340        .and_then(char::from_u32)
341        .ok_or_else(|| FrontendError::invalid_string_literal(span(start, end + 1)))?;
342    value.push(scalar);
343    Ok(end + 1)
344}

346/// Advances over bytes while the supplied ASCII predicate remains true.
347fn consume_while(source: &[u8], start: usize, predicate: fn(&u8) -> bool) -> usize {
348    let mut end = start;
349    while source.get(end).is_some_and(predicate) {
350        end += 1;
351    }
352    end
353}

355/// Converts validated ASCII token bytes into owned text.
356fn ascii_text(bytes: &[u8], start: usize) -> Result<String, FrontendError> {
357    if !bytes.is_ascii() {
358        return Err(FrontendError::other(span(start, start + bytes.len())));
359    }
360    Ok(String::from_utf8_lossy(bytes).into_owned())
361}

363/// Creates one raw token over an exact original-byte span.
364fn token(kind: TokenKind, start: usize, end: usize) -> Token {
365    Token {
366        kind,
367        span: span(start, end),
368    }
