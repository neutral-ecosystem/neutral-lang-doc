17/// Raw lexing result with nonsemantic trivia kept separate from parser tokens.
18#[derive(Clone, Debug, Eq, PartialEq)]
19struct LexedSource {
20    /// Semantic and physical-line tokens in original byte order.
21    tokens: Vec<Token>,
22    /// Exact source trivia retained privately for later formatter work.
23    trivia: Vec<Trivia>,
24}

26/// One exact nonsemantic source region.
27#[derive(Clone, Debug, Eq, PartialEq)]
28struct Trivia {
29    /// Trivia category.
30    kind: TriviaKind,
31    /// Exact half-open span in captured source bytes.
32    span: ByteSpan,
33}

35/// Compiler-private source trivia categories.
36#[derive(Clone, Copy, Debug, Eq, PartialEq)]
37enum TriviaKind {
38    /// One or more spaces or horizontal tabs.
39    HorizontalWhitespace,
40    /// A `//` comment excluding its physical line ending.
41    LineComment,
42    /// A non-nesting `/* ... */` comment.
43    BlockComment,
44}

46/// A raw or normalized token retained only inside the compiler frontend.
47#[derive(Clone, Debug, Eq, PartialEq)]
48struct Token {
49    /// Token category and safe decoded value, when applicable.
50    kind: TokenKind,
51    /// Exact half-open span in the original captured bytes.
52    span: ByteSpan,
53}

55/// Token categories required by the active source and scalar slices.
56#[derive(Clone, Debug, Eq, PartialEq)]
57enum TokenKind {
58    /// The `neu` document-header keyword.
59    Neu,
60    /// The `module` header keyword.
61    Module,
62    /// The `use` vocabulary-requirement keyword.
63    Use,
64    /// The `record` nominal-type declaration keyword.
65    Record,
66    /// The `List` invariant generic type constructor.
67    List,
68    /// The `Ref` invariant identity-reference type constructor.
69    RefType,
70    /// The `ref` identity-reference value constructor.
71    RefValue,
72    /// The `num` core-type keyword.
73    Num,
74    /// The `string` core-type keyword.
75    StringType,
76    /// The `bool` core-type keyword.
77    BoolType,
78    /// The `true` Boolean literal.
79    True,
80    /// The `false` Boolean literal.
81    False,
82    /// The `null` explicit null literal.
83    Null,
84    /// A protected core spelling appearing where an identifier can be diagnosed.
85    ProtectedName(String),
86    /// An ASCII identifier retained for later semantic validation.
87    Identifier(String),
88    /// A raw, unescaped string body used only for the version header.
89    StringLiteral(DecodedString),
90    /// A digits-only minimal numeric spelling retained for later semantics.
91    Number(String),
92    /// The binding initializer delimiter.
93    Equals,
94    /// Postfix outer-nullability delimiter.
95    Question,
96    /// Record declaration or contextual-value opening delimiter.
97    OpenBrace,
98    /// Record declaration or contextual-value closing delimiter.
99    CloseBrace,
100    /// Record value field-name separator.
101    Colon,
102    /// Vocabulary qualification delimiter.
103    DoubleColon,
104    /// Generic type-argument opening delimiter.
105    Less,
106    /// Generic type-argument closing delimiter.
107    Greater,
108    /// List value opening delimiter.
109    OpenBracket,
110    /// List value closing delimiter.
111    CloseBracket,
112    /// Identity-reference target opening delimiter.
113    OpenParen,
114    /// Identity-reference target closing delimiter.
115    CloseParen,
116    /// Required record field terminator.
117    Comma,
118    /// An original physical newline before layout normalization.
119    PhysicalLineEnd(PhysicalLineEnd),
120    /// A semantic declaration/header terminator.
121    LineEnd,
122    /// End of captured source.
123    EndOfFile,
124}

146/// A private successfully parsed minimal source unit.
147#[derive(Clone, Debug, Eq, PartialEq)]
148pub(super) struct ParsedUnit {

156    pub(super) vocabulary_use: Option<ParsedVocabularyUse>,
157    /// Parsed root declarations in source order.
158    pub(super) declarations: Vec<ParsedDeclaration>,

174/// One compiler-private root declaration.
175#[derive(Clone, Debug, Eq, PartialEq)]
176pub(super) enum ParsedDeclaration {
177    /// One nominal record schema declaration.
178    Record(ParsedRecord),
179    /// One immutable typed value binding.
180    Binding(ParsedBinding),
181}
182
183impl ParsedDeclaration {
184    /// Returns the declared root name.
185    pub(super) fn name(&self) -> &str {
186        match self {
187            Self::Record(record) => &record.name,
188            Self::Binding(binding) => &binding.name,
189        }
190    }
191
192    /// Returns the exact root-name span.
193    pub(super) const fn name_span(&self) -> ByteSpan {
194        match self {
195            Self::Record(record) => record.name_span,
196            Self::Binding(binding) => binding.name_span,
197        }
198    }
199}

348/// A private frontend failure with an optional frozen public diagnostic.
349#[derive(Clone, Debug, Eq, PartialEq)]
350pub(super) struct FrontendError {
351    /// Failure kind used to classify frozen and not-yet-active syntax.
352    kind: FrontendErrorKind,
353    /// Exact primary source span.
354    span: ByteSpan,
355}
356
357/// Private categories for active frontend failures.
358#[derive(Clone, Copy, Debug, Eq, PartialEq)]
359enum FrontendErrorKind {
360    /// The required module header was absent.
361    MissingModuleHeader,
362    /// The quoted language version was not exactly `0.1`.
363    UnsupportedLanguageVersion,
364    /// A malformed token or raw-newline boundary.
365    MalformedBoundary,
366    /// A symbol excluded by the frozen v0 grammar.
367    UnsupportedSymbol,
368    /// A block comment reached EOF without closing.
369    UnterminatedBlockComment,
370    /// A string contained an invalid escape, scalar, or raw control.
371    InvalidStringLiteral,
372    /// A string reached a raw newline or EOF without closing.
373    UnterminatedStringLiteral,
374    /// A record/declaration/depth limit was exceeded during bounded parsing.
375    RecordLimitExceeded,
376    /// A list item/depth/traversal limit was exceeded during bounded parsing.
377    ListLimitExceeded,
378    /// Source is invalid or beyond the currently active minimal slice.
379    Other,
380}
381
382impl FrontendError {
383    /// Creates a missing-module-header failure at its insertion position.
384    fn missing_module_header(span: ByteSpan) -> Self {
385        Self {
386            kind: FrontendErrorKind::MissingModuleHeader,
387            span,
388        }
389    }
390
391    /// Creates an unsupported-language-version failure over the quoted spelling.
392    fn unsupported_language_version(span: ByteSpan) -> Self {
393        Self {
394            kind: FrontendErrorKind::UnsupportedLanguageVersion,
395            span,
396        }
397    }
398
399    /// Creates a stable malformed-boundary failure.
400    fn malformed_boundary(span: ByteSpan) -> Self {
401        Self {
402            kind: FrontendErrorKind::MalformedBoundary,
403            span,
404        }
405    }
406
407    /// Creates a stable unsupported-symbol failure.
408    fn unsupported_symbol(span: ByteSpan) -> Self {
409        Self {
410            kind: FrontendErrorKind::UnsupportedSymbol,
411            span,
412        }
413    }
414
415    /// Creates a stable unterminated-block-comment failure.
416    fn unterminated_block_comment(span: ByteSpan) -> Self {
417        Self {
418            kind: FrontendErrorKind::UnterminatedBlockComment,
419            span,
420        }
421    }
422
423    /// Creates a stable invalid-string-literal failure.
424    fn invalid_string_literal(span: ByteSpan) -> Self {
425        Self {
426            kind: FrontendErrorKind::InvalidStringLiteral,
427            span,
428        }
429    }
430
431    /// Creates a stable unterminated-string-literal failure.
432    fn unterminated_string_literal(span: ByteSpan) -> Self {
433        Self {
434            kind: FrontendErrorKind::UnterminatedStringLiteral,
435            span,
436        }
437    }
438
439    /// Creates a bounded non-frozen failure for inactive or malformed syntax.
440    fn other(span: ByteSpan) -> Self {
441        Self {
442            kind: FrontendErrorKind::Other,
443            span,
444        }
445    }
446
447    /// Creates a stable record-structure resource-limit failure.
448    fn record_limit_exceeded(span: ByteSpan) -> Self {
449        Self {
450            kind: FrontendErrorKind::RecordLimitExceeded,
451            span,
452        }
453    }
454
455    /// Creates a stable list-structure resource-limit failure.
456    fn list_limit_exceeded(span: ByteSpan) -> Self {
457        Self {
458            kind: FrontendErrorKind::ListLimitExceeded,
459            span,
460        }
461    }
462
463    /// Returns the broad public failure class.
464    pub(super) const fn class(&self) -> neutral_core::ResultClass {
465        match self.kind {
466            FrontendErrorKind::RecordLimitExceeded | FrontendErrorKind::ListLimitExceeded => {
467                neutral_core::ResultClass::Resource
468            }
469            _ => neutral_core::ResultClass::Syntax,
470        }
471    }
472
473    /// Returns the public failure detail without exposing private token state.
474    pub(super) const fn detail(&self) -> crate::CompilationFailureDetail {
475        match self.kind {
476            FrontendErrorKind::MissingModuleHeader
477            | FrontendErrorKind::UnsupportedLanguageVersion
478            | FrontendErrorKind::MalformedBoundary
479            | FrontendErrorKind::UnsupportedSymbol
480            | FrontendErrorKind::UnterminatedBlockComment
481            | FrontendErrorKind::InvalidStringLiteral
482            | FrontendErrorKind::UnterminatedStringLiteral => {
483                crate::CompilationFailureDetail::SyntaxRejected
484            }
485            FrontendErrorKind::RecordLimitExceeded | FrontendErrorKind::ListLimitExceeded => {
486                crate::CompilationFailureDetail::ResourceLimitExceeded
487            }
488            FrontendErrorKind::Other => crate::CompilationFailureDetail::FrontendUnavailable,
489        }
490    }
491
492    /// Converts only frozen frontend failures into stable public diagnostics.
493    pub(super) fn into_diagnostics(self, source: SourceContentDigest) -> Vec<Diagnostic> {
494        let code = match self.kind {
495            FrontendErrorKind::MissingModuleHeader => diagnostics::MISSING_MODULE_HEADER,
496            FrontendErrorKind::UnsupportedLanguageVersion => {
497                diagnostics::UNSUPPORTED_LANGUAGE_VERSION
498            }
499            FrontendErrorKind::MalformedBoundary => diagnostics::MALFORMED_BOUNDARY,
500            FrontendErrorKind::UnsupportedSymbol => diagnostics::UNSUPPORTED_SYMBOL,
501            FrontendErrorKind::UnterminatedBlockComment => diagnostics::UNTERMINATED_BLOCK_COMMENT,
502            FrontendErrorKind::InvalidStringLiteral => diagnostics::INVALID_STRING_LITERAL,
503            FrontendErrorKind::UnterminatedStringLiteral => {
504                diagnostics::UNTERMINATED_STRING_LITERAL
505            }
506            FrontendErrorKind::RecordLimitExceeded => diagnostics::RECORD_LIMIT_EXCEEDED,
507            FrontendErrorKind::ListLimitExceeded => diagnostics::LIST_LIMIT_EXCEEDED,
508            FrontendErrorKind::Other => return Vec::new(),
509        };
510        let code = DiagnosticCode::new(code).expect("frozen diagnostic code must be valid ASCII");
511        vec![Diagnostic::new(
512            code,
513            if matches!(
514                self.kind,
515                FrontendErrorKind::RecordLimitExceeded | FrontendErrorKind::ListLimitExceeded
516            ) {
517                DiagnosticLayer::Resource
518            } else {
519                DiagnosticLayer::Syntax
520            },
521            DiagnosticSeverity::Error,
522            SourceLocation::new(source, self.span),
523            Vec::new(),
524            false,
525        )]
526    }
527}

529/// Runs raw lexing, layout normalization, and parsing without ambient authority.
530pub(super) fn parse(source: &[u8], limits: StructuralLimits) -> Result<ParsedUnit, FrontendError> {
531    let raw = lexer::lex(source)?;
532    let normalized = layout::normalize(raw)?;
533    parser::parse(normalized, limits)
534}

536/// Formats one successfully parsed unit without exposing private syntax types.
537pub(super) fn format_source(source: &[u8], unit: &ParsedUnit) -> Vec<u8> {
538    formatter::format_source(source, unit)
539}

541/// Creates a checked source span from validated in-memory indexes.
542fn span(start: usize, end: usize) -> ByteSpan {
543    ByteSpan::new(
544        u64::try_from(start).unwrap_or(u64::MAX),
545        u64::try_from(end).unwrap_or(u64::MAX),
546    )
547    .expect("frontend indexes must form an ordered span")
548}
