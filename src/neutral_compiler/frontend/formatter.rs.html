12/// Formats one valid parsed source unit under the frozen v0 style.
13pub(super) fn format_source(source: &[u8], unit: &ParsedUnit) -> Vec<u8> {
14    let comments = unit
15        .trivia
16        .iter()
17        .filter(|trivia| {
18            matches!(
19                trivia.kind,
20                TriviaKind::LineComment | TriviaKind::BlockComment
21            )
22        })
23        .collect::<Vec<_>>();
24    let mut formatter = Formatter {
25        source,
26        comments,
27        next_comment: 0,
28        blocks: Vec::new(),
29    };
30
31    formatter.push_root(
32        unit.language_header_span.end(),
33        &format!(
34            "{}{}{}",
35            style::LANGUAGE_HEADER_PREFIX,
36            names::SOURCE_LANGUAGE_VERSION,
37            style::LANGUAGE_HEADER_SUFFIX
38        ),
39    );
40    formatter.push_root(
41        unit.module.span.end(),
42        &format!("{}{}", style::MODULE_PREFIX, unit.module.name),
43    );
44    if let Some(vocabulary) = &unit.vocabulary_use {
45        formatter.push_root(
46            vocabulary.span.end(),
47            &format!("{}{}", style::USE_PREFIX, vocabulary.name),
48        );
49    }
50    for declaration in &unit.declarations {
51        let rendered = render_declaration(declaration);
52        formatter.push_root(declaration_span_end(declaration), &rendered);
53    }
54    formatter.finish()
55}

57/// Accumulates canonical root blocks while assigning retained comments.
58struct Formatter<'a> {
59    /// Original validated UTF-8 source used only to recover comment text.
60    source: &'a [u8],
61    /// Retained comments in original source order.
62    comments: Vec<&'a Trivia>,
63    /// Index of the next comment not yet assigned to output.
64    next_comment: usize,
65    /// Complete canonical top-level blocks.
66    blocks: Vec<String>,
67}
68
69impl Formatter<'_> {
70    /// Adds comments through one root span followed by that rendered construct.
71    fn push_root(&mut self, span_end: u64, construct: &str) {
72        let mut block = String::new();
73        while self
74            .comments
75            .get(self.next_comment)
76            .is_some_and(|comment| comment.span.start() <= span_end)
77        {
78            let comment = self.comments[self.next_comment];
79            append_comment(&mut block, self.source, comment);
80            self.next_comment += 1;
81        }
82        block.push_str(construct);
83        self.blocks.push(block);
84    }
85
86    /// Emits trailing comments and assembles the final LF-terminated source.
87    fn finish(mut self) -> Vec<u8> {
88        if self.next_comment < self.comments.len() {
89            let mut trailing = String::new();
90            for comment in &self.comments[self.next_comment..] {
91                append_comment(&mut trailing, self.source, comment);
92            }
93            while trailing.ends_with(style::NEWLINE) {
94                trailing.pop();
95            }
96            if !trailing.is_empty() {
97                self.blocks.push(trailing);
98            }
99        }
100        let mut blocks = self.blocks.into_iter();
101        let mut output = blocks.next().unwrap_or_default();
102        for (index, block) in blocks.enumerate() {
103            if index == 0 {
104                output.push(style::HEADER_SEPARATOR);
105            } else {
106                output.push_str(style::ROOT_SEPARATOR);
107            }
108            output.push_str(&block);
109        }
110        output.push(style::NEWLINE);
111        output.into_bytes()
112    }
113}

115/// Appends one exact comment with normalized physical newlines.
116fn append_comment(output: &mut String, source: &[u8], comment: &Trivia) {
117    let start = usize::try_from(comment.span.start()).unwrap_or(usize::MAX);
118    let end = usize::try_from(comment.span.end()).unwrap_or(usize::MAX);
119    let text = source
120        .get(start..end)
121        .and_then(|bytes| std::str::from_utf8(bytes).ok())
122        .expect("parsed trivia spans must select valid source UTF-8");
123    output.push_str(&normalize_newlines(text));
124    if !output.ends_with(style::NEWLINE) {
125        output.push(style::NEWLINE);
126    }
127}

129/// Converts every supported physical newline spelling to one line feed.
130fn normalize_newlines(value: &str) -> String {
131    value.replace("\r\n", "\n").replace('\r', "\n")
132}

134/// Returns the complete original end offset of one root declaration.
135fn declaration_span_end(declaration: &ParsedDeclaration) -> u64 {
136    match declaration {
137        ParsedDeclaration::Record(record) => record.span.end(),
138        ParsedDeclaration::Binding(binding) => binding.span.end(),
139    }
140}

142/// Renders one root declaration using canonical spacing and layout.
143fn render_declaration(declaration: &ParsedDeclaration) -> String {
144    let mut output = String::new();
145    match declaration {
146        ParsedDeclaration::Record(record) => write_record_declaration(&mut output, record),
147        ParsedDeclaration::Binding(binding) => {
148            write_type(&mut output, &binding.declared_type);
149            output.push(style::TYPE_NAME_SEPARATOR);
150            output.push_str(&binding.name);
151            output.push_str(style::INITIALIZER_SEPARATOR);
152            write_value(&mut output, &binding.value, 0);
153        }
154    }
155    output
156}

158/// Renders one nominal record declaration with one trailing-comma field per line.
159fn write_record_declaration(output: &mut String, record: &ParsedRecord) {
160    output.push_str(style::RECORD_PREFIX);
161    output.push_str(&record.name);
162    output.push_str(" {");
163    output.push(style::NEWLINE);
164    for field in &record.fields {
165        write_indent(output, 1);
166        write_type(output, &field.declared_type);
167        output.push(style::TYPE_NAME_SEPARATOR);
168        output.push_str(&field.name);
169        if let Some(default) = &field.default_value {
170            output.push_str(style::INITIALIZER_SEPARATOR);
171            write_value(output, default, 1);
172        }
173        output.push(style::MEMBER_SUFFIX);
174        output.push(style::NEWLINE);
175    }
176    output.push('}');
177}

179/// Renders one parsed type without exposing it as a public syntax model.
180fn write_type(output: &mut String, parsed_type: &ParsedType) {
181    match parsed_type {
182        ParsedType::Num => output.push_str(names::NUM),
183        ParsedType::String => output.push_str(names::STRING),
184        ParsedType::Bool => output.push_str(names::BOOL),
185        ParsedType::Record(name) => output.push_str(name),
186        ParsedType::VocabularyRecord { namespace, name } => {
187            output.push_str(namespace);
188            output.push_str(style::QUALIFIER_SEPARATOR);
189            output.push_str(name);
190        }
191        ParsedType::Nullable(inner) => {
192            write_type(output, inner);
193            output.push('?');
194        }
195        ParsedType::List(inner) => {
196            output.push_str(names::LIST);
197            output.push('<');
198            write_type(output, inner);
199            output.push('>');
200        }
201        ParsedType::Ref(inner) => {
202            output.push_str(names::REF_TYPE);
203            output.push('<');
204            write_type(output, inner);
205            output.push('>');
206        }
207    }
208}

210/// Renders one parsed value with canonical recursive indentation.
211fn write_value(output: &mut String, value: &ParsedValue, depth: usize) {
212    match value {
213        ParsedValue::Number(number) | ParsedValue::Name(number) => output.push_str(number),
214        ParsedValue::String(value) => write_string(output, value),
215        ParsedValue::Boolean(value) => {
216            output.push_str(if *value { names::TRUE } else { names::FALSE });
217        }
218        ParsedValue::Null => output.push_str(names::NULL),
219        ParsedValue::Reference { target, .. } => {
220            output.push_str(names::REF);
221            output.push('(');
222            output.push_str(target);
223            output.push(')');
224        }
225        ParsedValue::Record(fields) => {
226            output.push('{');
227            output.push(style::NEWLINE);
228            for field in fields {
229                write_indent(output, depth + 1);
230                output.push_str(&field.name);
231                output.push_str(style::FIELD_VALUE_SEPARATOR);
232                write_value(output, &field.value, depth + 1);
233                output.push(style::MEMBER_SUFFIX);
234                output.push(style::NEWLINE);
235            }
236            write_indent(output, depth);
237            output.push('}');
238        }
239        ParsedValue::List(items) if items.is_empty() => output.push_str("[]"),
240        ParsedValue::List(items) => {
241            output.push('[');
242            output.push(style::NEWLINE);
243            for item in items {
244                write_indent(output, depth + 1);
245                write_value(output, &item.value, depth + 1);
246                output.push(style::MEMBER_SUFFIX);
247                output.push(style::NEWLINE);
248            }
249            write_indent(output, depth);
250            output.push(']');
251        }
252    }
253}

255/// Renders one decoded string with the frozen canonical escape spellings.
256fn write_string(output: &mut String, value: &str) {
257    output.push('"');
258    for character in value.chars() {
259        match character {
260            '"' => output.push_str("\\\""),
261            '\\' => output.push_str("\\\\"),
262            '\n' => output.push_str("\\n"),
263            '\r' => output.push_str("\\r"),
264            '\t' => output.push_str("\\t"),
265            '\0' => output.push_str("\\0"),
266            character if character.is_control() => {
267                write!(output, "\\u{{{:x}}}", u32::from(character))
268                    .expect("writing to a string must be infallible");
269            }
270            character => output.push(character),
271        }
272    }
273    output.push('"');
274}
275
276/// Writes a checked number of canonical four-space indentation levels.
277fn write_indent(output: &mut String, depth: usize) {
278    for _ in 0..depth {
279        output.push_str(style::INDENT);
280    }
