3//! Semantic line-end normalization for active scalar frontend slices.
4
5use super::{FrontendError, LexedSource, Token, TokenKind};
6
7/// Replaces physical newlines with semantic line ends and inserts one at EOF.
8pub(super) fn normalize(raw: LexedSource) -> Result<LexedSource, FrontendError> {
9    let mut normalized = Vec::with_capacity(raw.tokens.len().saturating_add(1));
10    let mut construct_start = 0_usize;
11    let mut delimiter_depth = 0_u64;
12    for token in raw.tokens {
13        match token.kind {
14            TokenKind::PhysicalLineEnd(_) => {
15                if delimiter_depth > 0 {
16                    continue;
17                }
18                if construct_start == normalized.len() {
19                    continue;
20                }
21                if !is_complete_construct(&normalized[construct_start..]) {
22                    return Err(FrontendError::malformed_boundary(token.span));
23                }
24                normalized.push(Token {
25                    kind: TokenKind::LineEnd,
26                    span: token.span,
27                });
28                construct_start = normalized.len();
29            }
30            TokenKind::EndOfFile => {
31                if delimiter_depth != 0 {
32                    return Err(FrontendError::malformed_boundary(token.span));
33                }
34                if construct_start < normalized.len() {
35                    if !is_complete_construct(&normalized[construct_start..]) {
36                        return Err(FrontendError::malformed_boundary(token.span));
37                    }
38                    normalized.push(Token {
39                        kind: TokenKind::LineEnd,
40                        span: token.span,
41                    });
42                }
43                normalized.push(token);
44            }
45            TokenKind::OpenBrace | TokenKind::OpenBracket | TokenKind::OpenParen => {
46                delimiter_depth = delimiter_depth.saturating_add(1);
47                normalized.push(token);
48            }
49            TokenKind::CloseBrace | TokenKind::CloseBracket | TokenKind::CloseParen => {
50                delimiter_depth = delimiter_depth
51                    .checked_sub(1)
52                    .ok_or_else(|| FrontendError::malformed_boundary(token.span))?;
53                normalized.push(token);
54            }
55            _ => normalized.push(token),
56        }
57    }
58    if !normalized
59        .last()
60        .is_some_and(|token| matches!(token.kind, TokenKind::EndOfFile))
61    {
62        let span = normalized
63            .last()
64            .map(|token| token.span)
65            .ok_or_else(|| FrontendError::other(super::span(0, 0)))?;
66        return Err(FrontendError::other(span));
67    }
68    Ok(LexedSource {
69        tokens: normalized,
70        trivia: raw.trivia,
71    })
72}

74/// Returns whether one top-level physical construct is complete.
75fn is_complete_construct(tokens: &[Token]) -> bool {
76    match tokens {
77        [first, second] => {
78            matches!(first.kind, TokenKind::Neu)
79                && matches!(second.kind, TokenKind::StringLiteral(_))
80                || matches!(first.kind, TokenKind::Module) && is_name_token(&second.kind)
81                || matches!(first.kind, TokenKind::Use) && is_name_token(&second.kind)
82        }
83        [first, .., last] if matches!(first.kind, TokenKind::Record) => {
84            matches!(last.kind, TokenKind::CloseBrace)
85        }
86        [first, .., last] if is_type_start(&first.kind) => {
87            tokens
88                .iter()
89                .any(|token| matches!(token.kind, TokenKind::Equals))
90                && (is_value_end(&last.kind)
91                    || matches!(
92                        last.kind,
93                        TokenKind::CloseBrace | TokenKind::CloseBracket | TokenKind::CloseParen
94                    ))
95        }
96        _ => false,
97    }
98}

100/// Returns whether a token can begin an active scalar or nominal type.
101fn is_type_start(kind: &TokenKind) -> bool {
102    is_scalar_type(kind)
103        || matches!(
104            kind,
105            TokenKind::Identifier(_) | TokenKind::List | TokenKind::RefType
106        )
107}
108
109/// Returns whether a token is an active explicit scalar type.
110fn is_scalar_type(kind: &TokenKind) -> bool {
111    matches!(
112        kind,
113        TokenKind::Num | TokenKind::StringType | TokenKind::BoolType
114    )
115}
116
117/// Returns whether a token can end one active Stage 5.2 value construct.
118fn is_value_end(kind: &TokenKind) -> bool {
119    matches!(
120        kind,
121        TokenKind::Number(_)
122            | TokenKind::StringLiteral(_)
123            | TokenKind::True
124            | TokenKind::False
125            | TokenKind::Null
126            | TokenKind::Identifier(_)
127            | TokenKind::ProtectedName(_)
128            | TokenKind::CloseParen
129    )
130}
131
132/// Returns whether a token spelling can be diagnosed in an identifier position.
133fn is_name_token(kind: &TokenKind) -> bool {
134    matches!(
135        kind,
136        TokenKind::Identifier(_)
137            | TokenKind::ProtectedName(_)
138            | TokenKind::Neu
139            | TokenKind::Module
140            | TokenKind::Record
141            | TokenKind::Num
142            | TokenKind::StringType
143            | TokenKind::BoolType
144            | TokenKind::True
145            | TokenKind::False
146            | TokenKind::Null
147            | TokenKind::RefType
148            | TokenKind::RefValue
149    )
