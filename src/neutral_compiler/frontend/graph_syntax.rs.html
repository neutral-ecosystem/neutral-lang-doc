9/// One parsed, source-accounted logical import.
10#[derive(Clone, Debug, Eq, PartialEq)]
11pub(crate) struct GraphImport {
12    /// Exact target module ID.
13    pub(crate) target: String,
14    /// Required local alias.
15    pub(crate) alias: String,
16    /// Original-byte span of the complete import statement.
17    pub(crate) span: ByteSpan,
18}
19
20/// One syntax scan result before graph validation.
21pub(crate) struct GraphSourceSyntax {
22    /// Exact module header span.
23    pub(crate) header_span: ByteSpan,
24    /// Imported modules in original source order.
25    pub(crate) imports: Vec<GraphImport>,
26    /// Names already occupied by captured vocabulary requirements.
27    pub(crate) vocabulary_aliases: Vec<String>,
28}
29
30/// Classification of a source syntax failure at the graph boundary.
31#[derive(Clone, Copy, Debug, Eq, PartialEq)]
32pub(crate) enum GraphSyntaxErrorKind {
33    /// The required source grammar or placement is malformed.
34    InvalidSyntax,
35    /// An import attempts a forbidden form, including re-export or acquisition.
36    ForbiddenImport,
37    /// One vocabulary alias collides with another alias or the local module.
38    AliasCollision,
39    /// The per-module import count exceeded its explicit ceiling.
40    LimitExceeded,
41}
42
43/// One bounded source syntax error with an original-byte location.
44pub(crate) struct GraphSyntaxError {
45    /// Stable failure classification.
46    pub(crate) kind: GraphSyntaxErrorKind,
47    /// Exact original-byte location.
48    pub(crate) span: ByteSpan,
49}

51/// Scans one captured unit without consulting host locations or resolvers.
52pub(crate) fn scan_graph_source(
53    bytes: &[u8],
54    expected_module: &str,
55    import_limit: u64,
56) -> Result<GraphSourceSyntax, GraphSyntaxError> {
57    let lexed = lexer::lex(bytes).map_err(|error| GraphSyntaxError {
58        kind: if forbidden_line_at(bytes, error.span.start()) {
59            GraphSyntaxErrorKind::ForbiddenImport
60        } else {
61            GraphSyntaxErrorKind::InvalidSyntax
62        },
63        span: error.span,
64    })?;
65    let mut result = GraphSourceSyntax {
66        header_span: span(0, 0),
67        imports: Vec::new(),
68        vocabulary_aliases: Vec::new(),
69    };
70    let mut phase = 0_u8;
71    let mut start = 0_usize;
72    for (index, token) in lexed.tokens.iter().enumerate() {
73        if !matches!(
74            token.kind,
75            TokenKind::PhysicalLineEnd(_) | TokenKind::EndOfFile
76        ) {
77            continue;
78        }
79        let line = &lexed.tokens[start..index];
80        if !line.is_empty() {
81            scan_line(line, expected_module, import_limit, &mut phase, &mut result)?;
82        }
83        start = index + 1;
84    }
85    if phase < 2 {
86        return Err(GraphSyntaxError {
87            kind: GraphSyntaxErrorKind::InvalidSyntax,
88            span: span(bytes.len(), bytes.len()),
89        });
90    }
91    Ok(result)
92}

94/// Advances the header/import/declaration phases for one physical source line.
95fn scan_line(
96    line: &[Token],
97    expected_module: &str,
98    import_limit: u64,
99    phase: &mut u8,
100    result: &mut GraphSourceSyntax,
101) -> Result<(), GraphSyntaxError> {
102    let line_span = token_span(line);
103    if *phase == 0 {
104        if !matches!(line, [Token { kind: TokenKind::Neu, .. }, Token { kind: TokenKind::StringLiteral(value), .. }] if !value.had_escape && value.value == V1_SOURCE_PROFILE)
105        {
106            return Err(invalid(line_span));
107        }
108        *phase = 1;
109        return Ok(());
110    }
111    if *phase == 1 {
112        if !matches!(line[0].kind, TokenKind::Module)
113            || parse_module_name(&line[1..]).as_deref() != Some(expected_module)
114        {
115            return Err(invalid(line_span));
116        }
117        result.header_span = line_span;
118        *phase = 2;
119        return Ok(());
120    }
121    if matches!(line[0].kind, TokenKind::Use) {
122        if *phase > 2 {
123            return Err(invalid(line_span));
124        }
125        let [
126            _,
127            Token {
128                kind: TokenKind::Identifier(identity),
129                ..
130            },
131            Token {
132                kind: TokenKind::Identifier(as_word),
133                ..
134            },
135            Token {
136                kind: TokenKind::Identifier(alias),
137                ..
138            },
139        ] = line
140        else {
141            return Err(invalid(line_span));
142        };
143        if !identity.starts_with(|character: char| character.is_ascii_uppercase())
144            || as_word != graph_names::AS
145            || !valid_graph_name(alias)
146        {
147            return Err(invalid(line_span));
148        }
149        if alias == expected_module.rsplit("::").next().unwrap_or("")
150            || result.vocabulary_aliases.contains(alias)
151        {
152            return Err(GraphSyntaxError {
153                kind: GraphSyntaxErrorKind::AliasCollision,
154                span: line_span,
155            });
156        }
157        result.vocabulary_aliases.push(alias.clone());
158        return Ok(());
159    }
160    if matches!(&line[0].kind, TokenKind::Identifier(word) if word == graph_names::PUBLIC)
161        && line
162            .get(1)
163            .is_some_and(|token| matches!(&token.kind, TokenKind::Identifier(word) if word == graph_names::IMPORT))
164    {
165        return Err(GraphSyntaxError {
166            kind: GraphSyntaxErrorKind::ForbiddenImport,
167            span: line_span,
168        });
169    }
170    if matches!(&line[0].kind, TokenKind::Identifier(word) if word == graph_names::IMPORT) {
171        if *phase > 3 {
172            return Err(invalid(line_span));
173        }
174        *phase = 3;
175        let parsed = parse_import(line)?;
176        if u64::try_from(result.imports.len()).unwrap_or(u64::MAX) >= import_limit {
177            return Err(GraphSyntaxError {
178                kind: GraphSyntaxErrorKind::LimitExceeded,
179                span: line_span,
180            });
181        }
182        result.imports.push(parsed);
183        return Ok(());
184    }
185    if matches!(line[0].kind, TokenKind::Module | TokenKind::Neu) {
186        return Err(invalid(line_span));
187    }
188    *phase = 4;
189    Ok(())
190}

192/// Parses one exact aliased import statement from already lexed tokens.
193fn parse_import(line: &[Token]) -> Result<GraphImport, GraphSyntaxError> {
194    let line_span = token_span(line);
195    if line.len() < 4 {
196        return Err(invalid(line_span));
197    }
198    if line.get(1).is_some_and(|token| {
199        matches!(
200            token.kind,
201            TokenKind::DoubleColon | TokenKind::StringLiteral(_)
202        )
203    }) {
204        return Err(GraphSyntaxError {
205            kind: GraphSyntaxErrorKind::ForbiddenImport,
206            span: line_span,
207        });
208    }
209    let mut end = 2_usize;
210    while end + 1 < line.len() && matches!(line[end].kind, TokenKind::DoubleColon) {
211        end += 2;
212    }
213    let target = parse_module_name(&line[1..end]).ok_or_else(|| invalid(line_span))?;
214    let [
215        Token {
216            kind: TokenKind::Identifier(as_word),
217            ..
218        },
219        Token {
220            kind: TokenKind::Identifier(alias),
221            ..
222        },
223    ] = &line[end..]
224    else {
225        return Err(invalid(line_span));
226    };
227    if as_word != graph_names::AS || !valid_graph_name(alias) {
228        return Err(invalid(line_span));
229    }
230    Ok(GraphImport {
231        target,
232        alias: alias.clone(),
233        span: line_span,
234    })
235}

237/// Reads a qualified module ID from alternating name and `::` tokens.
238fn parse_module_name(tokens: &[Token]) -> Option<String> {
239    if tokens.is_empty() || tokens.len().is_multiple_of(2) {
240        return None;
241    }
242    let mut parts = Vec::new();
243    for (index, token) in tokens.iter().enumerate() {
244        if index % 2 == 0 {
245            let TokenKind::Identifier(name) = &token.kind else {
246                return None;
247            };
248            if !valid_graph_name(name) {
249                return None;
250            }
251            parts.push(name.as_str());
252        } else if !matches!(token.kind, TokenKind::DoubleColon) {
253            return None;
254        }
255    }
256    Some(parts.join("::"))
257}

259/// Applies the exact ASCII snake-name grammar and excludes language keywords.
260fn valid_graph_name(value: &str) -> bool {
261    value.split('_').all(|part| {
262        let mut bytes = part.bytes();
263        bytes.next().is_some_and(|byte| byte.is_ascii_lowercase())
264            && bytes.all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit())
265    }) && !names::is_protected_name(value)
266        && !matches!(
267            value,
268            graph_names::IMPORT | graph_names::AS | graph_names::PUBLIC
269        )
270}

272/// Classifies an unsupported symbol on an import line as a forbidden form.
273fn forbidden_line_at(bytes: &[u8], offset: u64) -> bool {
274    let end = usize::try_from(offset)
275        .unwrap_or(bytes.len())
276        .min(bytes.len());
277    let start = bytes[..end]
278        .iter()
279        .rposition(|byte| matches!(byte, b'\n' | b'\r'))
280        .map_or(0, |index| index + 1);
281    let prefix = bytes[start..end]
282        .iter()
283        .copied()
284        .skip_while(|byte| matches!(byte, b' ' | b'\t'))
285        .collect::<Vec<_>>();
286    prefix.starts_with(graph_names::IMPORT.as_bytes())
287        || prefix.starts_with(format!("{} {}", graph_names::PUBLIC, graph_names::IMPORT).as_bytes())
288}

290/// Creates one stable invalid-syntax error at the original-byte location.
291fn invalid(span: ByteSpan) -> GraphSyntaxError {
292    GraphSyntaxError {
293        kind: GraphSyntaxErrorKind::InvalidSyntax,
294        span,
295    }
296}

298/// Returns the exact original-byte span of a nonempty token line.
299fn token_span(line: &[Token]) -> ByteSpan {
300    ByteSpan::new(line[0].span.start(), line[line.len() - 1].span.end())
301        .expect("ordered token offsets must form a span")
302}
303
304/// Creates a checked original-byte half-open span from lexer offsets.
305fn span(start: usize, end: usize) -> ByteSpan {
306    ByteSpan::new(start as u64, end as u64).expect("ordered lexer offsets must form a span")
307}
