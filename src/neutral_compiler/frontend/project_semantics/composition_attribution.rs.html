16/// Traces one materialized path without fabricating source bytes for vocabulary defaults.
17pub(super) fn origin(
18    resolver: &mut Resolver<'_>,
19    root: &Root,
20    ty: &T,
21    path: &[P],
22    kind: K,
23) -> Result<Option<A>, E> {
24    let attribution = trace(resolver, root, ty, &root.tokens, path, 0)?;
25    // A copied materialized vocabulary default is not a newly applied default.
26    // Its original facts remain on the reuse target; unavailable is safer than relabelling supplied data.
27    if kind != K::Defaulted && matches!(attribution, Some(A::Vocabulary { .. })) {
28        return Ok(None);
29    }
30    Ok(attribution)
31}

33/// Builds a checked original-byte location for a complete initializer or selected subexpression.
34fn location(root: &Root, tokens: &[Token]) -> Result<Option<A>, E> {
35    let (Some(first), Some(last)) = (tokens.first(), tokens.last()) else {
36        return Ok(None);
37    };
38    let span = ByteSpan::new(first.span.start(), last.span.end())
39        .map_err(|_| fail(codes::INVALID_SOURCE, Some(root.location)))?;
40    Ok(Some(A::Source(SourceLocation::new(
41        root.location.source(),
42        span,
43    ))))
44}

46/// Selects one object member without confusing member-name tokens with its payload.
47fn member<'a>(tokens: &'a [Token], name: &str) -> Result<Option<&'a [Token]>, E> {
48    if tokens.len() < 2 {
49        return Ok(None);
50    }
51    for part in split(&tokens[1..tokens.len() - 1])? {
52        if matches!(part.first().map(|t| &t.kind), Some(TokenKind::Identifier(n)) if n == name) {
53            return part
54                .get(2..)
55                .map(Some)
56                .ok_or_else(|| fail(codes::INVALID_SOURCE, None));
57        }
58    }
59    Ok(None)
60}

62/// Follows bounded source reuse, lists, selected variants and closed default definitions.
63fn trace(
64    resolver: &mut Resolver<'_>,
65    root: &Root,
66    ty: &T,
67    tokens: &[Token],
68    path: &[P],
69    depth: usize,
70) -> Result<Option<A>, E> {
71    resolver.step(
72        tokens.len() as u64 + path.len() as u64 + 1,
73        Some(root.location),
74    )?;
75    if depth > PROJECT_MAX_DEPTH {
76        return Err(fail(codes::LIMIT, Some(root.location)));
77    }
78    if let Some((name, end)) = parse_name(tokens, 0)
79        && end == tokens.len()
80    {
81        let owner = resolver.name(root, name.alias, name.name, name.span, false)?;
82        let target = retain(resolver.roots[&owner].try_clone())?;
83        return trace(resolver, &target, ty, &target.tokens, path, depth + 1);
84    }
85    let Some((segment, rest)) = path.split_first() else {
86        return location(root, tokens);
87    };
88    let mut ty = ty;
89    while let T::Nullable(inner) = ty {
90        ty = inner;
91    }
92    match segment {
93        P::Element(index) => {
94            let T::List(inner) = ty else {
95                return Err(fail(codes::INCOMPATIBLE_VALUE, Some(root.location)));
96            };
97            let interior = tokens
98                .get(1..tokens.len().saturating_sub(1))
99                .ok_or_else(|| fail(codes::INVALID_SOURCE, Some(root.location)))?;
100            let items = split(interior)?;
101            let item = usize::try_from(*index)
102                .ok()
103                .and_then(|i| items.get(i))
104                .ok_or_else(|| fail(codes::INCOMPATIBLE_VALUE, Some(root.location)))?;
105            trace(resolver, root, inner, item, rest, depth + 1)
106        }
107        P::Payload => {
108            let B::Variant(alternatives) = resolver.body(ty)? else {
109                return Err(fail(codes::INCOMPATIBLE_VALUE, Some(root.location)));
110            };
111            let tag = member(tokens, profile::TAG)?
112                .ok_or_else(|| fail(codes::INVALID_SOURCE, Some(root.location)))?;
113            let Some(Token {
114                kind: TokenKind::StringLiteral(tag),
115                ..
116            }) = tag.first()
117            else {
118                return Err(fail(codes::INVALID_SOURCE, Some(root.location)));
119            };
120            let selected = alternatives
121                .iter()
122                .find(|a| a.tag == tag.value)
123                .ok_or_else(|| fail(codes::UNKNOWN_TAG, Some(root.location)))?;
124            let payload = member(tokens, profile::PAYLOAD)?
125                .ok_or_else(|| fail(codes::INVALID_SOURCE, Some(root.location)))?;
126            trace(resolver, root, &selected.ty, payload, rest, depth + 1)
127        }
128        P::Field(name) => field(resolver, root, ty, tokens, name, rest, depth),
129    }
130}

132/// Attributes supplied fields to their source subexpression and omitted defaults to their real contract owner.
133fn field(
134    resolver: &mut Resolver<'_>,
135    root: &Root,
136    ty: &T,
137    tokens: &[Token],
138    name: &str,
139    rest: &[P],
140    depth: usize,
141) -> Result<Option<A>, E> {
142    let B::Record(fields) = resolver.body(ty)? else {
143        return Err(fail(codes::INCOMPATIBLE_VALUE, Some(root.location)));
144    };
145    let field = fields
146        .iter()
147        .find(|f| f.name == name)
148        .ok_or_else(|| fail(codes::INCOMPATIBLE_VALUE, Some(root.location)))?;
149    if let Some(value) = member(tokens, name)? {
150        return trace(resolver, root, &field.ty, value, rest, depth + 1);
151    }
152    if field.default.is_none() {
153        return location(root, tokens.last().map_or(&[], std::slice::from_ref));
154    }
155    match ty {
156        T::VocabularyNominal {
157            identity,
158            version,
159            name: type_name,
160        } => Ok(Some(A::Vocabulary {
161            identity: retain(identity.try_clone())?,
162            version: retain(version.try_clone())?,
163            type_name: retain(type_name.try_clone())?,
164            field_name: retain(text(name))?,
165            span: None,
166        })),
167        T::Nominal(owner) => {
168            let definition = retain(resolver.roots[owner].try_clone())?;
169            for part in split(&definition.tokens)? {
170                let Some(eq) = part
171                    .iter()
172                    .position(|t| matches!(t.kind, TokenKind::Equals))
173                else {
174                    continue;
175                };
176                if matches!(eq.checked_sub(1).and_then(|i| part.get(i)).map(|t| &t.kind), Some(TokenKind::Identifier(n)) if n == name)
177                {
178                    return trace(
179                        resolver,
180                        &definition,
181                        &field.ty,
182                        &part[eq + 1..],
183                        rest,
184                        depth + 1,
185                    );
186                }
187            }
188            Ok(None)
189        }
190        _ => Err(fail(codes::INCOMPATIBLE_VALUE, Some(root.location))),
191    }
