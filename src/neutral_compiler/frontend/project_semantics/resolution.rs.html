3//! Resolve names, types and value dependencies across captured modules.
4//!
5//! Import aliases select visible symbols; nominal identities determine type
6//! compatibility. Public closure and exposed-reference checks prevent private
7//! targets leaking through exports, while dependency walks detect semantic cycles.
8//! Source occurrences are retained for diagnostics; no path or URL is acquired.
9
10use super::{
11    BTreeMap, BTreeSet, ByteSpan, ModuleContext, NameOccurrence, ProjectDependency,
12    ProjectDependencyKind, ProjectSemanticDiagnostic, ProjectSymbolKind, ProjectVocabularySet,
13    Root, SourceContentDigest, SourceLocation, Token, TokenKind, TypeExpr, diagnostic, diagnostics,
14    key,
15};

17/// Resolves nominal type occurrences and public signature closure.
18#[expect(
19    clippy::too_many_arguments,
20    reason = "resolution requires explicit source, declaration, and result sinks"
21)]
22pub(super) fn resolve_type(
23    ty: &TypeExpr,
24    owner: &(String, String),
25    root: &Root,
26    module: &ModuleContext,
27    roots: &BTreeMap<(String, String), Root>,
28    vocabularies: &ProjectVocabularySet,
29    edges: &mut Vec<ProjectDependency>,
30    errors: &mut Vec<ProjectSemanticDiagnostic>,
31    under_ref: bool,
32) {
33    match ty {
34        TypeExpr::Nominal(alias, name, span) => {
35            if let Some(vocabulary) = alias
36                .as_ref()
37                .and_then(|alias| vocabularies.resolve(&owner.0, alias))
38            {
39                if vocabulary.public_type(name).is_none() {
40                    errors.push(diagnostic(
41                        diagnostics::PRIVATE_VOCABULARY_TYPE,
42                        &owner.0,
43                        module.digest,
44                        *span,
45                    ));
46                }
47                return;
48            }
49            let occurrence = NameOccurrence {
50                alias: alias.clone(),
51                name: name.clone(),
52                span: *span,
53            };
54            if let Some(target) = resolve_name(&occurrence, owner, module, roots, errors) {
55                let target_root = &roots[&target];
56                if target_root.symbol.kind == ProjectSymbolKind::Record {
57                    if root.symbol.is_public && !target_root.symbol.is_public {
58                        errors.push(diagnostic(
59                            diagnostics::PRIVATE_PUBLIC_TYPE,
60                            &owner.0,
61                            module.digest,
62                            *span,
63                        ));
64                    }
65                    let kind = if under_ref {
66                        ProjectDependencyKind::ReferenceType
67                    } else {
68                        ProjectDependencyKind::Type
69                    };
70                    edges.push(dependency(root, target_root, kind, module.digest, *span));
71                } else {
72                    errors.push(diagnostic(
73                        diagnostics::INACCESSIBLE_NAME,
74                        &owner.0,
75                        module.digest,
76                        *span,
77                    ));
78                }
79            }
80        }
81        TypeExpr::List(inner) | TypeExpr::Nullable(inner) => {
82            resolve_type(
83                inner,
84                owner,
85                root,
86                module,
87                roots,
88                vocabularies,
89                edges,
90                errors,
91                under_ref,
92            );
93        }
94        TypeExpr::Ref(inner) => {
95            resolve_type(
96                inner,
97                owner,
98                root,
99                module,
100                roots,
101                vocabularies,
102                edges,
103                errors,
104                true,
105            );
106        }
107        _ => {}
108    }
109}

111/// Resolves every value and reference occurrence within one immutable binding.
112#[expect(
113    clippy::too_many_lines,
114    reason = "value and identity references share one ordered token walk"
115)]
116#[expect(
117    clippy::too_many_arguments,
118    reason = "explicit context prevents ambient resolution"
119)]
120pub(super) fn resolve_value(
121    root: &Root,
122    owner: &(String, String),
123    module: &ModuleContext,
124    modules: &BTreeMap<String, ModuleContext>,
125    roots: &BTreeMap<(String, String), Root>,
126    vocabularies: &ProjectVocabularySet,
127    edges: &mut Vec<ProjectDependency>,
128    errors: &mut Vec<ProjectSemanticDiagnostic>,
129) {
130    let tokens = &root.value;
131    let mut index = 0;
132    while index < tokens.len() {
133        if matches!(tokens[index].kind, TokenKind::RefValue) {
134            let Some((occurrence, end)) = parse_reference(tokens, index) else {
135                errors.push(diagnostic(
136                    diagnostics::INVALID_REFERENCE,
137                    &owner.0,
138                    module.digest,
139                    tokens[index].span,
140                ));
141                index += 1;
142                continue;
143            };
144            if let Some(target) = resolve_name(&occurrence, owner, module, roots, errors) {
145                let target_root = &roots[&target];
146                if target_root.symbol.kind == ProjectSymbolKind::Binding {
147                    if root.symbol.is_public && !target_root.symbol.is_public {
148                        errors.push(diagnostic(
149                            diagnostics::PRIVATE_REFERENCE,
150                            &owner.0,
151                            module.digest,
152                            occurrence.span,
153                        ));
154                    }
155                    if tokens.len() == end && index == 0 {
156                        if let Some(TypeExpr::Ref(inner)) = &root.declared_type {
157                            if !target_root
158                                .declared_type
159                                .as_ref()
160                                .is_some_and(|target_type| {
161                                    same_type(
162                                        inner,
163                                        &owner.0,
164                                        target_type,
165                                        &target.0,
166                                        modules,
167                                        vocabularies,
168                                    )
169                                })
170                            {
171                                errors.push(diagnostic(
172                                    diagnostics::TYPE_MISMATCH,
173                                    &owner.0,
174                                    module.digest,
175                                    occurrence.span,
176                                ));
177                            }
178                        } else {
179                            errors.push(diagnostic(
180                                diagnostics::TYPE_MISMATCH,
181                                &owner.0,
182                                module.digest,
183                                occurrence.span,
184                            ));
185                        }
186                    }
187                    edges.push(dependency(
188                        root,
189                        target_root,
190                        ProjectDependencyKind::Reference,
191                        module.digest,
192                        occurrence.span,
193                    ));
194                } else {
195                    errors.push(diagnostic(
196                        diagnostics::INVALID_REFERENCE,
197                        &owner.0,
198                        module.digest,
199                        occurrence.span,
200                    ));
201                }
202            }
203            index = end;
204            continue;
205        }
206        if let Some((occurrence, end)) = parse_name(tokens, index) {
207            if !matches!(
208                tokens.get(end).map(|token| &token.kind),
209                Some(TokenKind::Colon)
210            ) && let Some(target) = resolve_name(&occurrence, owner, module, roots, errors)
211            {
212                let target_root = &roots[&target];
213                if target_root.symbol.kind == ProjectSymbolKind::Binding {
214                    if index == 0
215                        && end == tokens.len()
216                        && !root
217                            .declared_type
218                            .as_ref()
219                            .zip(target_root.declared_type.as_ref())
220                            .is_some_and(|(left, right)| {
221                                same_type(left, &owner.0, right, &target.0, modules, vocabularies)
222                            })
223                    {
224                        errors.push(diagnostic(
225                            diagnostics::TYPE_MISMATCH,
226                            &owner.0,
227                            module.digest,
228                            occurrence.span,
229                        ));
230                    }
231                    edges.push(dependency(
232                        root,
233                        target_root,
234                        ProjectDependencyKind::Value,
235                        module.digest,
236                        occurrence.span,
237                    ));
238                } else {
239                    errors.push(diagnostic(
240                        diagnostics::INACCESSIBLE_NAME,
241                        &owner.0,
242                        module.digest,
243                        occurrence.span,
244                    ));
245                }
246            }
247            index = end;
248        } else {
249            index += 1;
250        }
251    }
252}

254/// Compares types by resolved nominal module identity, never alias spelling.
255fn same_type(
256    left: &TypeExpr,
257    left_module: &str,
258    right: &TypeExpr,
259    right_module: &str,
260    modules: &BTreeMap<String, ModuleContext>,
261    vocabularies: &ProjectVocabularySet,
262) -> bool {
263    match (left, right) {
264        (TypeExpr::Num, TypeExpr::Num)
265        | (TypeExpr::String, TypeExpr::String)
266        | (TypeExpr::Bool, TypeExpr::Bool)
267        | (TypeExpr::Url, TypeExpr::Url)
268        | (TypeExpr::Path, TypeExpr::Path) => true,
269        (
270            TypeExpr::Nominal(left_alias, left_name, _),
271            TypeExpr::Nominal(right_alias, right_name, _),
272        ) => {
273            let left_vocabulary = left_alias
274                .as_ref()
275                .and_then(|alias| vocabularies.resolve(left_module, alias));
276            let right_vocabulary = right_alias
277                .as_ref()
278                .and_then(|alias| vocabularies.resolve(right_module, alias));
279            if left_vocabulary.is_some() || right_vocabulary.is_some() {
280                return left_vocabulary
281                    .zip(right_vocabulary)
282                    .is_some_and(|(left, right)| {
283                        left.identity() == right.identity()
284                            && left.version() == right.version()
285                            && left_name == right_name
286                    });
287            }
288            let left_owner = left_alias
289                .as_ref()
290                .and_then(|alias| modules[left_module].aliases.get(alias))
291                .map_or(left_module, String::as_str);
292            let right_owner = right_alias
293                .as_ref()
294                .and_then(|alias| modules[right_module].aliases.get(alias))
295                .map_or(right_module, String::as_str);
296            left_owner == right_owner && left_name == right_name
297        }
298        (TypeExpr::List(left), TypeExpr::List(right))
299        | (TypeExpr::Ref(left), TypeExpr::Ref(right))
300        | (TypeExpr::Nullable(left), TypeExpr::Nullable(right)) => same_type(
301            left,
302            left_module,
303            right,
304            right_module,
305            modules,
306            vocabularies,
307        ),
308        _ => false,
309    }
310}

312/// Finds a private reference reachable by ordinary reuse from a public root.
313pub(super) fn exposed_private_reference(
314    roots: &BTreeMap<(String, String), Root>,
315    edges: &[ProjectDependency],
316) -> Option<(String, String)> {
317    let mut queue = roots
318        .iter()
319        .filter(|(_, root)| root.symbol.is_public && root.symbol.kind == ProjectSymbolKind::Binding)
320        .map(|(key, _)| key.clone())
321        .collect::<Vec<_>>();
322    let mut seen = BTreeSet::new();
323    while let Some(current) = queue.pop() {
324        if !seen.insert(current.clone()) {
325            continue;
326        }
327        for edge in edges.iter().filter(|edge| key(edge.from()) == current) {
328            let target = key(edge.to());
329            if edge.kind == ProjectDependencyKind::Reference && !roots[&target].symbol.is_public {
330                return Some(current);
331            }
332            if edge.kind == ProjectDependencyKind::Value {
333                queue.push(target);
334            }
335        }
336    }
337    None
338}

340/// Parses a `ref(name)` or `ref(alias::name)` target at one token offset.
341pub(super) fn parse_reference(tokens: &[Token], index: usize) -> Option<(NameOccurrence, usize)> {
342    if !matches!(tokens.get(index + 1)?.kind, TokenKind::OpenParen) {
343        return None;
344    }
345    let (name, end) = parse_name(tokens, index + 2)?;
346    matches!(tokens.get(end)?.kind, TokenKind::CloseParen).then_some((name, end + 1))
347}

349/// Parses one unqualified or alias-qualified name occurrence.
350pub(super) fn parse_name(tokens: &[Token], index: usize) -> Option<(NameOccurrence, usize)> {
351    let first = tokens.get(index)?;
352    let TokenKind::Identifier(name) = &first.kind else {
353        return None;
354    };
355    if matches!(
356        tokens.get(index + 1).map(|token| &token.kind),
357        Some(TokenKind::DoubleColon)
358    ) {
359        let second = tokens.get(index + 2)?;
360        let TokenKind::Identifier(target) = &second.kind else {
361            return None;
362        };
363        Some((
364            NameOccurrence {
365                alias: Some(name.clone()),
366                name: target.clone(),
367                span: second.span,
368            },
369            index + 3,
370        ))
371    } else {
372        Some((
373            NameOccurrence {
374                alias: None,
375                name: name.clone(),
376                span: first.span,
377            },
378            index + 1,
379        ))
380    }
381}

383/// Resolves one name using only the local module or one explicit import alias.
384pub(super) fn resolve_name(
385    occurrence: &NameOccurrence,
386    owner: &(String, String),
387    module: &ModuleContext,
388    roots: &BTreeMap<(String, String), Root>,
389    errors: &mut Vec<ProjectSemanticDiagnostic>,
390) -> Option<(String, String)> {
391    let target_module = match &occurrence.alias {
392        Some(alias) => module.aliases.get(alias).cloned(),
393        None => Some(owner.0.clone()),
394    };
395    let target = target_module.map(|module| (module, occurrence.name.clone()));
396    match target {
397        Some(ref key)
398            if roots
399                .get(key)
400                .is_some_and(|root| occurrence.alias.is_none() || root.symbol.is_public) =>
401        {
402            Some(key.clone())
403        }
404        _ => {
405            errors.push(diagnostic(
406                diagnostics::INACCESSIBLE_NAME,
407                &owner.0,
408                module.digest,
409                occurrence.span,
410            ));
411            None
412        }
413    }
414}

416/// Constructs an identity edge with separate source provenance.
417pub(super) fn dependency(
418    from: &Root,
419    to: &Root,
420    kind: ProjectDependencyKind,
421    digest: SourceContentDigest,
422    span: ByteSpan,
423) -> ProjectDependency {
424    ProjectDependency {
425        from: from.symbol.identity.clone(),
426        to: to.symbol.identity.clone(),
427        kind,
428        source: SourceLocation::new(digest, span),
429    }
430}

432/// Returns the least root in an ordinary value or embedded-type dependency cycle.
433pub(super) fn cycle_root(
434    roots: &BTreeMap<(String, String), Root>,
435    edges: &[ProjectDependency],
436) -> Option<(String, String)> {
437    let mut remaining = roots.keys().cloned().collect::<BTreeSet<_>>();
438    let mut dependencies = roots
439        .keys()
440        .cloned()
441        .map(|key| (key, BTreeSet::new()))
442        .collect::<BTreeMap<_, _>>();
443    for edge in edges {
444        if matches!(
445            edge.kind,
446            ProjectDependencyKind::Type | ProjectDependencyKind::Value
447        ) {
448            dependencies
449                .get_mut(&key(edge.from()))
450                .expect("edge owner is collected")
451                .insert(key(edge.to()));
452        }
453    }
454    loop {
455        let ready = remaining
456            .iter()
457            .filter(|node| dependencies[*node].is_disjoint(&remaining))
458            .cloned()
459            .collect::<Vec<_>>();
460        if ready.is_empty() {
461            break;
462        }
463        for node in ready {
464            remaining.remove(&node);
465        }
466    }
467    remaining.into_iter().next()
