13/// Private exact-input parse entry, never accepted from callers or deserialized.
14#[derive(Clone)]
15struct Entry {
16    /// Selected capture grammar and immutable feature context.
17    features: Arc<Vec<String>>,
18    /// Exact-byte hash is only a fast filter; bytes must also match.
19    digest: SourceContentDigest,
20    /// Original bytes reject stale entries and digest collisions.
21    bytes: Arc<Vec<u8>>,
22    /// Private parsed declarations with original byte spans.
23    roots: Vec<Root>,
24}

30/// Caller-owned, bounded successor syntax cache with atomic successful generations.
31///
32/// Separate from the old-profile cache: no API accepts external entries or stores
33/// graph, vocabulary, source IDs, controls, materialized values or identity results.
34pub struct CompositionCompilationCache {
35    /// Independent retained original-byte/unit controls.
36    limits: ProjectCacheLimits,
37    /// Previous successful generation in canonical logical-module order.
38    entries: BTreeMap<String, Entry>,
39    /// Exact total original source bytes retained.
40    source_bytes: u64,
41}
42impl CompositionCompilationCache {
43    /// Creates an empty cache with positive independent retention budgets.
44    #[must_use]
45    pub fn new(limits: ProjectCacheLimits) -> Option<Self> {
46        (limits.source_units != 0 && limits.source_bytes != 0).then(|| Self {
47            limits,
48            entries: BTreeMap::new(),
49            source_bytes: 0,
50        })
51    }
52    /// Returns retained units, not request or semantic-result counts.
53    #[must_use]
54    pub fn retained_units(&self) -> usize {
55        self.entries.len()
56    }
57    /// Returns exact retained original bytes, not a promise about total heap usage.
58    #[must_use]
59    pub const fn retained_source_bytes(&self) -> u64 {
60        self.source_bytes
61    }
62    /// Reuses only exact source/module/feature syntax, rebuilding every other phase.
63    ///
64    /// # Errors
65    /// Returns the clean pipeline's failure without publishing a pending cache
66    /// generation. Cancellation, invalid new vocabularies and changed controls
67    /// cannot be bypassed through a syntax hit.
68    pub fn compile(
69        &mut self,
70        captured: &CapturedCompositionProject,
71        cancel: &CancellationToken,
72    ) -> Result<(Arc<Ir>, ProjectCacheStats), E> {
73        self.compile_observed(captured, cancel, &mut |_| Ok(()))
74    }
75    /// Shares atomic generation construction with private phase-fault tests, never accepting public hooks.
76    fn compile_observed(
77        &mut self,
78        captured: &CapturedCompositionProject,
79        cancel: &CancellationToken,
80        observer: &mut impl FnMut(Phase) -> Result<(), E>,
81    ) -> Result<(Arc<Ir>, ProjectCacheStats), E> {
82        self.compile_retention_observed(captured, cancel, observer, &mut || Ok(()))
83    }
84
85    /// Runs request-local fault checkpoints before cache ownership transitions.
86    fn compile_retention_observed(
87        &mut self,
88        captured: &CapturedCompositionProject,
89        cancel: &CancellationToken,
90        observer: &mut impl FnMut(Phase) -> Result<(), E>,
91        retention: &mut impl FnMut() -> Result<(), E>,
92    ) -> Result<(Arc<Ir>, ProjectCacheStats), E> {
93        let mut pending = BTreeMap::new();
94        let mut stats = ProjectCacheStats::default();
95        let mut pending_bytes = 0_u64;
96        let ir = compile_pipeline(
97            captured,
98            cancel,
99            &mut |source| {
100                let previous = self.entries.get(source.module_id());
101                let reusable = previous.filter(|entry| {
102                    entry.features.as_slice() == captured.required_features()
103                        && entry.digest == source.digest()
104                        && entry.bytes.as_ref() == source.bytes()
105                        && entry.roots.iter().all(|root| {
106                            root.owner.module().module_name() == source.module_id()
107                                && root.owner.module().language_behavior_version()
108                                    == neutral_core::profile::V1_SOURCE_PROFILE
109                        })
110                });
111                let roots = if let Some(entry) = reusable {
112                    if cancel.is_cancelled() {
113                        return Err(fail(codes::CANCELLED, None));
114                    }
115                    retention()?;
116                    stats.reused_units += 1;
117                    entry
118                        .roots
119                        .try_clone()
120                        .map_err(|_| fail(codes::LIMIT, None))?
121                } else {
122                    stats.parsed_units += 1;
123                    stats.rejected_entries += u64::from(previous.is_some());
124                    parse(source, captured, cancel)?
125                };
126                let length = source.bytes().len() as u64;
127                if (pending.len() as u64) < self.limits.source_units
128                    && pending_bytes
129                        .checked_add(length)
130                        .is_some_and(|n| n <= self.limits.source_bytes)
131                {
132                    if cancel.is_cancelled() {
133                        return Err(fail(codes::CANCELLED, None));
134                    }
135                    retention()?;
136                    let key = text(source.module_id()).map_err(|_| fail(codes::LIMIT, None))?;
137                    retention()?;
138                    let retained_roots = roots.try_clone().map_err(|_| fail(codes::LIMIT, None))?;
139                    retention()?;
140                    pending
141                        .insert(
142                            key,
143                            Entry {
144                                features: captured.shared_features(),
145                                digest: source.digest(),
146                                bytes: source.shared_bytes(),
147                                roots: retained_roots,
148                            },
149                        )
150                        .map_err(|_| fail(codes::LIMIT, None))?;
151                    pending_bytes += length;
152                }
153                Ok(roots)
154            },
155            observer,
156        )?;
157        if cancel.is_cancelled() {
158            return Err(fail(codes::CANCELLED, None));
159        }
160        // This type's dedicated pipeline fixes capture/grammar selection; no
161        // package version or public fingerprint is ever a cache key.
162        debug_assert_eq!(ir.schema, profile::PROJECT_IR_SCHEMA);
163        self.entries = pending;
164        self.source_bytes = pending_bytes;
165        Ok((ir, stats))
166    }
167}
