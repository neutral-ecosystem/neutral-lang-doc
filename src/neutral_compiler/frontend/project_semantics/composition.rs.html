51/// Safe successor compilation failure; no variant publishes partial logical output.
52#[derive(Clone, Debug, Eq, PartialEq)]
53pub struct CompositionCompileFailure {
54    /// Frozen source diagnostic classification.
55    pub code: &'static str,
56    /// Checked original-byte occurrence when available.
57    pub location: Option<SourceLocation>,
58}
59impl CompositionCompileFailure {
60    /// Returns the explicitly selected successor whole-request envelope.
61    #[must_use]
62    pub const fn schema(&self) -> &'static str {
63        profile::PROJECT_RESULT_SCHEMA
64    }
65}

91/// Parsed declaration; initializer/member tokens are private and reused within this request only.
92#[derive(Clone)]
93struct Root {
94    /// Stable resolved owner.
95    owner: ModuleSymbolIdentity,
96    /// Explicit source visibility.
97    public: bool,
98    /// Complete original declaration span.
99    location: SourceLocation,
100    /// Resolved binding type, absent for nominal declarations.
101    ty: Option<TypeExpr>,
102    /// Record fields or variant alternatives in source order.
103    fields: Vec<(String, TypeExpr)>,
104    /// Initializer or field/default body tokens.
105    tokens: Vec<Token>,
106    /// Nominal variant rather than record.
107    variant: bool,
108}
109
110impl neutral_core::allocation::TryClone for Root {
111    /// Copies preflighted syntax into a new cache generation without infallible recursive clones.
112    fn try_clone(&self) -> Result<Self, neutral_core::allocation::AllocationError> {
113        Ok(Self {
114            owner: self.owner.try_clone()?,
115            public: self.public,
116            location: self.location,
117            ty: self.ty.try_clone()?,
118            fields: self.fields.try_clone()?,
119            tokens: self.tokens.try_clone()?,
120            variant: self.variant,
121        })
122    }
123}

125/// Request-local resolver; no filesystem, URL acquisition, ambient environment or persistent cache.
126struct Resolver<'a> {
127    /// Complete explicit captured closure.
128    captured: &'a CapturedCompositionProject,
129    /// Independently graph-checked explicit imports/aliases.
130    graph: graph::Graph,
131    /// Canonical parsed declarations, including private/disconnected units.
132    roots: BTreeMap<ModuleSymbolIdentity, Root>,
133    /// Exact resolved binding signatures.
134    types: BTreeMap<ModuleSymbolIdentity, T>,
135    /// Complete source contract bodies, before or after default materialization.
136    definitions: Vec<SourceCompositionDefinition>,
137    /// Request-only memoized immutable initializer values.
138    values: BTreeMap<ModuleSymbolIdentity, V>,
139    /// Active ordinary reuse chain; references never enter it.
140    active: BTreeSet<ModuleSymbolIdentity>,
141    /// Source-accounted ordinary reuse occurrences.
142    provenance: Vec<ProjectProvenance>,
143    /// Charged source/resolution traversal work.
144    work: u64,
145    /// Explicit cooperative cancellation.
146    cancel: &'a CancellationToken,
147}

149/// Compiles every capture/2 source under the shared record/variant/default/constraint model.
150///
151/// The caller must independently open the returned producer data through the
152/// successor reader before encoding it. No root selection or host acquisition is
153/// part of compilation, and no package release or source header selects this API.
154///
155/// # Errors
156/// Rejects graph, grammar, visibility, cycles, type/default/value, bounds or
157/// cancellation failures atomically without returning any successful prefix.
158pub fn compile_composition_project(
159    captured: &CapturedCompositionProject,
160    cancel: &CancellationToken,
161) -> Result<Arc<Ir>, E> {
162    compile_with_parser(captured, cancel, &mut |source| {
163        parse(source, captured, cancel)
164    })
165}

167/// Shares the complete clean pipeline with cache-backed private parsing, never cached semantic authority.
168fn compile_with_parser(
169    captured: &CapturedCompositionProject,
170    cancel: &CancellationToken,
171    parser: &mut impl FnMut(&crate::CapturedProjectSource) -> Result<Vec<Root>, E>,
172) -> Result<Arc<Ir>, E> {
173    compile_observed(captured, cancel, parser, &mut |_| Ok(()))
174}

176/// Private phase boundaries for cooperative publication checks and deterministic fault tests.
177#[derive(Clone, Copy, Debug, Eq, PartialEq)]
178enum Phase {
179    /// Complete import/SCC validation before private parsing.
180    Graph,
181    /// Real or exact-cache source parser invocation.
182    Parse,
183    /// Resolve source signatures and nominal owners.
184    Signatures,
185    /// Check embedded/public contract closure before defaults.
186    InitialScope,
187    /// Materialize source-owned closed defaults.
188    Defaults,
189    /// Recheck finalized source and vocabulary contracts together.
190    FinalScope,
191    /// Resolve complete immutable binding initializers.
192    Values,
193    /// Validate materialized values, restrictions and typed references.
194    Bindings,
195    /// Construct complete IR and independently retained companions.
196    Assemble,
197    /// Final atomic publication, also guarding pending cache generations.
198    Publish,
199}

201/// Runs a private observer and checks cancellation on both sides of a phase boundary.
202fn checkpoint(
203    phase: Phase,
204    cancel: &CancellationToken,
205    observer: &mut impl FnMut(Phase) -> Result<(), E>,
206) -> Result<(), E> {
207    if cancel.is_cancelled() {
208        return Err(fail(codes::CANCELLED, None));
209    }
210    observer(phase)?;
211    if cancel.is_cancelled() {
212        return Err(fail(codes::CANCELLED, None));
213    }
214    Ok(())
215}

217/// Executes every complete-project phase with no public hooks or partial-output publication.
218fn compile_observed(
219    captured: &CapturedCompositionProject,
220    cancel: &CancellationToken,
221    parser: &mut impl FnMut(&crate::CapturedProjectSource) -> Result<Vec<Root>, E>,
222    observer: &mut impl FnMut(Phase) -> Result<(), E>,
223) -> Result<Arc<Ir>, E> {
224    checkpoint(Phase::Graph, cancel, observer)?;
225    let graph = graph::build(captured, cancel)?;
226    let mut resolver = Resolver {
227        captured,
228        graph,
229        roots: BTreeMap::new(),
230        types: BTreeMap::new(),
231        definitions: Vec::new(),
232        values: BTreeMap::new(),
233        active: BTreeSet::new(),
234        provenance: Vec::new(),
235        work: 0,
236        cancel,
237    };
238    resolver.step(1, None)?;
239    checkpoint(Phase::Parse, cancel, observer)?;
240    for source in captured.sources() {
241        resolver.step(source.bytes().len() as u64, None)?;
242        for root in parser(source)? {
243            if resolver.roots.len() as u64 >= captured.limits().values().declarations {
244                return Err(fail(codes::LIMIT, Some(root.location)));
245            }
246            resolver.step(resolver.roots.len() as u64 + 1, Some(root.location))?;
247            let location = root.location;
248            if resolver
249                .roots
250                .insert(retain(root.owner.try_clone())?, root)
251                .map_err(|_| fail(codes::LIMIT, Some(location)))?
252                .is_some()
253            {
254                return Err(fail(codes::DUPLICATE_MEMBER, Some(location)));
255            }
256        }
257    }
258    let roots = resolver.retained_roots()?;
259    checkpoint(Phase::Signatures, cancel, observer)?;
260    resolver.signatures(&roots)?;
261    // Visibility and embedded closure precede all source default/value interpretation.
262    checkpoint(Phase::InitialScope, cancel, observer)?;
263    validate_composition_scope(
264        captured.catalogue(),
265        retain(resolver.definitions.try_clone())?,
266        captured.composition_limits(),
267        cancel,
268    )
269    .map_err(|e| contract(&e, None))?;
270    checkpoint(Phase::Defaults, cancel, observer)?;
271    resolver.source_defaults(&roots)?;
272    checkpoint(Phase::FinalScope, cancel, observer)?;
273    let scope = Arc::try_new(
274        validate_composition_scope(
275            captured.catalogue(),
276            retain(resolver.definitions.try_clone())?,
277            captured.composition_limits(),
278            cancel,
279        )
280        .map_err(|e| contract(&e, None))?,
281    )
282    .map_err(|_| fail(codes::LIMIT, None))?;
283    resolver.definitions = retain(copy_slice(scope.sources()))?;
284    checkpoint(Phase::Values, cancel, observer)?;
285    let mut bindings = Vec::new();
286    for root in roots.iter().filter(|r| r.ty.is_some()) {
287        let value = resolver.binding(&root.owner, 0)?;
288        resolver.step(1, Some(root.location))?;
289        reserve(&mut bindings, 1)?;
290        bindings.push(CompositionBinding {
291            owner: retain(root.owner.try_clone())?,
292            public: root.public,
293            ty: retain(resolver.types[&root.owner].try_clone())?,
294            value,
295        });
296    }
297    checkpoint(Phase::Bindings, cancel, observer)?;
298    let bindings = validate_composition_bindings(
299        scope.clone(),
300        &bindings,
301        captured.composition_limits(),
302        cancel,
303    )
304    .map_err(|e| contract(&e, None))?;
305    checkpoint(Phase::Assemble, cancel, observer)?;
306    let ir = assemble(&mut resolver, &scope, &bindings)?;
307    checkpoint(Phase::Publish, cancel, observer)?;
308    Ok(ir)
309}

311/// Maps fallible retention failures to the frozen bounded-resource source envelope.
312fn retain<T>(result: Result<T, neutral_core::allocation::AllocationError>) -> Result<T, E> {
313    result.map_err(|_| fail(codes::LIMIT, None))
314}
315/// Reserves unpublished vector growth before insertion, preserving existing state on failure.
316fn reserve<T>(values: &mut Vec<T>, additional: usize) -> Result<(), E> {
317    values
318        .try_retain(additional)
319        .map_err(|_| fail(codes::LIMIT, None))
320}
321/// Maps shared semantic failures without exposing private contract text.
322fn contract(e: &CompositionError, location: Option<SourceLocation>) -> E {
323    fail(
324        match e {
325            CompositionError::Cancelled => codes::CANCELLED,
326            CompositionError::Limit
327            | CompositionError::Allocation
328            | CompositionError::InvalidLimits => codes::LIMIT,
329            CompositionError::PrivateType => codes::PRIVATE_EXPOSURE,
330            CompositionError::EmbeddedCycle => codes::EMBEDDED_CYCLE,
331            CompositionError::InvalidRestrictions | CompositionError::DuplicateChoice => {
332                codes::RESTRICTION_VIOLATION
333            }
334            _ => codes::INCOMPATIBLE_VALUE,
335        },
336        location,
337    )
338}
339/// Creates one non-authoritative original-byte diagnostic.
340fn fail(code: &'static str, location: Option<SourceLocation>) -> E {
341    E { code, location }
