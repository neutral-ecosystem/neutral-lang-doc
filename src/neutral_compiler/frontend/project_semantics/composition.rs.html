56/// Borrowed source name; resolving a qualifier never allocates temporary spellings.
57struct BorrowedOccurrence<'a> {
58    /// Source-local qualifier, if present.
59    alias: Option<&'a str>,
60    /// Exact declaration spelling.
61    name: &'a str,
62    /// Original occurrence for checked diagnostics.
63    span: ByteSpan,
64}
65/// Recognizes a local or qualified name without copying token storage.
66fn parse_name(tokens: &[Token], index: usize) -> Option<(BorrowedOccurrence<'_>, usize)> {
67    let first = tokens.get(index)?;
68    let TokenKind::Identifier(name) = &first.kind else {
69        return None;
70    };
71    if matches!(
72        tokens.get(index + 1).map(|t| &t.kind),
73        Some(TokenKind::DoubleColon)
74    ) {
75        let second = tokens.get(index + 2)?;
76        let TokenKind::Identifier(target) = &second.kind else {
77            return None;
78        };
79        Some((
80            BorrowedOccurrence {
81                alias: Some(name),
82                name: target,
83                span: second.span,
84            },
85            index + 3,
86        ))
87    } else {
88        Some((
89            BorrowedOccurrence {
90                alias: None,
91                name,
92                span: first.span,
93            },
94            index + 1,
95        ))
96    }
97}
98/// Recognizes a reference's borrowed target without retaining an intermediate name.
99fn parse_reference(tokens: &[Token], index: usize) -> Option<(BorrowedOccurrence<'_>, usize)> {
100    if !matches!(tokens.get(index + 1)?.kind, TokenKind::OpenParen) {
101        return None;
102    }
103    let (name, end) = parse_name(tokens, index + 2)?;
104    matches!(tokens.get(end)?.kind, TokenKind::CloseParen).then_some((name, end + 1))
105}

205/// Private phase boundaries for cooperative publication checks and deterministic fault tests.
206#[derive(Clone, Copy, Debug, Eq, PartialEq)]
207enum Phase {
208    /// Complete import/SCC validation before private parsing.
209    Graph,
210    /// Real or exact-cache source parser invocation.
211    Parse,
212    /// Resolve source signatures and nominal owners.
213    Signatures,
214    /// Check embedded/public contract closure before defaults.
215    InitialScope,
216    /// Materialize source-owned closed defaults.
217    Defaults,
218    /// Recheck finalized source and vocabulary contracts together.
219    FinalScope,
220    /// Resolve complete immutable binding initializers.
221    Values,
222    /// Validate materialized values, restrictions and typed references.
223    Bindings,
224    /// Construct complete IR and independently retained companions.
225    Assemble,
226    /// Final atomic publication, also guarding pending cache generations.
227    Publish,
228}

230/// Runs a private observer and checks cancellation on both sides of a phase boundary.
231fn checkpoint(
232    phase: Phase,
233    cancel: &CancellationToken,
234    observer: &mut impl FnMut(Phase) -> Result<(), E>,
235) -> Result<(), E> {
236    if cancel.is_cancelled() {
237        return Err(fail(codes::CANCELLED, None));
238    }
239    observer(phase)?;
240    if cancel.is_cancelled() {
241        return Err(fail(codes::CANCELLED, None));
242    }
243    Ok(())
244}

246/// Executes every complete-project phase with no public hooks or partial-output publication.
247fn compile_observed(
248    captured: &CapturedCompositionProject,
249    cancel: &CancellationToken,
250    parser: &mut impl FnMut(&crate::CapturedProjectSource) -> Result<Vec<Root>, E>,
251    observer: &mut impl FnMut(Phase) -> Result<(), E>,
252) -> Result<Arc<Ir>, E> {
253    checkpoint(Phase::Graph, cancel, observer)?;
254    let graph = graph::build(captured, cancel)?;
255    let mut resolver = Resolver {
256        captured,
257        graph,
258        roots: BTreeMap::new(),
259        types: BTreeMap::new(),
260        definitions: Vec::new(),
261        values: BTreeMap::new(),
262        active: BTreeSet::new(),
263        provenance: Vec::new(),
264        work: 0,
265        cancel,
266    };
267    resolver.step(1, None)?;
268    checkpoint(Phase::Parse, cancel, observer)?;
269    for source in captured.sources() {
270        resolver.step(source.bytes().len() as u64, None)?;
271        for root in parser(source)? {
272            if resolver.roots.len() as u64 >= captured.limits().values().declarations {
273                return Err(fail(codes::LIMIT, Some(root.location)));
274            }
275            resolver.step(resolver.roots.len() as u64 + 1, Some(root.location))?;
276            let location = root.location;
277            if resolver
278                .roots
279                .insert(retain(root.owner.try_clone())?, root)
280                .map_err(|_| fail(codes::LIMIT, Some(location)))?
281                .is_some()
282            {
283                return Err(fail(codes::DUPLICATE_MEMBER, Some(location)));
284            }
285        }
286    }
287    let roots = resolver.retained_roots()?;
288    checkpoint(Phase::Signatures, cancel, observer)?;
289    resolver.signatures(&roots)?;
290    // Visibility and embedded closure precede all source default/value interpretation.
291    checkpoint(Phase::InitialScope, cancel, observer)?;
292    validate_composition_scope(
293        captured.catalogue(),
294        retain(resolver.definitions.try_clone())?,
295        captured.composition_limits(),
296        cancel,
297    )
298    .map_err(|e| contract(&e, None))?;
299    checkpoint(Phase::Defaults, cancel, observer)?;
300    resolver.source_defaults(&roots)?;
301    checkpoint(Phase::FinalScope, cancel, observer)?;
302    let scope = Arc::try_new(
303        validate_composition_scope(
304            captured.catalogue(),
305            retain(resolver.definitions.try_clone())?,
306            captured.composition_limits(),
307            cancel,
308        )
309        .map_err(|e| contract(&e, None))?,
310    )
311    .map_err(|_| fail(codes::LIMIT, None))?;
312    resolver.definitions = retain(copy_slice(scope.sources()))?;
313    checkpoint(Phase::Values, cancel, observer)?;
314    let mut bindings = Vec::new();
315    for root in roots.iter().filter(|r| r.ty.is_some()) {
316        let value = resolver.binding(&root.owner, 0)?;
317        resolver.step(1, Some(root.location))?;
318        reserve(&mut bindings, 1)?;
319        bindings.push(CompositionBinding {
320            owner: retain(root.owner.try_clone())?,
321            public: root.public,
322            ty: retain(resolver.types[&root.owner].try_clone())?,
323            value,
324        });
325    }
326    checkpoint(Phase::Bindings, cancel, observer)?;
327    let bindings = validate_composition_bindings(
328        scope.clone(),
329        &bindings,
330        captured.composition_limits(),
331        cancel,
332    )
333    .map_err(|e| contract(&e, None))?;
334    checkpoint(Phase::Assemble, cancel, observer)?;
335    let ir = assemble(&mut resolver, &scope, &bindings)?;
336    checkpoint(Phase::Publish, cancel, observer)?;
337    Ok(ir)
338}

340/// Builds complete canonical logical data and separately retained source/resource companions.
341fn assemble(
342    resolver: &mut Resolver<'_>,
343    scope: &ValidatedCompositionScope,
344    bindings: &neutral_vocabulary::composition::ValidatedCompositionBindings,
345) -> Result<Arc<Ir>, E> {
346    let captured = resolver.captured;
347    let cap = captured.limits().values();
348    let LoweredDeclarations {
349        declarations,
350        maps,
351        origins,
352    } = declarations(resolver, scope, bindings)?;
353    let CapturedCompanions {
354        modules,
355        sources,
356        vocabulary_sources,
357    } = captured_companions(resolver)?;
358    complete_provenance(resolver, &declarations, bindings)?;
359    let limits = ProjectLimits {
360        modules: cap.source_units,
361        declarations: cap.declarations,
362        import_edges: cap.import_edges,
363        nodes: captured.composition_limits().value_nodes,
364        text_bytes: cap
365            .total_source_bytes
366            .checked_add(cap.total_vocabulary_bytes)
367            .ok_or_else(|| fail(codes::LIMIT, None))?,
368        artifact_bytes: cap.output_bytes,
369    };
370    let facts = captured.resource_facts();
371    let mut ir = Ir {
372        schema: retain(text(profile::PROJECT_IR_SCHEMA))?,
373        modules,
374        declarations,
375        vocabularies: retain(copy_slice(scope.catalogue().bundles()))?,
376        interface_digest: SemanticDigest::from_raw_bytes([0; 32]),
377        sources,
378        source_maps: maps,
379        provenance: std::mem::take(&mut resolver.provenance),
380        limits,
381        resources: ProjectResourceFacts {
382            source_units: facts.source_units(),
383            source_bytes: facts.total_source_bytes(),
384            vocabulary_units: facts.vocabulary_units(),
385            vocabulary_bytes: facts.total_vocabulary_bytes(),
386            declarations: resolver.roots.len() as u64,
387            import_edges: resolver.graph.edges().len() as u64,
388            value_nodes: 0,
389        },
390        vocabulary_sources,
391        origins,
392        composition_limits: captured.composition_limits().policy(),
393        composition_resources: CompositionResourceFacts::from_values([0; 7]),
394    };
395    ir.composition_resources =
396        inspect_composition_ir(&ir, limits, ir.composition_limits, resolver.cancel).map_err(
397            |e| {
398                fail(
399                    match e {
400                        neutral_ir::composition::project::CompositionShapeError::Cancelled => {
401                            codes::CANCELLED
402                        }
403                        neutral_ir::composition::project::CompositionShapeError::Limit => {
404                            codes::LIMIT
405                        }
406                    },
407                    None,
408                )
409            },
410        )?;
411    ir.resources.value_nodes = ir.composition_resources.retained_value_nodes;
412    ir.interface_digest = composition_interface(
413        &ir,
414        IdentityLimits {
415            bytes: neutral_ir::project_identity::MAX_TRANSCRIPT_BYTES,
416            nodes: captured.composition_limits().work,
417        },
418        resolver.cancel,
419    )
420    .map_err(|e| {
421        fail(
422            match e {
423                neutral_ir::project_identity::IdentityError::Cancelled => codes::CANCELLED,
424                _ => codes::LIMIT,
425            },
426            None,
427        )
428    })?
429    .identity();
430    resolver.step(1, None)?;
431    Arc::try_new(ir).map_err(|_| fail(codes::LIMIT, None))
432}

434/// Completes canonical source type and reference edges after all values have been validated.
435fn complete_provenance(
436    resolver: &mut Resolver<'_>,
437    declarations: &[CompositionDeclaration],
438    bindings: &neutral_vocabulary::composition::ValidatedCompositionBindings,
439) -> Result<(), E> {
440    for d in declarations {
441        let location = resolver.roots[&d.identity].location;
442        match &d.signature {
443            S::Binding(t) => type_edges(&d.identity, t, false, location, resolver)?,
444            S::Definition(B::Record(fs)) => {
445                for f in fs {
446                    type_edges(&d.identity, &f.ty, false, location, resolver)?;
447                }
448            }
449            S::Definition(B::Variant(alternatives)) => {
450                for a in alternatives {
451                    type_edges(&d.identity, &a.ty, false, location, resolver)?;
452                }
453            }
454        }
455    }
456    for b in bindings.bindings() {
457        for r in b.references() {
458            resolver.step(1, None)?;
459            reserve(&mut resolver.provenance, 1)?;
460            resolver.provenance.push(ProjectProvenance {
461                from: retain(b.binding().owner.try_clone())?,
462                to: retain(r.target.try_clone())?,
463                kind: Edge::Reference,
464                location: resolver.roots[&b.binding().owner].location,
465            });
466        }
467    }
468    resolver.provenance.sort_unstable_by(|a, b| {
469        (&a.from, a.kind, &a.to, a.location).cmp(&(&b.from, b.kind, &b.to, b.location))
470    });
471    resolver.provenance.dedup();
472    resolver.step(1, None)
473}

475/// Canonical source declarations with separate occurrence and coverage companions.
476struct LoweredDeclarations {
477    /// Complete private/public signatures and values.
478    declarations: Vec<CompositionDeclaration>,
479    /// Complete original-byte declaration coverage.
480    maps: Vec<ProjectSourceMap>,
481    /// Non-semantic supplied/default/null/omitted occurrences.
482    origins: Vec<CompositionOrigin>,
483}
484/// Projects already validated complete source contracts and materialized bindings.
485fn declarations(
486    resolver: &mut Resolver<'_>,
487    scope: &ValidatedCompositionScope,
488    bindings: &neutral_vocabulary::composition::ValidatedCompositionBindings,
489) -> Result<LoweredDeclarations, E> {
490    let mut declarations = Vec::new();
491    let mut maps = Vec::new();
492    let mut origins = Vec::new();
493    let roots = resolver.retained_roots()?;
494    for root in &roots {
495        let (signature, value) = if root.ty.is_some() {
496            let b = bindings
497                .bindings()
498                .iter()
499                .find(|b| b.binding().owner == root.owner)
500                .ok_or_else(|| fail(codes::INCOMPATIBLE_VALUE, Some(root.location)))?;
501            for o in b.origins() {
502                resolver.step(1, Some(root.location))?;
503                reserve(&mut origins, 1)?;
504                origins.push(CompositionOrigin {
505                    binding: retain(root.owner.try_clone())?,
506                    path: retain(o.path.try_clone())?,
507                    kind: o.kind,
508                    attribution: attribution::origin(
509                        resolver,
510                        root,
511                        &b.binding().ty,
512                        &o.path,
513                        o.kind,
514                    )?,
515                });
516            }
517            (
518                S::Binding(retain(b.binding().ty.try_clone())?),
519                Some(retain(b.binding().value.try_clone())?),
520            )
521        } else {
522            let d = scope
523                .sources()
524                .iter()
525                .find(|d| d.owner == root.owner)
526                .ok_or_else(|| fail(codes::INCOMPATIBLE_VALUE, Some(root.location)))?;
527            (S::Definition(retain(d.definition.body.try_clone())?), None)
528        };
529        resolver.step(1, Some(root.location))?;
530        reserve(&mut declarations, 1)?;
531        reserve(&mut maps, 1)?;
532        declarations.push(CompositionDeclaration {
533            identity: retain(root.owner.try_clone())?,
534            public: root.public,
535            signature,
536            value,
537        });
538        maps.push(ProjectSourceMap {
539            declaration: retain(root.owner.try_clone())?,
540            location: root.location,
541        });
542    }
543    Ok(LoweredDeclarations {
544        declarations,
545        maps,
546        origins,
547    })
548}
549/// Exact captured accounting and complete logical module topology.
550struct CapturedCompanions {
551    /// All modules, including disconnected and private-only units.
552    modules: Vec<ProjectModule>,
553    /// Logical source identities and byte lengths.
554    sources: Vec<ProjectSource>,
555    /// Exact locked vocabulary capture evidence.
556    vocabulary_sources: Vec<ProjectVocabularySource>,
557}
558/// Derives checked captured companions without acquiring any input.
559fn captured_companions(resolver: &Resolver<'_>) -> Result<CapturedCompanions, E> {
560    let captured = resolver.captured;
561    let mut modules = Vec::new();
562    let mut sources = Vec::new();
563    reserve(&mut modules, captured.sources().len())?;
564    reserve(&mut sources, captured.sources().len())?;
565    for s in captured.sources() {
566        if resolver.cancel.is_cancelled() {
567            return Err(fail(codes::CANCELLED, None));
568        }
569        let mut imports = Vec::new();
570        for e in resolver
571            .graph
572            .edges()
573            .iter()
574            .filter(|e| e.from() == s.module_id())
575        {
576            if resolver.cancel.is_cancelled() {
577                return Err(fail(codes::CANCELLED, None));
578            }
579            reserve(&mut imports, 1)?;
580            imports.push(retain(text(e.target()))?);
581        }
582        imports.sort_unstable();
583        modules.push(ProjectModule {
584            identity: LogicalModuleIdentity::new(
585                retain(text(V1_SOURCE_PROFILE))?,
586                retain(text(s.module_id()))?,
587            ),
588            imports,
589        });
590        sources.push(ProjectSource {
591            module: retain(text(s.module_id()))?,
592            source_id: retain(text(s.source_id()))?,
593            digest: s.digest(),
594            byte_len: s.bytes().len() as u64,
595        });
596    }
597    let mut vocabulary_sources = Vec::new();
598    reserve(
599        &mut vocabulary_sources,
600        captured.catalogue().bundles().len(),
601    )?;
602    for b in captured.catalogue().bundles() {
603        if resolver.cancel.is_cancelled() {
604            return Err(fail(codes::CANCELLED, None));
605        }
606        let source = captured
607            .vocabularies()
608            .iter()
609            .find(|v| v.lock().identity() == b.identity.identity())
610            .ok_or_else(|| fail(codes::INCOMPATIBLE_VALUE, None))?;
611        vocabulary_sources.push(ProjectVocabularySource {
612            identity: retain(text(b.identity.identity()))?,
613            version: retain(text(b.identity.version()))?,
614            digest: b.identity.content_digest(),
615            byte_len: source.bytes().len() as u64,
616        });
617    }
618    Ok(CapturedCompanions {
619        modules,
620        sources,
621        vocabulary_sources,
622    })

624impl Resolver<'_> {
625    /// Retains bounded syntax for mutable semantic traversal without infallible deep copies.
626    fn retained_roots(&mut self) -> Result<Vec<Root>, E> {
627        let work = self.roots.values().try_fold(0_u64, |sum, root| {
628            sum.checked_add(root.tokens.len() as u64)
629                .and_then(|sum| {
630                    sum.checked_add(root.location.span().end() - root.location.span().start())
631                })
632                .and_then(|sum| sum.checked_add(symbol_work(&root.owner) + 1))
633                .ok_or_else(|| fail(codes::LIMIT, Some(root.location)))
634        })?;
635        self.step(work, None)?;
636        let mut roots = Vec::new();
637        reserve(&mut roots, self.roots.len())?;
638        for root in self.roots.values() {
639            if self.cancel.is_cancelled() {
640                return Err(fail(codes::CANCELLED, Some(root.location)));
641            }
642            roots.push(retain(root.try_clone())?);
643        }
644        self.step(1, None)?;
645        Ok(roots)
646    }
647    /// Resolves all source signature owners before any value evaluation.
648    fn signatures(&mut self, roots: &[Root]) -> Result<(), E> {
649        // Resolve every signature before interpreting a default or initializer.
650        for root in roots {
651            if let Some(ty) = &root.ty {
652                let ty = self.ty(root, ty)?;
653                self.step(self.types.len() as u64 + 1, Some(root.location))?;
654                self.types
655                    .insert(retain(root.owner.try_clone())?, ty)
656                    .map_err(|_| fail(codes::LIMIT, Some(root.location)))?;
657            } else {
658                let mut fields = Vec::new();
659                let mut alternatives = Vec::new();
660                for (name, ty) in &root.fields {
661                    let ty = self.ty(root, ty)?;
662                    self.step(1, Some(root.location))?;
663                    if root.variant {
664                        reserve(&mut alternatives, 1)?;
665                        alternatives.push(CompositionAlternative {
666                            tag: retain(name.try_clone())?,
667                            ty,
668                        });
669                    } else {
670                        reserve(&mut fields, 1)?;
671                        fields.push(CompositionField {
672                            name: retain(name.try_clone())?,
673                            ty,
674                            presence: FieldPresence::Required,
675                            restrictions: FieldRestrictions::default(),
676                            default: None,
677                        });
678                    }
679                }
680                fields.sort_unstable_by(|a, b| a.name.cmp(&b.name));
681                alternatives.sort_unstable_by(|a, b| a.tag.cmp(&b.tag));
682                reserve(&mut self.definitions, 1)?;
683                self.definitions.push(SourceCompositionDefinition {
684                    owner: retain(root.owner.try_clone())?,
685                    definition: CompositionDefinition {
686                        name: retain(text(root.owner.declaration_name()))?,
687                        public: root.public,
688                        body: if root.variant {
689                            B::Variant(alternatives)
690                        } else {
691                            B::Record(fields)
692                        },
693                    },
694                });
695            }
696        }
697        Ok(())
698    }

700    /// Resolves closed source defaults against the complete nominal type set.
701    fn source_defaults(&mut self, roots: &[Root]) -> Result<(), E> {
702        // Parse closed defaults against the complete set of source and vocabulary types.
703        let mut defaults = Vec::new();
704        for root in roots.iter().filter(|r| r.ty.is_none() && !r.variant) {
705            for member in split(&root.tokens)? {
706                if let Some(eq) = member
707                    .iter()
708                    .position(|t| matches!(t.kind, TokenKind::Equals))
709                {
710                    let Some(Token {
711                        kind: TokenKind::Identifier(name),
712                        ..
713                    }) = member.get(eq.saturating_sub(1))
714                    else {
715                        return Err(fail(codes::INVALID_SOURCE, Some(root.location)));
716                    };
717                    let ty = self
718                        .definitions
719                        .iter()
720                        .find(|d| d.owner == root.owner)
721                        .and_then(|d| match &d.definition.body {
722                            B::Record(fs) => fs.iter().find(|f| f.name == *name).map(|f| &f.ty),
723                            B::Variant(_) => None,
724                        })
725                        .ok_or_else(|| fail(codes::INVALID_SOURCE, Some(root.location)))?;
726                    let ty = retain(ty.try_clone())?;
727                    let value = self.value(root, &ty, &member[eq + 1..], true, 0)?;
728                    reserve(&mut defaults, 1)?;
729                    defaults.push((
730                        retain(root.owner.try_clone())?,
731                        retain(name.try_clone())?,
732                        closed(value)?,
733                    ));
734                }
735            }
736        }
737        for (owner, name, value) in defaults {
738            let d = self
739                .definitions
740                .iter_mut()
741                .find(|d| d.owner == owner)
742                .ok_or_else(|| fail(codes::INVALID_SOURCE, None))?;
743            let B::Record(fields) = &mut d.definition.body else {
744                return Err(fail(codes::INVALID_SOURCE, None));
745            };
746            let f = fields
747                .iter_mut()
748                .find(|f| f.name == name)
749                .ok_or_else(|| fail(codes::INVALID_SOURCE, None))?;
750            f.presence = FieldPresence::Defaulted;
751            f.default = Some(value);
752        }
753        Ok(())
754    }
755    /// Charges source/token/key/value work before recursion or proportional materialization.
756    fn step(&mut self, n: u64, location: Option<SourceLocation>) -> Result<(), E> {
757        if self.cancel.is_cancelled() {
758            return Err(fail(codes::CANCELLED, location));
759        }
760        self.work = self
761            .work
762            .checked_add(n)
763            .ok_or_else(|| fail(codes::LIMIT, location))?;
764        if self.work > self.captured.composition_limits().work {
765            return Err(fail(codes::LIMIT, location));
766        }
767        Ok(())
768    }
769    /// Resolves explicit local/import/vocabulary qualifiers without path/URL interpretation.
770    fn name(
771        &mut self,
772        root: &Root,
773        alias: Option<&str>,
774        name: &str,
775        span: ByteSpan,
776        nominal: bool,
777    ) -> Result<ModuleSymbolIdentity, E> {
778        let module = root.owner.module().module_name();
779        self.step(
780            (name.len() + alias.map_or(0, str::len)) as u64,
781            Some(SourceLocation::new(root.location.source(), span)),
782        )?;
783        let target = if let Some(alias) = alias {
784            self.graph
785                .edges()
786                .iter()
787                .find(|e| e.from() == module && e.alias() == alias)
788                .map(graph::Import::target)
789                .ok_or_else(|| {
790                    fail(
791                        codes::INCOMPATIBLE_VALUE,
792                        Some(SourceLocation::new(root.location.source(), span)),
793                    )
794                })?
795        } else {
796            module
797        };
798        let owner = ModuleSymbolIdentity::new(
799            LogicalModuleIdentity::new(retain(text(V1_SOURCE_PROFILE))?, retain(text(target))?),
800            retain(text(name))?,
801        );
802        let declaration = self.roots.get(&owner).ok_or_else(|| {
803            fail(
804                codes::INCOMPATIBLE_VALUE,
805                Some(SourceLocation::new(root.location.source(), span)),
806            )
807        })?;
808        if nominal == declaration.ty.is_some() {
809            return Err(fail(codes::INCOMPATIBLE_VALUE, Some(root.location)));
810        }
811        if target != module && !declaration.public {
812            return Err(fail(codes::PRIVATE_EXPOSURE, Some(root.location)));
813        }
814        Ok(owner)
815    }
816    /// Converts parsed type wrappers to exact source/vocabulary nominal owners.
817    fn ty(&mut self, root: &Root, ty: &TypeExpr) -> Result<T, E> {
818        self.step(1, Some(root.location))?;
819        Ok(match ty {
820            TypeExpr::Num => T::Num,
821            TypeExpr::String => T::String,
822            TypeExpr::Bool => T::Bool,
823            TypeExpr::Url => T::Url,
824            TypeExpr::Path => T::Path,
825            TypeExpr::List(t) => T::List(retain(boxed(self.ty(root, t)?))?),
826            TypeExpr::Ref(t) => T::Ref(retain(boxed(self.ty(root, t)?))?),
827            TypeExpr::Nullable(t) => T::Nullable(retain(boxed(self.ty(root, t)?))?),
828            TypeExpr::Nominal(alias, name, span) => {
829                if let Some(v) = alias.as_deref().and_then(|a| {
830                    self.captured
831                        .vocabulary_for_alias(root.owner.module().module_name(), a)
832                }) {
833                    let b = self.captured.catalogue();
834                    let public = b
835                        .bundles()
836                        .iter()
837                        .find(|b| b.identity == *v)
838                        .and_then(|b| b.definitions.iter().find(|d| d.name == *name))
839                        .is_some_and(|d| d.public);
840                    if !public {
841                        return Err(fail(
842                            codes::PRIVATE_EXPOSURE,
843                            Some(SourceLocation::new(root.location.source(), *span)),
844                        ));
845                    }
846                    T::VocabularyNominal {
847                        identity: retain(text(v.identity()))?,
848                        version: retain(text(v.version()))?,
849                        name: retain(name.try_clone())?,
850                    }
851                } else {
852                    let owner = self.name(root, alias.as_deref(), name, *span, true)?;
853                    T::Nominal(owner)
854                }
855            }
856        })
857    }
858    /// Looks up complete nominal bodies; all alternatives are known before value interpretation.
859    fn body(&mut self, ty: &T) -> Result<B, E> {
860        let catalogue = self.captured.catalogue();
861        let body = match ty {
862            T::Nominal(owner) => self
863                .definitions
864                .iter()
865                .find(|d| d.owner == *owner)
866                .map(|d| &d.definition.body),
867            T::VocabularyNominal {
868                identity,
869                version,
870                name,
871            } => catalogue
872                .bundles()
873                .iter()
874                .find(|b| b.identity.identity() == identity && b.identity.version() == version)
875                .and_then(|b| b.definitions.iter().find(|d| d.name == *name))
876                .map(|d| &d.body),
877            _ => None,
878        }
879        .ok_or_else(|| fail(codes::INCOMPATIBLE_VALUE, None))?;
880        let work = body_clone_work(body, self.cancel)?;
881        // Charge before retaining a recursive copy; repeated nominal use must not amplify unchecked allocation.
882        let remaining = self
883            .captured
884            .composition_limits()
885            .work
886            .saturating_sub(self.work);
887        if work > remaining {
888            return Err(fail(codes::LIMIT, None));
889        }
890        self.work = self
891            .work
892            .checked_add(work)
893            .ok_or_else(|| fail(codes::LIMIT, None))?;
894        let copied = retain(body.try_clone())?;
895        self.step(1, None)?;
896        Ok(copied)
897    }
898    /// Resolves ordinary reuse with explicit cycle/depth guards; references do not evaluate targets.
899    fn binding(&mut self, owner: &ModuleSymbolIdentity, depth: usize) -> Result<V, E> {
900        self.step(1, None)?;
901        if depth > PROJECT_MAX_DEPTH {
902            return Err(fail(codes::LIMIT, None));
903        }
904        if let Some(v) = self.values.get(owner) {
905            let work = value_clone_work(v, self.cancel, symbol_work)?;
906            self.step(work, None)?;
907            return retain(self.values[owner].try_clone());
908        }
909        self.step(self.active.len() as u64 + 1, None)?;
910        if !self
911            .active
912            .insert(retain(owner.try_clone())?)
913            .map_err(|_| fail(codes::LIMIT, None))?
914        {
915            return Err(fail(codes::REUSE_CYCLE, Some(self.roots[owner].location)));
916        }
917        let root = retain(self.roots[owner].try_clone())?;
918        let ty = retain(self.types[owner].try_clone())?;
919        let value = self.value(&root, &ty, &root.tokens, false, 0)?;
920        self.step(self.active.len() as u64 + self.values.len() as u64, None)?;
921        self.active.remove(owner);
922        self.step(
923            value_clone_work(&value, self.cancel, symbol_work)?,
924            Some(root.location),
925        )?;
926        self.values
927            .insert(retain(owner.try_clone())?, retain(value.try_clone())?)
928            .map_err(|_| fail(codes::LIMIT, None))?;
929        Ok(value)
930    }
931    /// Types exact scalar literals without host numeric or inert-location coercion.
932    fn scalar(&self, ty: &T, token: &Token, location: SourceLocation) -> Result<V, E> {
933        match &token.kind {
934            TokenKind::Null => {
935                if matches!(ty, T::Nullable(_)) {
936                    Ok(V::Null)
937                } else {
938                    Err(fail(codes::INCOMPATIBLE_VALUE, Some(location)))
939                }
940            }
941            TokenKind::True | TokenKind::False => {
942                let inner = if let T::Nullable(inner) = ty {
943                    inner.as_ref()
944                } else {
945                    ty
946                };
947                if !matches!(inner, T::Bool) {
948                    return Err(fail(codes::INCOMPATIBLE_VALUE, Some(location)));
949                }
950                Ok(V::Bool(matches!(token.kind, TokenKind::True)))
951            }
952            TokenKind::Number(s) => {
953                let inner = if let T::Nullable(inner) = ty {
954                    inner.as_ref()
955                } else {
956                    ty
957                };
958                if !matches!(inner, T::Num) {
959                    return Err(fail(codes::INCOMPATIBLE_VALUE, Some(location)));
960                }
961                ExactNumber::from_source(
962                    s,
963                    self.captured.composition_limits().json.numeric_digits(),
964                    self.captured.composition_limits().json.numeric_scale(),
965                )
966                .map(V::Number)
967                .map_err(|_| fail(codes::LIMIT, Some(location)))
968            }
969            TokenKind::StringLiteral(s) => {
970                let inner = if let T::Nullable(inner) = ty {
971                    inner.as_ref()
972                } else {
973                    ty
974                };
975                if !matches!(inner, T::String | T::Url | T::Path) {
976                    return Err(fail(codes::INCOMPATIBLE_VALUE, Some(location)));
977                }
978                Ok(match ty {
979                    T::Url => V::Url(retain(s.value.try_clone())?),
980                    T::Path => V::Path(retain(s.value.try_clone())?),
981                    T::Nullable(inner) if matches!(**inner, T::Url) => {
982                        V::Url(retain(s.value.try_clone())?)
983                    }
984                    T::Nullable(inner) if matches!(**inner, T::Path) => {
985                        V::Path(retain(s.value.try_clone())?)
986                    }
987                    _ => V::String(retain(s.value.try_clone())?),
988                })
989            }
990            _ => Err(fail(codes::INVALID_SOURCE, Some(location))),
991        }
992    }
993    /// Indexes borrowed object members with charged shifts and fallible storage.
994    fn record_members<'a>(
995        &mut self,
996        tokens: &'a [Token],
997        location: SourceLocation,
998    ) -> Result<BTreeMap<&'a str, &'a [Token]>, E> {
999        let mut members = BTreeMap::new();
1000        for field in split(&tokens[1..tokens.len() - 1])? {
1001            let [
1002                Token {
1003                    kind: TokenKind::Identifier(name),
1004                    ..
1005                },
1006                Token {
1007                    kind: TokenKind::Colon,
1008                    ..
1009                },
1010                rest @ ..,
1011            ] = field
1012            else {
1013                return Err(fail(codes::INVALID_SOURCE, Some(location)));
1014            };
1015            self.step(members.len() as u64 + 1, Some(location))?;
1016            if members
1017                .insert(name.as_str(), rest)
1018                .map_err(|_| fail(codes::LIMIT, Some(location)))?
1019                .is_some()
1020            {
1021                return Err(fail(codes::DUPLICATE_MEMBER, Some(location)));
1022            }
1023        }
1024        Ok(members)
1025    }

1027    /// Types a closed contextual record/variant object with complete member checks.
1028    fn record_value(
1029        &mut self,
1030        root: &Root,
1031        ty: &T,
1032        tokens: &[Token],
1033        closed: bool,
1034        depth: usize,
1035    ) -> Result<V, E> {
1036        let location = SourceLocation::new(root.location.source(), tokens[0].span);
1037        let members = self.record_members(tokens, location)?;
1038        match self.body(ty)? {
1039            B::Record(fields) => {
1040                if !closed
1041                    && fields.iter().any(|f| {
1042                        f.presence == FieldPresence::Required
1043                            && !members.contains_key(f.name.as_str())
1044                    })
1045                {
1046                    let end =
1047                        SourceLocation::new(root.location.source(), tokens[tokens.len() - 1].span);
1048                    return Err(fail(codes::MISSING_REQUIRED_FIELD, Some(end)));
1049                }
1050                let mut value = Vec::new();
1051                for (name, tokens) in members {
1052                    let field = fields
1053                        .iter()
1054                        .find(|f| f.name == name)
1055                        .ok_or_else(|| fail(codes::INCOMPATIBLE_VALUE, Some(location)))?;
1056                    let v = self.value(root, &field.ty, tokens, closed, depth + 1)?;
1057                    if field.restrictions != FieldRestrictions::default() {
1058                        neutral_vocabulary::composition::check_composition_field_restrictions(
1059                            field,
1060                            &v,
1061                            self.captured.composition_limits(),
1062                            self.cancel,
1063                        )
1064                        .map_err(|e| {
1065                            if e == CompositionError::InvalidValue {
1066                                fail(codes::RESTRICTION_VIOLATION, Some(location))
1067                            } else {
1068                                contract(&e, Some(location))
1069                            }
1070                        })?;
1071                    }
1072                    reserve(&mut value, 1)?;
1073                    value.push((retain(text(name))?, Some(v)));
1074                }
1075                Ok(V::Record(value))
1076            }
1077            B::Variant(alternatives) => {
1078                if members.len() != 2
1079                    || !members.contains_key(profile::TAG)
1080                    || !members.contains_key(profile::PAYLOAD)
1081                {
1082                    return Err(fail(codes::DUPLICATE_MEMBER, Some(location)));
1083                }
1084                let [
1085                    Token {
1086                        kind: TokenKind::StringLiteral(tag),
1087                        span,
1088                    },
1089                ] = members[profile::TAG]
1090                else {
1091                    return Err(fail(codes::INVALID_SOURCE, Some(location)));
1092                };
1093                let alternative = alternatives
1094                    .iter()
1095                    .find(|a| a.tag == tag.value)
1096                    .ok_or_else(|| {
1097                        fail(
1098                            codes::UNKNOWN_TAG,
1099                            Some(SourceLocation::new(root.location.source(), *span)),
1100                        )
1101                    })?;
1102                let payload = self.value(
1103                    root,
1104                    &alternative.ty,
1105                    members[profile::PAYLOAD],
1106                    closed,
1107                    depth + 1,
1108                )?;
1109                Ok(V::Variant {
1110                    tag: retain(tag.value.try_clone())?,
1111                    payload: retain(boxed(payload))?,
1112                })
1113            }
1114        }
1115    }

1117    /// Parses contextually typed values and exact selected variant payloads using the shared bodies.
1118    fn value(
1119        &mut self,
1120        root: &Root,
1121        ty: &T,
1122        tokens: &[Token],
1123        closed: bool,
1124        depth: usize,
1125    ) -> Result<V, E> {
1126        self.step(tokens.len() as u64 + 1, Some(root.location))?;
1127        if depth as u64 > self.captured.composition_limits().value_depth
1128            || depth > PROJECT_MAX_DEPTH
1129        {
1130            return Err(fail(codes::LIMIT, Some(root.location)));
1131        }
1132        let location = tokens.first().map_or(root.location, |t| {
1133            SourceLocation::new(root.location.source(), t.span)
1134        });
1135        if tokens.is_empty() {
1136            return Err(fail(codes::INVALID_SOURCE, Some(root.location)));
1137        }
1138        if let Some((name, end)) = parse_reference(tokens, 0)
1139            && end == tokens.len()
1140        {
1141            if closed {
1142                return Err(fail(codes::INCOMPATIBLE_VALUE, Some(location)));
1143            }
1144            let target = self.name(root, name.alias, name.name, name.span, false)?;
1145            let expected = if let T::Nullable(inner) = ty {
1146                inner.as_ref()
1147            } else {
1148                ty
1149            };
1150            let T::Ref(expected) = expected else {
1151                return Err(fail(codes::INCOMPATIBLE_VALUE, Some(location)));
1152            };
1153            if self.types.get(&target) != Some(expected.as_ref()) {
1154                return Err(fail(codes::INCOMPATIBLE_VALUE, Some(location)));
1155            }
1156            if root.public && !self.roots[&target].public {
1157                return Err(fail(codes::PRIVATE_EXPOSURE, Some(location)));
1158            }
1159            return Ok(V::Reference(target));
1160        }
1161        if let Some((name, end)) = parse_name(tokens, 0)
1162            && end == tokens.len()
1163        {
1164            if closed {
1165                return Err(fail(codes::INCOMPATIBLE_VALUE, Some(location)));
1166            }
1167            let target = self.name(root, name.alias, name.name, name.span, false)?;
1168            if self.types.get(&target) != Some(ty) {
1169                return Err(fail(codes::INCOMPATIBLE_VALUE, Some(location)));
1170            }
1171            reserve(&mut self.provenance, 1)?;
1172            self.provenance.push(ProjectProvenance {
1173                from: retain(root.owner.try_clone())?,
1174                to: retain(target.try_clone())?,
1175                kind: Edge::Value,
1176                location,
1177            });
1178            return self.binding(&target, self.active.len());
1179        }
1180        if let [token] = tokens
1181            && matches!(
1182                token.kind,
1183                TokenKind::Null
1184                    | TokenKind::True
1185                    | TokenKind::False
1186                    | TokenKind::Number(_)
1187                    | TokenKind::StringLiteral(_)
1188            )
1189        {
1190            return self.scalar(ty, token, location);
1191        }
1192        if let T::Nullable(inner) = ty {
1193            return self.value(root, inner, tokens, closed, depth);
1194        }
1195        if matches!(tokens[0].kind, TokenKind::OpenBracket)
1196            && matches!(
1197                tokens.last().map(|t| &t.kind),
1198                Some(TokenKind::CloseBracket)
1199            )
1200        {
1201            let T::List(inner) = ty else {
1202                return Err(fail(codes::INCOMPATIBLE_VALUE, Some(location)));
1203            };
1204            let mut values = Vec::new();
1205            for item in split(&tokens[1..tokens.len() - 1])? {
1206                reserve(&mut values, 1)?;
1207                values.push(self.value(root, inner, item, closed, depth + 1)?);
1208            }
1209            return Ok(V::List(values));
1210        }
1211        if matches!(tokens[0].kind, TokenKind::OpenBrace)
1212            && matches!(tokens.last().map(|t| &t.kind), Some(TokenKind::CloseBrace))
1213        {
1214            return self.record_value(root, ty, tokens, closed, depth);
1215        }
1216        Err(fail(codes::INVALID_SOURCE, Some(location)))
1217    }
1218}

1220/// Parses declarations once while preserving physical statement and original-byte boundaries.
1221fn parse(
1222    source: &crate::CapturedProjectSource,
1223    captured: &CapturedCompositionProject,
1224    cancel: &CancellationToken,
1225) -> Result<Vec<Root>, E> {
1226    let string_limit =
1227        usize::try_from(captured.composition_limits().json.string_bytes()).unwrap_or(usize::MAX);
1228    let tokens = lexer::lex_bounded(source.bytes(), string_limit, Some(cancel))
1229        .map_err(|e| {
1230            fail(
1231                if cancel.is_cancelled() {
1232                    codes::CANCELLED
1233                } else if matches!(e.kind, super::super::FrontendErrorKind::RecordLimitExceeded) {
1234                    codes::LIMIT
1235                } else {
1236                    codes::INVALID_SOURCE
1237                },
1238                Some(SourceLocation::new(source.digest(), e.span)),
1239            )
1240        })?
1241        .tokens;
1242    let mut statement = Vec::new();
1243    let mut roots = Vec::new();
1244    let mut depth = 0;
1245    let mut headers = 0;
1246    for token in tokens {
1247        if cancel.is_cancelled() {
1248            return Err(fail(codes::CANCELLED, None));
1249        }
1250        if matches!(
1251            token.kind,
1252            TokenKind::EndOfFile | TokenKind::PhysicalLineEnd(_)
1253        ) {
1254            if depth == 0 && !statement.is_empty() {
1255                if headers < 2 {
1256                    headers += 1;
1257                } else if !is_requirement_or_import(&statement) {
1258                    if roots.len() as u64 >= captured.limits().values().declarations {
1259                        return Err(fail(codes::LIMIT, None));
1260                    }
1261                    roots.try_retain(1).map_err(|_| fail(codes::LIMIT, None))?;
1262                    roots.push(parse_declaration(
1263                        source,
1264                        &statement,
1265                        captured.composition_limits(),
1266                    )?);
1267                }
1268                statement.clear();
1269            }
1270            continue;
1271        }
1272        match token.kind {
1273            TokenKind::OpenBrace | TokenKind::OpenBracket | TokenKind::OpenParen => depth += 1,
1274            TokenKind::CloseBrace | TokenKind::CloseBracket | TokenKind::CloseParen => {
1275                if depth == 0 {
1276                    return Err(fail(
1277                        codes::INVALID_SOURCE,
1278                        Some(SourceLocation::new(source.digest(), token.span)),
1279                    ));
1280                }
1281                depth -= 1;
1282            }
1283            _ => {}
1284        }
1285        if depth > PROJECT_MAX_DEPTH {
1286            return Err(fail(
1287                codes::LIMIT,
1288                Some(SourceLocation::new(source.digest(), token.span)),
1289            ));
1290        }
1291        statement
1292            .try_retain(1)
1293            .map_err(|_| fail(codes::LIMIT, None))?;
1294        statement.push(token);
1295    }
1296    if !statement.is_empty() {
1297        return Err(fail(
1298            codes::INVALID_SOURCE,
1299            Some(SourceLocation::new(source.digest(), statement[0].span)),
1300        ));
1301    }
1302    Ok(roots)
1303}

1313/// Bounds recursive value copies iteratively before reuse/default expansion can allocate them.
1314fn value_clone_work<R>(
1315    value: &CompositionValue<R>,
1316    cancel: &CancellationToken,
1317    reference: impl Fn(&R) -> u64,
1318) -> Result<u64, E> {
1319    if cancel.is_cancelled() {
1320        return Err(fail(codes::CANCELLED, None));
1321    }
1322    let mut work = 0_u64;
1323    let mut stack = Vec::new();
1324    reserve(&mut stack, 1)?;
1325    stack.push(value);
1326    while let Some(value) = stack.pop() {
1327        if cancel.is_cancelled() {
1328            return Err(fail(codes::CANCELLED, None));
1329        }
1330        let extra = match value {
1331            CompositionValue::Number(n) => n.coefficient().len() as u64,
1332            CompositionValue::String(s) | CompositionValue::Url(s) | CompositionValue::Path(s) => {
1333                s.len() as u64
1334            }
1335            CompositionValue::Reference(target) => reference(target),
1336            CompositionValue::List(items) => {
1337                stack
1338                    .try_retain(items.len())
1339                    .map_err(|_| fail(codes::LIMIT, None))?;
1340                stack.extend(items);
1341                items.len() as u64
1342            }
1343            CompositionValue::Record(fields) => {
1344                stack
1345                    .try_retain(fields.len())
1346                    .map_err(|_| fail(codes::LIMIT, None))?;
1347                let mut bytes = 0_u64;
1348                for (name, value) in fields {
1349                    bytes = bytes
1350                        .checked_add(name.len() as u64 + 1)
1351                        .ok_or_else(|| fail(codes::LIMIT, None))?;
1352                    if let Some(value) = value {
1353                        stack.push(value);
1354                    }
1355                }
1356                bytes
1357            }
1358            CompositionValue::Variant { tag, payload } => {
1359                reserve(&mut stack, 1)?;
1360                stack.push(payload);
1361                tag.len() as u64
1362            }
1363            CompositionValue::Null | CompositionValue::Bool(_) => 0,
1364        };
1365        work = work
1366            .checked_add(extra)
1367            .and_then(|n| n.checked_add(1))
1368            .ok_or_else(|| fail(codes::LIMIT, None))?;
1369        if work > profile::MAX_ITEMS {
1370            return Err(fail(codes::LIMIT, None));
1371        }
1372    }
1373    Ok(work)
1374}

1376/// Accounts for wrapper depth and all nominal owner text before cloning a type.
1377fn type_clone_work(mut ty: &T) -> u64 {
1378    let mut work = 1;
1379    loop {
1380        match ty {
1381            T::List(inner) | T::Ref(inner) | T::Nullable(inner) => {
1382                work += 1;
1383                ty = inner;
1384            }
1385            T::Nominal(owner) => return work + symbol_work(owner),
1386            T::VocabularyNominal {
1387                identity,
1388                version,
1389                name,
1390            } => return work + (identity.len() + version.len() + name.len()) as u64,
1391            _ => return work,
1392        }
1393    }
1394}

1396/// Charges copied contract members, choices and defaults independently of later materialization.
1397fn body_clone_work(body: &B, cancel: &CancellationToken) -> Result<u64, E> {
1398    let mut work = 1_u64;
1399    match body {
1400        B::Record(fields) => {
1401            for field in fields {
1402                work = work
1403                    .checked_add(field.name.len() as u64 + type_clone_work(&field.ty) + 1)
1404                    .ok_or_else(|| fail(codes::LIMIT, None))?;
1405                for value in field
1406                    .default
1407                    .iter()
1408                    .chain(field.restrictions.choices.iter().flatten())
1409                {
1410                    work = work
1411                        .checked_add(value_clone_work(value, cancel, |never| match *never {})?)
1412                        .ok_or_else(|| fail(codes::LIMIT, None))?;
1413                }
1414                for bound in [&field.restrictions.minimum, &field.restrictions.maximum]
1415                    .into_iter()
1416                    .flatten()
1417                {
1418                    work = work
1419                        .checked_add(bound.coefficient().len() as u64)
1420                        .ok_or_else(|| fail(codes::LIMIT, None))?;
1421                }
1422            }
1423        }
1424        B::Variant(alternatives) => {
1425            for alternative in alternatives {
1426                work = work
1427                    .checked_add(
1428                        alternative.tag.len() as u64 + type_clone_work(&alternative.ty) + 1,
1429                    )
1430                    .ok_or_else(|| fail(codes::LIMIT, None))?;
1431            }
1432        }
1433    }
1434    Ok(work)
1435}

1437/// Adds contextual variant syntax without reserving `variant` as a binding identifier.
1438fn parse_declaration(
1439    source: &crate::CapturedProjectSource,
1440    tokens: &[Token],
1441    limits: neutral_vocabulary::composition::CompositionLimits,
1442) -> Result<Root, E> {
1443    let public = matches!(&tokens[0].kind,TokenKind::Identifier(s) if s==crate::language::graph_names::PUBLIC);
1444    let start = usize::from(public);
1445    let variant = matches!(tokens.get(start).map(|t|&t.kind),Some(TokenKind::Identifier(s)) if s==profile::VARIANT);
1446    let span = ByteSpan::new(tokens[0].span.start(), tokens[tokens.len() - 1].span.end())
1447        .map_err(|_| fail(codes::INVALID_SOURCE, None))?;
1448    let location = SourceLocation::new(source.digest(), span);
1449    if variant || matches!(tokens.get(start).map(|t| &t.kind), Some(TokenKind::Record)) {
1450        let Some(Token {
1451            kind: TokenKind::Identifier(name),
1452            ..
1453        }) = tokens.get(start + 1)
1454        else {
1455            return Err(fail(codes::INVALID_SOURCE, Some(location)));
1456        };
1457        if !is_upper_name(name)
1458            || is_protected_name(name)
1459            || !matches!(
1460                tokens.get(start + 2).map(|t| &t.kind),
1461                Some(TokenKind::OpenBrace)
1462            )
1463            || !matches!(tokens.last().map(|t| &t.kind), Some(TokenKind::CloseBrace))
1464        {
1465            return Err(fail(codes::INVALID_SOURCE, Some(location)));
1466        }
1467        let body = &tokens[start + 3..tokens.len() - 1];
1468        if variant && body.iter().any(|t| matches!(t.kind, TokenKind::Equals)) {
1469            return Err(fail(codes::INVALID_SOURCE, Some(location)));
1470        }
1471        let fields = parsed_fields(
1472            body,
1473            if variant {
1474                limits.alternatives_per_type
1475            } else {
1476                limits.json.fields()
1477            },
1478        )?;
1479        if variant && fields.is_empty() {
1480            return Err(fail(codes::DUPLICATE_MEMBER, Some(location)));
1481        }
1482        return Ok(Root {
1483            owner: ModuleSymbolIdentity::new(
1484                LogicalModuleIdentity::new(
1485                    neutral_core::allocation::text(V1_SOURCE_PROFILE)
1486                        .map_err(|_| fail(codes::LIMIT, None))?,
1487                    neutral_core::allocation::text(source.module_id())
1488                        .map_err(|_| fail(codes::LIMIT, None))?,
1489                ),
1490                neutral_core::allocation::text(name).map_err(|_| fail(codes::LIMIT, None))?,
1491            ),
1492            public,
1493            location,
1494            ty: None,
1495            fields,
1496            tokens: neutral_core::allocation::copy_slice(body)
1497                .map_err(|_| fail(codes::LIMIT, None))?,
1498            variant,
1499        });
1500    }
1501    let eq = tokens
1502        .iter()
1503        .position(|t| matches!(t.kind, TokenKind::Equals))
1504        .ok_or_else(|| fail(codes::INVALID_SOURCE, Some(location)))?;
1505    if eq <= start + 1 || eq + 1 >= tokens.len() {
1506        return Err(fail(codes::INVALID_SOURCE, Some(location)));
1507    }
1508    let TokenKind::Identifier(name) = &tokens[eq - 1].kind else {
1509        return Err(fail(codes::INVALID_SOURCE, Some(location)));
1510    };
1511    if !is_snake_name(name) || is_protected_name(name) {
1512        return Err(fail(codes::INVALID_SOURCE, Some(location)));
1513    }
1514    let ty = parse_complete_type_fallible(&tokens[start..eq - 1])
1515        .map_err(|_| fail(codes::LIMIT, None))?
1516        .ok_or_else(|| fail(codes::INVALID_SOURCE, Some(location)))?;
1517    Ok(Root {
1518        owner: ModuleSymbolIdentity::new(
1519            LogicalModuleIdentity::new(
1520                neutral_core::allocation::text(V1_SOURCE_PROFILE)
1521                    .map_err(|_| fail(codes::LIMIT, None))?,
1522                neutral_core::allocation::text(source.module_id())
1523                    .map_err(|_| fail(codes::LIMIT, None))?,
1524            ),
1525            neutral_core::allocation::text(name).map_err(|_| fail(codes::LIMIT, None))?,
1526        ),
1527        public,
1528        location,
1529        ty: Some(ty),
1530        fields: Vec::new(),
1531        tokens: neutral_core::allocation::copy_slice(&tokens[eq + 1..])
1532            .map_err(|_| fail(codes::LIMIT, None))?,
1533        variant: false,
1534    })
1535}

1537/// Parses complete source member types without requiring old-profile value terminators.
1538fn parsed_fields(tokens: &[Token], limit: u64) -> Result<Vec<(String, TypeExpr)>, E> {
1539    let mut result = Vec::new();
1540    let mut names = neutral_core::ordered::OrderedSet::new();
1541    for member in split(tokens)? {
1542        let end = member
1543            .iter()
1544            .position(|t| matches!(t.kind, TokenKind::Equals))
1545            .unwrap_or(member.len());
1546        if end < 2 {
1547            return Err(fail(codes::INVALID_SOURCE, None));
1548        }
1549        let TokenKind::Identifier(name) = &member[end - 1].kind else {
1550            return Err(fail(codes::INVALID_SOURCE, None));
1551        };
1552        if result.len() as u64 >= limit {
1553            return Err(fail(codes::LIMIT, None));
1554        }
1555        if !is_snake_name(name)
1556            || is_protected_name(name)
1557            || !names
1558                .insert(name.as_str())
1559                .map_err(|_| fail(codes::LIMIT, None))?
1560        {
1561            return Err(fail(codes::DUPLICATE_MEMBER, None));
1562        }
1563        let ty = parse_complete_type_fallible(&member[..end - 1])
1564            .map_err(|_| fail(codes::LIMIT, None))?
1565            .ok_or_else(|| fail(codes::INVALID_SOURCE, None))?;
1566        if end < member.len() && end + 1 == member.len() {
1567            return Err(fail(codes::INVALID_SOURCE, None));
1568        }
1569        result.try_retain(1).map_err(|_| fail(codes::LIMIT, None))?;
1570        result.push((
1571            neutral_core::allocation::text(name).map_err(|_| fail(codes::LIMIT, None))?,
1572            ty,
1573        ));
1574    }
1575    Ok(result)
1576}

1578/// Splits top-level members only, permitting one final comma and rejecting empty interior entries.
1579fn split(tokens: &[Token]) -> Result<Vec<&[Token]>, E> {
1580    let mut result = Vec::new();
1581    let mut start = 0;
1582    let mut depth = 0;
1583    for (i, t) in tokens.iter().enumerate() {
1584        match t.kind {
1585            TokenKind::OpenBrace
1586            | TokenKind::OpenBracket
1587            | TokenKind::OpenParen
1588            | TokenKind::Less => depth += 1,
1589            TokenKind::CloseBrace
1590            | TokenKind::CloseBracket
1591            | TokenKind::CloseParen
1592            | TokenKind::Greater => {
1593                if depth == 0 {
1594                    return Err(fail(codes::INVALID_SOURCE, None));
1595                }
1596                depth -= 1;
1597            }
1598            _ => {}
1599        }
1600        if depth == 0 && matches!(t.kind, TokenKind::Comma) {
1601            if start == i {
1602                return Err(fail(codes::INVALID_SOURCE, None));
1603            }
1604            result.try_retain(1).map_err(|_| fail(codes::LIMIT, None))?;
1605            result.push(&tokens[start..i]);
1606            start = i + 1;
1607        }
1608    }
1609    if depth != 0 {
1610        return Err(fail(codes::INVALID_SOURCE, None));
1611    }
1612    if start < tokens.len() {
1613        result.try_retain(1).map_err(|_| fail(codes::LIMIT, None))?;
1614        result.push(&tokens[start..]);
1615    }
1616    Ok(result)
1617}

1619/// Converts parser-produced closed values while making identity references unrepresentable.
1620fn closed(v: V) -> Result<ClosedValue, E> {
1621    Ok(match v {
1622        V::Number(n) => CompositionValue::Number(n),
1623        V::String(s) => CompositionValue::String(s),
1624        V::Bool(b) => CompositionValue::Bool(b),
1625        V::Url(s) => CompositionValue::Url(s),
1626        V::Path(s) => CompositionValue::Path(s),
1627        V::Null => CompositionValue::Null,
1628        V::Reference(_) => return Err(fail(codes::INCOMPATIBLE_VALUE, None)),
1629        V::List(vs) => {
1630            let mut result = Vec::new();
1631            reserve(&mut result, vs.len())?;
1632            for v in vs {
1633                result.push(closed(v)?);
1634            }
1635            CompositionValue::List(result)
1636        }
1637        V::Record(fs) => {
1638            let mut result = Vec::new();
1639            reserve(&mut result, fs.len())?;
1640            for (n, v) in fs {
1641                result.push((n, v.map(closed).transpose()?));
1642            }
1643            CompositionValue::Record(result)
1644        }
1645        V::Variant { tag, payload } => CompositionValue::Variant {
1646            tag,
1647            payload: retain(boxed(closed(*payload)?))?,
1648        },
1649    })
1650}

1652/// Retains type/reference-type occurrence dependencies without embedding referenced targets.
1653fn type_edges(
1654    from: &ModuleSymbolIdentity,
1655    ty: &T,
1656    reference: bool,
1657    location: SourceLocation,
1658    resolver: &mut Resolver<'_>,
1659) -> Result<(), E> {
1660    resolver.step(1, Some(location))?;
1661    match ty {
1662        T::Nominal(to) => {
1663            reserve(&mut resolver.provenance, 1)?;
1664            resolver.provenance.push(ProjectProvenance {
1665                from: retain(from.try_clone())?,
1666                to: retain(to.try_clone())?,
1667                kind: if reference {
1668                    Edge::ReferenceType
1669                } else {
1670                    Edge::Type
1671                },
1672                location,
1673            });
1674        }
1675        T::Ref(t) => type_edges(from, t, true, location, resolver)?,
1676        T::List(t) | T::Nullable(t) => type_edges(from, t, reference, location, resolver)?,
1677        _ => {}
1678    }
1679    Ok(())
1680}
1681/// Maps fallible retention failures to the frozen bounded-resource source envelope.

1691/// Maps shared semantic failures without exposing private contract text.
1692fn contract(e: &CompositionError, location: Option<SourceLocation>) -> E {
1693    fail(
1694        match e {
1695            CompositionError::Cancelled => codes::CANCELLED,
1696            CompositionError::Limit
1697            | CompositionError::Allocation
1698            | CompositionError::InvalidLimits => codes::LIMIT,
1699            CompositionError::PrivateType => codes::PRIVATE_EXPOSURE,
1700            CompositionError::EmbeddedCycle => codes::EMBEDDED_CYCLE,
1701            CompositionError::InvalidRestrictions | CompositionError::DuplicateChoice => {
1702                codes::RESTRICTION_VIOLATION
1703            }
1704            _ => codes::INCOMPATIBLE_VALUE,
1705        },
1706        location,
1707    )
