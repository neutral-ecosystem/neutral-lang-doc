7//! the envelope, not logical meaning. Section helpers enforce encoding ceilings;
8//! reconstruction and hostile-input validation remain in the independent decoder.

44/// Encodes one already validated document into Neutral IR Framed CBOR 0.1.
45///
46/// The function borrows and never mutates the input. Only envelope bytes depend
47/// on `producer`; the logical and companion section projections do not.
48///
49/// # Errors
50///
51/// Returns [`EncodingError::EncodedSizeLimit`] if a frozen encoding ceiling is
52/// exceeded, or [`EncodingError::InternalDefect`] if validated input contradicts
53/// an encoder invariant.
54pub fn encode(
55    document: &ValidatedDocument,
56    producer: &ProducerInfo,
57) -> Result<EncodedArtifact, EncodingError> {
58    let artifacts = document.artifacts().as_ref();
59    let mut budget = TraversalBudget::new();
60    let capabilities = derive_capabilities(artifacts, &mut budget)?;
61    let logical = encode_logical(artifacts.logical_document())?;
62    let source_map = encode_source_map(artifacts.source_map())?;
63    let provenance = encode_provenance(artifacts)?;
64    let derivation = encode_derivation(artifacts.derivation())?;
65    let envelope = encode_envelope(
66        artifacts,
67        producer,
68        capabilities,
69        [&logical, &source_map, &provenance, &derivation],
70    )?;
71    assemble_frame(
72        capabilities,
73        [envelope, logical, source_map, provenance, derivation],
74    )
75}
