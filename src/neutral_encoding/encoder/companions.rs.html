3//! Serialize the nonlogical sections accompanying a document.
4//!
5//! Source maps preserve original-byte spans; provenance distinguishes origins,
6//! reuse and references; derivation retains processing inputs and resource facts.
7//! This module projects existing records into the wire schema. It does not infer
8//! missing evidence or merge companion data into the logical section.
9
10use super::{
11    AcceptancePartition, ByteSpan, CborWriter, CompilationArtifacts, DERIVATION_SCHEMA_VERSION,
12    DerivationManifest, DiagnosticPartition, EncodingError, FieldProvenanceRecord,
13    ProvenanceRecord, ReferenceProvenanceRecord, ResourceFacts, ReuseProvenanceRecord, SourceMap,
14    SourceMapEntry, constants, encode_text_array, encode_vocabulary_identity, field_key,
15    field_text, field_unsigned,
16};
17
18/// Encodes the complete source-map section.
19pub(super) fn encode_source_map(source_map: &SourceMap) -> Result<Vec<u8>, EncodingError> {
20    let mut writer = CborWriter::new();
21    writer.map(5)?;
22    field_text(
23        &mut writer,
24        constants::key::SOURCE_MAP_VERSION,
25        neutral_ir::SOURCE_MAP_VERSION,
26    )?;
27    field_key(&mut writer, constants::key::SOURCE_DIGEST)?;
28    writer.bytes(&source_map.source_digest().as_bytes())?;
29    field_unsigned(
30        &mut writer,
31        constants::key::SOURCE_BYTE_LENGTH,
32        source_map.source_byte_length(),
33    )?;
34    field_key(&mut writer, constants::key::MODULE_SPAN)?;
35    encode_span(&mut writer, source_map.module_span())?;
36    field_key(&mut writer, constants::key::ENTRIES)?;
37    writer.array(source_map.entries().len())?;
38    for entry in source_map.entries() {
39        encode_source_map_entry(&mut writer, *entry)?;
40    }
41    writer.finish()
42}

44/// Encodes one half-open byte span.
45fn encode_span(writer: &mut CborWriter, span: ByteSpan) -> Result<(), EncodingError> {
46    writer.map(2)?;
47    field_unsigned(writer, constants::key::START, span.start())?;
48    field_unsigned(writer, constants::key::END, span.end())
49}

51/// Encodes one complete source-map entry.
52fn encode_source_map_entry(
53    writer: &mut CborWriter,
54    entry: SourceMapEntry,
55) -> Result<(), EncodingError> {
56    writer.map(5)?;
57    field_unsigned(writer, constants::key::ELEMENT_ID, entry.element_id().get())?;
58    field_key(writer, constants::key::DECLARATION_SPAN)?;
59    encode_span(writer, entry.declaration_span())?;
60    field_key(writer, constants::key::TYPE_SPAN)?;
61    encode_span(writer, entry.type_span())?;
62    field_key(writer, constants::key::NAME_SPAN)?;
63    encode_span(writer, entry.name_span())?;
64    field_key(writer, constants::key::VALUE_SPAN)?;
65    encode_span(writer, entry.value_span())
66}

68/// Encodes the complete provenance section.
69pub(super) fn encode_provenance(
70    artifacts: &CompilationArtifacts,
71) -> Result<Vec<u8>, EncodingError> {
72    let mut writer = CborWriter::new();
73    writer.map(5)?;
74    field_text(
75        &mut writer,
76        constants::key::PROVENANCE_VERSION,
77        neutral_ir::PROVENANCE_VERSION,
78    )?;
79    field_key(&mut writer, constants::key::VALUES)?;
80    writer.array(artifacts.provenance().len())?;
81    for record in artifacts.provenance() {
82        encode_value_provenance(&mut writer, *record)?;
83    }
84    field_key(&mut writer, constants::key::FIELDS)?;
85    writer.array(artifacts.field_provenance().len())?;
86    for record in artifacts.field_provenance() {
87        encode_field_provenance(&mut writer, record)?;
88    }
89    field_key(&mut writer, constants::key::REUSES)?;
90    writer.array(artifacts.reuse_provenance().len())?;
91    for record in artifacts.reuse_provenance() {
92        encode_reuse_provenance(&mut writer, record)?;
93    }
94    field_key(&mut writer, constants::key::REFERENCES)?;
95    writer.array(artifacts.reference_provenance().len())?;
96    for record in artifacts.reference_provenance() {
97        encode_reference_provenance(&mut writer, record)?;
98    }
99    writer.finish()
100}

116        constants::key::NORMALIZATION,
117        record.normalization().as_str(),
118    )

177/// Encodes the complete derivation section.
178pub(super) fn encode_derivation(derivation: &DerivationManifest) -> Result<Vec<u8>, EncodingError> {
179    let mut writer = CborWriter::new();
180    writer.map(10)?;
181    field_text(
182        &mut writer,
183        constants::key::DERIVATION_SCHEMA_VERSION,
184        DERIVATION_SCHEMA_VERSION,
185    )?;
186    field_text(
187        &mut writer,
188        constants::key::LANGUAGE_BEHAVIOR_VERSION,
189        derivation.language_behavior_version(),
190    )?;
191    field_text(
192        &mut writer,
193        constants::key::LOGICAL_IR_SCHEMA_VERSION,
194        derivation.logical_ir_schema_version(),
195    )?;
196    field_text(
197        &mut writer,
198        constants::key::SOURCE_MAP_VERSION,
199        derivation.source_map_version(),
200    )?;
201    field_text(
202        &mut writer,
203        constants::key::PROVENANCE_VERSION,
204        derivation.provenance_version(),
205    )?;
206    field_key(&mut writer, constants::key::MEANING)?;
207    writer.map(1)?;
208    field_key(&mut writer, constants::key::SOURCE_DIGEST)?;
209    writer.bytes(&derivation.meaning().source_digest().as_bytes())?;
210    field_key(&mut writer, constants::key::ACCEPTANCE)?;
211    encode_acceptance(&mut writer, derivation.acceptance())?;
212    field_key(&mut writer, constants::key::DIAGNOSTIC_POLICY)?;
213    encode_diagnostic_policy(&mut writer, derivation.diagnostics())?;
214    field_key(&mut writer, constants::key::RESOURCE_FACTS)?;
215    encode_resource_facts(&mut writer, derivation.resource_facts())?;
216    field_key(&mut writer, constants::key::VOCABULARY)?;
217    match derivation.vocabulary() {
218        Some(identity) => encode_vocabulary_identity(&mut writer, identity)?,
219        None => writer.null()?,
220    }
221    writer.finish()
222}

224/// Encodes captured acceptance limits.
225fn encode_acceptance(
226    writer: &mut CborWriter,
227    acceptance: AcceptancePartition,
228) -> Result<(), EncodingError> {
229    writer.map(10)?;
230    field_unsigned(
231        writer,
232        constants::key::SOURCE_BYTES,
233        acceptance.source_byte_limit(),
234    )?;
235    field_unsigned(
236        writer,
237        constants::key::DIAGNOSTICS,
238        u64::from(acceptance.diagnostic_limit()),
239    )?;
240    field_unsigned(
241        writer,
242        constants::key::STRING_BYTES,
243        acceptance.string_byte_limit(),
244    )?;
245    field_unsigned(
246        writer,
247        constants::key::NUMERIC_DIGITS,
248        acceptance.numeric_digit_limit(),
249    )?;
250    field_unsigned(
251        writer,
252        constants::key::NUMERIC_SCALE,
253        acceptance.numeric_scale_limit(),
254    )?;
255    field_unsigned(
256        writer,
257        constants::key::DECLARATIONS,
258        acceptance.declaration_limit(),
259    )?;
260    field_unsigned(
261        writer,
262        constants::key::RECORD_FIELDS,
263        acceptance.record_field_limit(),
264    )?;
265    field_unsigned(
266        writer,
267        constants::key::NESTING_DEPTH,
268        acceptance.nesting_depth_limit(),
269    )?;
270    field_unsigned(
271        writer,
272        constants::key::LIST_ITEMS,
273        acceptance.list_item_limit(),
274    )?;
275    field_unsigned(
276        writer,
277        constants::key::TRAVERSAL_NODES,
278        acceptance.traversal_node_limit(),
279    )
280}

282/// Encodes captured diagnostic policy.
283fn encode_diagnostic_policy(
284    writer: &mut CborWriter,
285    diagnostics: DiagnosticPartition,
286) -> Result<(), EncodingError> {
287    writer.map(1)?;
288    field_key(writer, constants::key::SAFE_BOUNDED_OUTPUT)?;
289    writer.boolean(diagnostics.safe_bounded_output())
290}

292/// Encodes observed resource facts.
293fn encode_resource_facts(
294    writer: &mut CborWriter,
295    facts: ResourceFacts,
296) -> Result<(), EncodingError> {
297    writer.map(4)?;
298    field_unsigned(writer, constants::key::SOURCE_BYTES, facts.source_bytes())?;
299    field_unsigned(writer, constants::key::DECLARATIONS, facts.declarations())?;
300    field_unsigned(writer, constants::key::DIAGNOSTICS, facts.diagnostics())?;
301    field_unsigned(
302        writer,
303        constants::key::DECODED_STRING_BYTES,
304        facts.decoded_string_bytes(),
305    )
