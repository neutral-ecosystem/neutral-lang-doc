3//! Hostile frame and restricted-CBOR decoding primitives.
4
5use crate::{DecodeError, DecodeErrorClass, DecodeLimits};
6use neutral_core::CancellationToken;
7
8/// One duplicate-preserving untrusted CBOR value and its byte offset.
9pub(crate) struct LocatedValue {
10    /// Absolute encoded byte offset of the initial byte.
11    pub(crate) offset: u64,
12    /// Untrusted restricted-CBOR value.
13    pub(crate) value: CborValue,
14}
15
16/// Untrusted value forms admitted by the restricted CBOR lexical profile.
17pub(crate) enum CborValue {
18    /// Unsigned integer.
19    Unsigned(u64),
20    /// Signed negative integer.
21    Negative(i64),
22    /// Definite byte string.
23    Bytes(Vec<u8>),
24    /// Definite valid UTF-8 text string.
25    Text(String),
26    /// Definite array.
27    Array(Vec<LocatedValue>),
28    /// Definite map retaining member order and duplicates.
29    Map(Vec<(String, LocatedValue)>),
30    /// Boolean simple value.
31    Boolean(bool),
32    /// Null simple value.
33    Null,
34}
35
36/// Parses exactly one bounded restricted-CBOR section.
37pub(crate) fn parse_section(
38    bytes: &[u8],
39    base_offset: usize,
40    limits: DecodeLimits,
41    cancellation: &CancellationToken,
42) -> Result<LocatedValue, DecodeError> {
43    let mut parser = Parser {
44        bytes,
45        position: 0,
46        base_offset,
47        limits,
48        nodes: 0,
49        cancellation,
50    };
51    let value = parser.value(1)?;
52    if parser.position != bytes.len() {
53        return Err(parser.error(DecodeErrorClass::MalformedCbor));
54    }
55    Ok(value)
56}
57
58/// Stateful parser that checks bounds before every allocation.
59struct Parser<'a> {
60    /// Exact section bytes.
61    bytes: &'a [u8],
62    /// Next unread section-relative position.
63    position: usize,
64    /// Absolute offset of the section.
65    base_offset: usize,
66    /// Effective caller and hard limits.
67    limits: DecodeLimits,
68    /// Total decoded nodes.
69    nodes: usize,
70    /// Cooperative cancellation signal.
71    cancellation: &'a CancellationToken,
72}
73
74impl Parser<'_> {
75    /// Parses one value under depth, traversal, and cancellation limits.
76    fn value(&mut self, depth: usize) -> Result<LocatedValue, DecodeError> {
77        self.check_cancelled()?;
78        if depth > self.limits.maximum_nesting_depth() {
79            return Err(self.error(DecodeErrorClass::EncodedSizeLimit));
80        }
81        self.nodes = self
82            .nodes
83            .checked_add(1)
84            .ok_or_else(|| self.error(DecodeErrorClass::EncodedSizeLimit))?;
85        if self.nodes > self.limits.maximum_traversal_nodes() {
86            return Err(self.error(DecodeErrorClass::EncodedSizeLimit));
87        }
88        let offset = self.absolute_offset()?;
89        let initial = self.read_byte()?;
90        let major = initial >> 5;
91        let additional = initial & 0x1f;
92        let value = match major {
93            0 => CborValue::Unsigned(self.argument(additional)?),
94            1 => {
95                let argument = self.argument(additional)?;
96                if argument > i64::MAX.cast_unsigned() {
97                    return Err(Self::error_at(
98                        DecodeErrorClass::InvalidEncodedSchema,
99                        offset,
100                    ));
101                }
102                let magnitude = i64::try_from(argument)
103                    .map_err(|_| self.error(DecodeErrorClass::InvalidEncodedSchema))?;
104                CborValue::Negative(-1 - magnitude)
105            }
106            2 => CborValue::Bytes(
107                self.byte_string(additional, self.limits.maximum_byte_string_bytes())?,
108            ),
109            3 => CborValue::Text(self.text_string(additional)?),
110            4 => CborValue::Array(self.array(additional, depth)?),
111            5 => CborValue::Map(self.map(additional, depth)?),
112            7 if additional == 20 => CborValue::Boolean(false),
113            7 if additional == 21 => CborValue::Boolean(true),
114            7 if additional == 22 => CborValue::Null,
115            _ => return Err(Self::error_at(DecodeErrorClass::MalformedCbor, offset)),
116        };
117        Ok(LocatedValue { offset, value })
118    }
119
120    /// Parses one bounded definite array without trusting its length for reserve.
121    fn array(&mut self, additional: u8, depth: usize) -> Result<Vec<LocatedValue>, DecodeError> {
122        let length = self.container_length(additional)?;
123        let mut values = Vec::new();
124        for _ in 0..length {
125            values.push(self.value(depth + 1)?);
126        }
127        Ok(values)
128    }
129
130    /// Parses one text-keyed map while retaining duplicate members.
131    fn map(
132        &mut self,
133        additional: u8,
134        depth: usize,
135    ) -> Result<Vec<(String, LocatedValue)>, DecodeError> {
136        let length = self.container_length(additional)?;
137        let mut members = Vec::new();
138        for _ in 0..length {
139            let key = self.value(depth + 1)?;
140            let CborValue::Text(key) = key.value else {
141                return Err(Self::error_at(DecodeErrorClass::MalformedCbor, key.offset));
142            };
143            let value = self.value(depth + 1)?;
144            members.push((key, value));
145        }
146        Ok(members)
147    }
148
149    /// Reads and bounds one definite container length.
150    fn container_length(&mut self, additional: u8) -> Result<usize, DecodeError> {
151        let length = self.argument(additional)?;
152        let length =
153            usize::try_from(length).map_err(|_| self.error(DecodeErrorClass::EncodedSizeLimit))?;
154        if length > self.limits.maximum_container_items() {
155            return Err(self.error(DecodeErrorClass::EncodedSizeLimit));
156        }
157        if length > self.remaining() {
158            return Err(self.error(DecodeErrorClass::MalformedCbor));
159        }
160        Ok(length)
161    }
162
163    /// Reads one bounded byte string after validating remaining input.
164    fn byte_string(&mut self, additional: u8, maximum: usize) -> Result<Vec<u8>, DecodeError> {
165        let length = self.argument(additional)?;
166        let length =
167            usize::try_from(length).map_err(|_| self.error(DecodeErrorClass::EncodedSizeLimit))?;
168        if length > maximum {
169            return Err(self.error(DecodeErrorClass::EncodedSizeLimit));
170        }
171        if length > self.remaining() {
172            return Err(self.error(DecodeErrorClass::MalformedCbor));
173        }
174        let end = self
175            .position
176            .checked_add(length)
177            .ok_or_else(|| self.error(DecodeErrorClass::EncodedSizeLimit))?;
178        let value = self.bytes[self.position..end].to_vec();
179        self.position = end;
180        Ok(value)
181    }
182
183    /// Reads one bounded valid UTF-8 text string.
184    fn text_string(&mut self, additional: u8) -> Result<String, DecodeError> {
185        let bytes = self.byte_string(additional, self.limits.maximum_text_bytes())?;
186        String::from_utf8(bytes).map_err(|_| self.error(DecodeErrorClass::MalformedCbor))
187    }
188
189    /// Reads a supported CBOR argument and rejects indefinite/reserved forms.
190    fn argument(&mut self, additional: u8) -> Result<u64, DecodeError> {
191        match additional {
192            0..=23 => Ok(u64::from(additional)),
193            24 => Ok(u64::from(self.read_byte()?)),
194            25 => Ok(u64::from(u16::from_be_bytes(self.read_array()?))),
195            26 => Ok(u64::from(u32::from_be_bytes(self.read_array()?))),
196            27 => Ok(u64::from_be_bytes(self.read_array()?)),
197            _ => Err(self.error(DecodeErrorClass::MalformedCbor)),
198        }
199    }
200
201    /// Reads one fixed-size byte array without unchecked slicing.
202    fn read_array<const N: usize>(&mut self) -> Result<[u8; N], DecodeError> {
203        let end = self
204            .position
205            .checked_add(N)
206            .ok_or_else(|| self.error(DecodeErrorClass::MalformedCbor))?;
207        let bytes = self
208            .bytes
209            .get(self.position..end)
210            .ok_or_else(|| self.error(DecodeErrorClass::MalformedCbor))?;
211        self.position = end;
212        bytes
213            .try_into()
214            .map_err(|_| self.error(DecodeErrorClass::InternalDefect))
215    }
216
217    /// Reads one byte or returns a bounded truncation error.
218    fn read_byte(&mut self) -> Result<u8, DecodeError> {
219        let value = self
220            .bytes
221            .get(self.position)
222            .copied()
223            .ok_or_else(|| self.error(DecodeErrorClass::MalformedCbor))?;
224        self.position += 1;
225        Ok(value)
226    }
227
228    /// Returns the unread section byte count.
229    fn remaining(&self) -> usize {
230        self.bytes.len().saturating_sub(self.position)
231    }
232
233    /// Returns the current absolute offset with checked conversion.
234    fn absolute_offset(&self) -> Result<u64, DecodeError> {
235        let absolute = self
236            .base_offset
237            .checked_add(self.position)
238            .ok_or_else(|| self.error(DecodeErrorClass::EncodedSizeLimit))?;
239        u64::try_from(absolute).map_err(|_| self.error(DecodeErrorClass::EncodedSizeLimit))
240    }
241
242    /// Returns an error at the current position.
243    fn error(&self, class: DecodeErrorClass) -> DecodeError {
244        let offset = self
245            .base_offset
246            .checked_add(self.position)
247            .and_then(|value| u64::try_from(value).ok());
248        DecodeError::new(class, offset)
249    }
250
251    /// Returns an error at a previously captured absolute position.
252    const fn error_at(class: DecodeErrorClass, offset: u64) -> DecodeError {
253        DecodeError::new(class, Some(offset))
254    }
255
256    /// Stops promptly when caller cancellation is observed.
257    fn check_cancelled(&self) -> Result<(), DecodeError> {
258        if self.cancellation.is_cancelled() {
259            Err(self.error(DecodeErrorClass::Cancelled))
260        } else {
261            Ok(())
262        }
263    }
264}
