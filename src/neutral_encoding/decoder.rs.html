13/// One duplicate-preserving untrusted CBOR value and its byte offset.
14pub(crate) struct LocatedValue {
15    /// Absolute encoded byte offset of the initial byte.
16    pub(crate) offset: u64,
17    /// Untrusted restricted-CBOR value.
18    pub(crate) value: CborValue,
19}
20
21/// Untrusted value forms admitted by the restricted CBOR lexical profile.
22pub(crate) enum CborValue {
23    /// Unsigned integer.
24    Unsigned(u64),
25    /// Signed negative integer.
26    Negative(i64),
27    /// Definite byte string.
28    Bytes(Vec<u8>),
29    /// Definite valid UTF-8 text string.
30    Text(String),
31    /// Definite array.
32    Array(Vec<LocatedValue>),
33    /// Definite map retaining member order and duplicates.
34    Map(Vec<(String, LocatedValue)>),
35    /// Boolean simple value.
36    Boolean(bool),
37    /// Null simple value.
38    Null,
39}

41/// Parses exactly one bounded restricted-CBOR section.
42pub(crate) fn parse_section(
43    bytes: &[u8],
44    base_offset: usize,
45    limits: DecodeLimits,
46    cancellation: &CancellationToken,
47) -> Result<LocatedValue, DecodeError> {
48    parse_section_profile(bytes, base_offset, limits, cancellation, false, true)
49        .map(|(value, _)| value)
50}

52/// Parses successor wire data with minimal-width integer and length arguments.
53pub(crate) fn parse_canonical_section(

63/// Checks writer output with the same lexical budgets, without retaining a decode tree.
64pub(crate) fn validate_section(
65    bytes: &[u8],
66    limits: DecodeLimits,
67    cancellation: &CancellationToken,
68) -> Result<(), DecodeError> {
69    parse_section_profile(bytes, 0, limits, cancellation, true, false).map(|_| ())
70}

72/// Keeps legacy lexical acceptance separate from explicit canonical successor selection.
73fn parse_section_profile(
74    bytes: &[u8],
75    base_offset: usize,
76    limits: DecodeLimits,
77    cancellation: &CancellationToken,
78    canonical: bool,
79    materialize: bool,
80) -> Result<(LocatedValue, usize), DecodeError> {
81    let mut parser = Parser {
82        bytes,
83        position: 0,
84        base_offset,
85        limits,
86        nodes: 0,
87        cancellation,
88        canonical,
89        materialize,
90        allocations: 0,
91        allocation_fault: None,
92    };
93    let value = parser.value(1)?;
94    if parser.position != bytes.len() {
95        return Err(parser.error(DecodeErrorClass::MalformedCbor));
96    }
97    Ok((value, parser.allocations))
98}

100/// Stateful parser that checks bounds before every allocation.
101struct Parser<'a> {
102    /// Validation-only walks share grammar and budgets but retain no untrusted tree.
103    materialize: bool,
104    /// Request-local reservation count for tests, never a global allocator override.
105    allocations: usize,
106    /// One private synthetic allocation failure or cancellation checkpoint.
107    allocation_fault: Option<(usize, bool)>,
108    /// Exact section bytes.
109    bytes: &'a [u8],
110    /// Next unread section-relative position.
111    position: usize,
112    /// Absolute offset of the section.
113    base_offset: usize,
114    /// Effective caller and hard limits.
115    limits: DecodeLimits,
116    /// Total decoded nodes.
117    nodes: usize,
118    /// Cooperative cancellation signal.
119    cancellation: &'a CancellationToken,
120    /// Explicit successor minimal-width enforcement; old selectors remain unchanged.
121    canonical: bool,
122}
123
124impl<'a> Parser<'a> {
125    /// Checks cancellation before fallible storage growth; tests interrupt the same boundary.
126    fn reserve<T>(&mut self, values: &mut Vec<T>, count: usize) -> Result<(), DecodeError> {
127        self.check_cancelled()?;
128        if !self.materialize {
129            return Ok(());
130        }
131        {
132            let current = self.allocations;
133            self.allocations += 1;
134            if let Some((index, cancel)) = self.allocation_fault
135                && index == current
136            {
137                if cancel {
138                    self.cancellation.cancel();
139                    return Err(self.error(DecodeErrorClass::Cancelled));
140                }
141                return Err(self.error(DecodeErrorClass::EncodedSizeLimit));
142            }
143        }
144        values
145            .try_retain(count)
146            .map_err(|_| self.error(DecodeErrorClass::EncodedSizeLimit))?;
147        self.check_cancelled()
148    }
149    /// Parses one value under depth, traversal, and cancellation limits.
150    fn value(&mut self, depth: usize) -> Result<LocatedValue, DecodeError> {
151        self.check_cancelled()?;
152        if depth > self.limits.maximum_nesting_depth() {
153            return Err(self.error(DecodeErrorClass::EncodedSizeLimit));
154        }
155        self.nodes = self
156            .nodes
157            .checked_add(1)
158            .ok_or_else(|| self.error(DecodeErrorClass::EncodedSizeLimit))?;
159        if self.nodes > self.limits.maximum_traversal_nodes() {
160            return Err(self.error(DecodeErrorClass::EncodedSizeLimit));
161        }
162        let offset = self.absolute_offset()?;
163        let initial = self.read_byte()?;
164        let major = initial >> 5;
165        let additional = initial & 0x1f;
166        let value = match major {
167            0 => CborValue::Unsigned(self.argument(additional)?),
168            1 => {
169                let argument = self.argument(additional)?;
170                if argument > i64::MAX.cast_unsigned() {
171                    return Err(Self::error_at(
172                        DecodeErrorClass::InvalidEncodedSchema,
173                        offset,
174                    ));
175                }
176                let magnitude = i64::try_from(argument)
177                    .map_err(|_| self.error(DecodeErrorClass::InvalidEncodedSchema))?;
178                CborValue::Negative(-1 - magnitude)
179            }
180            2 => CborValue::Bytes(
181                self.byte_string(additional, self.limits.maximum_byte_string_bytes())?,
182            ),
183            3 => CborValue::Text(self.text_string(additional)?),
184            4 => CborValue::Array(self.array(additional, depth)?),
185            5 => CborValue::Map(self.map(additional, depth)?),
186            7 if additional == 20 => CborValue::Boolean(false),
187            7 if additional == 21 => CborValue::Boolean(true),
188            7 if additional == 22 => CborValue::Null,
189            _ => return Err(Self::error_at(DecodeErrorClass::MalformedCbor, offset)),
190        };
191        Ok(LocatedValue { offset, value })
192    }
193
194    /// Parses one bounded definite array without trusting its length for reserve.
195    fn array(&mut self, additional: u8, depth: usize) -> Result<Vec<LocatedValue>, DecodeError> {
196        let length = self.container_length(additional)?;
197        let mut values = Vec::new();
198        for _ in 0..length {
199            if self.nodes >= self.limits.maximum_traversal_nodes() {
200                return Err(self.error(DecodeErrorClass::EncodedSizeLimit));
201            }
202            self.reserve(&mut values, 1)?;
203            let value = self.value(depth + 1)?;
204            if self.materialize {
205                values.push(value);
206            }
207        }
208        Ok(values)
209    }
210
211    /// Parses one text-keyed map while retaining duplicate members.
212    fn map(
213        &mut self,
214        additional: u8,
215        depth: usize,
216    ) -> Result<Vec<(String, LocatedValue)>, DecodeError> {
217        let length = self.container_length(additional)?;
218        let mut members = Vec::new();
219        for _ in 0..length {
220            if self
221                .limits
222                .maximum_traversal_nodes()
223                .saturating_sub(self.nodes)
224                < 2
225            {
226                return Err(self.error(DecodeErrorClass::EncodedSizeLimit));
227            }
228            self.reserve(&mut members, 1)?;
229            let key = self.value(depth + 1)?;
230            let CborValue::Text(key) = key.value else {
231                return Err(Self::error_at(DecodeErrorClass::MalformedCbor, key.offset));
232            };
233            let value = self.value(depth + 1)?;
234            if self.materialize {
235                members.push((key, value));
236            }
237        }
238        Ok(members)
239    }
240
241    /// Reads and bounds one definite container length.
242    fn container_length(&mut self, additional: u8) -> Result<usize, DecodeError> {
243        let length = self.argument(additional)?;
244        let length =
245            usize::try_from(length).map_err(|_| self.error(DecodeErrorClass::EncodedSizeLimit))?;
246        if length > self.limits.maximum_container_items() {
247            return Err(self.error(DecodeErrorClass::EncodedSizeLimit));
248        }
249        if length > self.remaining() {
250            return Err(self.error(DecodeErrorClass::MalformedCbor));
251        }
252        Ok(length)
253    }
254
255    /// Borrows bounded string bytes after checking length, before any retention.
256    fn string_slice(&mut self, additional: u8, maximum: usize) -> Result<&'a [u8], DecodeError> {
257        let length = self.argument(additional)?;
258        let length =
259            usize::try_from(length).map_err(|_| self.error(DecodeErrorClass::EncodedSizeLimit))?;
260        if length > maximum {
261            return Err(self.error(DecodeErrorClass::EncodedSizeLimit));
262        }
263        if length > self.remaining() {
264            return Err(self.error(DecodeErrorClass::MalformedCbor));
265        }
266        let end = self
267            .position
268            .checked_add(length)
269            .ok_or_else(|| self.error(DecodeErrorClass::EncodedSizeLimit))?;
270        let start = self.position;
271        self.position = end;
272        Ok(&self.bytes[start..end])
273    }
274
275    /// Retains byte strings only when constructing an independent decode tree.
276    fn byte_string(&mut self, additional: u8, maximum: usize) -> Result<Vec<u8>, DecodeError> {
277        let bytes = self.string_slice(additional, maximum)?;
278        let length = bytes.len();
279        let mut value = Vec::new();
280        self.reserve(&mut value, length)?;
281        if self.materialize {
282            value.extend_from_slice(bytes);
283        }
284        Ok(value)
285    }
286
287    /// Reads one bounded valid UTF-8 text string.
288    fn text_string(&mut self, additional: u8) -> Result<String, DecodeError> {
289        let bytes = self.string_slice(additional, self.limits.maximum_text_bytes())?;
290        let length = bytes.len();
291        if std::str::from_utf8(bytes).is_err() {
292            return Err(self.error(DecodeErrorClass::MalformedCbor));
293        }
294        let mut retained = Vec::new();
295        self.reserve(&mut retained, length)?;
296        if self.materialize {
297            retained.extend_from_slice(bytes);
298        }
299        String::from_utf8(retained).map_err(|_| self.error(DecodeErrorClass::MalformedCbor))
300    }
301
302    /// Reads a supported CBOR argument and rejects indefinite/reserved forms.
303    fn argument(&mut self, additional: u8) -> Result<u64, DecodeError> {
304        let value = match additional {
305            0..=23 => u64::from(additional),
306            24 => u64::from(self.read_byte()?),
307            25 => u64::from(u16::from_be_bytes(self.read_array()?)),
308            26 => u64::from(u32::from_be_bytes(self.read_array()?)),
309            27 => u64::from_be_bytes(self.read_array()?),
310            _ => return Err(self.error(DecodeErrorClass::MalformedCbor)),
311        };
312        let minimum = match additional {
313            24 => 24,
314            25 => 256,
315            26 => 65_536,
316            27 => 4_294_967_296,
317            _ => 0,
318        };
319        if self.canonical && value < minimum {
320            return Err(self.error(DecodeErrorClass::MalformedCbor));
321        }
322        Ok(value)
323    }
324
325    /// Reads one fixed-size byte array without unchecked slicing.
326    fn read_array<const N: usize>(&mut self) -> Result<[u8; N], DecodeError> {
327        let end = self
328            .position
329            .checked_add(N)
330            .ok_or_else(|| self.error(DecodeErrorClass::MalformedCbor))?;
331        let bytes = self
332            .bytes
333            .get(self.position..end)
334            .ok_or_else(|| self.error(DecodeErrorClass::MalformedCbor))?;
335        self.position = end;
336        bytes
337            .try_into()
338            .map_err(|_| self.error(DecodeErrorClass::InternalDefect))
339    }
340
341    /// Reads one byte or returns a bounded truncation error.
342    fn read_byte(&mut self) -> Result<u8, DecodeError> {
343        let value = self
344            .bytes
345            .get(self.position)
346            .copied()
347            .ok_or_else(|| self.error(DecodeErrorClass::MalformedCbor))?;
348        self.position += 1;
349        Ok(value)
350    }
351
352    /// Returns the unread section byte count.
353    fn remaining(&self) -> usize {
354        self.bytes.len().saturating_sub(self.position)
355    }
356
357    /// Returns the current absolute offset with checked conversion.
358    fn absolute_offset(&self) -> Result<u64, DecodeError> {
359        let absolute = self
360            .base_offset
361            .checked_add(self.position)
362            .ok_or_else(|| self.error(DecodeErrorClass::EncodedSizeLimit))?;
363        u64::try_from(absolute).map_err(|_| self.error(DecodeErrorClass::EncodedSizeLimit))
364    }
365
366    /// Returns an error at the current position.
367    fn error(&self, class: DecodeErrorClass) -> DecodeError {
368        let offset = self
369            .base_offset
370            .checked_add(self.position)
371            .and_then(|value| u64::try_from(value).ok());
372        DecodeError::new(class, offset)
373    }
374
375    /// Returns an error at a previously captured absolute position.
376    const fn error_at(class: DecodeErrorClass, offset: u64) -> DecodeError {
377        DecodeError::new(class, Some(offset))
378    }
379
380    /// Stops promptly when caller cancellation is observed.
381    fn check_cancelled(&self) -> Result<(), DecodeError> {
382        if self.cancellation.is_cancelled() {
383            Err(self.error(DecodeErrorClass::Cancelled))
384        } else {
385            Ok(())
386        }
387    }
