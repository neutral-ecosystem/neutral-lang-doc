3//! External encoding and hostile decoding of validated Neutral documents.
4//!
5//! This codec implements Neutral IR Framed CBOR 0.1. The encoder accepts only
6//! trusted reader views; the bounds-first decoder returns one only after full
7//! validation. Deterministic emitted field order is an implementation property:
8//! encoded bytes are not logical identity, and producer/build facts remain
9//! confined to the envelope.

28/// Complete bounded result class for hostile external artifact decoding.
29#[derive(Clone, Copy, Debug, Eq, PartialEq)]
30pub enum DecodeErrorClass {
31    /// An artifact, section, string, container, depth, or traversal limit failed.
32    EncodedSizeLimit,
33    /// Fixed framing or directory structure was malformed.
34    MalformedFrame,
35    /// A fixed framing, section, or contract version is unsupported.
36    UnsupportedVersion,
37    /// Capability declarations are unsupported or inconsistent with content.
38    UnsupportedCapability,
39    /// Exact section integrity verification failed.
40    IntegrityMismatch,
41    /// Restricted-CBOR lexical/container validation failed.
42    MalformedCbor,
43    /// A closed encoded schema was malformed.
44    InvalidEncodedSchema,
45    /// Logical IR validation failed.
46    InvalidLogicalIr,
47    /// Source-map validation failed.
48    InvalidSourceMap,
49    /// Provenance validation failed.
50    InvalidProvenance,
51    /// Derivation or cross-section validation failed.
52    InvalidDerivation,
53    /// Cooperative caller cancellation was observed.
54    Cancelled,
55    /// An implementation invariant failed independently of input validity.
56    InternalDefect,
57}
58
59impl DecodeErrorClass {
60    /// Returns the stable bounded diagnostic code for this result class.
61    #[must_use]
62    pub const fn code(self) -> &'static str {
63        match self {
64            Self::EncodedSizeLimit => diagnostics::ENCODED_SIZE_LIMIT,
65            Self::MalformedFrame => diagnostics::MALFORMED_FRAME,
66            Self::UnsupportedVersion => diagnostics::UNSUPPORTED_VERSION,
67            Self::UnsupportedCapability => diagnostics::UNSUPPORTED_CAPABILITY,
68            Self::IntegrityMismatch => diagnostics::INTEGRITY_MISMATCH,
69            Self::MalformedCbor => diagnostics::MALFORMED_CBOR,
70            Self::InvalidEncodedSchema => diagnostics::INVALID_ENCODED_SCHEMA,
71            Self::InvalidLogicalIr => diagnostics::INVALID_LOGICAL_IR,
72            Self::InvalidSourceMap => diagnostics::INVALID_SOURCE_MAP,
73            Self::InvalidProvenance => diagnostics::INVALID_PROVENANCE,
74            Self::InvalidDerivation => diagnostics::INVALID_DERIVATION,
75            Self::Cancelled => diagnostics::CANCELLED,
76            Self::InternalDefect => diagnostics::INTERNAL_DEFECT,
77        }
78    }
79}

81/// Bounded safe decoder failure with an optional encoded-byte offset.
82#[derive(Clone, Copy, Debug, Eq, PartialEq)]
83pub struct DecodeError {
84    /// Stable failure class.
85    class: DecodeErrorClass,
86    /// Absolute byte offset when one is available.
87    offset: Option<u64>,
88}
89
90impl DecodeError {
91    /// Constructs one bounded internal decoder error.
92    const fn new(class: DecodeErrorClass, offset: Option<u64>) -> Self {
93        Self { class, offset }
94    }
95
96    /// Returns the stable result class.
97    #[must_use]
98    pub const fn class(self) -> DecodeErrorClass {
99        self.class
100    }
101
102    /// Returns the stable diagnostic code.
103    #[must_use]
104    pub const fn code(self) -> &'static str {
105        self.class.code()
106    }
107
108    /// Returns the absolute encoded-byte offset when available.
109    #[must_use]
110    pub const fn offset(self) -> Option<u64> {
111        self.offset
112    }
113}

115/// Host-configurable decoder ceilings, always clamped to frozen hard limits.
116#[derive(Clone, Copy, Debug, Eq, PartialEq)]
117pub struct DecodeLimits {
118    /// Complete artifact bytes.
119    artifact_bytes: usize,
120    /// Bytes in any one section.
121    section_bytes: usize,
122    /// Nested CBOR/logical containers.
123    nesting_depth: usize,
124    /// Items in one container.
125    container_items: usize,
126    /// Bytes in one text string.
127    text_bytes: usize,
128    /// Bytes in one byte string.
129    byte_string_bytes: usize,
130    /// Total traversed nodes.
131    traversal_nodes: usize,
132}
133
134impl DecodeLimits {
135    /// Returns the immutable frozen hard ceilings.
136    #[must_use]
137    pub const fn hard() -> Self {
138        Self {
139            artifact_bytes: constants::MAXIMUM_ARTIFACT_BYTES,
140            section_bytes: constants::MAXIMUM_SECTION_BYTES,
141            nesting_depth: constants::MAXIMUM_NESTING_DEPTH,
142            container_items: constants::MAXIMUM_CONTAINER_ITEMS,
143            text_bytes: constants::MAXIMUM_TEXT_BYTES,
144            byte_string_bytes: constants::MAXIMUM_BYTE_STRING_BYTES,
145            traversal_nodes: constants::MAXIMUM_TRAVERSAL_NODES,
146        }
147    }
148
149    /// Applies a lower host artifact-byte ceiling; zero rejects every artifact.
150    #[must_use]
151    pub const fn with_artifact_bytes(mut self, value: usize) -> Self {
152        self.artifact_bytes = min_usize(self.artifact_bytes, value);
153        self
154    }
155
156    /// Applies a lower host section-byte ceiling.
157    #[must_use]
158    pub const fn with_section_bytes(mut self, value: usize) -> Self {
159        self.section_bytes = min_usize(self.section_bytes, value);
160        self
161    }
162
163    /// Applies a lower host nesting-depth ceiling.
164    #[must_use]
165    pub const fn with_nesting_depth(mut self, value: usize) -> Self {
166        self.nesting_depth = min_usize(self.nesting_depth, value);
167        self
168    }
169
170    /// Applies a lower host per-container item ceiling.
171    #[must_use]
172    pub const fn with_container_items(mut self, value: usize) -> Self {
173        self.container_items = min_usize(self.container_items, value);
174        self
175    }
176
177    /// Applies a lower host text-string byte ceiling.
178    #[must_use]
179    pub const fn with_text_bytes(mut self, value: usize) -> Self {
180        self.text_bytes = min_usize(self.text_bytes, value);
181        self
182    }
183
184    /// Applies a lower host byte-string ceiling.
185    #[must_use]
186    pub const fn with_byte_string_bytes(mut self, value: usize) -> Self {
187        self.byte_string_bytes = min_usize(self.byte_string_bytes, value);
188        self
189    }
190
191    /// Applies a lower host total traversal ceiling.
192    #[must_use]
193    pub const fn with_traversal_nodes(mut self, value: usize) -> Self {
194        self.traversal_nodes = min_usize(self.traversal_nodes, value);
195        self
196    }
197
198    /// Returns the effective artifact-byte ceiling.
199    pub(crate) const fn maximum_artifact_bytes(self) -> usize {
200        self.artifact_bytes
201    }
202    /// Returns the effective per-section byte ceiling.
203    pub(crate) const fn maximum_section_bytes(self) -> usize {
204        self.section_bytes
205    }
206    /// Returns the effective nesting-depth ceiling.
207    pub(crate) const fn maximum_nesting_depth(self) -> usize {
208        self.nesting_depth
209    }
210    /// Returns the effective per-container item ceiling.
211    pub(crate) const fn maximum_container_items(self) -> usize {
212        self.container_items
213    }
214    /// Returns the effective text-string byte ceiling.
215    pub(crate) const fn maximum_text_bytes(self) -> usize {
216        self.text_bytes
217    }
218    /// Returns the effective byte-string ceiling.
219    pub(crate) const fn maximum_byte_string_bytes(self) -> usize {
220        self.byte_string_bytes
221    }
222    /// Returns the effective traversal ceiling.
223    pub(crate) const fn maximum_traversal_nodes(self) -> usize {
224        self.traversal_nodes
225    }
226}

228/// Returns the smaller of two `usize` values in constant contexts.
229const fn min_usize(left: usize, right: usize) -> usize {
230    if left < right { left } else { right }
231}

233/// Envelope-only facts describing the implementation that produced an artifact.
234#[derive(Clone, Debug, Eq, PartialEq)]
235pub struct ProducerInfo {
236    /// Producer name.
237    name: String,
238    /// Producer version.
239    version: String,
240    /// Optional producer build identity.
241    build: Option<String>,
242}
243
244impl ProducerInfo {
245    /// Creates producer facts without a build identity.
246    #[must_use]
247    pub fn new(name: impl Into<String>, version: impl Into<String>) -> Self {
248        Self {
249            name: name.into(),
250            version: version.into(),
251            build: None,
252        }
253    }
254
255    /// Attaches an envelope-only build identity.
256    #[must_use]
257    pub fn with_build(mut self, build: impl Into<String>) -> Self {
258        self.build = Some(build.into());
259        self
260    }
261
262    /// Returns the producer name.
263    #[must_use]
264    pub fn name(&self) -> &str {
265        &self.name
266    }
267
268    /// Returns the producer version.
269    #[must_use]
270    pub fn version(&self) -> &str {
271        &self.version
272    }
273
274    /// Returns the optional producer build identity.
275    #[must_use]
276    pub fn build(&self) -> Option<&str> {
277        self.build.as_deref()
278    }
279}

281/// One fixed external section kind.
282#[derive(Clone, Copy, Debug, Eq, PartialEq)]
283#[repr(u16)]
284pub enum SectionKind {
285    /// Encoding and producer envelope.
286    Envelope = 1,
287    /// Authoritative logical payload.
288    LogicalPayload = 2,
289    /// Original-byte source map.
290    SourceMap = 3,
291    /// Complete provenance companions.
292    Provenance = 4,
293    /// Complete derivation companions.
294    Derivation = 5,
295}
296
297impl SectionKind {
298    /// Returns the zero-based section position in the fixed directory.
299    const fn index(self) -> usize {
300        self as usize - 1
301    }
302}

304/// Complete immutable encoded Neutral artifact.
305#[derive(Clone, Debug, Eq, PartialEq)]
306pub struct EncodedArtifact {
307    /// Complete framed bytes.
308    bytes: Vec<u8>,
309    /// Exact byte range for each fixed section kind.
310    sections: [Range<usize>; constants::SECTION_COUNT],
311}
312
313impl EncodedArtifact {
314    /// Returns the complete framed artifact bytes.
315    #[must_use]
316    pub fn as_bytes(&self) -> &[u8] {
317        &self.bytes
318    }
319
320    /// Returns the exact restricted-CBOR bytes for one section.
321    #[must_use]
322    pub fn section_bytes(&self, kind: SectionKind) -> &[u8] {
323        &self.bytes[self.sections[kind.index()].clone()]
324    }
325
326    /// Consumes the wrapper and returns the complete frame.
327    #[must_use]
328    pub fn into_bytes(self) -> Vec<u8> {
329        self.bytes
330    }
331}
