13/// Independent schema for generated workflow summaries.
14const WORKFLOW_SCHEMA_VERSION: u32 = 1;
15use std::io::Write as _;

17/// One append-only JSONL event; optional errors retain their explicit null representation.
18#[derive(Serialize)]
19struct WorkflowEvent<'a> {
20    /// Workflow name.
21    workflow: &'a str,
22    /// Selected execution profile.
23    profile: &'a str,
24    /// Ordered step name.
25    step: &'a str,
26    /// Event state.
27    status: &'a str,
28    /// Event time in Unix milliseconds.
29    timestamp_unix_ms: u128,
30    /// Monotonic step elapsed time.
31    duration_ms: u128,
32    /// Failure details, or null for nonfailure events.
33    error: Option<&'a str>,
34}
35
36/// Current workflow state with source and tool provenance.
37#[derive(Serialize)]
38struct WorkflowSummary<'a> {
39    /// Independent generated-document schema.
40    schema_version: u32,
41    /// Workflow name.
42    workflow: &'a str,
43    /// Selected execution profile.
44    profile: &'a str,
45    /// Current state.
46    status: &'a str,
47    /// Successful steps so far.
48    completed_steps: usize,
49    /// Number of selected steps.
50    total_steps: usize,
51    /// Initial Unix time.
52    started_at_unix_ms: u128,
53    /// Most recent Unix time.
54    updated_at_unix_ms: u128,
55    /// Checkout identity at this summary update.
56    source_commit: String,
57    /// Whether the checkout has tracked or untracked changes.
58    worktree_clean: bool,
59    /// Workspace package version.
60    package_version: String,
61    /// Workspace license expression.
62    license: String,
63    /// Selected compiler version.
64    rustc: String,
65    /// Failure details, or null.
66    error: Option<&'a str>,
67}

69/// Runs ordered workflow steps while retaining start, pass, and failure events.
70pub(crate) fn run_recorded_workflow(
71    workflow: &str,
72    profile: &str,
73    steps: Vec<WorkflowStep<'_>>,
74) -> Result<(), String> {
75    let directory = unique_generated_directory(
76        &result_root()?
77            .join(constants::WORKFLOW_RESULT_DIRECTORY)
78            .join(workflow)
79            .join(profile),
80    )?;
81    let events = directory.join(constants::WORKFLOW_EVENTS_FILE);
82    let summary = directory.join(constants::WORKFLOW_SUMMARY_FILE);
83    let total = steps.len();
84    let started_at = unix_time_millis()?;
85    let started = Instant::now();
86    write_workflow_summary(
87        &summary, workflow, profile, "running", 0, total, started_at, None,
88    )?;
89    output::start(format!("{workflow} {profile} ({total} steps)"));
90    output::file("workflow log", &directory);
91
92    for (index, (name, step)) in steps.into_iter().enumerate() {
93        let step_started_at = unix_time_millis()?;
94        let step_started = Instant::now();
95        append_workflow_event(
96            &events,
97            workflow,
98            profile,
99            name,
100            "start",
101            step_started_at,
102            0,
103            None,
104        )?;
105        output::start(format!("{workflow} [{}/{total}] {name}", index + 1));
106        match step() {
107            Ok(()) => {
108                let elapsed = step_started.elapsed().as_millis();
109                append_workflow_event(
110                    &events,
111                    workflow,
112                    profile,
113                    name,
114                    "pass",
115                    unix_time_millis()?,
116                    elapsed,
117                    None,
118                )?;
119                write_workflow_summary(
120                    &summary,
121                    workflow,
122                    profile,
123                    "running",
124                    index + 1,
125                    total,
126                    started_at,
127                    None,
128                )?;
129                output::pass(format!(
130                    "{workflow} [{}/{total}] {name} ({})",
131                    index + 1,
132                    output::duration(step_started.elapsed())
133                ));
134            }
135            Err(error) => {
136                let elapsed = step_started.elapsed().as_millis();
137                append_workflow_event(
138                    &events,
139                    workflow,
140                    profile,
141                    name,
142                    "fail",
143                    unix_time_millis()?,
144                    elapsed,
145                    Some(&error),
146                )?;
147                write_workflow_summary(
148                    &summary,
149                    workflow,
150                    profile,
151                    "fail",
152                    index,
153                    total,
154                    started_at,
155                    Some(&error),
156                )?;
157                return Err(format!(
158                    "{workflow} [{}/{total}] {name} failed ({}): {error}; workflow log: {}",
159                    index + 1,
160                    output::duration(step_started.elapsed()),
161                    output::path(&directory)
162                ));
163            }
164        }
165    }
166
167    write_workflow_summary(
168        &summary, workflow, profile, "pass", total, total, started_at, None,
169    )?;
170    output::pass(format!(
171        "{workflow} {profile} ({total}/{total} steps, {})",
172        output::duration(started.elapsed())
173    ));
174    Ok(())
175}

177/// Appends one machine-readable workflow event without replacing earlier events.
178#[allow(clippy::too_many_arguments)]
179pub(crate) fn append_workflow_event(
180    path: &Path,
181    workflow: &str,
182    profile: &str,
183    step: &str,
184    status: &str,
185    timestamp_unix_ms: u128,
186    duration_ms: u128,
187    error: Option<&str>,
188) -> Result<(), String> {
189    let mut file = OpenOptions::new()
190        .create(true)
191        .append(true)
192        .open(path)
193        .map_err(|error| format!("could not open workflow event log: {error}"))?;
194    let event = WorkflowEvent {
195        workflow,
196        profile,
197        step,
198        status,
199        timestamp_unix_ms,
200        duration_ms,
201        error,
202    };
203    // JSONL must remain one complete compact object per physical line.
204    serde_json::to_writer(&mut file, &event)
205        .map_err(|error| format!("could not serialize workflow event: {error}"))?;
206    writeln!(file).map_err(|error| format!("could not write workflow event log: {error}"))
207}

209/// Writes the current machine-readable summary for an aggregate workflow.
210#[allow(clippy::too_many_arguments)]
211pub(crate) fn write_workflow_summary(
212    path: &Path,
213    workflow: &str,
214    profile: &str,
215    status: &str,
216    completed_steps: usize,
217    total_steps: usize,
218    started_at_unix_ms: u128,
219    error: Option<&str>,
220) -> Result<(), String> {
221    let root = workspace_root()?;
222    let manifest = read_workspace_text(&root, constants::WORKSPACE_MANIFEST_FILE)?;
223    let version = workspace_package_version(&manifest)?;
224    let license = workspace_package_license(&manifest)?;
225    let commit = command_output(constants::GIT_COMMAND, &["rev-parse", "HEAD"])?;
226    let worktree_clean =
227        command_output(constants::GIT_COMMAND, &["status", "--porcelain"])?.is_empty();
228    let rustc = command_output(&rustc_command()?, &[flags::VERSION])?;
229    let summary = WorkflowSummary {
230        schema_version: WORKFLOW_SCHEMA_VERSION,
231        workflow,
232        profile,
233        status,
234        completed_steps,
235        total_steps,
236        started_at_unix_ms,
237        updated_at_unix_ms: unix_time_millis()?,
238        source_commit: commit,
239        worktree_clean,
240        package_version: version,
241        license,
242        rustc,
243        error,
244    };
245    fs::write(path, super::json::pretty(&summary)?)
246        .map_err(|error| format!("could not write workflow summary: {error}"))
247}

249/// Returns milliseconds elapsed since the Unix epoch for generated evidence.
250pub(crate) fn unix_time_millis() -> Result<u128, String> {
251    SystemTime::now()
252        .duration_since(UNIX_EPOCH)
253        .map(|duration| duration.as_millis())
254        .map_err(|error| format!("system clock precedes Unix epoch: {error}"))
255}
