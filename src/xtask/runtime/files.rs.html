7/// Returns the SHA-256 digest of one exact file.
8pub(crate) fn sha256_file(path: &Path) -> Result<String, String> {
9    fs::read(path)
10        .map(|bytes| sha256_hex(&bytes))
11        .map_err(|error| format!("could not read {}: {error}", path.display()))
12}

14/// Returns whether text is one lowercase SHA-256 hexadecimal digest.
15pub(crate) fn is_sha256(value: &str) -> bool {
16    value.len() == 64
17        && value
18            .chars()
19            .all(|character| character.is_ascii_hexdigit() && !character.is_ascii_uppercase())
20}

22/// Extracts local, anchor-free Markdown link paths from document text.
23pub(crate) fn markdown_link_targets(content: &str) -> Vec<String> {
24    content
25        .split("](")
26        .skip(1)
27        .filter_map(|tail| tail.split_once(')').map(|(target, _)| target))
28        .map(str::trim)
29        .map(|target| target.trim_matches(['<', '>']))
30        .filter(|target| {
31            !target.is_empty()
32                && !target.starts_with('#')
33                && !target.contains("://")
34                && !target.starts_with("mailto:")
35        })
36        .map(|target| target.split('#').next().unwrap_or(target))
37        .filter(|target| !target.is_empty())
38        .map(str::to_owned)
39        .collect()
40}

42/// Reads one required UTF-8 workspace file with a path-safe error.
43pub(crate) fn read_workspace_text(root: &Path, relative: &str) -> Result<String, String> {
44    fs::read_to_string(root.join(relative))
45        .map_err(|error| format!("could not read {relative}: {error}"))

48/// Recursively collects regular files without following directory symlinks.
49pub(crate) fn collect_regular_files(
50    directory: &Path,
51    files: &mut Vec<PathBuf>,
52) -> Result<(), String> {
53    let entries = fs::read_dir(directory)
54        .map_err(|error| format!("could not inspect {}: {error}", directory.display()))?;
55    for entry in entries {
56        let entry = entry.map_err(|error| format!("could not inspect directory entry: {error}"))?;
57        let file_type = entry
58            .file_type()
59            .map_err(|error| format!("could not inspect {}: {error}", entry.path().display()))?;
60        if file_type.is_dir() {
61            collect_regular_files(&entry.path(), files)?;
62        } else if file_type.is_file() {
63            files.push(entry.path());
64        }
65    }
66    Ok(())
67}

69/// Creates a process-unique directory below one approved generated-output root.
70pub(crate) fn unique_generated_directory(parent: &Path) -> Result<PathBuf, String> {
71    fs::create_dir_all(parent).map_err(|error| {
72        format!(
73            "could not create generated evidence directory {}: {error}",
74            parent.display()
75        )
76    })?;
77    for suffix in 0_u16..1000 {
78        let directory = parent.join(format!("run-{}-{suffix}", std::process::id()));
79        match fs::create_dir(&directory) {
80            Ok(()) => return Ok(directory),
81            Err(error) if error.kind() == std::io::ErrorKind::AlreadyExists => {}
82            Err(error) => return Err(format!("could not create {}: {error}", directory.display())),
83        }
84    }
85    Err("could not allocate a unique result directory".to_owned())
86}

88/// Requires a clean checkout and returns its exact current Git commit.
89pub(crate) fn require_clean_checkout() -> Result<String, String> {
90    let commit = command_output(constants::GIT_COMMAND, &["rev-parse", "HEAD"])?;
91    let status = command_output(constants::GIT_COMMAND, &["status", "--porcelain"])?;
92    if !status.is_empty() {
93        return Err("quality evaluation requires a clean worktree".to_owned());
94    }
95    Ok(commit)
96}

98/// Requires a clean checked-out `main` and returns its exact current candidate commit.
99pub(crate) fn require_main_head_checkout() -> Result<String, String> {
100    let branch = command_output("git", &["branch", "--show-current"])?;
101    if branch != "main" {
102        return Err(format!(
103            "release qualification requires checked-out main; found {branch:?}"
104        ));
105    }
106    let head = command_output("git", &["rev-parse", "HEAD"])?;
107    let status = command_output("git", &["status", "--porcelain"])?;
108    if !status.is_empty() {
109        return Err("release qualification requires a clean main worktree".to_owned());
110    }
111    Ok(head)
112}
