104/// One parsed repository-automation request.
105#[derive(Clone, Debug, Eq, PartialEq)]
106pub(crate) enum Task {
107    /// Render stable command help.
108    Help,
109    /// Verify the contributor environment and initialize ignored output roots.
110    Bootstrap,
111    /// Run the ordered, auto-formatting local-development workflow.
112    Dev,
113    /// Verify the selected environment.
114    EnvironmentVerify,
115    /// Print a machine-readable environment manifest.
116    EnvironmentManifest,
117    /// Check or write Rust formatting.
118    Format {
119        /// Whether formatting changes should be applied.
120        write: bool,
121    },
122    /// Run warning-free linting.
123    Lint,
124    /// Run compilation and repository-coherence checks.
125    Check,
126    /// Build the workspace using a durable profile.
127    Build(BuildProfile),
128    /// Generate workspace documentation.
129    Docs,
130    /// Run one durable test level.
131    Test(TestLevel),
132    /// Run a controlled performance profile.
133    Performance(PerformanceProfile),
134    /// Run one fuzz mode.
135    Fuzz(FuzzMode),
136    /// Run configured LLVM coverage.
137    Coverage,
138    /// Run configured mutation analysis.
139    Mutate,
140    /// Operate the managed quality ledger or run an aggregate profile.
141    Quality(QualityAction),
142    /// Validate released binaries or one encoded artifact.
143    Validate(ValidationTarget),
144    /// Assemble the selected release distribution.
145    Package,
146    /// Prepare a release without publishing it.
147    ReleasePrepare,
148    /// Inspect or prepare package versioning.
149    Version(VersionAction),
150    /// Verify or snapshot the active portable package.
151    Portable(PortableAction),
152    /// Remove ignored generated evidence.
153    Clean,
154    /// Run one internal CI composition.
155    Ci(CiProfile),
156}

274/// Parses exact command arguments into one stable task.
275pub(crate) fn parse(arguments: &[String]) -> Result<Task, String> {
276    let values = arguments.iter().map(String::as_str).collect::<Vec<_>>();
277    match values.as_slice() {
278        [] | [HELP_OPTION | SHORT_HELP_OPTION] => Ok(Task::Help),
279        [BOOTSTRAP_COMMAND] => Ok(Task::Bootstrap),
280        [DEV_COMMAND] => Ok(Task::Dev),
281        [ENVIRONMENT_COMMAND, "verify"] => Ok(Task::EnvironmentVerify),
282        [ENVIRONMENT_COMMAND, "manifest"] => Ok(Task::EnvironmentManifest),
283        [FORMAT_COMMAND] => Ok(Task::Format { write: false }),
284        [FORMAT_COMMAND, "--write"] => Ok(Task::Format { write: true }),
285        [LINT_COMMAND] => Ok(Task::Lint),
286        [CHECK_COMMAND] => Ok(Task::Check),
287        [DOCS_COMMAND] => Ok(Task::Docs),
288        [BUILD_COMMAND, PROFILE_OPTION, "dev"] => Ok(Task::Build(BuildProfile::Dev)),
289        [BUILD_COMMAND, PROFILE_OPTION, "release"] => Ok(Task::Build(BuildProfile::Release)),
290        [TEST_COMMAND, "performance", PROFILE_OPTION, profile] => {
291            parse_performance_profile(profile).map(Task::Performance)
292        }
293        [TEST_COMMAND, level] => parse_test_level(level).map(Task::Test),
294        [FUZZ_COMMAND, mode] => parse_fuzz_mode(mode).map(Task::Fuzz),
295        [COVERAGE_COMMAND] => Ok(Task::Coverage),
296        [MUTATE_COMMAND] => Ok(Task::Mutate),
297        [QUALITY_COMMAND] => Ok(Task::Quality(QualityAction::Run(QualityProfile::Pr))),
298        [QUALITY_COMMAND, PROFILE_OPTION, profile] => parse_quality_profile(profile)
299            .map(QualityAction::Run)
300            .map(Task::Quality),
301        [QUALITY_COMMAND, "status"] => Ok(Task::Quality(QualityAction::Status)),
302        [QUALITY_COMMAND, "render"] => Ok(Task::Quality(QualityAction::Render)),
303        [QUALITY_COMMAND, "verify"] => Ok(Task::Quality(QualityAction::Verify)),
304        [QUALITY_COMMAND, "evaluate", PROFILE_OPTION, profile] => parse_quality_profile(profile)
305            .map(QualityAction::Evaluate)
306            .map(Task::Quality),
307        [QUALITY_COMMAND, "approve", "--release", release] => {
308            Ok(Task::Quality(QualityAction::Approve((*release).to_owned())))
309        }
310        [VALIDATE_COMMAND, "binaries"] => Ok(Task::Validate(ValidationTarget::Binaries)),
311        [VALIDATE_COMMAND, artifact] => Ok(Task::Validate(ValidationTarget::Artifact(
312            PathBuf::from(artifact),
313        ))),
314        [PACKAGE_COMMAND] => Ok(Task::Package),
315        [RELEASE_COMMAND, "prepare"] => Ok(Task::ReleasePrepare),
316        [VERSION_COMMAND, "show"] => Ok(Task::Version(VersionAction::Show)),
317        [VERSION_COMMAND, "check"] => Ok(Task::Version(VersionAction::Check)),
318        [VERSION_COMMAND, "prepare", version] => {
319            Ok(Task::Version(VersionAction::Prepare((*version).to_owned())))
320        }
321        [PORTABLE_COMMAND, "verify"] => Ok(Task::Portable(PortableAction::Verify)),
322        [PORTABLE_COMMAND, "snapshot"] => Ok(Task::Portable(PortableAction::Snapshot)),
323        [PORTABLE_COMMAND, "install", source] => Ok(Task::Portable(PortableAction::Install(
324            PathBuf::from(source),
325        ))),
326        [CLEAN_COMMAND] => Ok(Task::Clean),
327        [CI_COMMAND, "pr"] => Ok(Task::Ci(CiProfile::Pr)),
328        [CI_COMMAND, "release"] => Ok(Task::Ci(CiProfile::Release)),
329        _ => Err(format!(
330            "unsupported command: {}; run `cargo xtask --help`",
331            values.join(" ")
332        )),
333    }
334}

336/// Parses one durable test-level name.
337fn parse_test_level(value: &str) -> Result<TestLevel, String> {
338    match value {
339        "unit" => Ok(TestLevel::Unit),
340        "smoke" => Ok(TestLevel::Smoke),
341        "integration" => Ok(TestLevel::Integration),
342        "system" => Ok(TestLevel::System),
343        "conformance" => Ok(TestLevel::Conformance),
344        "property" => Ok(TestLevel::Property),
345        "security" => Ok(TestLevel::Security),
346        "all" => Ok(TestLevel::All),
347        _ => Err(format!("unknown test level: {value}")),
348    }
349}

351/// Parses one controlled performance-profile name.
352fn parse_performance_profile(value: &str) -> Result<PerformanceProfile, String> {
353    match value {
354        "pr" => Ok(PerformanceProfile::Pr),
355        "release" => Ok(PerformanceProfile::Release),
356        "soak" => Ok(PerformanceProfile::Soak),
357        _ => Err(format!("unknown performance profile: {value}")),
358    }
359}

361/// Parses one fuzz-mode name.
362fn parse_fuzz_mode(value: &str) -> Result<FuzzMode, String> {
363    match value {
364        "smoke" => Ok(FuzzMode::Smoke),
365        "campaign" => Ok(FuzzMode::Campaign),
366        _ => Err(format!("unknown fuzz mode: {value}")),
367    }
368}

370/// Parses one aggregate quality-profile name.

372    match value {
373        "pr" => Ok(QualityProfile::Pr),
374        "release" => Ok(QualityProfile::Release),
375        _ => Err(format!("unknown quality profile: {value}")),
