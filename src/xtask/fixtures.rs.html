3//! Automated synchronization and verification of test fixtures, oracles, and freeze hashes.
4//!
5//! This module eliminates the manual burden of calculating SHA-256 digests across
6//! `portable/conformance/manifest.toml` and `portable/specs/contracts/freeze.toml`.
7
8use std::{
9    collections::BTreeSet,
10    fs,
11    path::{Path, PathBuf},
12};
13
14use sha2::{Digest, Sha256};
15
16/// Summary of synchronization operations performed across manifests and freeze contracts.
17#[derive(Default)]
18struct SyncSummary {
19    /// Number of fixture cases verified.
20    fixture_count: usize,
21    /// Number of oracle cases verified.
22    oracle_count: usize,
23    /// Number of digest changes made to the conformance manifest.
24    manifest_changes: usize,
25    /// Number of digest changes made to the contract freeze manifest.
26    freeze_changes: usize,
27    /// Mismatch errors encountered during check mode.
28    mismatch_errors: Vec<String>,
29}
30
31/// Synchronizes or verifies all test fixtures, oracles, and freeze hashes in the repository.
32pub(crate) fn sync_fixtures(workspace_root: &Path, check_only: bool) -> Result<(), String> {
33    let mode = if check_only {
34        "verifying"
35    } else {
36        "synchronizing"
37    };
38    println!("[info] fixtures: {mode} fixture hashes and contract freeze...");
39
40    let manifest_rel = "portable/conformance/manifest.toml";
41    let freeze_rel = "portable/specs/contracts/freeze.toml";
42
43    let manifest_path = workspace_root.join(manifest_rel);
44    let freeze_path = workspace_root.join(freeze_rel);
45
46    if !manifest_path.is_file() {
47        return Err(format!(
48            "manifest file not found: {}",
49            manifest_path.display()
50        ));
51    }
52    if !freeze_path.is_file() {
53        return Err(format!("freeze file not found: {}", freeze_path.display()));
54    }
55
56    let mut summary = SyncSummary::default();
57
58    let registered_fixtures =
59        sync_manifest(workspace_root, &manifest_path, check_only, &mut summary)?;
60    sync_freeze(workspace_root, &freeze_path, check_only, &mut summary)?;
61    discover_untracked_fixtures(workspace_root, &registered_fixtures)?;
62
63    if check_only && !summary.mismatch_errors.is_empty() {
64        return Err(format!(
65            "fixture verification failed with {} mismatch(es):\n{}",
66            summary.mismatch_errors.len(),
67            summary.mismatch_errors.join("\n\n")
68        ));
69    }
70
71    println!(
72        "[info] fixtures: verified {} fixtures, {} oracles. (manifest updates: {}, freeze updates: {})",
73        summary.fixture_count,
74        summary.oracle_count,
75        summary.manifest_changes,
76        summary.freeze_changes
77    );
78
79    Ok(())
80}

82/// Synchronizes the conformance manifest and returns the set of registered fixture paths.
83fn sync_manifest(
84    workspace_root: &Path,
85    manifest_path: &Path,
86    check_only: bool,
87    summary: &mut SyncSummary,
88) -> Result<BTreeSet<String>, String> {
89    let manifest_raw = fs::read_to_string(manifest_path)
90        .map_err(|e| format!("could not read {}: {e}", manifest_path.display()))?;
91
92    let mut registered_fixtures = BTreeSet::new();
93    let mut updated_lines = Vec::new();
94    let mut current_fixture: Option<String> = None;
95    let mut current_oracle: Option<String> = None;
96
97    for line in manifest_raw.lines() {
98        let trimmed = line.trim();
99
100        if trimmed == "[[case]]" || trimmed == "[[suite]]" {
101            current_fixture = None;
102            current_oracle = None;
103            updated_lines.push(line.to_string());
104            continue;
105        }
106
107        if let Some(path) = trimmed
108            .strip_prefix("fixture =")
109            .and_then(|s| extract_string_value(s.trim()))
110        {
111            registered_fixtures.insert(path.clone());
112            current_fixture = Some(path);
113            updated_lines.push(line.to_string());
114            continue;
115        }
116
117        if let Some(path) = trimmed
118            .strip_prefix("oracle =")
119            .and_then(|s| extract_string_value(s.trim()))
120        {
121            current_oracle = Some(path);
122            updated_lines.push(line.to_string());
123            continue;
124        }
125
126        if trimmed.starts_with("fixture_sha256 =")
127            && let Some(ref rel_path) = current_fixture
128        {
129            let new_line = process_digest_line(
130                workspace_root,
131                line,
132                trimmed,
133                rel_path,
134                "fixture",
135                check_only,
136                &mut summary.fixture_count,
137                &mut summary.manifest_changes,
138                &mut summary.mismatch_errors,
139            )?;
140            updated_lines.push(new_line);
141            continue;
142        }
143
144        if trimmed.starts_with("oracle_sha256 =")
145            && let Some(ref rel_path) = current_oracle
146        {
147            let new_line = process_digest_line(
148                workspace_root,
149                line,
150                trimmed,
151                rel_path,
152                "oracle",
153                check_only,
154                &mut summary.oracle_count,
155                &mut summary.manifest_changes,
156                &mut summary.mismatch_errors,
157            )?;
158            updated_lines.push(new_line);
159            continue;
160        }
161
162        updated_lines.push(line.to_string());
163    }
164
165    if !check_only && summary.manifest_changes > 0 {
166        let mut new_content = updated_lines.join("\n");
167        if manifest_raw.ends_with('\n') {
168            new_content.push('\n');
169        }
170        fs::write(manifest_path, new_content)
171            .map_err(|e| format!("could not write {}: {e}", manifest_path.display()))?;
172        println!(
173            "[info] fixtures: updated {} digests in {}",
174            summary.manifest_changes,
175            manifest_path.display()
176        );
177    }
178
179    Ok(registered_fixtures)
180}

182/// Processes a single digest line, computing its hash and checking for mismatches.
183#[allow(clippy::too_many_arguments)]
184fn process_digest_line(
185    workspace_root: &Path,
186    original_line: &str,
187    trimmed: &str,
188    rel_path: &str,
189    kind: &str,
190    check_only: bool,
191    counter: &mut usize,
192    change_counter: &mut usize,
193    mismatches: &mut Vec<String>,
194) -> Result<String, String> {
195    let full_path = workspace_root.join(rel_path);
196    let actual_digest = hash_file(&full_path)?;
197    let existing_digest = extract_string_value(trimmed).unwrap_or_default();
198    *counter += 1;
199
200    if actual_digest != existing_digest {
201        *change_counter += 1;
202        if check_only {
203            mismatches.push(format!(
204                "{kind} `{rel_path}` digest mismatch:\n  expected: {existing_digest}\n  actual:   {actual_digest}"
205            ));
206        }
207    }
208
209    let key = if kind == "fixture" {
210        "fixture_sha256"
211    } else {
212        "oracle_sha256"
213    };
214    let indent = original_line
215        .find(key.chars().next().unwrap_or(' '))
216        .unwrap_or(0);
217    Ok(format!("{:indent$}{key} = \"{actual_digest}\"", ""))
218}

220/// Synchronizes the contract freeze manifest digests.
221fn sync_freeze(
222    workspace_root: &Path,
223    freeze_path: &Path,
224    check_only: bool,
225    summary: &mut SyncSummary,
226) -> Result<(), String> {
227    let freeze_raw = fs::read_to_string(freeze_path)
228        .map_err(|e| format!("could not read {}: {e}", freeze_path.display()))?;
229
230    let mut updated_lines = Vec::new();
231    let mut last_path_entry: Option<(String, String)> = None;
232
233    for line in freeze_raw.lines() {
234        let trimmed = line.trim();
235
236        if trimmed.is_empty() || trimmed.starts_with('#') || trimmed.starts_with('[') {
237            if trimmed.starts_with('[') {
238                last_path_entry = None;
239            }
240            updated_lines.push(line.to_string());
241            continue;
242        }
243
244        if let Some((key, val)) = trimmed.split_once('=') {
245            let key = key.trim();
246            let val = val.trim().trim_matches('"');
247
248            if key.ends_with("_path") {
249                let prefix = key.strip_suffix("_path").unwrap_or(key).to_string();
250                last_path_entry = Some((prefix, val.to_string()));
251                updated_lines.push(line.to_string());
252                continue;
253            }
254
255            if key.ends_with("_sha256")
256                && let Some((ref last_prefix, ref rel_path)) = last_path_entry
257                && last_prefix == key.strip_suffix("_sha256").unwrap_or(key)
258            {
259                let full_path = workspace_root.join(rel_path);
260                let actual_digest = hash_file(&full_path)?;
261
262                if actual_digest != val {
263                    summary.freeze_changes += 1;
264                    if check_only {
265                        summary.mismatch_errors.push(format!(
266                            "freeze entry `{key}` ({rel_path}) mismatch:\n  expected: {val}\n  actual:   {actual_digest}"
267                        ));
268                    }
269                }
270
271                let indent = line.find(key.chars().next().unwrap_or(' ')).unwrap_or(0);
272                updated_lines.push(format!("{:indent$}{key} = \"{actual_digest}\"", ""));
273                continue;
274            }
275        }
276
277        updated_lines.push(line.to_string());
278    }
279
280    if !check_only && summary.freeze_changes > 0 {
281        let mut new_content = updated_lines.join("\n");
282        if freeze_raw.ends_with('\n') {
283            new_content.push('\n');
284        }
285        fs::write(freeze_path, new_content)
286            .map_err(|e| format!("could not write {}: {e}", freeze_path.display()))?;
287        println!(
288            "[info] fixtures: updated {} contract hashes in {}",
289            summary.freeze_changes,
290            freeze_path.display()
291        );
292    }
293
294    Ok(())
295}

297/// Discovers fixture files on disk that are not registered in the manifest.
298fn discover_untracked_fixtures(
299    workspace_root: &Path,
300    registered: &BTreeSet<String>,
301) -> Result<(), String> {
302    let fixtures_dir = workspace_root.join("portable/specs/fixtures");
303    if !fixtures_dir.is_dir() {
304        return Ok(());
305    }
306
307    let mut on_disk = Vec::new();
308    collect_fixture_files(&fixtures_dir, &mut on_disk)?;
309
310    for path in on_disk {
311        if let Ok(rel) = path.strip_prefix(workspace_root) {
312            let rel_str = rel.to_string_lossy().replace('\\', "/");
313            if !registered.contains(&rel_str) {
314                println!("[warn] fixtures: file on disk not yet in manifest: {rel_str}");
315            }
316        }
317    }
318
319    Ok(())
320}

322/// Recursively collects fixture test files (.neu, .toml).
323fn collect_fixture_files(dir: &Path, files: &mut Vec<PathBuf>) -> Result<(), String> {
324    if !dir.is_dir() {
325        return Ok(());
326    }
327    let entries =
328        fs::read_dir(dir).map_err(|e| format!("could not read {}: {e}", dir.display()))?;
329    for entry in entries.flatten() {
330        let path = entry.path();
331        if path.is_dir() {
332            collect_fixture_files(&path, files)?;
333        } else if let Some(ext) = path.extension().and_then(|s| s.to_str())
334            && (ext == "neu" || ext == "toml")
335            && path.file_name().and_then(|s| s.to_str()) != Some("README.md")
336        {
337            files.push(path);
338        }
339    }
340    Ok(())
341}

343/// Extracts the quoted string value from a line or token.
344fn extract_string_value(token: &str) -> Option<String> {
345    let raw = token.split_once('=').map_or(token, |(_, v)| v).trim();
346    raw.strip_prefix('"')?.strip_suffix('"').map(str::to_owned)
347}

349/// Returns the lowercase hex SHA-256 digest of a file.
350fn hash_file(path: &Path) -> Result<String, String> {
351    let bytes = fs::read(path).map_err(|e| format!("could not read {}: {e}", path.display()))?;
352    let mut hasher = Sha256::new();
353    hasher.update(&bytes);
354    Ok(format!("{:x}", hasher.finalize()))
355}
