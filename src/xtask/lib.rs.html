98/// Runs an `xtask` subcommand.
99///
100/// # Errors
101///
102/// Returns one contextual error when parsing or any selected task fails.
103pub fn run(arguments: impl IntoIterator<Item = String>) -> Result<(), String> {
104    let arguments = arguments.into_iter().collect::<Vec<_>>();
105    let label = arguments.join(" ");
106    let started = Instant::now();
107    output::start(format!("xtask {label}"));
108    let result = interface::parse(&arguments).and_then(execute);
109    let duration = output::duration(started.elapsed());
110    if result.is_ok() {
111        output::pass(format!("xtask {label} ({duration})"));
112    }
113    result.map_err(|error| format!("xtask {label} ({duration}): {error}"))
114}

116/// Executes one parsed stable automation task.
117fn execute(task: Task) -> Result<(), String> {
118    match task {
119        Task::Help => {
120            output::help(interface::HELP);
121            Ok(())
122        }
123        Task::Bootstrap => bootstrap(),
124        Task::Dev => develop(),
125        Task::EnvironmentVerify => verify_complete_environment(),
126        Task::EnvironmentManifest => print_environment_manifest(),
127        Task::Format { write } => format_workspace(write),
128        Task::Lint => lint(),
129        Task::Check => check(),
130        Task::Build(profile) => build(profile),
131        Task::Docs => documentation(),
132        Task::Test(level) => test_suite(level),
133        Task::Performance(profile) => performance(profile),
134        Task::Fuzz(mode) => fuzz(mode),
135        Task::Coverage => coverage(),
136        Task::Mutate => mutate(),
137        Task::Quality(action) => quality_action(action),
138        Task::Validate(target) => validate(target),
139        Task::Package => package(),
140        Task::ReleasePrepare(version) => release::lifecycle::prepare(version.as_deref()),
141        Task::ReleasePublish => release::lifecycle::publish(),
142        Task::ReleaseQualify => release_qualify(),
143        Task::ReleaseTag => output::machine(release_plan()?.release_tag),
144        Task::Version(action) => version(action),
145        Task::Portable(action) => portable(action),
146        Task::Clean => clean_results(),
147        Task::Ci(profile) => ci(profile),
148        Task::Fixtures { check } => fixtures::sync_fixtures(&workspace_root()?, check),
149    }
150}

152/// Finds the owning Cargo workspace even if the automation crate moves.
153fn workspace_root() -> Result<PathBuf, String> {
154    let current = env::current_dir()
155        .map_err(|error| format!("could not locate the current directory: {error}"))?;
156    workspace::discover(&current)
157}
