8/// Executes the stable version command family.
9pub(super) fn version(action: VersionAction) -> Result<(), String> {
10    match action {
11        VersionAction::Show => show_versions(),
12        VersionAction::Check => check_versions(),
13        VersionAction::Prepare(requested) => prepare_version(&requested),
14    }
15}

17/// Displays the package version separately from every frozen contract version.
18pub(super) fn show_versions() -> Result<(), String> {
19    let root = workspace_root()?;
20    let package = workspace_package_version(&read_workspace_text(
21        &root,
22        constants::WORKSPACE_MANIFEST_FILE,
23    )?)?;
24    crate::output::info(format!("package-release {package}"));
25    let freeze = read_workspace_text(
26        &root,
27        &ReleasedBundle::load(&root)?.member(crate::constants::BUNDLE_CONTRACT_FREEZE_MEMBER),
28    )?;
29    for (name, value) in configuration_section(&freeze, "contract_versions")? {
30        crate::output::info(format!("contract {name}={value}"));
31    }
32    Ok(())
33}

35/// Checks that package versions inherit the one workspace release version.
36pub(super) fn check_versions() -> Result<(), String> {
37    let root = workspace_root()?;
38    let package_version = workspace_package_version(&read_workspace_text(
39        &root,
40        constants::WORKSPACE_MANIFEST_FILE,
41    )?)?;
42    for manifest in workspace_package_manifests(&root)? {
43        let content = fs::read_to_string(&manifest)
44            .map_err(|error| format!("could not read {}: {error}", manifest.display()))?;
45        let document: toml::Table =
46            configuration::parse(&content, &manifest.display().to_string())?;
47        for inherited in ["version", "license", "repository"] {
48            if document
49                .get("package")
50                .and_then(|package| package.get(inherited))
51                .and_then(|value| value.get("workspace"))
52                .and_then(toml::Value::as_bool)
53                != Some(true)
54            {
55                return Err(format!(
56                    "workspace package must inherit {inherited}.workspace: {}",
57                    manifest.display()
58                ));
59            }
60        }
61    }
62    verify_dependency_lock(&root, &package_version)?;
63    let freeze = read_workspace_text(
64        &root,
65        &ReleasedBundle::load(&root)?.member(crate::constants::BUNDLE_CONTRACT_FREEZE_MEMBER),
66    )?;
67    if configuration_value(&freeze, "status").as_deref() != Some("approved")
68        || configuration_section(&freeze, "contract_versions")?.is_empty()
69    {
70        return Err("contract freeze must remain approved and version-complete".to_owned());
71    }
72    crate::output::pass("centralized package versions");
73    Ok(())
74}

76/// Rejects ordinary version work mixed with unreviewed normative changes.
77pub(super) fn ensure_no_unreviewed_contract_changes(root: &Path) -> Result<(), String> {
78    let bundle = ReleasedBundle::load(root)?;
79    let freeze = bundle.member(crate::constants::BUNDLE_CONTRACT_FREEZE_MEMBER);
80    let specs = bundle.member("specs");
81    let manifest = bundle.member("conformance/manifest.toml");
82    let oracles = bundle.member("conformance/oracles");
83    let review = bundle.member("conformance/fixture-oracle-review.toml");
84    let output = Command::new("git")
85        .current_dir(root)
86        .args([
87            "diff",
88            "--name-only",
89            "HEAD",
90            "--",
91            constants::CONFORMANCE_CONFIG_FILE,
92            &freeze,
93            &specs,
94            &manifest,
95            &oracles,
96            &review,
97            "config/ir-encoding.toml",
98        ])
99        .output()
100        .map_err(|error| format!("could not inspect normative changes: {error}"))?;
101    if !output.status.success() {
102        return Err(format!(
103            "could not inspect normative changes: {}",
104            String::from_utf8_lossy(&output.stderr).trim()
105        ));
106    }
107    let changed = String::from_utf8(output.stdout)
108        .map_err(|error| format!("Git emitted non-UTF-8 paths: {error}"))?;
109    if changed.trim().is_empty() {
110        Ok(())
111    } else {
112        Err(format!(
113            "ordinary package-version work includes normative changes requiring contract-freeze review: {}",
114            changed.lines().collect::<Vec<_>>().join(", ")
115        ))
116    }
117}

119/// Propagates one reviewed package-release version without tagging or publishing.
120pub(super) fn prepare_version(requested: &str) -> Result<(), String> {
121    validate_semver(requested)?;
122    let root = workspace_root()?;
123    let manifest_path = root.join(constants::WORKSPACE_MANIFEST_FILE);
124    let manifest = read_workspace_text(&root, constants::WORKSPACE_MANIFEST_FILE)?;
125    let current = workspace_package_version(&manifest)?;
126    validate_version_transition(&current, requested)?;
127    ensure_no_unreviewed_contract_changes(&root)?;
128    check_versions()?;
129    require_clean_checkout()?;
130
131    let updated_manifest = replace_workspace_package_version(&manifest, &current, requested)?;
132    let lock_path = root.join(constants::CARGO_LOCK_FILE);
133    let lock = read_workspace_text(&root, constants::CARGO_LOCK_FILE)?;
134    let package_names = workspace_package_names(&root)?;
135    let updated_lock = replace_workspace_lock_versions(&lock, &package_names, &current, requested)?;
136    let evidence_directory = root
137        .join(constants::QUALITY_EVIDENCE_DIRECTORY)
138        .join(format!("v{requested}"));
139    let evidence_readme = evidence_directory.join("README.md");
140    if evidence_readme.exists() {
141        return Err(format!(
142            "release evidence is already prepared: {}",
143            evidence_readme.display()
144        ));
145    }
146
147    let freeze_bytes = fs::read(root.join(
148        ReleasedBundle::load(&root)?.member(crate::constants::BUNDLE_CONTRACT_FREEZE_MEMBER),
149    ))
150    .map_err(|error| format!("could not read contract freeze: {error}"))?;
151    let freeze_digest = sha256_hex(&freeze_bytes);
152
153    fs::create_dir_all(&evidence_directory).map_err(|error| {
154        format!(
155            "could not create release evidence directory {}: {error}",
156            evidence_directory.display()
157        )
158    })?;
159    fs::write(&manifest_path, updated_manifest)
160        .map_err(|error| format!("could not update {}: {error}", manifest_path.display()))?;
161    fs::write(&lock_path, updated_lock)
162        .map_err(|error| format!("could not update {}: {error}", lock_path.display()))?;
163    fs::write(
164        &evidence_readme,
165        release_evidence_readme(requested, &project_license(&root)?),
166    )
167    .map_err(|error| format!("could not write {}: {error}", evidence_readme.display()))?;
168
169    let directory = result_root()?.join(constants::VERSION_RESULT_DIRECTORY);
170    fs::create_dir_all(&directory)
171        .map_err(|error| format!("could not create {}: {error}", directory.display()))?;
172    let output = directory.join(format!("prepare-{requested}.json"));
173    fs::write(
174        &output,
175        format!(
176            "{{\n  \"schema_version\": 1,\n  \"current\": \"{}\",\n  \"requested\": \"{}\",\n  \"derived_updates\": [],\n  \"contract_freeze_sha256\": \"{}\",\n  \"frozen_contracts_changed\": false,\n  \"actions\": [\n    \"edit-workspace-package-version\",\n    \"run-version-check\",\n    \"review\"\n  ],\n  \"status\": \"review-required\"\n}}\n",
177            json_string(&current),
178            json_string(requested),
179            freeze_digest
180        ),
181    )
182    .map_err(|error| format!("could not write {}: {error}", output.display()))?;
183    crate::output::pass(format!(
184        "package release prepared: {current} -> {requested}"
185    ));
186    crate::output::info(format!(
187        "updated: {}, {}, {}",
188        constants::WORKSPACE_MANIFEST_FILE,
189        constants::CARGO_LOCK_FILE,
190        evidence_readme
191            .strip_prefix(&root)
192            .unwrap_or(&evidence_readme)
193            .display()
194    ));
195    crate::output::file("version plan", &output);
196    Ok(())
197}

226/// Checks the root release lock and its declared dependency-source policy.
227pub(super) fn verify_dependency_lock(root: &Path, package_version: &str) -> Result<(), String> {
228    let policy = read_workspace_text(root, constants::DEPENDENCY_SOURCES_FILE)?;
229    if configuration_value(&policy, "lockfile").as_deref() != Some(constants::CARGO_LOCK_FILE)
230        || configuration_value(&policy, "allow_crates_io_registry").as_deref() != Some("true")
231        || configuration_value(&policy, "allow_git_sources").as_deref() != Some("false")
232        || configuration_value(&policy, "allow_external_paths").as_deref() != Some("false")
233    {
234        return Err("dependency-source policy is incomplete or unsafe".to_owned());
235    }
236    let review_path = configuration_value(&policy, "review")
237        .ok_or_else(|| "dependency-source policy has no review path".to_owned())?;
238    if !is_safe_relative_path(Path::new(&review_path)) {
239        return Err("dependency-source review path must stay within the workspace".to_owned());
240    }
241    read_workspace_text(root, &review_path)?;
242    if !matches!(
243        quality_document_status(root, &review_path)?.as_str(),
244        "approved" | "approved-with-limitation"
245    ) {
246        return Err("dependency review is not approved in the quality manifest".to_owned());
247    }
248    let lock = read_workspace_text(root, constants::CARGO_LOCK_FILE)?;
249    let lock_document: toml::Table =
250        configuration::parse(&lock, crate::constants::CARGO_LOCK_FILE)?;
251    let locked_packages = lock_document
252        .get("package")
253        .and_then(toml::Value::as_array)
254        .ok_or("Cargo.lock has no package records")?;
255    for package in locked_packages {
256        let source = package.get("source").and_then(toml::Value::as_str);
257        if source.is_some_and(|source| !source.starts_with("registry+")) {
258            return Err("Cargo.lock contains a forbidden non-registry dependency".to_owned());
259        }
260        if source.is_some()
261            && package
262                .get("checksum")
263                .and_then(toml::Value::as_str)
264                .is_none()
265        {
266            return Err("Cargo.lock contains a registry package without a checksum".to_owned());
267        }
268    }
269    verify_manifest_dependency_paths(root, &root.join(constants::WORKSPACE_MANIFEST_FILE))?;
270    for manifest in workspace_package_manifests(root)? {
271        verify_manifest_dependency_paths(root, &manifest)?;
272        let package_content = fs::read_to_string(&manifest)
273            .map_err(|error| format!("could not read {}: {error}", manifest.display()))?;
274        let package_name = configuration::quality_value_from(&package_content, "package", "name")
275            .ok_or_else(|| {
276            format!(
277                "package manifest has no package name: {}",
278                manifest.display()
279            )
280        })?;
281        if !locked_packages.iter().any(|package| {
282            package.get("name").and_then(toml::Value::as_str) == Some(&package_name)
283                && package.get("version").and_then(toml::Value::as_str) == Some(package_version)
284                && package.get("source").is_none()
285        }) {
286            return Err(format!(
287                "Cargo.lock is stale for workspace package {package_name} {package_version}"
288            ));
289        }
290    }
291    let output = Command::new(cargo_command()?)
292        .current_dir(root)
293        .args([
294            "metadata",
295            flags::LOCKED,
296            "--offline",
297            "--format-version",
298            "1",
299            "--no-deps",
300        ])
301        .output()
302        .map_err(|error| format!("could not validate Cargo.lock: {error}"))?;
303    if !output.status.success() {
304        return Err(format!(
305            "Cargo.lock is stale or unavailable offline: {}",
306            String::from_utf8_lossy(&output.stderr).trim()
307        ));
308    }
309    let stderr = String::from_utf8_lossy(&output.stderr);
310    if stderr.lines().any(|line| line.contains("warning:")) {
311        return Err(format!(
312            "Cargo metadata emitted a release-relevant warning: {}",
313            stderr.trim()
314        ));
315    }
316    Ok(())
317}

319/// Requires every manifest path dependency to resolve inside the workspace.
320pub(super) fn verify_manifest_dependency_paths(root: &Path, manifest: &Path) -> Result<(), String> {
321    let content = fs::read_to_string(manifest)
322        .map_err(|error| format!("could not read {}: {error}", manifest.display()))?;
323    let document: toml::Table = configuration::parse(&content, &manifest.display().to_string())?;
324    let canonical_root = fs::canonicalize(root)
325        .map_err(|error| format!("could not canonicalize workspace root: {error}"))?;
326    verify_dependency_sections(&canonical_root, manifest, &document)?;
327    if let Some(workspace) = document.get("workspace").and_then(toml::Value::as_table) {
328        verify_dependency_sections(&canonical_root, manifest, workspace)?;
329    }
330    if let Some(targets) = document.get("target").and_then(toml::Value::as_table) {
331        for target in targets.values() {
332            let table = target.as_table().ok_or("manifest target must be a table")?;
333            verify_dependency_sections(&canonical_root, manifest, table)?;
334        }
335    }
336    Ok(())
337}

339/// Checks actual dependency tables, ignoring incidental comments and unrelated path metadata.
340fn verify_dependency_sections(
341    root: &Path,
342    manifest: &Path,
343    table: &toml::Table,
344) -> Result<(), String> {
345    for kind in ["dependencies", "build-dependencies", "dev-dependencies"] {
346        let Some(dependencies) = table.get(kind) else {
347            continue;
348        };
349        let dependencies = dependencies
350            .as_table()
351            .ok_or_else(|| format!("manifest {kind} must be a table"))?;
352        for dependency in dependencies.values() {
353            let Some(path) = dependency.as_table().and_then(|table| table.get("path")) else {
354                continue;
355            };
356            let path = path.as_str().ok_or("dependency path must be a string")?;
357            let path = manifest.parent().unwrap_or(root).join(path);
358            let canonical = fs::canonicalize(&path).map_err(|error| {
359                format!(
360                    "could not resolve path dependency {}: {error}",
361                    path.display()
362                )
363            })?;
364            if !canonical.starts_with(root) {
365                return Err(format!(
366                    "manifest path dependency escapes the workspace: {}",
367                    path.display()
368                ));
369            }
370        }
371    }
372    Ok(())
373}

411/// Derives a safe distribution name from the workspace repository URL.
412pub(super) fn project_slug(root: &Path) -> Result<String, String> {
413    let manifest = read_workspace_text(root, constants::WORKSPACE_MANIFEST_FILE)?;
414    let repository = workspace_package_value(&manifest, "repository")?;
415    let slug = repository
416        .trim_end_matches('/')
417        .rsplit('/')
418        .next()
419        .unwrap_or_default()
420        .trim_end_matches(".git");
421    if slug.is_empty()
422        || !slug
423            .bytes()
424            .all(|byte| byte.is_ascii_alphanumeric() || byte == b'-' || byte == b'_')
425    {
426        return Err("workspace repository has no safe distribution name".to_owned());
427    }
428    Ok(slug.to_owned())
429}

466/// Collects regular files with one exact filename below a directory.
467pub(super) fn collect_named_files(
468    directory: &Path,
469    name: &str,
470    files: &mut Vec<PathBuf>,
471) -> Result<(), String> {
472    for entry in fs::read_dir(directory)
473        .map_err(|error| format!("could not inspect {}: {error}", directory.display()))?
474    {
475        let entry = entry.map_err(|error| format!("could not inspect directory entry: {error}"))?;
476        let path = entry.path();
477        if path.is_dir() {
478            collect_named_files(&path, name, files)?;
479        } else if path.file_name().and_then(std::ffi::OsStr::to_str) == Some(name) {
480            files.push(path);
481        }
482    }
483    Ok(())

486/// Validates the supported numeric `SemVer` core with an optional prerelease suffix.
487pub(super) fn validate_semver(value: &str) -> Result<(), String> {
488    let (core, suffix) = value
489        .split_once('-')
490        .map_or((value, None), |(core, suffix)| (core, Some(suffix)));
491    let components = core.split('.').collect::<Vec<_>>();
492    let numeric = components.len() == 3
493        && components.iter().all(|component| {
494            !component.is_empty()
495                && component.bytes().all(|byte| byte.is_ascii_digit())
496                && (component == &"0" || !component.starts_with('0'))
497        });
498    let valid_suffix = suffix.is_none_or(|suffix| {
499        !suffix.is_empty()
500            && suffix.split('.').all(|identifier| {
501                !identifier.is_empty()
502                    && identifier
503                        .bytes()
504                        .all(|byte| byte.is_ascii_alphanumeric() || byte == b'-')
505                    && (!identifier.bytes().all(|byte| byte.is_ascii_digit())
506                        || identifier == "0"
507                        || !identifier.starts_with('0'))
508            })
509    });
510    if numeric && valid_suffix {
511        Ok(())
512    } else {
513        Err(format!("invalid package SemVer: {value}"))
514    }
515}

517/// Rejects package-version downgrades and invalid prerelease transitions.
518pub(super) fn validate_version_transition(current: &str, requested: &str) -> Result<(), String> {
519    let current = parsed_semver(current)?;
520    let requested = parsed_semver(requested)?;
521    if compare_semver(&requested, &current) != std::cmp::Ordering::Greater {
522        return Err(
523            "requested package version must be greater than the current version".to_owned(),
524        );
525    }
526    Ok(())
527}

529/// Parses supported `SemVer` into numeric core and prerelease identifiers.
530pub(super) fn parsed_semver(value: &str) -> Result<(u64, u64, u64, Vec<String>), String> {
531    validate_semver(value)?;
532    let (core, prerelease) = value
533        .split_once('-')
534        .map_or((value, ""), |(core, prerelease)| (core, prerelease));
535    let mut numbers = core.split('.').map(|value| {
536        value
537            .parse::<u64>()
538            .map_err(|error| format!("invalid package SemVer component: {error}"))
539    });
540    let parsed = (
541        numbers.next().transpose()?.unwrap_or_default(),
542        numbers.next().transpose()?.unwrap_or_default(),
543        numbers.next().transpose()?.unwrap_or_default(),
544        prerelease
545            .split('.')
546            .filter(|value| !value.is_empty())
547            .map(str::to_owned)
548            .collect(),
549    );
550    Ok(parsed)
551}

553/// Compares two parsed `SemVer` values using numeric prerelease precedence.
554pub(super) fn compare_semver(
555    left: &(u64, u64, u64, Vec<String>),
556    right: &(u64, u64, u64, Vec<String>),
557) -> std::cmp::Ordering {
558    let core = (left.0, left.1, left.2).cmp(&(right.0, right.1, right.2));
559    if core != std::cmp::Ordering::Equal {
560        return core;
561    }
562    match (left.3.is_empty(), right.3.is_empty()) {
563        (true, false) => std::cmp::Ordering::Greater,
564        (false, true) => std::cmp::Ordering::Less,
565        (true, true) => std::cmp::Ordering::Equal,
566        (false, false) => compare_prerelease(&left.3, &right.3),
567    }
568}

570/// Compares `SemVer` prerelease identifier sequences.
571pub(super) fn compare_prerelease(left: &[String], right: &[String]) -> std::cmp::Ordering {
572    for (left, right) in left.iter().zip(right) {
573        let ordering = match (left.parse::<u64>(), right.parse::<u64>()) {
574            (Ok(left), Ok(right)) => left.cmp(&right),
575            (Ok(_), Err(_)) => std::cmp::Ordering::Less,
576            (Err(_), Ok(_)) => std::cmp::Ordering::Greater,
577            (Err(_), Err(_)) => left.cmp(right),
578        };
579        if ordering != std::cmp::Ordering::Equal {
580            return ordering;
581        }
582    }
583    left.len().cmp(&right.len())
584}

586/// Returns the lowercase SHA-256 digest of exact bytes.
587pub(super) fn sha256_hex(bytes: &[u8]) -> String {
588    let digest = Sha256::digest(bytes);
589    let mut output = String::with_capacity(digest.len() * 2);
590    for byte in digest {
591        write!(&mut output, "{byte:02x}").expect("writing to a String cannot fail");
592    }
593    output
594}
