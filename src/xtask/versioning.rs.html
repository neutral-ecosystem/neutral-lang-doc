8/// Executes the stable version command family.
9pub(super) fn version(action: VersionAction) -> Result<(), String> {
10    match action {
11        VersionAction::Show => show_versions(),
12        VersionAction::Check => check_versions(),
13        VersionAction::Prepare(requested) => prepare_version(&requested),
14    }
15}

17/// Displays the package version separately from every frozen contract version.
18pub(super) fn show_versions() -> Result<(), String> {
19    let root = workspace_root()?;
20    let package = workspace_package_version(&read_workspace_text(
21        &root,
22        constants::WORKSPACE_MANIFEST_FILE,
23    )?)?;
24    crate::output::info(format!("package-release {package}"));
25    let freeze = read_workspace_text(
26        &root,
27        &ReleasedBundle::load(&root)?.member(crate::constants::BUNDLE_CONTRACT_FREEZE_MEMBER),
28    )?;
29    for (name, value) in configuration_section(&freeze, "contract_versions")? {
30        crate::output::info(format!("contract {name}={value}"));
31    }
32    Ok(())
33}

35/// Checks that package versions inherit the one workspace release version.
36pub(super) fn check_versions() -> Result<(), String> {
37    let root = workspace_root()?;
38    let package_version = workspace_package_version(&read_workspace_text(
39        &root,
40        constants::WORKSPACE_MANIFEST_FILE,
41    )?)?;
42    for manifest in workspace_package_manifests(&root)? {
43        let content = fs::read_to_string(&manifest)
44            .map_err(|error| format!("could not read {}: {error}", manifest.display()))?;
45        let document: toml::Table =
46            configuration::parse(&content, &manifest.display().to_string())?;
47        for inherited in ["version", "license", "repository"] {
48            if document
49                .get("package")
50                .and_then(|package| package.get(inherited))
51                .and_then(|value| value.get("workspace"))
52                .and_then(toml::Value::as_bool)
53                != Some(true)
54            {
55                return Err(format!(
56                    "workspace package must inherit {inherited}.workspace: {}",
57                    manifest.display()
58                ));
59            }
60        }
61    }
62    verify_dependency_lock(&root, &package_version)?;
63    let freeze = read_workspace_text(
64        &root,
65        &ReleasedBundle::load(&root)?.member(crate::constants::BUNDLE_CONTRACT_FREEZE_MEMBER),
66    )?;
67    if configuration_value(&freeze, "status").as_deref() != Some("approved")
68        || configuration_section(&freeze, "contract_versions")?.is_empty()
69    {
70        return Err("contract freeze must remain approved and version-complete".to_owned());
71    }
72    crate::output::pass("centralized package versions");
73    Ok(())
74}

76/// Rejects ordinary version work mixed with unreviewed normative changes.
77pub(super) fn ensure_no_unreviewed_contract_changes(root: &Path) -> Result<(), String> {
78    let bundle = ReleasedBundle::load(root)?;
79    let freeze = bundle.member(crate::constants::BUNDLE_CONTRACT_FREEZE_MEMBER);
80    let specs = bundle.member("specs");
81    let manifest = bundle.member("conformance/manifest.toml");
82    let oracles = bundle.member("conformance/oracles");
83    let review = bundle.member("conformance/fixture-oracle-review.toml");
84    let output = Command::new(constants::GIT_COMMAND)
85        .current_dir(root)
86        .args([
87            "diff",
88            "--name-only",
89            "HEAD",
90            "--",
91            constants::CONFORMANCE_CONFIG_FILE,
92            &freeze,
93            &specs,
94            &manifest,
95            &oracles,
96            &review,
97            "config/ir-encoding.toml",
98        ])
99        .output()
100        .map_err(|error| format!("could not inspect normative changes: {error}"))?;
101    if !output.status.success() {
102        return Err(format!(
103            "could not inspect normative changes: {}",
104            String::from_utf8_lossy(&output.stderr).trim()
105        ));
106    }
107    let changed = String::from_utf8(output.stdout)
108        .map_err(|error| format!("Git emitted non-UTF-8 paths: {error}"))?;
109    if changed.trim().is_empty() {
110        Ok(())
111    } else {
112        Err(format!(
113            "ordinary package-version work includes normative changes requiring contract-freeze review: {}",
114            changed.lines().collect::<Vec<_>>().join(", ")
115        ))
116    }
117}

119/// Propagates one reviewed package-release version without tagging or publishing.
120pub(super) fn prepare_version(requested: &str) -> Result<(), String> {
121    validate_semver(requested)?;
122    let root = workspace_root()?;
123    if crate::release::version_update::recover(&root)?.is_some() {
124        let current = workspace_package_version(&read_workspace_text(
125            &root,
126            constants::WORKSPACE_MANIFEST_FILE,
127        )?)?;
128        if current == requested {
129            crate::output::pass(format!(
130                "package release {requested} metadata already prepared"
131            ));
132            return Ok(());
133        }
134        require_clean_checkout()?;
135        crate::release::version_update::acknowledge(&root)?;
136    }
137    let manifest = read_workspace_text(&root, constants::WORKSPACE_MANIFEST_FILE)?;
138    let current = workspace_package_version(&manifest)?;
139    validate_version_transition(&current, requested)?;
140    ensure_no_unreviewed_contract_changes(&root)?;
141    check_versions()?;
142    require_clean_checkout()?;
143
144    let updated_manifest = replace_workspace_package_version(&manifest, &current, requested)?;
145    let lock = read_workspace_text(&root, constants::CARGO_LOCK_FILE)?;
146    let package_names = workspace_package_names(&root)?;
147    let updated_lock = replace_workspace_lock_versions(&lock, &package_names, &current, requested)?;
148    let evidence_directory = root
149        .join(constants::QUALITY_EVIDENCE_DIRECTORY)
150        .join(format!("v{requested}"));
151    let evidence_readme = evidence_directory.join("README.md");
152    if evidence_readme.exists() {
153        return Err(format!(
154            "release evidence is already prepared: {}",
155            evidence_readme.display()
156        ));
157    }
158
159    let freeze_bytes = fs::read(root.join(
160        ReleasedBundle::load(&root)?.member(crate::constants::BUNDLE_CONTRACT_FREEZE_MEMBER),
161    ))
162    .map_err(|error| format!("could not read contract freeze: {error}"))?;
163    let freeze_digest = sha256_hex(&freeze_bytes);
164
165    let changes = vec![
166        (
167            constants::WORKSPACE_MANIFEST_FILE.to_owned(),
168            updated_manifest.into_bytes(),
169        ),
170        (
171            constants::CARGO_LOCK_FILE.to_owned(),
172            updated_lock.into_bytes(),
173        ),
174        (
175            format!(
176                "{}/v{requested}/README.md",
177                constants::QUALITY_EVIDENCE_DIRECTORY
178            ),
179            release_evidence_readme(requested, &project_license(&root)?).into_bytes(),
180        ),
181    ];
182
183    let directory = result_root()?.join(constants::VERSION_RESULT_DIRECTORY);
184    fs::create_dir_all(&directory)
185        .map_err(|error| format!("could not create {}: {error}", directory.display()))?;
186    let output = directory.join(format!("prepare-{requested}.json"));
187    fs::write(
188        &output,
189        format!(
190            "{{\n  \"schema_version\": 1,\n  \"current\": \"{}\",\n  \"requested\": \"{}\",\n  \"derived_updates\": [],\n  \"contract_freeze_sha256\": \"{}\",\n  \"frozen_contracts_changed\": false,\n  \"actions\": [\n    \"edit-workspace-package-version\",\n    \"run-version-check\",\n    \"review\"\n  ],\n  \"status\": \"review-required\"\n}}\n",
191            json_string(&current),
192            json_string(requested),
193            freeze_digest
194        ),
195    )
196    .map_err(|error| format!("could not write {}: {error}", output.display()))?;
197    crate::release::version_update::apply(&root, &changes)?;
198    crate::output::pass(format!(
199        "package release prepared: {current} -> {requested}"
200    ));
201    crate::output::info(format!(
202        "updated: {}, {}, {}",
203        constants::WORKSPACE_MANIFEST_FILE,
204        constants::CARGO_LOCK_FILE,
205        evidence_readme
206            .strip_prefix(&root)
207            .unwrap_or(&evidence_readme)
208            .display()
209    ));
210    crate::output::file("version plan", &output);
211    Ok(())
212}

241/// Checks the root release lock and its declared dependency-source policy.
242pub(super) fn verify_dependency_lock(root: &Path, package_version: &str) -> Result<(), String> {
243    let policy = read_workspace_text(root, constants::DEPENDENCY_SOURCES_FILE)?;
244    if configuration_value(&policy, "lockfile").as_deref() != Some(constants::CARGO_LOCK_FILE)
245        || configuration_value(&policy, "allow_crates_io_registry").as_deref() != Some("true")
246        || configuration_value(&policy, "allow_git_sources").as_deref() != Some("false")
247        || configuration_value(&policy, "allow_external_paths").as_deref() != Some("false")
248    {
249        return Err("dependency-source policy is incomplete or unsafe".to_owned());
250    }
251    let review_path = configuration_value(&policy, "review")
252        .ok_or_else(|| "dependency-source policy has no review path".to_owned())?;
253    if !is_safe_relative_path(Path::new(&review_path)) {
254        return Err("dependency-source review path must stay within the workspace".to_owned());
255    }
256    read_workspace_text(root, &review_path)?;
257    if !matches!(
258        quality_document_status(root, &review_path)?.as_str(),
259        "approved" | "approved-with-limitation"
260    ) {
261        return Err("dependency review is not approved in the quality manifest".to_owned());
262    }
263    let lock = read_workspace_text(root, constants::CARGO_LOCK_FILE)?;
264    let lock_document: toml::Table =
265        configuration::parse(&lock, crate::constants::CARGO_LOCK_FILE)?;
266    let locked_packages = lock_document
267        .get("package")
268        .and_then(toml::Value::as_array)
269        .ok_or("Cargo.lock has no package records")?;
270    for package in locked_packages {
271        let source = package.get("source").and_then(toml::Value::as_str);
272        if source.is_some_and(|source| !source.starts_with("registry+")) {
273            return Err("Cargo.lock contains a forbidden non-registry dependency".to_owned());
274        }
275        if source.is_some()
276            && package
277                .get("checksum")
278                .and_then(toml::Value::as_str)
279                .is_none()
280        {
281            return Err("Cargo.lock contains a registry package without a checksum".to_owned());
282        }
283    }
284    verify_manifest_dependency_paths(root, &root.join(constants::WORKSPACE_MANIFEST_FILE))?;
285    for manifest in workspace_package_manifests(root)? {
286        verify_manifest_dependency_paths(root, &manifest)?;
287        let package_content = fs::read_to_string(&manifest)
288            .map_err(|error| format!("could not read {}: {error}", manifest.display()))?;
289        let package_name = configuration::quality_value_from(&package_content, "package", "name")
290            .ok_or_else(|| {
291            format!(
292                "package manifest has no package name: {}",
293                manifest.display()
294            )
295        })?;
296        if !locked_packages.iter().any(|package| {
297            package.get("name").and_then(toml::Value::as_str) == Some(&package_name)
298                && package.get("version").and_then(toml::Value::as_str) == Some(package_version)
299                && package.get("source").is_none()
300        }) {
301            return Err(format!(
302                "Cargo.lock is stale for workspace package {package_name} {package_version}"
303            ));
304        }
305    }
306    let output = Command::new(cargo_command()?)
307        .current_dir(root)
308        .args([
309            "metadata",
310            flags::LOCKED,
311            "--offline",
312            "--format-version",
313            "1",
314            "--no-deps",
315        ])
316        .output()
317        .map_err(|error| format!("could not validate Cargo.lock: {error}"))?;
318    if !output.status.success() {
319        return Err(format!(
320            "Cargo.lock is stale or unavailable offline: {}",
321            String::from_utf8_lossy(&output.stderr).trim()
322        ));
323    }
324    let stderr = String::from_utf8_lossy(&output.stderr);
325    if stderr.lines().any(|line| line.contains("warning:")) {
326        return Err(format!(
327            "Cargo metadata emitted a release-relevant warning: {}",
328            stderr.trim()
329        ));
330    }
331    Ok(())
332}

334/// Requires every manifest path dependency to resolve inside the workspace.
335pub(super) fn verify_manifest_dependency_paths(root: &Path, manifest: &Path) -> Result<(), String> {
336    let content = fs::read_to_string(manifest)
337        .map_err(|error| format!("could not read {}: {error}", manifest.display()))?;
338    let document: toml::Table = configuration::parse(&content, &manifest.display().to_string())?;
339    let canonical_root = fs::canonicalize(root)
340        .map_err(|error| format!("could not canonicalize workspace root: {error}"))?;
341    verify_dependency_sections(&canonical_root, manifest, &document)?;
342    if let Some(workspace) = document.get("workspace").and_then(toml::Value::as_table) {
343        verify_dependency_sections(&canonical_root, manifest, workspace)?;
344    }
345    if let Some(targets) = document.get("target").and_then(toml::Value::as_table) {
346        for target in targets.values() {
347            let table = target.as_table().ok_or("manifest target must be a table")?;
348            verify_dependency_sections(&canonical_root, manifest, table)?;
349        }
350    }
351    Ok(())
352}

354/// Checks actual dependency tables, ignoring incidental comments and unrelated path metadata.
355fn verify_dependency_sections(
356    root: &Path,
357    manifest: &Path,
358    table: &toml::Table,
359) -> Result<(), String> {
360    for kind in ["dependencies", "build-dependencies", "dev-dependencies"] {
361        let Some(dependencies) = table.get(kind) else {
362            continue;
363        };
364        let dependencies = dependencies
365            .as_table()
366            .ok_or_else(|| format!("manifest {kind} must be a table"))?;
367        for dependency in dependencies.values() {
368            let Some(path) = dependency.as_table().and_then(|table| table.get("path")) else {
369                continue;
370            };
371            let path = path.as_str().ok_or("dependency path must be a string")?;
372            let path = manifest.parent().unwrap_or(root).join(path);
373            let canonical = fs::canonicalize(&path).map_err(|error| {
374                format!(
375                    "could not resolve path dependency {}: {error}",
376                    path.display()
377                )
378            })?;
379            if !canonical.starts_with(root) {
380                return Err(format!(
381                    "manifest path dependency escapes the workspace: {}",
382                    path.display()
383                ));
384            }
385        }
386    }
387    Ok(())
388}

406/// Reads one quoted value from the root workspace package section.
407pub(super) fn workspace_package_value(manifest: &str, key: &str) -> Result<String, String> {
408    let document: toml::Table = configuration::parse(manifest, "Cargo.toml")?;
409    document
410        .get("workspace")

412        .and_then(|package| package.get(key))

414        .map(str::to_owned)
415        .ok_or_else(|| format!("root Cargo.toml has no [workspace.package] string {key}"))
416}

426/// Derives a safe distribution name from the workspace repository URL.
427pub(super) fn project_slug(root: &Path) -> Result<String, String> {
428    let manifest = read_workspace_text(root, constants::WORKSPACE_MANIFEST_FILE)?;
429    let repository = workspace_package_value(&manifest, "repository")?;
430    let slug = repository
431        .trim_end_matches('/')
432        .rsplit('/')
433        .next()
434        .unwrap_or_default()
435        .trim_end_matches(".git");
436    if slug.is_empty()
437        || !slug
438            .bytes()
439            .all(|byte| byte.is_ascii_alphanumeric() || byte == b'-' || byte == b'_')
440    {
441        return Err("workspace repository has no safe distribution name".to_owned());
442    }
443    Ok(slug.to_owned())
444}

481/// Collects regular files with one exact filename below a directory.
482pub(super) fn collect_named_files(
483    directory: &Path,
484    name: &str,
485    files: &mut Vec<PathBuf>,
486) -> Result<(), String> {
487    for entry in fs::read_dir(directory)
488        .map_err(|error| format!("could not inspect {}: {error}", directory.display()))?
489    {
490        let entry = entry.map_err(|error| format!("could not inspect directory entry: {error}"))?;
491        let path = entry.path();
492        if path.is_dir() {
493            collect_named_files(&path, name, files)?;
494        } else if path.file_name().and_then(std::ffi::OsStr::to_str) == Some(name) {
495            files.push(path);
496        }
497    }
498    Ok(())
499}

501/// Validates package `SemVer` without enabling unsupported build metadata.
502pub(super) fn validate_semver(value: &str) -> Result<(), String> {
503    parsed_semver(value).map(|_| ())
504}
505
506/// Rejects package-version downgrades and invalid prerelease transitions.
507pub(super) fn validate_version_transition(current: &str, requested: &str) -> Result<(), String> {
508    if parsed_semver(requested)? <= parsed_semver(current)? {
509        return Err(
510            "requested package version must be greater than the current version".to_owned(),
511        );
512    }
513    Ok(())
514}
515
516/// Uses Cargo's `SemVer` implementation, including arbitrarily long prerelease numbers.
517fn parsed_semver(value: &str) -> Result<cargo_metadata::semver::Version, String> {
518    let version = cargo_metadata::semver::Version::parse(value)
519        .map_err(|error| format!("invalid package SemVer: {value}: {error}"))?;
520    if !version.build.is_empty() {
521        return Err(format!(
522            "invalid package SemVer: {value}; build metadata is unsupported"
523        ));
524    }
525    Ok(version)
526}

528/// Returns the lowercase SHA-256 digest of exact bytes.
529pub(super) fn sha256_hex(bytes: &[u8]) -> String {
530    let digest = Sha256::digest(bytes);
531    let mut output = String::with_capacity(digest.len() * 2);
532    for byte in digest {
533        write!(&mut output, "{byte:02x}").expect("writing to a String cannot fail");
534    }
535    output
