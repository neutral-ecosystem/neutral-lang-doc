5use cargo_metadata::{DependencyKind, Metadata, MetadataCommand, NodeDep, Package, PackageId};
6use std::{
7    collections::BTreeSet,
8    path::{Path, PathBuf},
9};

11/// Queries the selected Cargo executable without modifying its lockfile or using the network.
12pub(crate) fn metadata(root: &Path, dependencies: bool) -> Result<Metadata, String> {
13    let mut options = vec!["--locked".to_owned(), "--offline".to_owned()];
14    if dependencies {
15        let compiler = crate::command_output(&crate::configuration::rustc_command()?, &["-vV"])?;
16        let host = compiler
17            .lines()
18            .find_map(|line| line.strip_prefix("host: "))
19            .ok_or("Rust compiler did not identify its host platform")?;
20        options.extend(["--filter-platform".to_owned(), host.to_owned()]);
21    }
22    let mut command = MetadataCommand::new();
23    command
24        .cargo_path(crate::configuration::cargo_command()?)
25        .current_dir(root)
26        .manifest_path(root.join(crate::constants::WORKSPACE_MANIFEST_FILE))
27        .other_options(options);
28    if !dependencies {
29        command.no_deps();
30    }
31    let metadata = command.exec().map_err(|error| format!("Cargo discovery failed (run `cargo fetch --locked` if dependencies are missing): {error}"))?;
32    if metadata.workspace_root.as_std_path() != root {
33        return Err(format!(
34            "Cargo resolved a different workspace: {}",
35            metadata.workspace_root
36        ));
37    }
38    Ok(metadata)
39}

41/// Returns Cargo's resolved member manifests, including globbed and root-package members.
42pub(crate) fn manifests(root: &Path) -> Result<Vec<PathBuf>, String> {
43    let metadata = metadata(root, false)?;
44    let canonical_root = root
45        .canonicalize()
46        .map_err(|error| format!("could not resolve workspace root: {error}"))?;
47    let mut manifests = BTreeSet::new();
48    for package in metadata.workspace_packages() {
49        let manifest = package
50            .manifest_path
51            .canonicalize()
52            .map_err(|error| format!("could not resolve {}: {error}", package.manifest_path))?;
53        if !manifest.starts_with(&canonical_root) || !manifests.insert(manifest) {
54            return Err(format!(
55                "workspace package is external or repeated: {}",
56                package.name
57            ));
58        }
59    }
60    if manifests.is_empty() {
61        return Err("Cargo workspace has no packages".to_owned());
62    }
63    Ok(manifests.into_iter().collect())
64}

66/// Finds a workspace package by its declared name, not its directory or dependency alias.
67pub(crate) fn package<'a>(metadata: &'a Metadata, name: &str) -> Result<&'a Package, String> {
68    metadata
69        .workspace_packages()
70        .into_iter()
71        .find(|package| package.name.as_str() == name)
72        .ok_or_else(|| format!("Cargo workspace has no package {name}"))
73}

75/// Resolves a graph package ID to its real declared package name.
76fn package_name(metadata: &Metadata, id: &PackageId) -> Result<String, String> {
77    metadata
78        .packages
79        .iter()
80        .find(|package| &package.id == id)
81        .map(|package| package.name.to_string())
82        .ok_or_else(|| format!("Cargo graph references an unknown package {id}"))
83}

85/// Matches Cargo traversal: normal edges, optionally build edges and root-only development edges.
86fn selected(dependency: &NodeDep, all: bool, root: bool) -> bool {
87    dependency.dep_kinds.iter().any(|kind| {
88        kind.kind == DependencyKind::Normal
89            || (all && (kind.kind != DependencyKind::Development || root))
90    })
91}

93/// Returns the direct normal dependency names from one resolved Cargo graph.
94pub(crate) fn direct_dependencies(
95    metadata: &Metadata,
96    name: &str,
97) -> Result<BTreeSet<String>, String> {
98    let id = &package(metadata, name)?.id;
99    let resolve = metadata
100        .resolve
101        .as_ref()
102        .ok_or("Cargo metadata has no resolved dependency graph")?;
103    let node = resolve
104        .nodes
105        .iter()
106        .find(|node| &node.id == id)
107        .ok_or_else(|| format!("Cargo graph has no node for {name}"))?;
108    node.deps
109        .iter()
110        .filter(|dependency| selected(dependency, false, true))
111        .map(|dependency| package_name(metadata, &dependency.pkg))
112        .collect()
113}

115/// Traverses a resolved dependency closure once per ID, retaining the root package.
116pub(crate) fn closure(
117    metadata: &Metadata,
118    name: &str,
119    all: bool,
120) -> Result<BTreeSet<String>, String> {
121    let resolve = metadata
122        .resolve
123        .as_ref()
124        .ok_or("Cargo metadata has no resolved dependency graph")?;
125    let root = package(metadata, name)?.id.clone();
126    let mut pending = vec![root.clone()];
127    let mut visited = BTreeSet::new();
128    let mut names = BTreeSet::new();
129    while let Some(id) = pending.pop() {
130        if !visited.insert(id.clone()) {
131            continue;
132        }
133        names.insert(package_name(metadata, &id)?);
134        let node = resolve
135            .nodes
136            .iter()
137            .find(|node| node.id == id)
138            .ok_or_else(|| format!("Cargo graph has no node for {id}"))?;
139        pending.extend(
140            node.deps
141                .iter()
142                .filter(|dependency| selected(dependency, all, id == root))
143                .map(|dependency| dependency.pkg.clone()),
144        );
145    }
146    Ok(names)
147}

149/// Finds a benchmark executable through Cargo's typed build-message stream.
150pub(crate) fn benchmark_executable(output: &str, target: &str) -> Result<String, String> {
151    for message in cargo_metadata::Message::parse_stream(output.as_bytes()) {
152        let message = message.map_err(|error| format!("invalid Cargo build message: {error}"))?;
153        if let cargo_metadata::Message::CompilerArtifact(artifact) = message
154            && artifact.target.name == target
155            && let Some(executable) = artifact.executable
156        {
157            return Ok(executable.to_string());
158        }
159    }
160    Err(format!(
161        "Cargo did not identify the benchmark executable for {target}"
162    ))
163}
