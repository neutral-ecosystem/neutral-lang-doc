15/// Typed executable acceptance policy; historical status annotations confer no pass.
16#[derive(Clone, Debug, Deserialize)]
17#[serde(deny_unknown_fields)]
18pub(crate) struct QualityPolicy {
19    /// Coverage command inputs and thresholds.
20    pub(crate) coverage: CoverageSettings,
21    /// Mutation target and acceptance threshold.
22    pub(crate) mutation: MutationSettings,
23    /// Fuzzer targets, seeds, and duration.
24    pub(crate) fuzz: FuzzSettings,
25    /// Controlled performance harness.
26    pub(crate) performance: PerformanceSettings,
27}

87/// Validated Cargo benchmark selection, parsed once at the configuration boundary.
88#[derive(Clone, Debug, Deserialize)]
89#[serde(try_from = "String")]
90pub(crate) struct BenchmarkHarness {
91    /// Owning workspace package.
92    pub(crate) package: String,
93    /// Benchmark target within that package.
94    pub(crate) target: String,
95}
96
97impl TryFrom<String> for BenchmarkHarness {
98    type Error = String;
99
100    /// Rejects unsupported package names and non-target syntax before dispatch.
101    fn try_from(value: String) -> Result<Self, Self::Error> {
102        let (package, target) = value
103            .split_once('/')
104            .ok_or("quality performance harness must be package/target")?;
105        if package != constants::NEUTRAL_BENCH
106            || target.is_empty()
107            || !target
108                .bytes()
109                .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'-' | b'_'))
110        {
111            return Err("invalid quality performance harness".to_owned());
112        }
113        Ok(Self {
114            package: package.to_owned(),
115            target: target.to_owned(),
116        })
117    }
118}

120impl QualityPolicy {
121    /// Parses and validates all operational values before any measurement starts.
122    pub(crate) fn parse(content: &str) -> Result<Self, String> {
123        let policy: Self = configuration::parse(content, constants::QUALITY_GATES_FILE)?;
124        for percentage in [
125            policy.coverage.minimum_line_percent,
126            policy.coverage.minimum_function_percent,
127            policy.coverage.minimum_region_percent,
128            policy.mutation.minimum_caught_percent,
129        ] {
130            if !percentage.is_finite() || !(0.0..=100.0).contains(&percentage) {
131                return Err("quality percentages must be finite and between 0 and 100".to_owned());
132            }
133        }
134        for path in [&policy.coverage.html_output, &policy.coverage.json_output] {
135            if !crate::results::is_safe_result_path(path) {
136                return Err("coverage outputs must be safe relative result paths".to_owned());
137            }
138        }
139        if !is_safe_relative_path(Path::new(&policy.mutation.critical_target))
140            || !is_safe_relative_path(&policy.fuzz.corpus_root)
141            || !is_safe_relative_path(&policy.fuzz.seed_root)
142            || policy.fuzz.targets.is_empty()
143            || policy
144                .fuzz
145                .targets
146                .iter()
147                .any(|t| !is_safe_relative_path(Path::new(t)))
148            || policy.fuzz.minimum_seconds_per_target == 0
149        {
150            return Err(
151                "quality targets and paths must be safe and the fuzz budget positive".to_owned(),
152            );
153        }
154        Ok(policy)
155    }
156}

158/// Immutable settings and exact policy bytes bound to one quality command.
159#[derive(Clone)]
160pub(crate) struct QualitySettings {
161    /// Typed operational values.
162    pub(crate) policy: QualityPolicy,
163    /// Exact configuration bytes, including reviewed annotations.
164    pub(crate) policy_sha256: String,
165    /// Advisory freshness from typed automation settings.
166    pub(crate) advisory_max_age_seconds: u64,
167    /// Exact lock coverage from dependency policy.
168    pub(crate) advisory_locks: Vec<String>,
169    /// Evidence directory relative to the configured generated-result root.
170    pub(crate) evidence_root: PathBuf,
171}
172
173impl QualitySettings {
174    /// Reads quality, automation, and dependency policy once for this invocation.
175    pub(crate) fn load(root: &Path) -> Result<Self, String> {
176        let content = read_workspace_text(root, constants::QUALITY_GATES_FILE)?;
177        let policy = QualityPolicy::parse(&content)?;
178        let automation: crate::configuration_models::Automation =
179            configuration::read(root, constants::AUTOMATION_CONFIG_FILE)?;
180        configuration::require_schema(automation.schema_version, "automation configuration")?;
181        if automation.quality.advisory_max_age_seconds == 0 {
182            return Err("advisory freshness must be positive".to_owned());
183        }
184        let evidence_root = PathBuf::from(automation.quality.evidence_root);
185        if !is_safe_relative_path(&evidence_root) {
186            return Err("quality evidence root must be relative".to_owned());
187        }
188        let dependency = read_workspace_text(root, constants::DEPENDENCY_SOURCES_FILE)?;
189        let mut advisory_locks = vec![constants::CARGO_LOCK_FILE.to_owned()];
190        advisory_locks.extend(configuration_array_from(
191            &dependency,
192            "",
193            "isolated_tool_lockfiles",
194        )?);
195        if advisory_locks
196            .iter()
197            .any(|lock| !is_safe_relative_path(Path::new(lock)))
198        {
199            return Err("unsafe advisory lockfile".to_owned());
200        }
201        Ok(Self {
202            policy,
203            policy_sha256: sha256_hex(content.as_bytes()),
204            advisory_max_age_seconds: automation.quality.advisory_max_age_seconds,
205            advisory_locks,
206            evidence_root,
207        })
208    }
209}
