15/// Typed executable acceptance policy; historical status annotations confer no pass.
16#[derive(Clone, Debug, Deserialize)]
17#[serde(deny_unknown_fields)]
18pub(crate) struct QualityPolicy {
19    /// Coverage command inputs and thresholds.
20    pub(crate) coverage: CoverageSettings,
21    /// Mutation target and acceptance threshold.
22    pub(crate) mutation: MutationSettings,
23    /// Fuzzer targets, seeds, and duration.
24    pub(crate) fuzz: FuzzSettings,
25    /// Controlled performance harness.
26    pub(crate) performance: PerformanceSettings,
27}

99/// Validated Cargo benchmark selection, parsed once at the configuration boundary.
100#[derive(Clone, Debug, Deserialize)]
101#[serde(try_from = "String")]
102pub(crate) struct BenchmarkHarness {
103    /// Owning workspace package.
104    pub(crate) package: String,
105    /// Benchmark target within that package.
106    pub(crate) target: String,
107}
108
109impl TryFrom<String> for BenchmarkHarness {
110    type Error = String;
111
112    /// Rejects unsupported package names and non-target syntax before dispatch.
113    fn try_from(value: String) -> Result<Self, Self::Error> {
114        let (package, target) = value
115            .split_once('/')
116            .ok_or("quality performance harness must be package/target")?;
117        if package != constants::NEUTRAL_BENCH
118            || target.is_empty()
119            || !target
120                .bytes()
121                .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'-' | b'_'))
122        {
123            return Err("invalid quality performance harness".to_owned());
124        }
125        Ok(Self {
126            package: package.to_owned(),
127            target: target.to_owned(),
128        })
129    }
130}

132impl QualityPolicy {
133    /// Parses and validates all operational values before any measurement starts.
134    pub(crate) fn parse(content: &str) -> Result<Self, String> {
135        let policy: Self = configuration::parse(content, constants::QUALITY_GATES_FILE)?;
136        for percentage in [
137            policy.coverage.minimum_line_percent,
138            policy.coverage.minimum_function_percent,
139            policy.coverage.minimum_region_percent,
140            policy.mutation.minimum_caught_percent,
141        ] {
142            if !(percentage.is_finite() && percentage > 0.0 && percentage <= 100.0) {
143                return Err("quality percentages must be finite and within (0, 100]".to_owned());
144            }
145        }
146        for path in [&policy.coverage.html_output, &policy.coverage.json_output] {
147            if !crate::results::is_safe_result_path(path) {
148                return Err("coverage outputs must be safe relative result paths".to_owned());
149            }
150        }
151        let coverage = &policy.coverage;
152        if coverage.toolchain != "nightly-only"
153            || !coverage.html_output.ends_with("html/index.html")
154            || coverage
155                .json_output
156                .extension()
157                .and_then(std::ffi::OsStr::to_str)
158                != Some("json")
159            || coverage.exclusion_regex.is_empty()
160            || !coverage.exclusion_policy.starts_with("reviewed:")
161        {
162            return Err("coverage requires nightly-only, HTML/JSON outputs, and reviewed explicit exclusions".to_owned());
163        }
164        if policy.fuzz.finding_policy.trim().is_empty()
165            || policy.fuzz.mutable_state.trim().is_empty()
166            || policy.fuzz.targets.iter().any(|target| {
167                policy
168                    .fuzz
169                    .annotations
170                    .get(&format!("{target}_owner"))
171                    .and_then(toml::Value::as_str)
172                    .is_none_or(|owner| owner.trim().is_empty())
173            })
174        {
175            return Err(
176                "fuzz targets require owners, finding policy, and mutable-state policy".to_owned(),
177            );
178        }
179        let targets = &policy.mutation.critical_targets;
180        if targets.is_empty()
181            || targets
182                .iter()
183                .collect::<std::collections::BTreeSet<_>>()
184                .len()
185                != targets.len()
186            || targets.iter().any(|target| {
187                !is_safe_relative_path(Path::new(target))
188                    || Path::new(target)
189                        .extension()
190                        .is_none_or(|extension| extension != "rs")
191            })
192            || !is_safe_relative_path(&policy.fuzz.corpus_root)
193            || !is_safe_relative_path(&policy.fuzz.seed_root)
194            || policy.fuzz.targets.is_empty()
195            || policy
196                .fuzz
197                .targets
198                .iter()
199                .any(|t| !is_safe_relative_path(Path::new(t)))
200            || policy.fuzz.minimum_seconds_per_target == 0
201        {
202            return Err(
203                "quality targets and paths must be safe and the fuzz budget positive".to_owned(),
204            );
205        }
206        let phases = &policy.performance.maximum_ns_per_iteration;
207        if phases.is_empty()
208            || policy.performance.maximum_heap_bytes == 0
209            || phases.iter().any(|(phase, maximum)| {
210                *maximum == 0
211                    || phase.is_empty()
212                    || !phase
213                        .bytes()
214                        .all(|byte| byte.is_ascii_alphanumeric() || byte == b'-')
215            })
216        {
217            return Err(
218                "performance requires named phases with positive timing and heap ceilings"
219                    .to_owned(),
220            );
221        }
222        Ok(policy)
223    }
224}

226/// Immutable settings and exact policy bytes bound to one quality command.
227#[derive(Clone)]
228pub(crate) struct QualitySettings {
229    /// Typed operational values.
230    pub(crate) policy: QualityPolicy,
231    /// Exact configuration bytes, including reviewed annotations.
232    pub(crate) policy_sha256: String,
233    /// Advisory freshness from typed automation settings.
234    pub(crate) advisory_max_age_seconds: u64,
235    /// Exact lock coverage from dependency policy.
236    pub(crate) advisory_locks: Vec<String>,
237    /// Evidence directory relative to the configured generated-result root.
238    pub(crate) evidence_root: PathBuf,
239}
240
241impl QualitySettings {
242    /// Reads quality, automation, and dependency policy once for this invocation.
243    pub(crate) fn load(root: &Path) -> Result<Self, String> {
244        let content = read_workspace_text(root, constants::QUALITY_GATES_FILE)?;
245        let policy = QualityPolicy::parse(&content)?;
246        let automation = configuration::automation_at(root)?;
247        let evidence_root = PathBuf::from(automation.quality.evidence_root);
248        let dependency = read_workspace_text(root, constants::DEPENDENCY_SOURCES_FILE)?;
249        let mut advisory_locks = vec![constants::CARGO_LOCK_FILE.to_owned()];
250        advisory_locks.extend(configuration_array_from(
251            &dependency,
252            "",
253            "isolated_tool_lockfiles",
254        )?);
255        if advisory_locks
256            .iter()
257            .any(|lock| !is_safe_relative_path(Path::new(lock)))
258        {
259            return Err("unsafe advisory lockfile".to_owned());
260        }
261        Ok(Self {
262            policy,
263            policy_sha256: sha256_hex(content.as_bytes()),
264            advisory_max_age_seconds: automation.quality.advisory_max_age_seconds,
265            advisory_locks,
266            evidence_root,
267        })
268    }
269}
