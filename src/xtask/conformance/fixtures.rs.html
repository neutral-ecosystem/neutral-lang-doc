13/// Counts checked inputs and changed digests without treating synchronization as approval.
14#[derive(Default)]
15struct SyncSummary {
16    /// Checked fixture records.
17    fixture_count: usize,
18    /// Checked oracle records.
19    oracle_count: usize,
20    /// Changed conformance hashes.
21    manifest_changes: usize,
22    /// Changed contract hashes.
23    freeze_changes: usize,
24    /// Check-mode mismatches.
25    mismatch_errors: Vec<String>,
26}
27
28/// Preflights both documents and all their inputs before writing any changed manifest.
29pub(crate) fn sync_fixtures(root: &Path, check: bool) -> Result<(), String> {
30    crate::output::start(format!(
31        "fixtures: {} fixture and contract hashes",
32        if check { "verifying" } else { "synchronizing" }
33    ));
34    let manifest_path = root.join(constants::PORTABLE_CONFORMANCE_MANIFEST_FILE);
35    let freeze_path = root.join(constants::PORTABLE_CONTRACT_FREEZE_FILE);
36    let manifest_raw =
37        crate::read_workspace_text(root, constants::PORTABLE_CONFORMANCE_MANIFEST_FILE)?;
38    let freeze_raw = crate::read_workspace_text(root, constants::PORTABLE_CONTRACT_FREEZE_FILE)?;
39    let mut manifest = manifest_updates::parse(&manifest_raw, "fixture manifest")?;
40    let mut freeze = manifest_updates::parse(&freeze_raw, "contract freeze")?;
41    let mut summary = SyncSummary::default();
42    let registered = sync_manifest(root, &mut manifest, check, &mut summary)?;
43    let updated_manifest = manifest.to_string();
44    sync_freeze_table(
45        root,
46        freeze.as_table_mut(),
47        check,
48        &mut summary,
49        &manifest_path,
50        updated_manifest.as_bytes(),
51    )?;
52    discover_untracked_fixtures(root, &registered)?;
53    if check && !summary.mismatch_errors.is_empty() {
54        return Err(format!(
55            "fixture verification failed with {} mismatch(es):\n{}",
56            summary.mismatch_errors.len(),
57            summary.mismatch_errors.join("\n\n")
58        ));
59    }
60    if !check {
61        if summary.manifest_changes > 0 {
62            fs::write(&manifest_path, updated_manifest)
63                .map_err(|error| format!("could not write {}: {error}", manifest_path.display()))?;
64        }
65        if summary.freeze_changes > 0 {
66            fs::write(&freeze_path, freeze.to_string())
67                .map_err(|error| format!("could not write {}: {error}", freeze_path.display()))?;
68        }
69    }
70    crate::output::pass(format!(
71        "fixtures: verified {} fixtures, {} oracles (manifest updates: {}, freeze updates: {})",
72        summary.fixture_count,
73        summary.oracle_count,
74        summary.manifest_changes,
75        summary.freeze_changes
76    ));
77    Ok(())
78}

80/// Verifies fixture and oracle pairs regardless of field ordering, preserving unrelated extensions.
81fn sync_manifest(
82    root: &Path,
83    document: &mut DocumentMut,
84    check: bool,
85    summary: &mut SyncSummary,
86) -> Result<BTreeSet<String>, String> {
87    let cases = document
88        .get_mut("case")
89        .and_then(Item::as_array_of_tables_mut)
90        .ok_or("fixture manifest has no [[case]] records")?;
91    let mut registered = BTreeSet::new();
92    for case in cases.iter_mut() {
93        for kind in ["fixture", "oracle"] {
94            let path = case
95                .get(kind)
96                .and_then(Item::as_str)
97                .ok_or_else(|| format!("fixture case has no string {kind} path"))?
98                .to_owned();
99            let actual = hash_file(&registered_path(root, &path)?)?;
100            let key = format!("{kind}_sha256");
101            let item = case
102                .get_mut(&key)
103                .ok_or_else(|| format!("fixture case has no {key}"))?;
104            if update_digest(item, &actual, &path, check, &mut summary.mismatch_errors)? {
105                summary.manifest_changes += 1;
106            }
107            if kind == "fixture" {
108                registered.insert(path);
109                summary.fixture_count += 1;
110            } else {
111                summary.oracle_count += 1;
112            }
113        }
114    }
115    Ok(registered)
116}

118/// Resolves only regular workspace-owned files, rejecting traversal and symlink escapes.
119fn registered_path(root: &Path, relative: &str) -> Result<PathBuf, String> {
120    if !crate::configuration::is_safe_relative_path(Path::new(relative)) {
121        return Err(format!("unsafe fixture/contract path: {relative}"));
122    }
123    let path = root
124        .join(relative)
125        .canonicalize()
126        .map_err(|error| format!("could not resolve {relative}: {error}"))?;
127    let root = root
128        .canonicalize()
129        .map_err(|error| format!("could not resolve workspace root: {error}"))?;
130    if !path.starts_with(root) || !path.is_file() {
131        return Err(format!(
132            "fixture/contract path is external or not a file: {relative}"
133        ));
134    }
135    Ok(path)
136}

138/// Checks or edits a digest string while retaining the original comment and spacing.
139fn update_digest(
140    item: &mut Item,
141    actual: &str,
142    context: &str,
143    check: bool,
144    mismatches: &mut Vec<String>,
145) -> Result<bool, String> {
146    let expected = item
147        .as_str()
148        .ok_or_else(|| format!("{context} digest must be a string"))?
149        .to_owned();
150    if expected == actual {
151        return Ok(false);
152    }
153    if check {
154        mismatches.push(format!(
155            "{context} digest mismatch:\n  expected: {expected}\n  actual:   {actual}"
156        ));
157    } else {
158        manifest_updates::replace_string(item, &expected, actual, context)?;
159    }
160    Ok(true)
161}

163/// Visits contract path/hash pairs in their owning table, including nested and repeated tables.
164fn sync_freeze_table(
165    root: &Path,
166    table: &mut dyn TableLike,
167    check: bool,
168    summary: &mut SyncSummary,
169    manifest_path: &Path,
170    manifest_bytes: &[u8],
171) -> Result<(), String> {
172    let pairs = table
173        .iter()
174        .filter_map(|(key, item)| {
175            let prefix = key.strip_suffix("_sha256")?;
176            table.get(&format!("{prefix}_path")).map(|path| {
177                (
178                    key.to_owned(),
179                    path.as_str().map(str::to_owned),
180                    item.as_str().is_some(),
181                )
182            })
183        })
184        .collect::<Vec<_>>();
185    for (key, path, valid_digest) in pairs {
186        let relative = path.ok_or_else(|| format!("{key} companion path must be a string"))?;
187        if !valid_digest {
188            return Err(format!("{key} digest must be a string"));
189        }
190        let path = registered_path(root, &relative)?;
191        let actual = if path == manifest_path {
192            crate::sha256_hex(manifest_bytes)
193        } else {
194            hash_file(&path)?
195        };
196        let item = table
197            .get_mut(&key)
198            .ok_or_else(|| format!("missing freeze digest {key}"))?;
199        if update_digest(
200            item,
201            &actual,
202            &relative,
203            check,
204            &mut summary.mismatch_errors,
205        )? {
206            summary.freeze_changes += 1;
207        }
208    }
209    for (_, item) in table.iter_mut() {
210        if let Some(tables) = item.as_array_of_tables_mut() {
211            for child in tables.iter_mut() {
212                sync_freeze_table(root, child, check, summary, manifest_path, manifest_bytes)?;
213            }
214        } else if let Some(child) = item.as_table_like_mut() {
215            sync_freeze_table(root, child, check, summary, manifest_path, manifest_bytes)?;
216        }
217    }
218    Ok(())
219}

221/// Warns about fixture files not registered in the reviewed manifest.
222fn discover_untracked_fixtures(root: &Path, registered: &BTreeSet<String>) -> Result<(), String> {
223    let mut files = Vec::new();
224    collect_fixture_files(
225        &root.join(constants::PORTABLE_FIXTURE_DIRECTORY),
226        &mut files,
227    )?;
228    for path in files {
229        let relative = path
230            .strip_prefix(root)
231            .map_err(|error| error.to_string())?
232            .to_string_lossy()
233            .replace('\\', "/");
234        if !registered.contains(&relative) {
235            crate::output::warn(format!(
236                "fixtures: file on disk not yet in manifest: {relative}"
237            ));
238        }
239    }
240    Ok(())
241}

243/// Collects test files without following directory symlinks or swallowing directory errors.
244fn collect_fixture_files(directory: &Path, files: &mut Vec<PathBuf>) -> Result<(), String> {
245    if !directory.is_dir() {
246        return Ok(());
247    }
248    for entry in fs::read_dir(directory)
249        .map_err(|error| format!("could not read {}: {error}", directory.display()))?
250    {
251        let entry =
252            entry.map_err(|error| format!("could not read fixture directory entry: {error}"))?;
253        let kind = entry
254            .file_type()
255            .map_err(|error| format!("could not inspect fixture: {error}"))?;
256        if kind.is_symlink() {
257            return Err(format!(
258                "fixture inventory cannot follow symlinks: {}",
259                entry.path().display()
260            ));
261        }
262        if kind.is_dir() {
263            collect_fixture_files(&entry.path(), files)?;
264        } else if kind.is_file()
265            && matches!(
266                entry
267                    .path()
268                    .extension()
269                    .and_then(|extension| extension.to_str()),
270                Some("neu" | "toml")
271            )
272        {
273            files.push(entry.path());
274        }
275    }
276    Ok(())
277}

279/// Hashes the exact captured bytes, never a normalized text representation.
280fn hash_file(path: &Path) -> Result<String, String> {
281    let bytes =
282        fs::read(path).map_err(|error| format!("could not read {}: {error}", path.display()))?;
283    Ok(crate::sha256_hex(&bytes))
284}
