3//! Typed, fail-closed parsing for the explicit release-authority selection.
4
5use std::{collections::BTreeSet, fs, path::Path};
6
7/// One explicitly selected release distribution channel.
8#[derive(Clone, Copy, Debug, Eq, Ord, PartialEq, PartialOrd)]
9pub(crate) enum DistributionChannel {
10    /// Annotated source tag.
11    SourceTag,
12    /// GitHub-hosted binary assets.
13    GithubBinaries,
14    /// crates.io package publication.
15    CratesIo,
16}
17
18/// One reviewed distribution selection for the `main`-head release candidate.
19#[derive(Clone, Debug, Eq, PartialEq)]
20pub(crate) struct ReleasePlan {
21    /// Release tag that will name the publication after qualification completes.
22    pub(crate) release_tag: String,
23    /// Explicitly selected distribution channels.
24    pub(crate) channels: BTreeSet<DistributionChannel>,
25    /// Binary package names selected for GitHub distribution.
26    pub(crate) binaries: Vec<String>,
27}
28
29impl ReleasePlan {
30    /// Reads release scope and derives its eventual publication tag from the workspace version.
31    pub(crate) fn read(path: &Path, package_version: &str) -> Result<Self, String> {
32        let content = fs::read_to_string(path)
33            .map_err(|error| format!("could not read {}: {error}", path.display()))?;
34        let mut channels = BTreeSet::new();
35        if required_bool(&content, "source_tag")? {
36            channels.insert(DistributionChannel::SourceTag);
37        }
38        if required_bool(&content, "github_binaries")? {
39            channels.insert(DistributionChannel::GithubBinaries);
40        }
41        if required_bool(&content, "crates_io")? {
42            channels.insert(DistributionChannel::CratesIo);
43        }
44        let plan = Self {
45            release_tag: format!("v{package_version}"),
46            channels,
47            binaries: required_array(&content, "binaries")?,
48        };
49        plan.validate()?;
50        Ok(plan)
51    }
52
53    /// Rejects incomplete or malformed release-authority selections.
54    fn validate(&self) -> Result<(), String> {
55        if !self.release_tag.starts_with('v') || self.release_tag.chars().any(char::is_whitespace) {
56            return Err("release tag must be a whitespace-free v-prefixed tag".to_owned());
57        }
58        if self.channels.is_empty() {
59            return Err("at least one release distribution channel must be selected".to_owned());
60        }
61        if self.channels.contains(&DistributionChannel::GithubBinaries) && self.binaries.is_empty()
62        {
63            return Err("GitHub binary distribution requires at least one binary".to_owned());
64        }
65        if self.binaries.iter().any(|binary| {
66            binary.is_empty()
67                || binary.contains('/')
68                || binary.contains('\\')
69                || binary == "."
70                || binary == ".."
71        }) {
72            return Err("release binary names must be plain package names".to_owned());
73        }
74        Ok(())
75    }
76}
77
78/// Reads one required Boolean from the constrained release TOML.
79fn required_bool(content: &str, key: &str) -> Result<bool, String> {
80    match required_value(content, key)? {
81        "true" => Ok(true),
82        "false" => Ok(false),
83        _ => Err(format!("release {key} must be true or false")),
84    }
85}
86
87/// Reads one required quoted-string array from the constrained release TOML.
88fn required_array(content: &str, key: &str) -> Result<Vec<String>, String> {
89    let value = required_value(content, key)?;
90    let inner = value
91        .strip_prefix('[')
92        .and_then(|value| value.strip_suffix(']'))
93        .ok_or_else(|| format!("release {key} must be an array"))?;
94    if inner.trim().is_empty() {
95        return Ok(Vec::new());
96    }
97    inner
98        .split(',')
99        .map(str::trim)
100        .map(|entry| {
101            entry
102                .strip_prefix('"')
103                .and_then(|entry| entry.strip_suffix('"'))
104                .map(str::to_owned)
105                .ok_or_else(|| format!("release {key} entries must be quoted strings"))
106        })
107        .collect()
108}
109
110/// Finds one exact key in the constrained release TOML.
111fn required_value<'a>(content: &'a str, key: &str) -> Result<&'a str, String> {
112    content
113        .lines()
114        .map(str::trim)
115        .filter(|line| !line.is_empty() && !line.starts_with('#') && !line.starts_with('['))
116        .find_map(|line| {
117            let (candidate, value) = line.split_once('=')?;
118            (candidate.trim() == key).then_some(value.trim())
119        })
120        .ok_or_else(|| format!("release configuration has no {key}"))
121}
