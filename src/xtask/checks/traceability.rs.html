10/// Checks accepted IDs, completed syntax items, and fixture/oracle inventories.
11pub(crate) fn check_traceability() -> Result<(), String> {
12    let root = workspace_root()?;
13    let bundle = ReleasedBundle::load(&root)?;
14    check_traceability_bundle(
15        &bundle.member("specs/REQUIREMENTS.md"),
16        &bundle.member("specs/contracts/syntax.md"),
17        &bundle.member("specs/contracts/syntax-checklist.md"),
18        &bundle.member("specs/TRACEABILITY.md"),
19        &bundle.member("conformance/manifest.toml"),
20        &bundle.member("specs/fixtures"),
21        &bundle.member("conformance/oracles"),
22    )?;
23    verify_frozen_input_digests(&root, &bundle.member("specs/contracts/freeze.toml"))
24}

26/// Checks the version-independent conformance inventory of an installed portable package.
27pub(crate) fn check_portable_traceability_at(root: &Path) -> Result<(), String> {
28    let manifest = read_workspace_text(root, constants::PORTABLE_CONFORMANCE_MANIFEST_FILE)?;
29    ensure_inventory_registered(root, constants::PORTABLE_FIXTURE_DIRECTORY, &manifest)?;
30    ensure_inventory_registered(root, constants::PORTABLE_ORACLE_DIRECTORY, &manifest)?;
31    ensure_registered_paths_exist(root, &manifest)?;
32    crate::output::pass("traceability coherence");
33    Ok(())
34}

36/// Checks one self-contained contract, fixture, and oracle bundle.
37pub(crate) fn check_traceability_bundle(
38    requirements_file: &str,
39    syntax_file: &str,
40    checklist_file: &str,
41    traceability_file: &str,
42    manifest_file: &str,
43    fixture_directory: &str,
44    oracle_directory: &str,
45) -> Result<(), String> {
46    let root = workspace_root()?;
47    let requirements = read_workspace_text(&root, requirements_file)?;
48    let syntax = read_workspace_text(&root, syntax_file)?;
49    let checklist = read_workspace_text(&root, checklist_file)?;
50    let traceability = read_workspace_text(&root, traceability_file)?;
51    let manifest = read_workspace_text(&root, manifest_file)?;
52
53    let requirement_ids = contract_ids(&requirements, "NL-");
54    let syntax_ids = contract_ids(&syntax, "SYN-");
55    let checklist_ids = contract_ids(&checklist, "SYN-");
56    ensure_ids_covered("requirements", &requirement_ids, &traceability)?;
57    ensure_ids_covered("syntax", &syntax_ids, &traceability)?;
58    if syntax_ids != checklist_ids {
59        return Err("master syntax and implementation checklist IDs differ".to_owned());
60    }
61    ensure_syntax_complete(syntax_file, &syntax)?;
62    ensure_syntax_complete(checklist_file, &checklist)?;
63    ensure_inventory_registered(&root, fixture_directory, &manifest)?;
64    ensure_inventory_registered(&root, oracle_directory, &manifest)?;
65    ensure_registered_paths_exist(&root, &manifest)?;
66
67    crate::output::pass("traceability coherence");
68    Ok(())
69}

71/// Extracts unique contract identifiers beginning with `prefix`.
72pub(crate) fn contract_ids(content: &str, prefix: &str) -> BTreeSet<String> {
73    content
74        .split(|character: char| !(character.is_ascii_alphanumeric() || character == '-'))
75        .filter(|token| token.starts_with(prefix))
76        .map(str::to_owned)
77        .collect()
78}

80/// Requires every accepted identifier to occur in the evidence index.
81pub(crate) fn ensure_ids_covered(
82    category: &str,
83    identifiers: &BTreeSet<String>,
84    traceability: &str,
85) -> Result<(), String> {
86    if identifiers.is_empty() {
87        return Err(format!("{category} contains no contract identifiers"));
88    }
89    let missing = identifiers
90        .iter()
91        .filter(|identifier| !traceability.contains(identifier.as_str()))
92        .cloned()
93        .collect::<Vec<_>>();
94    if missing.is_empty() {
95        Ok(())
96    } else {
97        Err(format!(
98            "traceability is missing {category} IDs: {}",
99            missing.join(", ")
100        ))
101    }
102}

104/// Rejects an unchecked master syntax item after traceability closure.
105pub(crate) fn ensure_syntax_complete(path: &str, content: &str) -> Result<(), String> {
106    let unchecked = content
107        .lines()
108        .filter(|line| line.trim_start().starts_with("- [ ]") && line.contains("SYN-"))
109        .collect::<Vec<_>>();
110    if unchecked.is_empty() {
111        Ok(())
112    } else {
113        Err(format!(
114            "{path} contains {} unchecked syntax items",
115            unchecked.len()
116        ))
117    }
118}

120/// Requires every normative file beneath `directory` to occur in the manifest.
121pub(crate) fn ensure_inventory_registered(
122    root: &Path,
123    directory: &str,
124    manifest: &str,
125) -> Result<(), String> {
126    let mut files = Vec::new();
127    collect_regular_files(&root.join(directory), &mut files)?;
128    let missing = files
129        .iter()
130        .filter_map(|path| path.strip_prefix(root).ok())
131        .map(|path| path.to_string_lossy().replace('\\', "/"))
132        .filter(|path| {
133            matches!(
134                Path::new(path).extension().and_then(|value| value.to_str()),
135                Some("neu" | "json" | "toml")
136            )
137        })
138        .filter(|path| !manifest.contains(path))
139        .collect::<Vec<_>>();
140    if missing.is_empty() {
141        Ok(())
142    } else {
143        Err(format!(
144            "conformance manifest omits normative files: {}",
145            missing.join(", ")
146        ))
147    }
148}

150/// Requires every workspace-relative path registered by the manifest to exist.
151pub(crate) fn ensure_registered_paths_exist(root: &Path, manifest: &str) -> Result<(), String> {
152    let missing = manifest
153        .split('"')
154        .filter(|value| {
155            value.starts_with("conformance/releases/") || value.starts_with("portable/")
156        })
157        .filter(|value| !root.join(value).is_file())
158        .collect::<Vec<_>>();
159    if missing.is_empty() {
160        Ok(())
161    } else {
162        Err(format!(
163            "manifest paths do not exist: {}",
164            missing.join(", ")
165        ))
166    }
