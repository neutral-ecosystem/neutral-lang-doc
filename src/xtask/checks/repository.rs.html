12/// Verifies every tracked repository-local Markdown link stays within the workspace.
13pub(crate) fn verify_repository_markdown_links() -> Result<(), String> {
14    let root = workspace_root()?;
15    let canonical_root = fs::canonicalize(&root)
16        .map_err(|error| format!("could not canonicalize workspace root: {error}"))?;
17    let mut files = vec![root.join(constants::ROOT_README_FILE)];
18    for directory in repository_directories(&root)? {
19        collect_regular_files(&root.join(directory.path), &mut files)?;
20    }
21    let mut failures = Vec::new();
22    for file in files
23        .iter()
24        .filter(|path| path.extension().and_then(|value| value.to_str()) == Some("md"))
25    {
26        let content = fs::read_to_string(file)
27            .map_err(|error| format!("could not read {}: {error}", file.display()))?;
28        for target in markdown_link_targets(&content) {
29            let candidate = file.parent().unwrap_or(&root).join(&target);
30            let valid = fs::canonicalize(&candidate)
31                .is_ok_and(|resolved| resolved.starts_with(&canonical_root));
32            if !valid {
33                failures.push(format!(
34                    "{} -> {target}",
35                    file.strip_prefix(&root).unwrap_or(file).display()
36                ));
37            }
38        }
39    }
40    if failures.is_empty() {
41        crate::output::pass("repository Markdown links");
42        Ok(())
43    } else {
44        failures.sort();
45        failures.dedup();
46        Err(format!(
47            "repository Markdown links are missing or escape the workspace: {}",
48            failures.join(", ")
49        ))
50    }
51}

53/// Verifies command documentation, platform adapters, and CI stay synchronized.
54pub(crate) fn check_workflow_contract() -> Result<(), String> {
55    let root = workspace_root()?;
56    for relative in ["README.md"] {
57        let content = read_workspace_text(&root, relative)?;
58        for command in interface::DOCUMENTED_COMMANDS {
59            if !content.contains(command) {
60                return Err(format!("{relative} does not document `{command}`"));
61            }
62        }
63    }
64
65    let ci = read_workspace_text(&root, ".github/workflows/ci.yml")?;
66    if !ci.contains("cargo xtask ci pr") {
67        return Err("ci.yml does not delegate to `cargo xtask ci pr`".to_owned());
68    }
69    let retained_workflows = format!(
70        "{}/workflows",
71        crate::configuration::automation_at(&root)?
72            .output
73            .results_root
74    );
75    for retention_requirement in [
76        "if: always()",
77        "actions/upload-artifact@",
78        &retained_workflows,
79    ] {
80        if !ci.contains(retention_requirement) {
81            return Err(format!(
82                "ci.yml does not retain workflow evidence with `{retention_requirement}`"
83            ));
84        }
85    }
86    let release = read_workspace_text(&root, ".github/workflows/release.yml")?;
87    if !release.contains("cargo xtask release qualify") {
88        return Err("release.yml does not delegate to `cargo xtask release qualify`".to_owned());
89    }
90    for requirement in [
91        "tags: ['v*']",
92        "github.event_name == 'push' && github.ref || 'main'",
93        "Verify tag identifies main HEAD",
94        "git switch -C main",
95        "cargo xtask release tag",
96        "sha256sum --check SHA256SUMS",
97        "contents: write",
98        "if: github.event_name == 'push'",
99        "gh release create",
100        "--verify-tag",
101        "--draft",
102    ] {
103        if !release.contains(requirement) {
104            return Err(format!(
105                "release.yml does not enforce publication requirement `{requirement}`"
106            ));
107        }
108    }
109    if ci.contains("pull_request:") || ci.contains("contents: write") {
110        return Err("push CI must not expose release credentials or write permission".to_owned());
111    }
112    if ci.contains("cargo xtask docs") || ci.contains("cargo xtask quality") {
113        return Err("ci.yml must use the single logged CI composition".to_owned());
114    }
115
116    for (relative, command) in [
117        ("scripts/linux/bootstrap.sh", "xtask bootstrap"),
118        ("scripts/linux/environment.sh", "xtask environment"),
119        ("scripts/linux/release.sh", "xtask release prepare"),
120        ("scripts/win/bootstrap.ps1", "xtask bootstrap"),
121        ("scripts/win/environment.ps1", "xtask environment"),
122        ("scripts/win/release.ps1", "xtask release prepare"),
123    ] {
124        if !read_workspace_text(&root, relative)?.contains(command) {
125            return Err(format!("{relative} does not delegate to `{command}`"));
126        }
127    }
128    crate::output::pass("stable workflow contract");
129    Ok(())
130}

132/// Rejects production or executable-test references to archived portable inputs.
133pub(crate) fn ensure_no_portable_archive_dependencies(root: &Path) -> Result<(), String> {
134    let mut files = Vec::new();
135    let mut production_roots = Vec::new();
136    for manifest in workspace_package_manifests(root)? {
137        let package_root = manifest
138            .parent()
139            .ok_or_else(|| format!("workspace manifest has no parent: {}", manifest.display()))?;
140        collect_regular_files(package_root, &mut files)?;
141        let content = fs::read_to_string(&manifest)
142            .map_err(|error| format!("could not read {}: {error}", manifest.display()))?;
143        if configuration::quality_value_from(&content, "package", "name").as_deref()
144            != Some(constants::XTASK)
145        {
146            production_roots.push(package_root.to_path_buf());
147        }
148    }
149    let forbidden = [
150        concat!("portable", "/archive/"),
151        concat!("portable", "/archived/"),
152    ];
153    let violations = files
154        .iter()
155        .filter(|path| {
156            matches!(
157                path.extension().and_then(|value| value.to_str()),
158                Some("rs" | "toml")
159            )
160        })
161        .filter_map(|path| {
162            let content = fs::read_to_string(path).ok()?;
163            let uses_active_portable = production_roots
164                .iter()
165                .any(|package_root| path.starts_with(package_root))
166                && content.contains("portable/");
167            (uses_active_portable || forbidden.iter().any(|value| content.contains(value))).then(
168                || {
169                    path.strip_prefix(root)
170                        .unwrap_or(path)
171                        .display()
172                        .to_string()
173                },
174            )
175        })
176        .collect::<Vec<_>>();
177    if violations.is_empty() {
178        Ok(())
179    } else {
180        Err(format!(
181            "code depends on archived portable material: {}",
182            violations.join(", ")
183        ))
184    }
185}

187/// Verifies generated-output ownership and ignored tracking policy.
188pub(crate) fn check_generated_outputs() -> Result<(), String> {
189    let root = workspace_root()?;
190    let inventory = read_workspace_text(&root, constants::GENERATED_OUTPUTS_FILE)?;
191    let result_directory = crate::configuration::automation_at(&root)?
192        .output
193        .results_root;
194    if !is_safe_relative_path(Path::new(&result_directory)) {
195        return Err("configured generated-results root is not a safe relative path".to_owned());
196    }
197    for name in [
198        "rustdoc",
199        "coverage",
200        "quality-measurements",
201        "package",
202        "workflow-runs",
203        "portable-snapshot",
204        "portable-rejected",
205        "quality-evaluation",
206    ] {
207        if !inventory.contains(&format!("name = \"{name}\"")) {
208            return Err(format!("generated-output inventory is missing {name}"));
209        }
210    }
211    for entry in inventory.split("[[output]]").skip(1) {
212        let path = configuration_value(entry, "path")
213            .ok_or_else(|| "generated output has no path".to_owned())?;
214        let tracking = configuration_value(entry, "tracking");
215        let ephemeral_output = (path.starts_with("target/")
216            || path.starts_with(&format!("{result_directory}/")))
217            && tracking.as_deref() == Some("ignored");
218        if !ephemeral_output
219            || configuration_value(entry, "owner").is_none()
220            || configuration_value(entry, "generate").is_none()
221            || configuration_value(entry, "validate").is_none()
222        {
223            return Err(format!(
224                "generated output {path} has an unsafe or incomplete policy"
225            ));
226        }
227    }
228    let ignore = read_workspace_text(&root, ".gitignore")?;
229    if !ignore.lines().any(|line| line.trim() == "target")
230        || !ignore
231            .lines()
232            .any(|line| line.trim() == format!("{result_directory}/"))
233    {
234        return Err("Cargo target and configured results must remain ignored".to_owned());
235    }
236    crate::output::pass("generated-output ownership");
237    Ok(())
238}

240/// Verifies the categorized inventory of durable quality documents.
241pub(crate) fn check_quality_inventory() -> Result<(), String> {
242    let root = workspace_root()?;
243    let expected_license_marker = html_spdx_marker(&project_license(&root)?);
244    let manifest = read_workspace_text(&root, constants::QUALITY_MANIFEST_FILE)?;
245    let mut registered = BTreeSet::new();
246    for entry in manifest.split("[[document]]").skip(1) {
247        let path = configuration_value(entry, "path")
248            .ok_or_else(|| "quality document has no path".to_owned())?;
249        let kind = configuration_value(entry, "kind")
250            .ok_or_else(|| format!("quality document {path} has no kind"))?;
251        let owner = configuration_value(entry, "owner")
252            .ok_or_else(|| format!("quality document {path} has no owner"))?;
253        let status = configuration_value(entry, "status")
254            .ok_or_else(|| format!("quality document {path} has no status"))?;
255        let expected_prefix = match kind.as_str() {
256            "policy" => "quality/policy/",
257            "review" => "quality/reviews/",
258            "release-evidence" => "quality/evidence/",
259            _ => return Err(format!("quality document {path} has unknown kind {kind:?}")),
260        };
261        if owner.is_empty() || status.is_empty() || !path.starts_with(expected_prefix) {
262            return Err(format!(
263                "quality document {path} has incomplete or inconsistent metadata"
264            ));
265        }
266        if !registered.insert(path.clone()) {
267            return Err(format!("quality document is registered twice: {path}"));
268        }
269        let content = read_workspace_text(&root, &path)?;
270        if !content.starts_with(&expected_license_marker) {
271            return Err(format!("quality document lacks its license marker: {path}"));
272        }
273    }
274
275    let mut files = Vec::new();
276    collect_regular_files(&root.join("quality"), &mut files)?;
277    let actual = files
278        .iter()
279        .filter(|path| path.extension().and_then(|value| value.to_str()) == Some("md"))
280        .filter(|path| path.file_name().and_then(|value| value.to_str()) != Some("README.md"))
281        .filter(|path| path.as_path() != root.join(constants::QUALITY_STATUS_FILE))
282        .map(|path| {
283            path.strip_prefix(&root)
284                .unwrap_or(path)
285                .to_string_lossy()
286                .replace('\\', "/")
287        })
288        .collect::<BTreeSet<_>>();
289    if actual != registered {
290        let unregistered = actual.difference(&registered).collect::<Vec<_>>();
291        let missing = registered.difference(&actual).collect::<Vec<_>>();
292        return Err(format!(
293            "quality inventory differs from durable documents; unregistered: {unregistered:?}; missing: {missing:?}"
294        ));
295    }
296    crate::output::pass("quality document inventory");
297    Ok(())
298}

300/// Verifies directory ownership, durable test levels, fuzz ownership, and hygiene.
301pub(crate) fn check_repository_structure() -> Result<(), String> {
302    let root = workspace_root()?;
303    let expected_license_marker = html_spdx_marker(&project_license(&root)?);
304    let mut configured = BTreeSet::new();
305    for directory in repository_directories(&root)? {
306        let path = directory.path;
307        let readme = directory.readme;
308        if !is_safe_relative_path(Path::new(&path))
309            || Path::new(&path).components().count() != 1
310            || !is_safe_relative_path(Path::new(&readme))
311            || !readme.starts_with(&format!("{path}/"))
312            || directory.owner.is_empty()
313            || directory.lifecycle.is_empty()
314            || !root.join(&path).is_dir()
315            || !root.join(&readme).is_file()
316        {
317            return Err(format!(
318                "repository directory {path} has incomplete ownership"
319            ));
320        }
321        let readme_content = read_workspace_text(&root, &readme)?;
322        if !readme_content.starts_with(&expected_license_marker) {
323            return Err(format!(
324                "repository README lacks its license marker: {readme}"
325            ));
326        }
327        if !configured.insert(path.clone()) {
328            return Err(format!("repository directory is declared twice: {path}"));
329        }
330    }
331    let tracked = Command::new(constants::GIT_COMMAND)
332        .current_dir(&root)
333        .args(["ls-files", "--cached", "-z"])
334        .output()
335        .map_err(|error| format!("could not enumerate tracked repository roots: {error}"))?;
336    if !tracked.status.success() {
337        return Err("could not enumerate tracked repository roots".to_owned());
338    }
339    let mut expected = BTreeSet::new();
340    for file in tracked
341        .stdout
342        .split(|byte| *byte == 0)
343        .filter(|file| !file.is_empty())
344    {
345        let name = std::str::from_utf8(file)
346            .map_err(|error| format!("tracked repository path is not UTF-8: {error}"))?;
347        if let Some((directory, _)) = name.split_once('/')
348            && directory != constants::PORTABLE_DIRECTORY
349        {
350            expected.insert(directory.to_owned());
351        }
352    }
353    if configured != expected {
354        return Err(format!(
355            "repository layout differs from required tracked roots: expected {expected:?}, found {configured:?}"
356        ));
357    }
358    for manifest in workspace_package_manifests(&root)? {
359        let readme = manifest.parent().unwrap_or(&root).join("README.md");
360        if !readme.is_file() {
361            return Err(format!(
362                "workspace package has no responsibility README: {}",
363                manifest.display()
364            ));
365        }
366    }
367    for readme in ["scripts/linux/README.md", "scripts/win/README.md"] {
368        if !root.join(readme).is_file() {
369            return Err(format!(
370                "script directory has no responsibility README: {readme}"
371            ));
372        }
373    }
374    verify_test_level_inventory(&root)?;
375    let policy = crate::config::quality_settings::QualityPolicy::parse(&read_workspace_text(
376        &root,
377        constants::QUALITY_GATES_FILE,
378    )?)?;
379    verify_fuzz_ownership(&root, &policy.fuzz)?;
380    verify_generated_file_hygiene(&root)?;
381    verify_ignore_policy(&root)?;
382    crate::output::pass("repository ownership and hygiene");
383    Ok(())
384}

386/// Verifies generated-state ignores without allowing source-of-truth inputs.
387pub(crate) fn verify_ignore_policy(root: &Path) -> Result<(), String> {
388    let ignore = read_workspace_text(root, ".gitignore")?;
389    for required in [
390        "target",
391        "**/mutants.out*/",
392        "fuzz/artifacts/",
393        "fuzz/corpus/*",
394        "!fuzz/corpus/README.md",
395        "fuzz/coverage/",
396        "/quality/evidence/*/gates/",
397        "massif.out.*",
398        "callgrind.out.*",
399        "perf.data*",
400        "*.profraw",
401        "/dist/",
402        "/release/",
403        ".idea/",
404        ".vscode/",
405        "*.swp",
406    ] {
407        if !ignore.lines().any(|line| line.trim() == required) {
408            return Err(format!("generated-state ignore is missing {required}"));
409        }
410    }
411    let results_ignore = format!(
412        "{}/",
413        crate::configuration::automation_at(root)?
414            .output
415            .results_root
416    );
417    if !ignore.lines().any(|line| line.trim() == results_ignore) {
418        return Err(format!(
419            "generated-state ignore is missing {results_ignore}"
420        ));
421    }
422    let bundle = ReleasedBundle::load(root)?;
423    let freeze = bundle.member(crate::constants::BUNDLE_CONTRACT_FREEZE_MEMBER);
424    let manifest = bundle.member("conformance/manifest.toml");
425    for required in [
426        crate::constants::CARGO_LOCK_FILE,
427        "rust-toolchain.toml",
428        constants::AUTOMATION_CONFIG_FILE,
429        constants::CONFORMANCE_CONFIG_FILE,
430        "config/release.toml",
431        &freeze,
432        &manifest,
433        "scripts/linux/bootstrap.sh",
434    ] {
435        let output = Command::new(constants::GIT_COMMAND)
436            .current_dir(root)
437            .args(["ls-files", "--error-unmatch", required])
438            .output()
439            .map_err(|error| format!("could not inspect tracked input {required}: {error}"))?;
440        if !output.status.success() {
441            return Err(format!(
442                "required source-of-truth input is not tracked: {required}"
443            ));
444        }
445    }
446    Ok(())
447}

449/// Verifies every durable test purpose has one documented stable command.
450pub(crate) fn verify_test_level_inventory(root: &Path) -> Result<(), String> {
451    let inventory: crate::configuration_models::TestInventory =
452        crate::configuration::read(root, constants::TEST_LEVELS_FILE)?;
453    crate::configuration::require_schema(inventory.schema_version, "test inventory")?;
454    let mut required = BTreeSet::from([
455        "unit",
456        "smoke",
457        "integration",
458        "system",
459        "conformance",
460        "property",
461        "security",
462        "fuzz-regression",
463        "performance",
464        "all",
465    ]);
466    for entry in inventory.level {
467        if !required.remove(entry.name.as_str()) || entry.owner.trim().is_empty() {
468            return Err(format!(
469                "unknown, duplicate, or unowned test purpose: {}",
470                entry.name
471            ));
472        }
473        let expected = match entry.name.as_str() {
474            "fuzz-regression" => "fuzz smoke".to_owned(),
475            "performance" => "test performance --profile pr".to_owned(),
476            name => format!("test {name}"),
477        };
478        let words = entry.command.split_whitespace().collect::<Vec<_>>();
479        if !words.starts_with(&["cargo", "xtask"]) {
480            return Err(format!("invalid test command for {}", entry.name));
481        }
482        let arguments = words[2..]
483            .iter()
484            .map(|word| (*word).to_owned())
485            .collect::<Vec<_>>();
486        let expected = expected
487            .split_whitespace()
488            .map(str::to_owned)
489            .collect::<Vec<_>>();
490        if crate::interface::parse(&arguments)? != crate::interface::parse(&expected)? {
491            return Err(format!(
492                "test command does not match purpose {}",
493                entry.name
494            ));
495        }
496    }
497    if !required.is_empty() {
498        return Err(format!("test inventory is missing purposes: {required:?}"));
499    }
500    Ok(())
501}

503/// Verifies configured fuzz targets have harnesses, owners, and state policy.
504fn verify_fuzz_ownership(
505    root: &Path,
506    settings: &crate::config::quality_settings::FuzzSettings,
507) -> Result<(), String> {
508    for target in &settings.targets {
509        if !root
510            .join("fuzz/fuzz_targets")
511            .join(format!("{target}.rs"))
512            .is_file()
513        {
514            return Err(format!(
515                "fuzz target {target} has no harness or subsystem owner"
516            ));
517        }
518    }
519    Ok(())
520}

522/// Rejects tracked generated products while retaining documented empty roots.
523pub(crate) fn verify_generated_file_hygiene(root: &Path) -> Result<(), String> {
524    git_hygiene::verify_no_ignored_tracked_files(root)?;
525    let result_directory = crate::configuration::automation_at(root)?
526        .output
527        .results_root;
528    let output = Command::new(constants::GIT_COMMAND)
529        .current_dir(root)
530        .args([
531            "ls-files",
532            "--",
533            "target",
534            "mutants.out",
535            "mutants.out.old",
536            "fuzz/artifacts",
537            "fuzz/corpus",
538            "portable/archive",
539            "portable/archived",
540        ])
541        .arg(result_directory)
542        .output()
543        .map_err(|error| format!("could not inspect tracked generated files: {error}"))?;
544    if !output.status.success() {
545        return Err(format!(
546            "could not inspect tracked generated files: {}",
547            String::from_utf8_lossy(&output.stderr).trim()
548        ));
549    }
550    let tracked = String::from_utf8(output.stdout)
551        .map_err(|error| format!("Git emitted non-UTF-8 paths: {error}"))?
552        .lines()
553        .filter(|path| *path != "fuzz/corpus/README.md")
554        .map(str::to_owned)
555        .collect::<Vec<_>>();
556    if tracked.is_empty() {
557        Ok(())
558    } else {
559        Err(format!(
560            "generated or archived products are tracked: {}",
561            tracked.join(", ")
562        ))
563    }
