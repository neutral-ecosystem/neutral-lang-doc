11/// Builds every workspace crate's local API documentation.
12pub(crate) fn documentation() -> Result<(), String> {
13    run_rustdoc()?;
14    let metadata = serde_json::to_string(&cargo_discovery::metadata(&workspace_root()?, false)?)
15        .map_err(|error| format!("could not serialize Cargo metadata: {error}"))?;
16    let index = render_rustdoc_index(&metadata)?;
17    let output_directory = cargo_target_directory()?.join("doc");
18    fs::create_dir_all(&output_directory).map_err(|error| {
19        format!(
20            "could not create rustdoc directory {}: {error}",
21            output_directory.display()
22        )
23    })?;
24    let output_path = output_directory.join(constants::RUSTDOC_INDEX_FILE);
25    fs::write(&output_path, index)
26        .map_err(|error| format!("could not write {}: {error}", output_path.display()))?;
27    copy_documentation_assets(&workspace_root()?, &output_directory)?;
28    output::file("workspace documentation", &output_path);
29    Ok(())
30}

32/// Copies repository-owned brand assets into generated documentation output.
33///
34/// # Errors
35///
36/// Returns an error when a required source asset is missing or cannot be copied.
37pub(crate) fn copy_documentation_assets(
38    root: &Path,
39    output_directory: &Path,
40) -> Result<(), String> {
41    let source_directory = root.join(constants::ASSET_DIRECTORY);
42    let destination_directory = output_directory.join(constants::DOCUMENTATION_ASSET_DIRECTORY);
43    fs::create_dir_all(&destination_directory).map_err(|error| {
44        format!(
45            "could not create documentation asset directory {}: {error}",
46            destination_directory.display()
47        )
48    })?;
49    for filename in constants::DOCUMENTATION_ASSET_FILES {
50        let source = source_directory.join(filename);
51        let destination = destination_directory.join(filename);
52        if !source.is_file() {
53            return Err(format!(
54                "required documentation asset is missing: {}",
55                source.display()
56            ));
57        }
58        fs::copy(&source, &destination).map_err(|error| {
59            format!(
60                "could not copy documentation asset {} to {}: {error}",
61                source.display(),
62                destination.display()
63            )
64        })?;
65    }
66    Ok(())
67}

69/// Runs workspace rustdoc with the shared navigation header on every HTML page.
70pub(crate) fn run_rustdoc() -> Result<(), String> {
71    let header_path = workspace_root()?.join(constants::RUSTDOC_HEADER_FILE);
72    if !header_path.is_file() {
73        return Err(format!(
74            "rustdoc navigation header is missing: {}",
75            header_path.display()
76        ));
77    }
78    let header_path = header_path
79        .to_str()
80        .ok_or_else(|| "rustdoc navigation header path is not valid UTF-8".to_owned())?;
81    let mut rustdoc_flags = env::var(constants::CARGO_ENCODED_RUSTDOCFLAGS).unwrap_or_default();
82    if !rustdoc_flags.is_empty() {
83        rustdoc_flags.push(constants::RUSTDOC_FLAG_SEPARATOR);
84    }
85    rustdoc_flags.push_str(constants::RUSTDOC_HTML_HEADER_FLAG);
86    rustdoc_flags.push(constants::RUSTDOC_FLAG_SEPARATOR);
87    rustdoc_flags.push_str(header_path);
88    rustdoc_flags.push(constants::RUSTDOC_FLAG_SEPARATOR);
89    rustdoc_flags.push_str("-D");
90    rustdoc_flags.push(constants::RUSTDOC_FLAG_SEPARATOR);
91    rustdoc_flags.push_str("warnings");
92    rustdoc_flags.push(constants::RUSTDOC_FLAG_SEPARATOR);
93    rustdoc_flags.push_str("--cfg");
94    rustdoc_flags.push(constants::RUSTDOC_FLAG_SEPARATOR);
95    rustdoc_flags.push_str(&rustdoc_header_configuration());
96
97    let arguments = ["doc", flags::WORKSPACE, "--no-deps"];
98    let cargo = cargo_command()?;
99    output::invocation(&cargo, &arguments);
100    let status = Command::new(&cargo)
101        .current_dir(workspace_root()?)
102        .env(constants::CARGO_TERM_COLOR_ENV, output::child_color())
103        .env(constants::CARGO_ENCODED_RUSTDOCFLAGS, rustdoc_flags)
104        .args(arguments)
105        .status()
106        .map_err(|error| format!("could not run {} {}: {error}", cargo, arguments.join(" ")))?;
107    status
108        .success()
109        .then_some(())
110        .ok_or_else(|| format!("{} {} failed with {status}", cargo, arguments.join(" ")))
111}

113/// Derives a stable non-security cache token from the shared rustdoc header.
114pub(crate) fn rustdoc_header_configuration() -> String {
115    let mut hash = constants::RUSTDOC_HEADER_HASH_OFFSET;
116    for byte in RUSTDOC_HEADER_TEMPLATE.bytes() {
117        hash ^= u64::from(byte);
118        hash = hash.wrapping_mul(constants::RUSTDOC_HEADER_HASH_PRIME);
119    }
120    format!("{}_{hash:016x}", constants::RUSTDOC_HEADER_CFG_PREFIX)
121}

123/// Embeds Cargo metadata safely into the workspace rustdoc index template.
124pub(crate) fn render_rustdoc_index(metadata: &str) -> Result<String, String> {
125    if !RUSTDOC_INDEX_TEMPLATE.contains(constants::CARGO_METADATA_PLACEHOLDER) {
126        return Err("rustdoc index template has no Cargo metadata placeholder".to_owned());
127    }
128    let safe_metadata = metadata.replace('<', "\\u003c");
129    Ok(RUSTDOC_INDEX_TEMPLATE.replacen(constants::CARGO_METADATA_PLACEHOLDER, &safe_metadata, 1))
130}
