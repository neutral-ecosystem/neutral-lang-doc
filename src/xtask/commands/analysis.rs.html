11/// Runs the selected durable fuzz mode.
12pub(crate) fn fuzz(mode: FuzzMode) -> Result<(), String> {
13    match mode {
14        FuzzMode::Smoke => run_active_test_filter("fuzz_smoke"),
15        FuzzMode::Campaign => coverage_guided_fuzz_campaign(),
16    }
17}

19/// Runs every configured coverage-guided fuzz target for its approved budget.
20pub(crate) fn coverage_guided_fuzz_campaign() -> Result<(), String> {
21    let mut measurement = quality_evidence::Measurement::begin("fuzz")?;
22    let targets = quality_array("fuzz", "targets")?;
23    let seconds = quality_value("fuzz", "minimum_seconds_per_target")?
24        .parse::<u64>()
25        .map_err(|error| format!("invalid fuzz budget: {error}"))?;
26    if seconds == 0 {
27        return Err("fuzz budget must be positive".to_owned());
28    }
29    let root = workspace_root()?;
30    let corpus_root = PathBuf::from(quality_value("fuzz", "corpus_root")?);
31    let seed_root = PathBuf::from(quality_value("fuzz", "seed_root")?);
32    if !is_safe_relative_path(&corpus_root) || !is_safe_relative_path(&seed_root) {
33        return Err("fuzz corpus and seed roots must be safe workspace-relative paths".to_owned());
34    }
35    let total = targets.len();
36    output::info(format!(
37        "fuzz campaign: {total} targets | budget {} each | estimated total {} + build/startup",
38        output::duration(std::time::Duration::from_secs(seconds)),
39        output::duration(std::time::Duration::from_secs(
40            seconds.saturating_mul(total as u64)
41        ))
42    ));
43    for (index, target) in targets.into_iter().enumerate() {
44        output::info(format!("fuzz [{}/{total}]: {target}", index + 1));
45        if !is_safe_relative_path(Path::new(&target)) {
46            return Err(format!("unsafe fuzz target name: {target}"));
47        }
48        let seed_directory = root.join(&seed_root).join(&target);
49        let corpus_directory = corpus_root.join(&target);
50        let corpus = corpus_directory
51            .to_str()
52            .ok_or_else(|| "fuzz corpus path is not UTF-8".to_owned())?;
53        let budget = format!("-max_total_time={seconds}");
54        if seed_directory.is_dir() {
55            let seed = seed_directory
56                .to_str()
57                .ok_or_else(|| "fuzz seed path is not UTF-8".to_owned())?;
58            measurement.cargo_with_progress(
59                &["fuzz", "run", &target, corpus, seed, "--", &budget],
60                &target,
61                seconds,
62            )?;
63        } else {
64            measurement.cargo_with_progress(
65                &["fuzz", "run", &target, "--", &budget],
66                &target,
67                seconds,
68            )?;
69        }
70    }
71    measurement.finish()
72}

74/// Runs workspace coverage and enforces every configured percentage threshold.
75pub(crate) fn coverage() -> Result<(), String> {
76    let mut measurement = quality_evidence::Measurement::begin("coverage")?;
77    let lines = quality_value("coverage", "minimum_line_percent")?;
78    let functions = quality_value("coverage", "minimum_function_percent")?;
79    let regions = quality_value("coverage", "minimum_region_percent")?;
80    let exclusions = quality_value("coverage", "exclusion_regex")?;
81    let html_index = quality_output_path("coverage", "html_output")?;
82    let html = html_index
83        .parent()
84        .and_then(Path::parent)
85        .ok_or_else(|| "coverage HTML output must be below a report directory".to_owned())?;
86    let json = quality_output_path("coverage", "json_output")?;
87    fs::create_dir_all(
88        json.parent()
89            .ok_or_else(|| "coverage JSON output has no parent".to_owned())?,
90    )
91    .map_err(|error| format!("could not create coverage result directory: {error}"))?;
92    let html = html
93        .to_str()
94        .ok_or_else(|| "coverage HTML path is not valid UTF-8".to_owned())?;
95    let json = json
96        .to_str()
97        .ok_or_else(|| "coverage JSON path is not valid UTF-8".to_owned())?;
98    measurement.cargo(
99        &["llvm-cov", "--workspace", "--all-targets", "--no-report"],
100        "tests",
101    )?;
102    measurement.cargo(
103        &[
104            "llvm-cov",
105            "report",
106            "--html",
107            "--output-dir",
108            html,
109            "--ignore-filename-regex",
110            &exclusions,
111            "--fail-under-lines",
112            &lines,
113            "--fail-under-functions",
114            &functions,
115            "--fail-under-regions",
116            &regions,
117        ],
118        "html",
119    )?;
120    measurement.cargo(
121        &[
122            "llvm-cov",
123            "report",
124            "--json",
125            "--summary-only",
126            "--output-path",
127            json,
128            "--ignore-filename-regex",
129            &exclusions,
130            "--fail-under-lines",
131            &lines,
132            "--fail-under-functions",
133            &functions,
134            "--fail-under-regions",
135            &regions,
136        ],
137        "json",
138    )?;
139    output::file("coverage HTML", &html_index);
140    output::file("coverage JSON", Path::new(json));
141    measurement.copy_report(Path::new(json), "coverage.json")?;
142    measurement.finish()
143}

145/// Runs mutation analysis for the configured critical production target.
146pub(crate) fn mutate() -> Result<(), String> {
147    let target = quality_value("mutation", "critical_target")?;
148    let mut measurement = quality_evidence::Measurement::begin("mutation")?;
149    let output = measurement.directory.to_string_lossy().into_owned();
150    measurement.cargo(
151        &["mutants", "--file", &target, "--output", &output],
152        "mutation",
153    )?;
154    let outcomes = measurement.directory.join("mutants.out/outcomes.json");
155    measurement.copy_report(&outcomes, "outcomes.json")?;
156    measurement.finish()
157}

159/// Runs one controlled benchmark, stress, or soak profile.
160pub(crate) fn performance(profile: PerformanceProfile) -> Result<(), String> {
161    let measured = !matches!(profile, PerformanceProfile::Pr);
162    let gate = if matches!(profile, PerformanceProfile::Soak) {
163        "performance-soak"
164    } else {
165        "performance-release"
166    };
167    let profile = match profile {
168        PerformanceProfile::Pr => "pr",
169        PerformanceProfile::Release => "release",
170        PerformanceProfile::Soak => "extended-soak",
171    };
172    match profile {
173        "pr" | "release" | "extended-soak" => {
174            let harness = quality_value("performance", "harness")?;
175            let (package, target) = harness
176                .split_once('/')
177                .ok_or_else(|| "quality performance harness must be package/target".to_owned())?;
178            if package != constants::NEUTRAL_BENCH
179                || target.is_empty()
180                || !target
181                    .bytes()
182                    .all(|byte| byte.is_ascii_alphanumeric() || byte == b'-' || byte == b'_')
183            {
184                return Err(format!("invalid quality performance harness: {harness}"));
185            }
186            let arguments = [
187                "bench",
188                "--package",
189                package,
190                "--bench",
191                target,
192                "--",
193                profile,
194            ];
195            if !measured {
196                return run_cargo(&arguments);
197            }
198            let mut measurement = quality_evidence::Measurement::begin(gate)?;
199            measurement.cargo(&arguments, "benchmark")?;
200            measurement.cargo(
201                &[
202                    "bench",
203                    "--package",
204                    package,
205                    "--bench",
206                    target,
207                    "--no-run",
208                    "--message-format=json",
209                ],
210                "build",
211            )?;
212            let build = fs::read_to_string(measurement.directory.join("build.stdout"))
213                .map_err(|error| format!("could not read benchmark build: {error}"))?;
214            let executable = cargo_discovery::benchmark_executable(&build, target)?;
215            let heap_report = measurement.directory.join("massif.out");
216            let heap_argument = format!("--massif-out-file={}", heap_report.display());
217            measurement.run(
218                constants::VALGRIND_COMMAND,
219                &["--tool=massif", &heap_argument, &executable, profile],
220                "massif",
221            )?;
222            measurement.copy_report(&heap_report, "massif.out")?;
223            measurement.run(
224                constants::VALGRIND_COMMAND,
225                &[
226                    "--tool=memcheck",
227                    "--leak-check=full",
228                    "--errors-for-leak-kinds=definite,indirect,possible",
229                    "--error-exitcode=1",
230                    &executable,
231                    profile,
232                ],
233                "memcheck",
234            )?;
235            measurement.finish()
236        }
237        _ => unreachable!("performance profile is closed by command parsing"),
238    }
