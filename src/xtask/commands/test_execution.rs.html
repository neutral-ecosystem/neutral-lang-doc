11/// Existing explicit categories excluded by the focused crate-local unit selection.
12const NON_UNIT_CATEGORIES: &[&str] = &[
13    "integration_",
14    "system_",
15    "conformance_",
16    "property_",
17    "security_",
18    "smoke_",
19    "determinism_",
20    "fuzz_",
21];

23/// One invocation's backend, profiles, output policy and executable; never cached globally.
24pub(crate) struct TestExecution {
25    /// Effective backend after environment overrides.
26    pub(crate) backend: TestRunner,
27    /// Parsed profiles and selection policy for the invocation.
28    pub(crate) testing: crate::configuration_models::Testing,
29    /// Effective successful-test reporting policy.
30    verbose: bool,
31    /// Effective Cargo executable, resolved once.
32    cargo: String,
33}
34impl TestExecution {
35    /// Loads configuration and environment overrides once for a complete test invocation.
36    pub(crate) fn load() -> Result<Self, String> {
37        let config = configuration::automation()?;
38        let backend = match env::var(constants::TEST_RUNNER_ENV) {
39            Ok(value) => parse_runner(&value)?,
40            Err(env::VarError::NotPresent) => config.testing.runner,
41            Err(error) => return Err(format!("invalid {}: {error}", constants::TEST_RUNNER_ENV)),
42        };
43        let verbose = match env::var(constants::TEST_VERBOSE_ENV) {
44            Ok(value) => parse_verbose(&value)?,
45            Err(env::VarError::NotPresent) => config.testing.verbose,
46            Err(error) => return Err(format!("invalid {}: {error}", constants::TEST_VERBOSE_ENV)),
47        };
48        let cargo =
49            configuration::configured_command(constants::CARGO_COMMAND_ENV, config.tools.cargo)?;
50        Ok(Self {
51            backend,
52            testing: config.testing,
53            verbose,
54            cargo,
55        })
56    }
57    /// Checks the selected backend using the same executable and config as execution.
58    pub(crate) fn verify(&self) -> Result<(), String> {
59        if self.backend == TestRunner::Nextest {
60            crate::command_output(&self.cargo, &["nextest", flags::VERSION])
61                .map_err(|error| format!("nextest is required: {error}; run cargo install cargo-nextest --locked (or explicitly set {}=cargo)", constants::TEST_RUNNER_ENV))?;
62            crate::read_workspace_text(&crate::workspace_root()?, &self.testing.config)?;
63        }
64        Ok(())
65    }
66    /// Runs through the shared subprocess adapter without resolving configuration again.
67    fn run(&self, arguments: &[String]) -> Result<(), String> {
68        crate::run_program(
69            std::path::Path::new(&self.cargo),
70            &arguments
71                .iter()
72                .map(std::ffi::OsStr::new)
73                .collect::<Vec<_>>(),
74        )
75    }
76}
77
78/// Validates the supported backend names for command-line environment overrides.
79fn parse_runner(value: &str) -> Result<TestRunner, String> {
80    match value {
81        "nextest" => Ok(TestRunner::Nextest),
82        "cargo" => Ok(TestRunner::Cargo),
83        _ => Err(format!(
84            "{} must be nextest or cargo; found {value:?}",
85            constants::TEST_RUNNER_ENV
86        )),
87    }
88}
89
90/// Verifies the selected runner early and explains how to install it.
91pub(crate) fn verify_runner() -> Result<(), String> {
92    TestExecution::load()?.verify()
93}

95impl TestExecution {
96    /// Builds a locked command using the repository's configured runner/profile and target selection.
97    fn arguments(
98        &self,
99        runner: TestRunner,
100        action: &str,
101        unit: bool,
102        full_gate: bool,
103        filter: Option<&str>,
104    ) -> Result<Vec<String>, String> {
105        let mut arguments: Vec<String> = match runner {
106            TestRunner::Cargo => vec!["test".to_owned()],
107            TestRunner::Nextest => {
108                let testing = &self.testing;
109                vec![
110                    "nextest".to_owned(),
111                    action.to_owned(),
112                    "--config-file".to_owned(),
113                    crate::workspace_root()?
114                        .join(&testing.config)
115                        .to_string_lossy()
116                        .into_owned(),
117                    "--profile".to_owned(),
118                    if full_gate {
119                        testing.ci_profile.clone()
120                    } else {
121                        testing.profile.clone()
122                    },
123                ]
124            }
125        };
126        arguments.extend([flags::WORKSPACE, flags::LOCKED, "--lib", "--bins"].map(str::to_owned));
127        if unit {
128            arguments.extend(["--exclude", constants::NEUTRAL_TEST_SUITE].map(str::to_owned));
129        }
130        if !unit {
131            arguments.push("--tests".to_owned());
132        }
133        if runner == TestRunner::Nextest {
134            // Default-filter customization must not turn a full gate into a subset run.
135            arguments.push("--ignore-default-filter".to_owned());
136            if action == "run" {
137                arguments.extend(["--no-tests", "fail"].map(str::to_owned));
138            }
139            if let Some(filter) = filter {
140                arguments.extend(["-E".to_owned(), format!("test(/(^|::){filter}_/)")]);
141            }
142            if unit {
143                let excluded = NON_UNIT_CATEGORIES
144                    .iter()
145                    .map(|category| format!("test(~{category})"))
146                    .collect::<Vec<_>>()
147                    .join(" | ");
148                arguments.extend(["-E".to_owned(), format!("not ({excluded})")]);
149            }
150        } else if let Some(filter) = filter {
151            arguments.extend(["--".to_owned(), format!("{filter}_")]);
152        }
153        if runner == TestRunner::Cargo && unit {
154            arguments.push("--".to_owned());
155            for category in NON_UNIT_CATEGORIES {
156                arguments.extend(["--skip".to_owned(), (*category).to_owned()]);
157            }
158        }
159        if action == "run" {
160            reporting_arguments(&mut arguments, runner, self.verbose);
161        }
162        Ok(arguments)
163    }
164
165    /// Instruments the complete target set while reusing the selected test runner's policy.
166    pub(crate) fn coverage_arguments(&self, backend: TestRunner) -> Result<Vec<String>, String> {
167        let tests = self.arguments(backend, "run", false, true, None)?;
168        let mut command = vec!["llvm-cov".to_owned()];
169        let prefix = if backend == TestRunner::Nextest {
170            command.push("nextest".to_owned());
171            2
172        } else {
173            1
174        };
175        command.extend(["--all-targets", "--no-report"].map(str::to_owned));
176        command.extend(
177            tests
178                .into_iter()
179                .skip(prefix)
180                .filter(|argument| !matches!(argument.as_str(), "--lib" | "--bins" | "--tests")),
181        );
182        Ok(command)
183    }
184}

186/// Selects cargo-mutants' native integration with the configured test backend.
187pub(crate) fn mutation_arguments(
188    backend: TestRunner,
189    targets: &[String],
190    output: &str,
191    testing: &crate::configuration_models::Testing,
192) -> Vec<String> {
193    let tool = backend.as_str();
194    let mut command = ["mutants", "--test-tool", tool, "--output", output]
195        .map(str::to_owned)
196        .to_vec();
197    for target in targets {
198        command.extend(["--file".to_owned(), target.clone()]);
199    }
200    command.push("--no-config".to_owned());
201    // Baseline timing must include the same workspace tested for each mutant.
202    command.push("--cargo-arg=--workspace".to_owned());
203    command.extend(["--jobs".to_owned(), testing.mutation_jobs.to_string()]);
204    // Generated nested build trees can exceed the temporary workspace quota.
205    command.extend(["--gitignore".to_owned(), "true".to_owned()]);
206    // Workspace tests must resolve Git policy within the copy, not its parent checkout.
207    command.extend(["--copy-vcs".to_owned(), "true".to_owned()]);
208    if backend == TestRunner::Nextest {
209        // cargo-mutants tests copied workspaces: resolve config relative to that copy.
210        command.extend([
211            "--".to_owned(),
212            "--config-file".to_owned(),
213            testing.config.clone(),
214            "--profile".to_owned(),
215            testing.mutation_profile.clone(),
216            "--fail-fast".to_owned(),
217            "--ignore-default-filter".to_owned(),
218            "--no-tests".to_owned(),
219            "fail".to_owned(),
220        ]);
221    }
222    command
223}

225/// Accepts conventional Boolean spellings for the per-command verbosity override.
226fn parse_verbose(value: &str) -> Result<bool, String> {
227    match value {
228        "true" | "1" => Ok(true),
229        "false" | "0" => Ok(false),
230        _ => Err(format!(
231            "{} must be true, false, 1, or 0",
232            constants::TEST_VERBOSE_ENV
233        )),
234    }
235}

237/// Reduces successful-test noise while preserving summaries and failure diagnostics.
238fn reporting_arguments(arguments: &mut Vec<String>, runner: TestRunner, verbose: bool) {
239    if runner == TestRunner::Nextest {
240        let level = if verbose { "pass" } else { "slow" };
241        arguments.extend(
242            [
243                "--status-level",
244                level,
245                "--final-status-level",
246                "fail",
247                "--show-progress",
248                "auto",
249            ]
250            .map(str::to_owned),
251        );
252    } else if !verbose {
253        // Libtest options belong after the separator, including filtered runs.
254        if !arguments.iter().any(|argument| argument == "--") {
255            arguments.push("--".to_owned());
256        }
257        arguments.push("--quiet".to_owned());
258    }
259}

261/// Executes test binaries with the selected backend; shell smoke stays in its host adapter.
262pub(crate) fn run(level: TestLevel, full_gate: bool) -> Result<(), String> {
263    if level == TestLevel::Smoke {
264        return crate::run_shell_smoke();
265    }
266    let filter = match level {
267        TestLevel::Integration => Some("integration"),
268        TestLevel::System => Some("system"),
269        TestLevel::Conformance => Some("conformance"),
270        TestLevel::Property => Some("property"),
271        TestLevel::Security => Some("security"),
272        _ => None,
273    };
274    let execution = TestExecution::load()?;
275    execute(&execution, level == TestLevel::Unit, full_gate, filter)?;
276    if level == TestLevel::All {
277        verify_counts(&execution, configuration::active_test_profile(), full_gate)?;
278        // Nextest cannot run Rustdoc tests; every complete run includes them separately.
279        execution.run(&["test", flags::WORKSPACE, "--doc", flags::LOCKED].map(str::to_owned))?;
280    }
281    Ok(())
282}

284/// Runs one named regression category, including the bounded fuzz-smoke suite.
285pub(crate) fn run_filter(filter: &str) -> Result<(), String> {
286    if !filter
287        .bytes()
288        .all(|byte| byte.is_ascii_alphanumeric() || byte == b'_')
289        || filter.is_empty()
290    {
291        return Err("test filter must be a nonempty category identifier".to_owned());
292    }
293    execute(&TestExecution::load()?, false, false, Some(filter))
294}

296/// Executes a command with inherited streams and a visible backend/profile selection.
297fn execute(
298    execution: &TestExecution,
299    unit: bool,
300    full_gate: bool,
301    filter: Option<&str>,
302) -> Result<(), String> {
303    execution.verify()?;
304    let backend = execution.backend;
305    let arguments = execution.arguments(backend, "run", unit, full_gate, filter)?;
306    crate::output::info(format!(
307        "tests: {:?}{}",
308        backend,
309        if full_gate { " (full gate)" } else { "" }
310    ));
311    execution.run(&arguments)
312}

314/// Counts actual runnable test names, never occurrences in binary paths, banners, or metadata.
315fn counts<'a>(
316    minimums: &BTreeMap<String, usize>,
317    names: impl Iterator<Item = &'a str>,
318) -> BTreeMap<String, usize> {
319    let mut counts = minimums
320        .keys()
321        .map(|name| (name.clone(), 0))
322        .collect::<BTreeMap<_, _>>();
323    for name in names {
324        let leaf = name.rsplit("::").next().unwrap_or(name);
325        for (category, count) in &mut counts {
326            if leaf
327                .strip_prefix(category.as_str())
328                .is_some_and(|suffix| suffix.starts_with('_'))
329            {
330                *count += 1;
331            }
332        }
333    }
334    counts
335}

337/// Parses Cargo's explicit libtest records when the compatibility backend is selected.
338fn cargo_test_names(list: &str) -> impl Iterator<Item = &str> {
339    list.lines().filter_map(|line| line.strip_suffix(": test"))
340}

342/// Verifies category minima using nextest's own typed listing format or explicit libtest records.
343fn verify_counts(execution: &TestExecution, profile: &str, full_gate: bool) -> Result<(), String> {
344    let minimums = configuration::test_minimums(profile)?;
345    let backend = execution.backend;
346    let mut args = execution.arguments(backend, "list", false, full_gate, None)?;
347    if backend == TestRunner::Nextest {
348        args.extend(["--message-format", "json-pretty"].map(str::to_owned));
349    } else {
350        args.extend(["--", "--list"].map(str::to_owned));
351    }
352    let output = crate::command_output(
353        &execution.cargo,
354        &args.iter().map(String::as_str).collect::<Vec<_>>(),
355    )?;
356    let discovered = if backend == TestRunner::Nextest {
357        let listing = TestListSummary::parse_json(&output)
358            .map_err(|error| format!("invalid nextest inventory: {error}"))?;
359        let actual = listing
360            .rust_suites
361            .values()
362            .map(|suite| suite.test_cases.len())
363            .sum::<usize>();
364        if actual != listing.test_count {
365            return Err("nextest inventory test count is inconsistent".to_owned());
366        }
367        let names = listing
368            .rust_suites
369            .values()
370            .filter(|suite| suite.status == RustTestSuiteStatusSummary::LISTED)
371            .flat_map(|suite| &suite.test_cases)
372            .filter(|(_, case)| !case.ignored && case.filter_match.is_match())
373            .map(|(name, _)| name.as_str());
374        let discovered = counts(&minimums, names);
375        let directory = crate::unique_generated_directory(&crate::result_root()?.join("tests"))?;
376        std::fs::write(directory.join("inventory.json"), format!("{output}\n"))
377            .map_err(|error| format!("could not write test inventory: {error}"))?;
378        discovered
379    } else {
380        let mut discovered = counts(&minimums, cargo_test_names(&output));
381        // Libtest's default listing includes ignored tests; subtract its explicit ignored-only listing.
382        args.push("--ignored".to_owned());
383        let ignored = crate::command_output(
384            &configuration::cargo_command()?,
385            &args.iter().map(String::as_str).collect::<Vec<_>>(),
386        )?;
387        let ignored = counts(&minimums, cargo_test_names(&ignored));
388        for (category, count) in &mut discovered {
389            *count = count
390                .checked_sub(ignored[category])
391                .ok_or("Cargo test inventory has inconsistent ignored counts")?;
392        }
393        discovered
394    };
395    crate::validate_test_minimums(&minimums, &discovered)
396}
