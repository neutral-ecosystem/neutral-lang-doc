14/// Executes the active portable lifecycle command family.
15pub(crate) fn portable(action: PortableAction) -> Result<(), String> {
16    match action {
17        PortableAction::Install(source) => install_portable(&source),
18        PortableAction::Verify => verify_portable(),
19        PortableAction::Snapshot => snapshot_portable(),
20    }
21}

23/// Atomically installs and verifies one reviewed active portable package.
24pub(crate) fn install_portable(source: &Path) -> Result<(), String> {
25    let root = workspace_root()?;
26    let destination = root.join(constants::PORTABLE_DIRECTORY);
27    if destination.exists() {
28        return Err(format!(
29            "active portable destination already exists: {}",
30            destination.display()
31        ));
32    }
33    let source = fs::canonicalize(source).map_err(|error| {
34        format!(
35            "could not resolve portable source {}: {error}",
36            source.display()
37        )
38    })?;
39    if !source.is_dir() {
40        return Err(format!(
41            "portable source is not a directory: {}",
42            source.display()
43        ));
44    }
45    let staging = root.join(format!(".portable.install-{}", std::process::id()));
46    if staging.exists() {
47        return Err(format!(
48            "portable installation staging path already exists: {}",
49            staging.display()
50        ));
51    }
52    if let Err(error) = copy_portable_tree(&source, &staging) {
53        let _ = fs::remove_dir_all(&staging);
54        return Err(error);
55    }
56    fs::rename(&staging, &destination).map_err(|error| {
57        format!(
58            "could not publish portable package {} as {}: {error}",
59            staging.display(),
60            destination.display()
61        )
62    })?;
63    if let Err(error) = verify_portable() {
64        let rejected_root = result_root()?.join("portable/rejected");
65        let rejected_run = match unique_generated_directory(&rejected_root) {
66            Ok(directory) => directory,
67            Err(move_error) => {
68                fs::rename(&destination, &staging).map_err(|restore_error| {
69                    format!(
70                        "{error}; could not allocate rejected-package storage: {move_error}; could not retain the package at {}: {restore_error}",
71                        staging.display()
72                    )
73                })?;
74                return Err(format!(
75                    "{error}; could not allocate rejected-package storage: {move_error}; rejected package retained at {}",
76                    staging.display()
77                ));
78            }
79        };
80        let rejected = rejected_run.join(constants::PORTABLE_DIRECTORY);
81        fs::rename(&destination, &rejected).map_err(|move_error| {
82            format!(
83                "{error}; could not move rejected package {} to {}: {move_error}",
84                destination.display(),
85                rejected.display()
86            )
87        })?;
88        return Err(format!(
89            "{error}; rejected package retained at {}",
90            rejected.display()
91        ));
92    }
93    crate::output::pass("active portable installed");
94    crate::output::file("portable source", &source);
95    Ok(())
96}

98/// Copies a portable directory without following symbolic links or special files.
99pub(crate) fn copy_portable_tree(source: &Path, destination: &Path) -> Result<(), String> {
100    fs::create_dir(destination)
101        .map_err(|error| format!("could not create {}: {error}", destination.display()))?;
102    for entry in fs::read_dir(source)
103        .map_err(|error| format!("could not inspect {}: {error}", source.display()))?
104    {
105        let entry = entry.map_err(|error| format!("could not inspect portable entry: {error}"))?;
106        let file_type = entry
107            .file_type()
108            .map_err(|error| format!("could not inspect {}: {error}", entry.path().display()))?;
109        let target = destination.join(entry.file_name());
110        if file_type.is_dir() {
111            copy_portable_tree(&entry.path(), &target)?;
112        } else if file_type.is_file() {
113            fs::copy(entry.path(), &target).map_err(|error| {
114                format!(
115                    "could not copy {} to {}: {error}",
116                    entry.path().display(),
117                    target.display()
118                )
119            })?;
120        } else {
121            return Err(format!(
122                "portable source contains a symbolic link or special file: {}",
123                entry.path().display()
124            ));
125        }
126    }
127    Ok(())
128}

130/// Verifies the active portable package's required roots and traceability.
131pub(crate) fn verify_portable() -> Result<(), String> {
132    let root = workspace_root()?;
133    verify_portable_layout(&root)?;
134    let lifecycle = read_workspace_text(&root, constants::PORTABLE_LIFECYCLE_FILE)?;
135    let active_series = configuration_value(&lifecycle, "active_series")
136        .ok_or_else(|| "active portable lifecycle has no active_series".to_owned())?;
137    let status = configuration_value(&lifecycle, "status")
138        .ok_or_else(|| "active portable lifecycle has no status".to_owned())?;
139    if status != "active" {
140        return Err(format!(
141            "installed portable status must be active; found {status:?}"
142        ));
143    }
144    if !is_portable_series(&active_series) {
145        return Err(format!(
146            "active portable series must be a numeric v-prefixed identifier; found {active_series:?}"
147        ));
148    }
149    check_portable_traceability_at(&root)?;
150    verify_frozen_input_digests(&root, constants::PORTABLE_CONTRACT_FREEZE_FILE)?;
151    verify_portable_links(&root)?;
152    ensure_no_portable_archive_dependencies(&root)?;
153    verify_existing_portable_snapshots(&root)?;
154    crate::output::pass("active portable package");
155    Ok(())
156}

158/// Verifies the version-independent files and directories required from every portable package.
159pub(crate) fn verify_portable_layout(root: &Path) -> Result<(), String> {
160    for relative in [
161        constants::PORTABLE_PLAN_FILE,
162        constants::PORTABLE_LIFECYCLE_FILE,
163        constants::PORTABLE_REQUIREMENTS_FILE,
164        constants::PORTABLE_TRACEABILITY_FILE,
165        constants::PORTABLE_CONTRACT_FREEZE_FILE,
166        constants::PORTABLE_CONFORMANCE_MANIFEST_FILE,
167    ] {
168        if !root.join(relative).is_file() {
169            return Err(format!("active portable file is missing: {relative}"));
170        }
171        if read_workspace_text(root, relative)?.trim().is_empty() {
172            return Err(format!("active portable file is empty: {relative}"));
173        }
174    }
175    for relative in [
176        constants::PORTABLE_CONTRACT_DIRECTORY,
177        constants::PORTABLE_FIXTURE_DIRECTORY,
178        constants::PORTABLE_ORACLE_DIRECTORY,
179    ] {
180        if !root.join(relative).is_dir() {
181            return Err(format!("active portable directory is missing: {relative}"));
182        }
183    }
184    Ok(())
185}

187/// Returns whether a portable series is a nonempty numeric `v` identifier.
188pub(crate) fn is_portable_series(value: &str) -> bool {
189    value.strip_prefix('v').is_some_and(|digits| {
190        !digits.is_empty() && digits.chars().all(|character| character.is_ascii_digit())
191    })
192}

194/// Verifies an active portable package when one is installed for development.
195pub(crate) fn verify_optional_portable() -> Result<(), String> {
196    let root = workspace_root()?;
197    if root.join(constants::PORTABLE_DIRECTORY).exists() {
198        verify_portable()
199    } else {
200        crate::output::info(
201            "no active portable package installed; release conformance remains available",
202        );
203        Ok(())
204    }
205}

207/// Verifies every locally retained digest-addressed portable snapshot.
208pub(crate) fn verify_existing_portable_snapshots(root: &Path) -> Result<(), String> {
209    let snapshot_root = result_root()?.join(constants::PORTABLE_SNAPSHOT_DIRECTORY);
210    if !snapshot_root.exists() {
211        return Ok(());
212    }
213    for entry in fs::read_dir(&snapshot_root)
214        .map_err(|error| format!("could not inspect {}: {error}", snapshot_root.display()))?
215    {
216        let entry = entry.map_err(|error| format!("could not inspect snapshot entry: {error}"))?;
217        if !entry
218            .file_type()
219            .map_err(|error| format!("could not inspect {}: {error}", entry.path().display()))?
220            .is_dir()
221            || entry.file_name().to_string_lossy().starts_with('.')
222        {
223            continue;
224        }
225        verify_portable_snapshot_directory(root, &entry.path())?;
226    }
227    Ok(())
228}

230/// Verifies one snapshot manifest, directory digest, and copied file set.
231pub(crate) fn verify_portable_snapshot_directory(
232    root: &Path,
233    directory: &Path,
234) -> Result<(), String> {
235    let manifest = fs::read_to_string(directory.join("manifest.sha256"))
236        .map_err(|error| format!("could not read snapshot manifest: {error}"))?;
237    let expected_license_marker = line_spdx_marker(&project_license(root)?);
238    if !manifest.starts_with(&expected_license_marker) {
239        return Err("portable snapshot manifest has the wrong license marker".to_owned());
240    }
241    let mut records = String::new();
242    for line in manifest
243        .lines()
244        .filter(|line| !line.starts_with('#') && !line.is_empty())
245    {
246        writeln!(&mut records, "{line}").expect("writing to a String cannot fail");
247    }
248    let expected_tree = directory
249        .file_name()
250        .and_then(std::ffi::OsStr::to_str)
251        .ok_or_else(|| "portable snapshot directory has no UTF-8 digest name".to_owned())?;
252    if sha256_hex(records.as_bytes()) != expected_tree {
253        return Err(format!(
254            "portable snapshot directory {expected_tree} does not match its manifest"
255        ));
256    }
257    for record in records.lines() {
258        let mut fields = record.splitn(3, "  ");
259        let expected_digest = fields
260            .next()
261            .ok_or_else(|| format!("invalid portable snapshot record: {record}"))?;
262        let expected_size = fields
263            .next()
264            .ok_or_else(|| format!("invalid portable snapshot record: {record}"))?
265            .parse::<usize>()
266            .map_err(|error| format!("invalid portable snapshot size: {error}"))?;
267        let relative = fields
268            .next()
269            .ok_or_else(|| format!("invalid portable snapshot record: {record}"))?;
270        let path = directory.join("content").join(relative);
271        let bytes = fs::read(&path)
272            .map_err(|error| format!("could not read snapshot file {}: {error}", path.display()))?;
273        if bytes.len() != expected_size || sha256_hex(&bytes) != expected_digest {
274            return Err(format!(
275                "portable snapshot file differs from manifest: {}",
276                path.strip_prefix(root).unwrap_or(&path).display()
277            ));
278        }
279    }
280    Ok(())
281}

283/// Verifies every frozen input path and any digest retained beside it.
284///
285/// Active single-maintainer plans may track the input by path alone. Released
286/// freezes can retain a matching `*_sha256` field for immutable archival
287/// verification.
288pub(crate) fn verify_frozen_input_digests(root: &Path, freeze_file: &str) -> Result<(), String> {
289    let freeze = read_workspace_text(root, freeze_file)?;
290    let values = configuration_section(&freeze, "fixture_corpus")?
291        .into_iter()
292        .collect::<BTreeMap<_, _>>();
293    let frozen_paths = values
294        .iter()
295        .filter_map(|(key, path)| {
296            key.strip_suffix("_path")
297                .map(|name| (name.to_owned(), path.to_owned()))
298        })
299        .collect::<Vec<_>>();
300    if frozen_paths.is_empty() {
301        return Err("contract freeze has no frozen input paths".to_owned());
302    }
303    for (name, path) in frozen_paths {
304        let digest_key = format!("{name}_sha256");
305        let relative = Path::new(&path);
306        if relative.as_os_str().is_empty()
307            || !relative
308                .components()
309                .all(|component| matches!(component, Component::Normal(_)))
310        {
311            return Err(format!(
312                "frozen input path must be a safe workspace-relative path: {path}"
313            ));
314        }
315        let bytes = fs::read(root.join(relative))
316            .map_err(|error| format!("could not read frozen input {path}: {error}"))?;
317        if let Some(expected) = values.get(&digest_key) {
318            if !is_sha256(expected) {
319                return Err(format!("contract freeze has an invalid {digest_key}"));
320            }
321            let actual = sha256_hex(&bytes);
322            if actual != expected.as_str() {
323                return Err(format!(
324                    "frozen input {path} has SHA-256 {actual}, expected {expected}; contract review is required"
325                ));
326            }
327        }
328    }
329    Ok(())
330}

332/// Verifies repository-local links in every active portable Markdown file.
333pub(crate) fn verify_portable_links(root: &Path) -> Result<(), String> {
334    let portable_root = root.join(constants::PORTABLE_DIRECTORY);
335    let canonical_root = fs::canonicalize(root)
336        .map_err(|error| format!("could not canonicalize workspace root: {error}"))?;
337    let mut files = Vec::new();
338    collect_regular_files(&portable_root, &mut files)?;
339    let mut missing = Vec::new();
340    for file in files
341        .iter()
342        .filter(|path| path.extension().and_then(|value| value.to_str()) == Some("md"))
343    {
344        let content = fs::read_to_string(file)
345            .map_err(|error| format!("could not read {}: {error}", file.display()))?;
346        for target in markdown_link_targets(&content) {
347            let candidate = file.parent().unwrap_or(&portable_root).join(&target);
348            let valid = fs::canonicalize(&candidate)
349                .is_ok_and(|candidate| candidate.starts_with(&canonical_root));
350            if !valid {
351                missing.push(format!(
352                    "{} -> {target}",
353                    file.strip_prefix(root).unwrap_or(file).display()
354                ));
355            }
356        }
357    }
358    if missing.is_empty() {
359        Ok(())
360    } else {
361        Err(format!(
362            "active portable links are missing: {}",
363            missing.join(", ")
364        ))
365    }
366}

368/// Creates an atomic digest-addressed copy of the active portable package.
369pub(crate) fn snapshot_portable() -> Result<(), String> {
370    verify_portable()?;
371    let root = workspace_root()?;
372    let portable_root = root.join(constants::PORTABLE_DIRECTORY);
373    let mut files = Vec::new();
374    collect_regular_files(&portable_root, &mut files)?;
375    files.sort();
376    let records = portable_digest_records(&root, &files)?;
377    let tree_digest = sha256_hex(records.as_bytes());
378    let parent = result_root()?.join(constants::PORTABLE_SNAPSHOT_DIRECTORY);
379    let output = parent.join(&tree_digest);
380    let license_marker = line_spdx_marker(&project_license(&root)?);
381    let manifest =
382        format!("{license_marker}\n# sha256  bytes  repository-relative-path\n{records}");
383    if output.exists() {
384        let existing = fs::read_to_string(output.join("manifest.sha256"))
385            .map_err(|error| format!("could not read existing snapshot manifest: {error}"))?;
386        if existing != manifest {
387            return Err(format!(
388                "portable snapshot collision at {}",
389                output.display()
390            ));
391        }
392        crate::output::file("portable snapshot", &output);
393        return Ok(());
394    }
395    fs::create_dir_all(&parent)
396        .map_err(|error| format!("could not create {}: {error}", parent.display()))?;
397    let partial = parent.join(format!(".{tree_digest}.partial-{}", std::process::id()));
398    fs::create_dir(&partial)
399        .map_err(|error| format!("could not create {}: {error}", partial.display()))?;
400    let content_root = partial.join("content").join(constants::PORTABLE_DIRECTORY);
401    for source in &files {
402        let relative = source
403            .strip_prefix(&portable_root)
404            .map_err(|error| format!("portable path escaped its root: {error}"))?;
405        let destination = content_root.join(relative);
406        if let Some(parent) = destination.parent() {
407            fs::create_dir_all(parent)
408                .map_err(|error| format!("could not create {}: {error}", parent.display()))?;
409        }
410        fs::copy(source, &destination).map_err(|error| {
411            format!(
412                "could not copy {} to {}: {error}",
413                source.display(),
414                destination.display()
415            )
416        })?;
417    }
418    fs::write(partial.join("manifest.sha256"), &manifest)
419        .map_err(|error| format!("could not write portable snapshot manifest: {error}"))?;
420    let lifecycle = read_workspace_text(&root, constants::PORTABLE_LIFECYCLE_FILE)?;
421    let report = format!(
422        "{{\n  \"schema_version\": 1,\n  \"tree_sha256\": \"{tree_digest}\",\n  \"file_count\": {},\n  \"next_series\": \"{}\",\n  \"status\": \"review-required\"\n}}\n",
423        files.len(),
424        json_string(&configuration_value(&lifecycle, "next_series").unwrap_or_default())
425    );
426    fs::write(partial.join("migration-report.json"), report)
427        .map_err(|error| format!("could not write portable migration report: {error}"))?;
428    fs::rename(&partial, &output).map_err(|error| {
429        format!(
430            "could not publish portable snapshot {} as {}: {error}",
431            partial.display(),
432            output.display()
433        )
434    })?;
435    crate::output::file("portable snapshot", &output);
436    Ok(())
437}

439/// Builds sorted exact-byte digest records for active portable files.
440pub(crate) fn portable_digest_records(root: &Path, files: &[PathBuf]) -> Result<String, String> {
441    let mut records = String::new();
442    for path in files {
443        let bytes = fs::read(path)
444            .map_err(|error| format!("could not read {}: {error}", path.display()))?;
445        let relative = path
446            .strip_prefix(root)
447            .map_err(|error| format!("portable path escaped workspace: {error}"))?
448            .to_string_lossy()
449            .replace('\\', "/");
450        writeln!(
451            &mut records,
452            "{}  {}  {relative}",
453            sha256_hex(&bytes),
454            bytes.len()
455        )
456        .expect("writing to a String cannot fail");
457    }
458    Ok(records)
