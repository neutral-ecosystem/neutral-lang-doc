3//! Recoverable writes limited to owned release-version metadata.
4
5use crate::{Path, constants, fs, runtime::files::ExclusiveFileLock};
6use serde::{Deserialize, Serialize};
7use std::io::Write as _;
8
9/// Stable ignored location, independent of output-directory environment overrides.
10const STORE: &str = ".neutral-version-update";
11/// Durable metadata transition; completion is marked before reporting success.
12#[derive(Serialize, Deserialize)]
13#[serde(deny_unknown_fields)]
14struct Journal {
15    /// Supported recovery representation.
16    schema_version: u32,
17    /// Whether every replacement was durably applied.
18    complete: bool,
19    /// Owned relative paths and exact old/new contents.
20    entries: Vec<Replacement>,
21}
22/// One recoverable release metadata replacement.
23#[derive(Serialize, Deserialize)]
24#[serde(deny_unknown_fields)]
25struct Replacement {
26    /// Restricted workspace-relative metadata path.
27    path: String,
28    /// Absence is retained distinctly from an empty file.
29    old: Option<Vec<u8>>,
30    /// Prepared replacement contents.
31    new: Vec<u8>,
32}
33
34/// Rejects traversal, symlinks, and destinations not owned by version preparation.
35fn target(root: &Path, path: &str) -> Result<std::path::PathBuf, String> {
36    let relative = Path::new(path);
37    let evidence = relative
38        .strip_prefix(constants::QUALITY_EVIDENCE_DIRECTORY)
39        .ok();
40    let evidence_readme = evidence.is_some_and(|suffix| {
41        let parts = suffix.iter().collect::<Vec<_>>();
42        parts.len() == 2
43            && parts[1] == "README.md"
44            && parts[0]
45                .to_str()
46                .and_then(|version| version.strip_prefix('v'))
47                .is_some_and(|version| crate::validate_semver(version).is_ok())
48    });
49    if !crate::is_safe_relative_path(relative)
50        || !(path == constants::WORKSPACE_MANIFEST_FILE
51            || path == constants::CARGO_LOCK_FILE
52            || evidence_readme)
53    {
54        return Err(format!("unsafe version recovery target: {path}"));
55    }
56    let mut checked = root.to_owned();
57    for part in relative.components() {
58        checked.push(part);
59        if let Ok(metadata) = fs::symlink_metadata(&checked)
60            && metadata.file_type().is_symlink()
61        {
62            return Err(format!(
63                "version recovery target contains a symlink: {path}"
64            ));
65        }
66    }
67    Ok(checked)
68}

70/// Claims the stable recovery store without deleting or replacing its lock inode.
71fn lock(root: &Path) -> Result<ExclusiveFileLock, String> {
72    let directory = root.join(STORE);
73    if let Ok(metadata) = fs::symlink_metadata(&directory)
74        && metadata.file_type().is_symlink()
75    {
76        return Err("version recovery directory must not be a symlink".to_owned());
77    }
78    fs::create_dir_all(&directory).map_err(|error| error.to_string())?;
79    sync_directory(root)?;
80    ExclusiveFileLock::claim(&directory.join("lock"))
81}

83/// Flushes directory entries where the platform exposes directory synchronization.
84fn sync_directory(path: &Path) -> Result<(), String> {
85    #[cfg(unix)]
86    fs::File::open(path)
87        .and_then(|file| file.sync_all())
88        .map_err(|error| error.to_string())?;
89    #[cfg(not(unix))]
90    let _ = path;
91    Ok(())
92}

94/// Stages complete bytes in the ignored store and replaces one destination.
95fn replace(root: &Path, destination: &Path, bytes: &[u8]) -> Result<(), String> {
96    let parent = destination
97        .parent()
98        .ok_or("version destination has no parent")?;
99    fs::create_dir_all(parent).map_err(|error| error.to_string())?;
100    let staging = root.join(STORE).join("replacement");
101    if fs::symlink_metadata(&staging).is_ok_and(|metadata| !metadata.is_file()) {
102        return Err("version recovery staging must be a regular file".to_owned());
103    }
104    let mut file = fs::File::create(&staging).map_err(|error| error.to_string())?;
105    file.write_all(bytes)
106        .and_then(|()| file.sync_all())
107        .map_err(|error| error.to_string())?;
108    if let Ok(metadata) = fs::metadata(destination) {
109        fs::set_permissions(&staging, metadata.permissions()).map_err(|error| error.to_string())?;
110    }
111    drop(file);
112    fs::rename(staging, destination).map_err(|error| error.to_string())?;
113    sync_directory(parent)
114}

116/// Publishes a complete journal before any destination is modified.
117fn write_journal(root: &Path, journal: &Journal) -> Result<(), String> {
118    let bytes = serde_json::to_vec_pretty(journal).map_err(|error| error.to_string())?;
119    replace(root, &root.join(STORE).join("journal.json"), &bytes)
120}

122/// Reads regular files only, retaining absence for a newly created scaffold.
123fn contents(path: &Path) -> Result<Option<Vec<u8>>, String> {
124    match fs::symlink_metadata(path) {
125        Err(error) if error.kind() == std::io::ErrorKind::NotFound => Ok(None),
126        Err(error) => Err(error.to_string()),
127        Ok(metadata) if metadata.is_file() => {
128            fs::read(path).map(Some).map_err(|error| error.to_string())
129        }
130        Ok(_) => Err(format!(
131            "version metadata must be a regular file: {}",
132            path.display()
133        )),
134    }
135}

137/// Validates every current state before recovery can touch any destination.
138fn validate(root: &Path, journal: &Journal) -> Result<(), String> {
139    if journal.schema_version != 1 || journal.entries.is_empty() || journal.entries.len() > 3 {
140        return Err("unsupported or empty version recovery journal".to_owned());
141    }
142    let mut seen = std::collections::BTreeSet::new();
143    for entry in &journal.entries {
144        let destination = target(root, &entry.path)?;
145        let current = contents(&destination)?;
146        if !seen.insert(&entry.path)
147            || (current.as_deref() != Some(entry.new.as_slice())
148                && (journal.complete || current != entry.old))
149        {
150            return Err(format!(
151                "version recovery refuses modified or duplicate target: {}",
152                entry.path
153            ));
154        }
155    }
156    Ok(())
157}

159/// Restores an interrupted update only when all destinations have expected bytes.
160fn rollback(root: &Path, journal: &Journal) -> Result<(), String> {
161    validate(root, journal)?;
162    for entry in journal.entries.iter().rev() {
163        let destination = target(root, &entry.path)?;
164        match &entry.old {
165            Some(bytes) => replace(root, &destination, bytes)?,
166            None if destination.exists() => {
167                fs::remove_file(&destination).map_err(|error| error.to_string())?;
168                sync_directory(
169                    destination
170                        .parent()
171                        .ok_or("version destination has no parent")?,
172                )?;
173            }
174            None => {}
175        }
176    }
177    fs::remove_file(root.join(STORE).join("journal.json")).map_err(|error| error.to_string())?;
178    sync_directory(&root.join(STORE))
179}

181/// Recovers interruption, or returns completed owned paths for safe release-commit retry.
182pub(crate) fn recover(root: &Path) -> Result<Option<Vec<String>>, String> {
183    let _lock = lock(root)?;
184    let Some(bytes) = contents(&root.join(STORE).join("journal.json"))? else {
185        return Ok(None);
186    };
187    let journal: Journal = serde_json::from_slice(&bytes)
188        .map_err(|error| format!("invalid version recovery journal: {error}"))?;
189    validate(root, &journal)?;
190    if journal.complete {
191        return Ok(Some(
192            journal
193                .entries
194                .into_iter()
195                .map(|entry| entry.path)
196                .collect(),
197        ));
198    }
199    rollback(root, &journal)?;
200    Ok(None)
201}

203/// Acknowledges a complete metadata update after the caller has safely consumed it.
204pub(crate) fn acknowledge(root: &Path) -> Result<(), String> {
205    let _lock = lock(root)?;
206    let path = root.join(STORE).join("journal.json");
207    let Some(bytes) = contents(&path)? else {
208        return Ok(());
209    };
210    let journal: Journal = serde_json::from_slice(&bytes).map_err(|error| error.to_string())?;
211    validate(root, &journal)?;
212    if !journal.complete {
213        return Err("version update still needs recovery".to_owned());
214    }
215    fs::remove_file(path).map_err(|error| error.to_string())?;
216    sync_directory(&root.join(STORE))
217}

219/// Applies the prepared metadata bytes to their workspace-relative destinations.
220pub(crate) fn apply(root: &Path, changes: &[(String, Vec<u8>)]) -> Result<(), String> {
221    apply_with(root, changes, replace)
222}
223
224/// Keeps installation failures injectable while recovery always uses the real replacement path.
225fn apply_with(
226    root: &Path,
227    changes: &[(String, Vec<u8>)],
228    mut install: impl FnMut(&Path, &Path, &[u8]) -> Result<(), String>,
229) -> Result<(), String> {
230    let _lock = lock(root)?;
231    if root.join(STORE).join("journal.json").exists() {
232        return Err("version update needs recovery first".to_owned());
233    }
234    let mut journal = Journal {
235        schema_version: 1,
236        complete: false,
237        entries: Vec::new(),
238    };
239    for (path, bytes) in changes {
240        let destination = target(root, path)?;
241        journal.entries.push(Replacement {
242            path: path.clone(),
243            old: contents(&destination)?,
244            new: bytes.clone(),
245        });
246    }
247    validate(root, &journal)?;
248    write_journal(root, &journal)?;
249    let result = journal
250        .entries
251        .iter()
252        .try_for_each(|entry| install(root, &root.join(&entry.path), &entry.new));
253    if let Err(error) = result {
254        return match rollback(root, &journal) {
255            Ok(()) => Err(format!("version update rolled back: {error}")),
256            Err(recovery) => Err(format!(
257                "version update failed: {error}; rerun preparation to recover: {recovery}"
258            )),
259        };
260    }
261    journal.complete = true;
262    write_journal(root, &journal)?;
263    Ok(())
264}
