10/// Reads and verifies the explicit release-authority selection for a `main`-head candidate.
11pub(crate) fn release_plan() -> Result<release::ReleasePlan, String> {
12    let root = workspace_root()?;
13    let package_version = workspace_package_version(&read_workspace_text(
14        &root,
15        constants::WORKSPACE_MANIFEST_FILE,
16    )?)?;
17    let plan =
18        release::ReleasePlan::read(&root.join(constants::RELEASE_CONFIG_FILE), &package_version)?;
19    let residual_risk_status = quality_document_status(&root, constants::RESIDUAL_RISKS_FILE)?;
20    if !matches!(
21        residual_risk_status.as_str(),
22        "approved" | "approved-with-limitation"
23    ) {
24        return Err(format!(
25            "residual-risk review is not approved in the quality manifest; found {residual_risk_status}"
26        ));
27    }
28    if plan
29        .channels
30        .contains(&release::DistributionChannel::GithubBinaries)
31    {
32        let expected = BTreeSet::from([
33            constants::NEUTRAL_CLI.to_owned(),
34            constants::NEUTRAL_PROBE.to_owned(),
35        ]);
36        let selected = plan.binaries.iter().cloned().collect::<BTreeSet<_>>();
37        if selected != expected {
38            return Err(format!(
39                "GitHub binary scope must select exactly {expected:?}; found {selected:?}"
40            ));
41        }
42    }
43    Ok(plan)
44}

46/// Reads the lifecycle status of one durable document from the quality manifest.
47pub(crate) fn quality_document_status(root: &Path, document: &str) -> Result<String, String> {
48    let manifest = read_workspace_text(root, constants::QUALITY_MANIFEST_FILE)?;
49    for entry in manifest.split("[[document]]").skip(1) {
50        if configuration_value(entry, "path").as_deref() == Some(document) {
51            return configuration_value(entry, "status")
52                .ok_or_else(|| format!("quality document {document} has no status"));
53        }
54    }
55    Err(format!(
56        "quality document is not registered in the manifest: {document}"
57    ))
58}

60/// Requires current main to descend from the approved candidate under the same quality policy.
61pub(crate) fn verify_release_approval() -> Result<(), String> {
62    let plan = release_plan()?;
63    let approvals = read_quality_approvals()?;
64    let approval = approvals
65        .iter()
66        .find(|record| record.release == plan.release_tag)
67        .ok_or_else(|| {
68            format!(
69                "release {} has no quality approval record",
70                plan.release_tag
71            )
72        })?;
73    if approval.status != "approved" {
74        return Err(format!(
75            "release {} quality approval is not approved",
76            plan.release_tag
77        ));
78    }
79    let root = workspace_root()?;
80    if approval.quality_gates_sha256 != sha256_file(&root.join(constants::QUALITY_GATES_FILE))? {
81        return Err("quality gates changed after release approval".to_owned());
82    }
83    let head = require_main_head_checkout()?;
84    release::verify_approval_lineage(&root, &approval.commit, &head)
85}
