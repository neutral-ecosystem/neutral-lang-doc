3//! Decide whether retained reports satisfy their configured quality gate.
4//!
5//! Verify receipt bindings and exact report hashes, then interpret native
6//! coverage, mutation, fuzz, performance or advisory results against policy.
7//! A successful tool exit or a receipt alone is insufficient: required reports
8//! and their measured outcomes must also pass the gate-specific checks.
9
10use super::{
11    BTreeSet, Path, QualityGate, QualitySettings, Receipt, is_sha256, measurement_store,
12    plain_filename, reports, sha256_file, unix_time_millis,
13};
14use std::fs;
15
16/// Loads one strict receipt schema.
17pub(super) fn read_receipt(directory: &Path) -> Result<Receipt, String> {
18    serde_json::from_slice(
19        &fs::read(directory.join(reports::RECEIPT))
20            .map_err(|error| format!("{}: {error}", directory.display()))?,
21    )
22    .map_err(|error| format!("invalid quality receipt: {error}"))
23}
24
25/// Rejects wrong-source, wrong-policy, modified, or incomplete measurements.
26pub(super) fn verify(
27    directory: &Path,
28    gate: QualityGate,
29    inputs: &str,
30    settings: &QualitySettings,
31) -> Result<(), String> {
32    let resolved = measurement_store::resolve(directory)?;
33    let directory = resolved.as_path();
34    let receipt = read_receipt(directory)?;
35    if receipt.schema_version != 2
36        || receipt.gate != gate
37        || receipt.inputs_sha256 != inputs
38        || receipt.policy_sha256 != settings.policy_sha256
39        || receipt.source_commit.len() != 40
40        || receipt.toolchain.is_empty()
41        || receipt.finished_at_unix_ms == 0
42        || receipt.invocations.is_empty()
43    {
44        return Err(format!("{gate} receipt is stale or incomplete"));
45    }
46    validate_reports(&receipt, directory, settings)
47}

49/// Parses actual retained JSON output from a measurement tool.
50fn json_report(directory: &Path, name: &str) -> Result<serde_json::Value, String> {
51    serde_json::from_slice(
52        &fs::read(directory.join(name)).map_err(|error| format!("missing {name}: {error}"))?,
53    )
54    .map_err(|error| format!("invalid {name}: {error}"))
55}

57/// Checks exact report bytes and independent measured acceptance criteria.
58pub(super) fn validate_reports(
59    receipt: &Receipt,
60    directory: &Path,
61    settings: &QualitySettings,
62) -> Result<(), String> {
63    if receipt.gate.requires_nightly() && !receipt.toolchain.contains("-nightly") {
64        return Err("analysis measurement was not produced with nightly".to_owned());
65    }
66    if receipt.invocations.is_empty() {
67        return Err("measurement has no invocations".to_owned());
68    }
69    for (filename, expected) in &receipt.reports {
70        if !plain_filename(filename)
71            || !is_sha256(expected)
72            || sha256_file(&directory.join(filename))? != *expected
73        {
74            return Err(format!("modified or unsafe report: {filename}"));
75        }
76    }
77    for invocation in &receipt.invocations {
78        if invocation.tool_version.trim().is_empty() {
79            return Err("measurement has no tool version".to_owned());
80        }
81        if !plain_filename(&invocation.report) {
82            return Err("unsafe invocation report".to_owned());
83        }
84        for suffix in ["stdout", "stderr"] {
85            if !receipt
86                .reports
87                .contains_key(&format!("{}.{suffix}", invocation.report))
88            {
89                return Err("invocation has no captured report".to_owned());
90            }
91        }
92    }
93    match receipt.gate {
94        QualityGate::Coverage => validate_coverage(receipt, directory, settings),
95        QualityGate::Mutation => validate_mutation(receipt, directory, settings),
96        QualityGate::Fuzz => validate_fuzz(receipt, directory, settings),
97        QualityGate::PerformanceRelease | QualityGate::PerformanceSoak => {
98            validate_performance(receipt, directory, settings)
99        }
100        QualityGate::Advisories => validate_advisories(receipt, directory, settings),
101    }
102}

104/// Validates native coverage measurements against the configured acceptance policy.
105fn validate_coverage(
106    receipt: &Receipt,
107    directory: &Path,
108    settings: &QualitySettings,
109) -> Result<(), String> {
110    require_report(receipt, "coverage.json")?;
111    let report = json_report(directory, "coverage.json")?;
112    for (metric, threshold) in [
113        ("lines", settings.policy.coverage.minimum_line_percent),
114        (
115            "functions",
116            settings.policy.coverage.minimum_function_percent,
117        ),
118        ("regions", settings.policy.coverage.minimum_region_percent),
119    ] {
120        let observed = report
121            .pointer(&format!("/data/0/totals/{metric}/percent"))
122            .and_then(serde_json::Value::as_f64)
123            .ok_or_else(|| format!("coverage has no {metric} percentage"))?;
124        let minimum = threshold;
125        if !observed.is_finite() || observed < minimum || observed > 100.0 {
126            return Err(format!(
127                "measured coverage {metric} {observed} fails minimum {minimum}"
128            ));
129        }
130    }
131    Ok(())
132}

134/// Validates native mutation measurements against the configured acceptance policy.
135fn validate_mutation(
136    receipt: &Receipt,
137    directory: &Path,
138    settings: &QualitySettings,
139) -> Result<(), String> {
140    require_report(receipt, "outcomes.json")?;
141    let targets = &settings.policy.mutation.critical_targets;
142    if !receipt.invocations.iter().any(|run| {
143        let selected = run
144            .arguments
145            .windows(2)
146            .filter(|args| args[0] == "--file")
147            .map(|args| args[1].as_str())
148            .collect::<BTreeSet<_>>();
149        selected == targets.iter().map(String::as_str).collect::<BTreeSet<_>>()
150            && run.arguments.contains(&"--no-config".to_owned())
151            && run
152                .arguments
153                .contains(&"--cargo-arg=--workspace".to_owned())
154            && !run.arguments.iter().any(|argument| {
155                matches!(
156                    argument.split('=').next().unwrap_or(argument),
157                    "--exclude"
158                        | "--exclude-re"
159                        | "--re"
160                        | "--shard"
161                        | "--limit"
162                        | "--package"
163                        | "-p"
164                        | "--config"
165                        | "--skip-calls"
166                )
167            })
168    }) {
169        return Err("mutation does not cover the complete configured critical scope".to_owned());
170    }
171    let report = json_report(directory, "outcomes.json")?;
172    let count = |name: &str| {
173        report[name]
174            .as_u64()
175            .ok_or_else(|| format!("mutation has no {name}"))
176    };
177    let caught = count("caught")?;
178    let missed = count("missed")?;
179    let minimum = settings.policy.mutation.minimum_caught_percent;
180    #[allow(clippy::cast_precision_loss)]
181    let percentage = 100.0 * caught as f64
182        / caught
183            .checked_add(missed)
184            .ok_or("mutation count overflow")? as f64;
185    if caught == 0 || count("timeout")? != 0 || count("success")? != 0 || percentage < minimum {
186        return Err("mutation results fail the caught threshold".to_owned());
187    }
188    Ok(())
189}

191/// Validates native fuzz measurements against the configured acceptance policy.
192fn validate_fuzz(
193    receipt: &Receipt,
194    directory: &Path,
195    settings: &QualitySettings,
196) -> Result<(), String> {
197    let minimum = u128::from(settings.policy.fuzz.minimum_seconds_per_target);
198    for target in &settings.policy.fuzz.targets {
199        if !receipt.invocations.iter().any(|run| {
200            run.report == *target
201                && run.elapsed_ms >= minimum * 1_000
202                && run
203                    .arguments
204                    .contains(&format!("-max_total_time={minimum}"))
205        }) {
206            return Err(format!(
207                "fuzz target {target} has no full measured campaign"
208            ));
209        }
210        let output = fs::read_to_string(directory.join(format!("{target}.stderr")))
211            .map_err(|error| format!("missing fuzz log: {error}"))?;
212        let seconds = output
213            .lines()
214            .filter_map(|line| line.strip_prefix("Done "))
215            .filter_map(|line| line.split_once(" runs in ").map(|(_, time)| time))
216            .filter_map(|time| time.split_whitespace().next()?.parse::<u128>().ok())
217            .max();
218        if seconds.is_none_or(|seconds| seconds < minimum) {
219            return Err(format!(
220                "fuzzer {target} did not confirm its full run duration"
221            ));
222        }
223    }
224    Ok(())
225}

227/// Validates native performance measurements against the configured acceptance policy.
228fn validate_performance(
229    receipt: &Receipt,
230    directory: &Path,
231    settings: &QualitySettings,
232) -> Result<(), String> {
233    for report in [
234        reports::BENCHMARK_STDOUT,
235        reports::MASSIF,
236        reports::MEMCHECK_STDERR,
237    ] {
238        require_report(receipt, report)?;
239    }
240    let profile = if receipt.gate == QualityGate::PerformanceSoak {
241        "extended-soak"
242    } else {
243        "release"
244    };
245    if !receipt.invocations.iter().any(|run| {
246        run.report == "benchmark" && run.arguments.last().is_some_and(|arg| arg == profile)
247    }) {
248        return Err("performance profile does not match its gate".to_owned());
249    }
250    let output = fs::read_to_string(directory.join(reports::BENCHMARK_STDOUT))
251        .map_err(|error| format!("missing benchmark: {error}"))?;
252    validate_phase_times(
253        &output,
254        &settings.policy.performance.maximum_ns_per_iteration,
255    )?;
256    let massif = fs::read_to_string(directory.join(reports::MASSIF))
257        .map_err(|error| format!("missing heap profile: {error}"))?;
258    let memcheck = fs::read_to_string(directory.join(reports::MEMCHECK_STDERR))
259        .map_err(|error| format!("missing allocation review: {error}"))?;
260    let peak = massif
261        .lines()
262        .filter_map(|line| line.strip_prefix("mem_heap_B="))
263        .try_fold(None, |peak: Option<u64>, value| {
264            let bytes = value
265                .parse::<u64>()
266                .map_err(|_| "invalid Massif heap sample".to_owned())?;
267            Ok::<_, String>(Some(peak.map_or(bytes, |previous| previous.max(bytes))))
268        })?
269        .ok_or("performance has no heap samples")?;
270    if peak > settings.policy.performance.maximum_heap_bytes {
271        return Err(format!(
272            "peak heap {peak} exceeds {} bytes",
273            settings.policy.performance.maximum_heap_bytes
274        ));
275    }
276    if !memcheck.contains("ERROR SUMMARY: 0 errors")
277        || !receipt.invocations.iter().any(|run| {
278            run.report == "memcheck" && run.arguments.contains(&"--error-exitcode=1".to_owned())
279        })
280    {
281        return Err("performance has no successful heap/allocation review".to_owned());
282    }
283    Ok(())
284}

286/// Parses exact native phase records; every required operation must pass once, without averaging away failures.
287fn validate_phase_times(
288    output: &str,
289    budgets: &std::collections::BTreeMap<String, u64>,
290) -> Result<(), String> {
291    let mut observed = BTreeSet::new();
292    for line in output.lines() {
293        let mut words = line.split_whitespace();
294        if words.next() != Some("PASS") || words.next() != Some("performance") {
295            continue;
296        }
297        let field = words.next().ok_or("malformed performance record")?;
298        if field.starts_with("artifact-bytes=") {
299            continue;
300        }
301        let phase = field
302            .strip_prefix("name=")
303            .ok_or("malformed performance record")?;
304        let maximum = budgets
305            .get(phase)
306            .ok_or_else(|| format!("unconfigured performance phase {phase}"))?;
307        let iterations = words
308            .next()
309            .and_then(|word| word.strip_prefix("iterations="))
310            .and_then(|value| value.parse::<u64>().ok())
311            .filter(|n| *n > 0)
312            .ok_or_else(|| format!("invalid iterations for {phase}"))?;
313        let elapsed = words
314            .next()
315            .and_then(|word| word.strip_prefix("elapsed-ns="))
316            .and_then(|value| value.parse::<u64>().ok())
317            .filter(|n| *n > 0)
318            .ok_or_else(|| format!("invalid elapsed time for {phase}"))?;
319        if words.next().is_some() || !observed.insert(phase) {
320            return Err(format!("duplicate or malformed performance phase {phase}"));
321        }
322        if u128::from(elapsed) > u128::from(*maximum) * u128::from(iterations) {
323            return Err(format!(
324                "performance {phase} exceeded {maximum} ns/iteration ({elapsed} ns / {iterations})"
325            ));
326        }
327    }
328    for phase in budgets.keys() {
329        if !observed.contains(phase.as_str()) {
330            return Err(format!("benchmark has no {phase} measurement"));
331        }
332    }
333    Ok(())
334}

336/// Validates native advisories measurements against the configured acceptance policy.
337fn validate_advisories(
338    receipt: &Receipt,
339    directory: &Path,
340    settings: &QualitySettings,
341) -> Result<(), String> {
342    for lock in &settings.advisory_locks {
343        if !receipt.invocations.iter().any(|run| {
344            run.arguments
345                .windows(2)
346                .any(|args| args == ["--file", lock.as_str()])
347        }) {
348            return Err(format!("advisory scan omitted {lock}"));
349        }
350    }
351    let age = u128::from(settings.advisory_max_age_seconds);
352    let now = unix_time_millis()?;
353    if receipt.finished_at_unix_ms > now || now - receipt.finished_at_unix_ms > age * 1_000 {
354        return Err("advisory scan is stale; rerun release quality".to_owned());
355    }
356    for invocation in &receipt.invocations {
357        let report = json_report(directory, &format!("{}.stdout", invocation.report))?;
358        if report
359            .pointer("/vulnerabilities/found")
360            .and_then(serde_json::Value::as_bool)
361            != Some(false)
362            || report
363                .pointer("/vulnerabilities/count")
364                .and_then(serde_json::Value::as_u64)
365                != Some(0)
366        {
367            return Err("advisory scan does not prove zero vulnerabilities".to_owned());
368        }
369    }
370    Ok(())
371}

373/// Requires a native acceptance report to be covered by the receipt's checksums.
374fn require_report(receipt: &Receipt, filename: &str) -> Result<(), String> {
375    if receipt.reports.contains_key(filename) {
376        Ok(())
377    } else {
378        Err(format!("unhashed acceptance report: {filename}"))
379    }
