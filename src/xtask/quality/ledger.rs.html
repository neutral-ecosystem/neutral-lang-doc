15/// One immutable release-quality record.
16#[derive(Debug, Eq, Ord, PartialEq, PartialOrd)]
17pub(crate) struct QualityApproval {
18    /// Release identifier without a `v` prefix.
19    pub(crate) release: String,
20    /// Exact approved Git commit.
21    pub(crate) commit: String,
22    /// Approval state.
23    pub(crate) status: String,
24    /// UTC-independent Unix timestamp or retained historical date.
25    pub(crate) approved_at: String,
26    /// Evaluation identifier that supported the approval.
27    pub(crate) evaluation: String,
28    /// SHA-256 of the quality-gate configuration used for evaluation.
29    pub(crate) quality_gates_sha256: String,
30}

32/// Runs a quality profile and writes a deterministic commit-bound evaluation.
33pub(crate) fn evaluate_quality(profile: QualityProfile) -> Result<(), String> {
34    let commit = require_clean_checkout()?;
35    quality(profile)?;
36    let root = workspace_root()?;
37    let profile_name = quality_profile_name(profile);
38    let output = result_root()?
39        .join(constants::QUALITY_EVALUATION_DIRECTORY)
40        .join(&commit)
41        .join(format!("{profile_name}.toml"));
42    if let Some(parent) = output.parent() {
43        fs::create_dir_all(parent)
44            .map_err(|error| format!("could not create {}: {error}", parent.display()))?;
45    }
46    let manifest_sha256 = sha256_file(&root.join(constants::QUALITY_MANIFEST_FILE))?;
47    let quality_gates_sha256 = sha256_file(&root.join(constants::QUALITY_GATES_FILE))?;
48    let toolchain = command_output(&rustc_command()?, &["--version"])?;
49    let license_marker = line_spdx_marker(&project_license(&root)?);
50    fs::write(
51        &output,
52        format!(
53            "{license_marker}\n\nschema_version = 1\ncommit = \"{}\"\nprofile = \"{profile_name}\"\nstatus = \"pass\"\ntoolchain = \"{}\"\nquality_manifest_sha256 = \"{manifest_sha256}\"\nquality_gates_sha256 = \"{quality_gates_sha256}\"\n",
54            json_string(&commit),
55            json_string(&toolchain)
56        ),
57    )
58    .map_err(|error| format!("could not write {}: {error}", output.display()))?;
59    crate::output::file("quality evaluation", &output);
60    Ok(())
61}

63/// Approves a passing release evaluation for the clean checked-out `main` HEAD.
64pub(crate) fn approve_quality_release(release: &str) -> Result<(), String> {
65    let version = release.strip_prefix('v').unwrap_or(release);
66    validate_semver(version)?;
67    let root = workspace_root()?;
68    let workspace_version = workspace_package_version(&read_workspace_text(
69        &root,
70        constants::WORKSPACE_MANIFEST_FILE,
71    )?)?;
72    if version != workspace_version {
73        return Err(format!(
74            "quality approval release {version} does not match workspace version {workspace_version}"
75        ));
76    }
77    let commit = require_main_head_checkout()?;
78    let evaluation_relative = PathBuf::from(constants::QUALITY_EVALUATION_DIRECTORY)
79        .join(&commit)
80        .join("release.toml");
81    let evaluation = result_root()?.join(&evaluation_relative);
82    let evaluation_content = fs::read_to_string(&evaluation).map_err(|error| {
83        format!(
84            "passing release evaluation is missing at {}: {error}; run `cargo xtask quality evaluate --profile release`",
85            evaluation.display()
86        )
87    })?;
88    if configuration_value(&evaluation_content, "commit").as_deref() != Some(commit.as_str())
89        || configuration_value(&evaluation_content, "profile").as_deref() != Some("release")
90        || configuration_value(&evaluation_content, "status").as_deref() != Some("pass")
91    {
92        return Err("release evaluation does not identify a passing current HEAD".to_owned());
93    }
94    let evaluated_manifest = configuration_value(&evaluation_content, "quality_manifest_sha256")
95        .ok_or_else(|| "release evaluation has no quality-manifest digest".to_owned())?;
96    let evaluated_gates = configuration_value(&evaluation_content, "quality_gates_sha256")
97        .ok_or_else(|| "release evaluation has no quality-gate digest".to_owned())?;
98    if evaluated_manifest != sha256_file(&root.join(constants::QUALITY_MANIFEST_FILE))?
99        || evaluated_gates != sha256_file(&root.join(constants::QUALITY_GATES_FILE))?
100    {
101        return Err(
102            "quality policy changed after evaluation; rerun the release evaluation".to_owned(),
103        );
104    }
105    let evidence_directory = root
106        .join(constants::QUALITY_EVIDENCE_DIRECTORY)
107        .join(format!("v{version}"));
108    if !evidence_directory.join("README.md").is_file() {
109        return Err(format!(
110            "release evidence directory is not prepared: {}",
111            evidence_directory.display()
112        ));
113    }
114    let approved_at = SystemTime::now()
115        .duration_since(UNIX_EPOCH)
116        .map_err(|error| format!("system clock precedes Unix epoch: {error}"))?
117        .as_secs();
118    let record = evidence_directory.join("record.toml");
119    if record.exists() {
120        verify_release_approval()?;
121        quality_evidence::retain()?;
122        crate::output::info(format!(
123            "retained current measurement evidence; existing approval remains immutable: {}",
124            record.display()
125        ));
126        return Ok(());
127    }
128    quality_evidence::retain()?;
129    let quality_gates_sha256 = evaluated_gates;
130    let license_marker = line_spdx_marker(&project_license(&root)?);
131    fs::write(
132        &record,
133        format!(
134            "{license_marker}\n\nschema_version = 1\nrelease = \"v{version}\"\ncommit = \"{commit}\"\nstatus = \"approved\"\napproved_at = \"{approved_at}\"\nevaluation = \"{}\"\nquality_gates_sha256 = \"{quality_gates_sha256}\"\n",
135            json_string(&evaluation_relative.to_string_lossy())
136        ),
137    )
138    .map_err(|error| format!("could not write {}: {error}", record.display()))?;
139    render_quality_status()?;
140    crate::output::pass(format!("quality release v{version}"));
141    Ok(())
142}

144/// Prints the current approved-release quality ledger.
145pub(crate) fn quality_status() -> Result<(), String> {
146    let approvals = read_quality_approvals()?;
147    if approvals.is_empty() {
148        crate::output::info("no approved quality releases");
149    }
150    for approval in approvals {
151        crate::output::info(format!("{} {}", approval.release, approval.status));
152    }
153    Ok(())
154}

156/// Regenerates the checked-in human-readable quality status document.
157pub(crate) fn render_quality_status() -> Result<(), String> {
158    let approvals = read_quality_approvals()?;
159    let root = workspace_root()?;
160    let rendered = quality_status_markdown(&approvals, &project_license(&root)?)?;
161    let path = root.join(constants::QUALITY_STATUS_FILE);
162    fs::write(&path, rendered)
163        .map_err(|error| format!("could not write {}: {error}", path.display()))?;
164    crate::output::file("quality status", &path);
165    Ok(())
166}

168/// Verifies release records and generated status documentation.
169pub(crate) fn verify_quality_ledger() -> Result<(), String> {
170    check_quality_inventory()?;
171    let root = workspace_root()?;
172    let approvals = read_quality_approvals()?;
173    for approval in &approvals {
174        if approval.status != "approved"
175            || approval.approved_at.is_empty()
176            || approval.evaluation.is_empty()
177            || !is_sha256(&approval.quality_gates_sha256)
178            || approval.commit.len() != 40
179            || !approval
180                .commit
181                .chars()
182                .all(|character| character.is_ascii_hexdigit())
183        {
184            return Err(format!(
185                "quality approval v{} is incomplete or invalid",
186                approval.release
187            ));
188        }
189        let version = approval.release.strip_prefix('v').ok_or_else(|| {
190            format!(
191                "quality approval release must start with v: {}",
192                approval.release
193            )
194        })?;
195        validate_semver(version)?;
196    }
197    let expected = quality_status_markdown(&approvals, &project_license(&root)?)?;
198    let status = read_workspace_text(&root, constants::QUALITY_STATUS_FILE)?;
199    if status != expected {
200        return Err(
201            "quality status documentation is stale; run `cargo xtask quality render`".to_owned(),
202        );
203    }
204    crate::output::pass("quality approval ledger");
205    Ok(())
206}

208/// Reads every immutable release-quality record in release order.
209pub(crate) fn read_quality_approvals() -> Result<Vec<QualityApproval>, String> {
210    let root = workspace_root()?;
211    let mut records = Vec::new();
212    collect_named_files(
213        &root.join(constants::QUALITY_EVIDENCE_DIRECTORY),
214        "record.toml",
215        &mut records,
216    )?;
217    let mut approvals = Vec::new();
218    for path in records {
219        let content = fs::read_to_string(&path)
220            .map_err(|error| format!("could not read {}: {error}", path.display()))?;
221        let field = |name: &str| {
222            configuration_value(&content, name)
223                .ok_or_else(|| format!("quality approval {} has no {name}", path.display()))
224        };
225        let approval = QualityApproval {
226            release: field("release")?,
227            commit: field("commit")?,
228            status: field("status")?,
229            approved_at: field("approved_at")?,
230            evaluation: field("evaluation")?,
231            quality_gates_sha256: field("quality_gates_sha256")?,
232        };
233        let directory_release = path
234            .parent()
235            .and_then(Path::file_name)
236            .and_then(std::ffi::OsStr::to_str)
237            .ok_or_else(|| {
238                format!(
239                    "quality approval has no release directory: {}",
240                    path.display()
241                )
242            })?;
243        if approval.release != directory_release {
244            return Err(format!(
245                "quality approval release {} does not match directory {directory_release}",
246                approval.release
247            ));
248        }
249        approvals.push(approval);
250    }
251    approvals.sort();
252    Ok(approvals)
253}

255/// Renders the deterministic human-readable release-quality table.
256pub(crate) fn quality_status_markdown(
257    approvals: &[QualityApproval],
258    license: &str,
259) -> Result<String, String> {
260    let mut output = format!(
261        "{}\n<!-- Generated by `cargo xtask quality render`; do not edit. -->\n\n# Release quality status\n\n| Release | Status | Approved on (UTC, DD-MM-YYYY) |\n| --- | --- | --- |\n",
262        html_spdx_marker(license)
263    );
264    for approval in approvals {
265        let date = quality_approval_date(&approval.approved_at)?;
266        writeln!(
267            &mut output,
268            "| `v{}` | {} | `{}` |",
269            approval
270                .release
271                .strip_prefix('v')
272                .unwrap_or(&approval.release),
273            approval.status,
274            date
275        )
276        .expect("writing to a String cannot fail");
277    }
278    output.push_str("\nRelease records are verified by `cargo xtask quality verify`.\n");
279    Ok(output)
280}

282/// Formats Unix-second or historical ISO approval dates as UTC day-month-year dates.
283pub(crate) fn quality_approval_date(value: &str) -> Result<String, String> {
284    if value.len() == 10
285        && value.as_bytes()[4] == b'-'
286        && value.as_bytes()[7] == b'-'
287        && value
288            .bytes()
289            .enumerate()
290            .all(|(index, byte)| index == 4 || index == 7 || byte.is_ascii_digit())
291    {
292        return Ok(format!(
293            "{}-{}-{}",
294            &value[8..10],
295            &value[5..7],
296            &value[..4]
297        ));
298    }
299    let seconds = value
300        .parse::<u64>()
301        .map_err(|_| format!("invalid quality approval timestamp: {value}"))?;
302    let days = i64::try_from(seconds / 86_400)
303        .map_err(|_| format!("quality approval timestamp is out of range: {value}"))?;
304    let day = days
305        .checked_add(719_468)
306        .ok_or_else(|| format!("quality approval timestamp is out of range: {value}"))?;
307    let era = day / 146_097;
308    let day_of_era = day - era * 146_097;
309    let year_of_era =
310        (day_of_era - day_of_era / 1_460 + day_of_era / 36_524 - day_of_era / 146_096) / 365;
311    let mut year = year_of_era + era * 400;
312    let day_of_year = day_of_era - (365 * year_of_era + year_of_era / 4 - year_of_era / 100);
313    let month_piece = (5 * day_of_year + 2) / 153;
314    let day_of_month = day_of_year - (153 * month_piece + 2) / 5 + 1;
315    let month = month_piece + if month_piece < 10 { 3 } else { -9 };
316    if month <= 2 {
317        year += 1;
318    }
319    if year > 9_999 {
320        return Err(format!(
321            "quality approval timestamp is out of range: {value}"
322        ));
323    }
324    Ok(format!("{day_of_month:02}-{month:02}-{year:04}"))
