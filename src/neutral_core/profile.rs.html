35/// One exact source-language profile recognized by this release train.
36#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
37pub enum LanguageProfile {
38    /// Frozen, implemented v0.1 source behavior.
39    V0_1,
40    /// Reserved v1.0 source behavior, recognized but not implemented in Stage 1.
41    V1_0,
42}
43
44impl LanguageProfile {
45    /// Selects a recognized profile from an exact unescaped header value.
46    #[must_use]
47    pub const fn from_source_version(value: &str) -> Option<Self> {
48        if const_str_eq(value, V0_SOURCE_PROFILE) {
49            Some(Self::V0_1)
50        } else if const_str_eq(value, V1_SOURCE_PROFILE) {
51            Some(Self::V1_0)
52        } else {
53            None
54        }
55    }
56
57    /// Returns the exact source-header version spelling.
58    #[must_use]
59    pub const fn source_version(self) -> &'static str {
60        match self {
61            Self::V0_1 => V0_SOURCE_PROFILE,
62            Self::V1_0 => V1_SOURCE_PROFILE,
63        }
64    }
65}

67/// Availability of a recognized language profile in the current implementation.
68#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
69pub enum ProfileAvailability {
70    /// The profile can produce authoritative logical IR.
71    Available,
72    /// The profile is recognized but must fail closed before interpretation.
73    Unavailable,
74}
75
76impl ProfileAvailability {
77    /// Returns the stable lowercase reporting spelling.
78    #[must_use]
79    pub const fn as_str(self) -> &'static str {
80        match self {
81            Self::Available => "available",
82            Self::Unavailable => "unavailable",
83        }
84    }
85}

87/// One stable capability identifier reported for an implemented profile.
88#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
89pub enum LanguageCapability {
90    /// One captured source unit with no project acquisition.
91    SingleSourceUnit,
92    /// Immutable typed scalar, list, and record values.
93    ImmutableData,
94    /// Nominal record declarations and contextual record values.
95    NominalRecords,
96    /// Typed identity references between bindings.
97    IdentityReferences,
98    /// One exact captured data-only vocabulary bundle.
99    CapturedVocabulary,
100}
101
102impl LanguageCapability {
103    /// Returns the stable lowercase capability identifier.
104    #[must_use]
105    pub const fn as_str(self) -> &'static str {
106        match self {
107            Self::SingleSourceUnit => "single-source-unit",
108            Self::ImmutableData => "immutable-data",
109            Self::NominalRecords => "nominal-records",
110            Self::IdentityReferences => "identity-references",
111            Self::CapturedVocabulary => "captured-vocabulary",
112        }
113    }
114}

116/// Public immutable discovery record for one recognized language profile.
117#[derive(Clone, Copy, Debug, Eq, PartialEq)]
118pub struct LanguageProfileDescriptor {
119    /// Exact source-level profile.
120    profile: LanguageProfile,
121    /// Current fail-open or fail-closed availability.
122    availability: ProfileAvailability,
123    /// Stable implemented capabilities, empty for unavailable profiles.
124    capabilities: &'static [LanguageCapability],
125}
126
127impl LanguageProfileDescriptor {
128    /// Returns the exact source-level profile.
129    #[must_use]
130    pub const fn profile(self) -> LanguageProfile {
131        self.profile
132    }
133
134    /// Returns whether this profile can currently compile.
135    #[must_use]
136    pub const fn availability(self) -> ProfileAvailability {
137        self.availability
138    }
139
140    /// Returns stable capabilities implemented for this profile.
141    #[must_use]
142    pub const fn capabilities(self) -> &'static [LanguageCapability] {
143        self.capabilities
144    }
145}

185/// Compares ASCII profile spellings in a constant context.
186const fn const_str_eq(left: &str, right: &str) -> bool {
187    let left = left.as_bytes();
188    let right = right.as_bytes();
189    if left.len() != right.len() {
190        return false;
191    }
192    let mut index = 0;
193    while index < left.len() {
194        if left[index] != right[index] {
195            return false;
196        }
197        index += 1;
198    }
199    true
200}
