3//! Fallibly growing canonical indexes. Callers charge insertion shifts and key comparisons to work policy.
4
5use crate::allocation::{AllocationError as E, RetainCapacity, TryClone};
6use std::borrow::Borrow;
7
8/// Canonically ordered request-local index with no infallible allocating insertion.
9///
10/// Insertion shifts up to `len()` entries; callers must charge that work before
11/// inserting attacker-controlled data. Read-only lookup uses binary search.
12#[derive(Debug, Eq, PartialEq)]
13pub struct OrderedMap<K, V> {
14    /// Sorted unique keys and their owned or borrowed values.
15    entries: Vec<(K, V)>,
16}
17impl<K, V> Default for OrderedMap<K, V> {
18    /// Starts empty without allocating storage.
19    fn default() -> Self {
20        Self {
21            entries: Vec::new(),
22        }
23    }
24}
25impl<K: Ord, V> OrderedMap<K, V> {
26    /// Starts empty without allocating storage.
27    #[must_use]
28    pub fn new() -> Self {
29        Self::default()
30    }
31    /// Returns the exact unique entry count, also an upper bound on one insertion shift.
32    #[must_use]
33    pub fn len(&self) -> usize {
34        self.entries.len()
35    }
36    /// Returns whether the index is empty.
37    #[must_use]
38    pub fn is_empty(&self) -> bool {
39        self.entries.is_empty()
40    }
41    /// Finds a borrowed key without allocating.
42    fn position<Q: Ord + ?Sized>(&self, key: &Q) -> Result<usize, usize>
43    where
44        K: Borrow<Q>,
45    {
46        self.entries
47            .binary_search_by(|(existing, _)| existing.borrow().cmp(key))
48    }
49    /// Returns immutable value access for one exact borrowed key.
50    pub fn get<Q: Ord + ?Sized>(&self, key: &Q) -> Option<&V>
51    where
52        K: Borrow<Q>,
53    {
54        self.position(key).ok().map(|index| &self.entries[index].1)
55    }
56    /// Returns mutable request-local value access without growing storage.
57    pub fn get_mut<Q: Ord + ?Sized>(&mut self, key: &Q) -> Option<&mut V>
58    where
59        K: Borrow<Q>,
60    {
61        self.position(key)
62            .ok()
63            .map(|index| &mut self.entries[index].1)
64    }
65    /// Checks exact membership without allocating.
66    pub fn contains_key<Q: Ord + ?Sized>(&self, key: &Q) -> bool
67    where
68        K: Borrow<Q>,
69    {
70        self.position(key).is_ok()
71    }
72    /// Inserts with fallible growth, returning replaced data if the key already existed.
73    ///
74    /// # Errors
75    /// Returns allocation failure without changing the index when growth cannot be reserved.
76    pub fn insert(&mut self, key: K, value: V) -> Result<Option<V>, E> {
77        match self.position(&key) {
78            Ok(index) => Ok(Some(std::mem::replace(&mut self.entries[index].1, value))),
79            Err(index) => {
80                self.entries.try_retain(1)?;
81                self.entries.insert(index, (key, value));
82                Ok(None)
83            }
84        }
85    }
86    /// Reserves a vacant entry before any nonallocating default is installed.
87    ///
88    /// # Errors
89    /// Returns allocation failure before changing entry membership.
90    pub fn entry(&mut self, key: K) -> Result<Entry<'_, K, V>, E> {
91        let position = self.position(&key);
92        if position.is_err() {
93            self.entries.try_retain(1)?;
94        }
95        Ok(Entry {
96            map: self,
97            key,
98            position,
99        })
100    }
101    /// Removes one exact key without allocating.
102    pub fn remove<Q: Ord + ?Sized>(&mut self, key: &Q) -> Option<V>
103    where
104        K: Borrow<Q>,
105    {
106        self.position(key)
107            .ok()
108            .map(|index| self.entries.remove(index).1)
109    }
110    /// Enumerates keys and values in canonical order without copying them.
111    #[must_use]
112    pub fn iter(&self) -> impl DoubleEndedIterator<Item = (&K, &V)> {
113        self.entries.iter().map(|(k, v)| (k, v))
114    }
115    /// Enumerates canonical borrowed keys without allocating.
116    #[must_use]
117    pub fn keys(&self) -> impl DoubleEndedIterator<Item = &K> {
118        self.entries.iter().map(|(k, _)| k)
119    }
120    /// Enumerates canonical borrowed values without allocating.
121    #[must_use]
122    pub fn values(&self) -> impl DoubleEndedIterator<Item = &V> {
123        self.entries.iter().map(|(_, v)| v)
124    }
125    /// Moves canonical values out without allocating or cloning them.
126    pub fn into_values(self) -> impl Iterator<Item = V> {
127        self.entries.into_iter().map(|(_, v)| v)
128    }
129}
130impl<K: TryClone, V: TryClone> TryClone for OrderedMap<K, V> {
131    /// Copies retained canonical entries fallibly without repeating insertion work.
132    fn try_clone(&self) -> Result<Self, E> {
133        Ok(Self {
134            entries: self.entries.try_clone()?,
135        })
136    }
137}
138impl<K: Ord + Borrow<Q>, Q: Ord + ?Sized, V> std::ops::Index<&Q> for OrderedMap<K, V> {
139    type Output = V;
140
141    /// Accesses an already established key; untrusted membership must use `get` first.
142    ///
143    /// # Panics
144    /// Panics if the caller has not established that the key is present.
145    fn index(&self, key: &Q) -> &V {
146        self.get(key).expect("established ordered index key")
147    }
148}

159    type IntoIter = std::slice::Iter<'a, (K, V)>;
160    /// Borrows entries in canonical order without allocating.

165/// A reserved occupied/vacant slot; no allocation occurs when its value is installed.
166pub struct Entry<'a, K, V> {
167    /// Exclusively borrowed request-local index.
168    map: &'a mut OrderedMap<K, V>,
169    /// Owned insertion key, discarded when already present.
170    key: K,
171    /// Canonical occupied index or pre-reserved insertion position.
172    position: Result<usize, usize>,
173}
174impl<'a, K, V> Entry<'a, K, V> {
175    /// Installs the supplied nonallocating value after successful reservation.
176    pub fn or_insert(self, value: V) -> &'a mut V {
177        let index = match self.position {
178            Ok(index) => index,
179            Err(index) => {
180                self.map.entries.insert(index, (self.key, value));
181                index
182            }
183        };
184        &mut self.map.entries[index].1
185    }
186}
187impl<'a, K, V: Default> Entry<'a, K, V> {
188    /// Installs an empty default; callers must use a nonallocating `Default` implementation.
189    pub fn or_default(self) -> &'a mut V {
190        self.or_insert(V::default())
191    }
192}

194/// Canonically ordered unique membership using the same fallible storage as an index.
195#[derive(Debug, Eq, PartialEq)]
196pub struct OrderedSet<K>(OrderedMap<K, ()>);
197impl<K> Default for OrderedSet<K> {
198    /// Starts empty without allocating.
199    fn default() -> Self {
200        Self(OrderedMap::default())
201    }
202}
203impl<K: Ord> OrderedSet<K> {
204    /// Starts empty without allocating.
205    #[must_use]
206    pub fn new() -> Self {
207        Self::default()
208    }
209    /// Returns the unique key count and insertion-shift upper bound.
210    #[must_use]
211    pub fn len(&self) -> usize {
212        self.0.len()
213    }
214    /// Checks whether the set is empty.
215    #[must_use]
216    pub fn is_empty(&self) -> bool {
217        self.0.is_empty()
218    }
219    /// Inserts membership after a fallible growth reservation.
220    ///
221    /// # Errors
222    /// Returns allocation failure without adding a key.
223    pub fn insert(&mut self, key: K) -> Result<bool, E> {
224        self.0.insert(key, ()).map(|old| old.is_none())
225    }
226    /// Checks borrowed exact membership without allocating.
227    pub fn contains<Q: Ord + ?Sized>(&self, key: &Q) -> bool
228    where
229        K: Borrow<Q>,
230    {
231        self.0.contains_key(key)
232    }
233    /// Removes exact membership without allocating.
234    pub fn remove<Q: Ord + ?Sized>(&mut self, key: &Q) -> bool
235    where
236        K: Borrow<Q>,
237    {
238        self.0.remove(key).is_some()
239    }
240    /// Borrows all keys in canonical order without allocating.
241    #[must_use]
242    pub fn iter(&self) -> impl DoubleEndedIterator<Item = &K> {
243        self.0.keys()
244    }
245}
246impl<K> IntoIterator for OrderedSet<K> {
247    type Item = K;
248    type IntoIter = std::iter::Map<std::vec::IntoIter<(K, ())>, fn((K, ())) -> K>;
249    /// Moves keys out in canonical order without allocating.
250    fn into_iter(self) -> Self::IntoIter {
251        self.0.entries.into_iter().map(|(k, ())| k)
252    }
253}
254impl<'a, K> IntoIterator for &'a OrderedSet<K> {
255    type Item = &'a K;
256    type IntoIter = std::iter::Map<std::slice::Iter<'a, (K, ())>, fn(&'a (K, ())) -> &'a K>;
257    /// Borrows keys in canonical order without allocating.
258    fn into_iter(self) -> Self::IntoIter {
259        self.0.entries.iter().map(|(k, ())| k)
260    }
261}
