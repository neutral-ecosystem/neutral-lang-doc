3//! Safe fallible ownership primitives. Semantic count, depth and cancellation checks belong to callers.
4
5/// Allocation or capacity failure without formatting, heap-owned error text or partial output.
6#[derive(Clone, Copy, Debug, Eq, PartialEq)]
7pub struct AllocationError;
8
9/// Checks the test-only request-local allocation policy before an actual reservation.
10///
11/// # Errors
12/// Returns an injected failure in instrumented tests; production builds always continue.
13pub fn checkpoint() -> Result<(), AllocationError> {
14    #[cfg(feature = "allocation-testing")]
15    testing::checkpoint()?;
16    Ok(())
17}

19/// Fallible capacity growth shared by vectors and strings, with test-only failure observation.
20pub trait RetainCapacity {
21    /// Reserves additional elements/bytes before changing any retained content.
22    ///
23    /// # Errors
24    /// Returns allocation/capacity failure or a deterministic test failure.
25    fn try_retain(&mut self, additional: usize) -> Result<(), AllocationError>;
26    /// Reserves exactly the additional capacity requested, without changing retained content.
27    ///
28    /// # Errors
29    /// Returns allocation/capacity failure or a deterministic test failure.
30    fn try_retain_exact(&mut self, additional: usize) -> Result<(), AllocationError>;
31}
32
33impl<T> RetainCapacity for Vec<T> {
34    /// Checks the local policy before forwarding to the vector's fallible reservation.
35    fn try_retain(&mut self, additional: usize) -> Result<(), AllocationError> {
36        checkpoint()?;
37        self.try_reserve(additional).map_err(|_| AllocationError)
38    }
39    /// Checks the local policy before forwarding to the vector's exact reservation.
40    fn try_retain_exact(&mut self, additional: usize) -> Result<(), AllocationError> {
41        checkpoint()?;
42        self.try_reserve_exact(additional)
43            .map_err(|_| AllocationError)
44    }
45}
46
47impl RetainCapacity for String {
48    /// Checks the local policy before forwarding to fallible UTF-8 byte retention.
49    fn try_retain(&mut self, additional: usize) -> Result<(), AllocationError> {
50        checkpoint()?;
51        self.try_reserve(additional).map_err(|_| AllocationError)
52    }
53    /// Checks the local policy before forwarding to exact fallible UTF-8 byte retention.
54    fn try_retain_exact(&mut self, additional: usize) -> Result<(), AllocationError> {
55        checkpoint()?;
56        self.try_reserve_exact(additional)
57            .map_err(|_| AllocationError)
58    }
59}

68/// Fallibly constructed immutable shared ownership for successor-only public contracts.
69///
70/// Unlike standard `Arc` construction, this type offers no infallible allocating
71/// constructor or slice conversion. Cloning shares an existing allocation.
72pub struct Shared<T>(triomphe::Arc<T>);
73
74impl<T> Shared<T> {
75    /// Creates shared ownership, returning allocation failure instead of aborting.
76    ///
77    /// # Errors
78    /// Returns [`AllocationError`] if the shared header/data allocation fails.
79    pub fn try_new(value: T) -> Result<Self, AllocationError> {
80        checkpoint()?;
81        triomphe::Arc::try_new(value)
82            .map(Self)
83            .map_err(|_| AllocationError)
84    }
85}
86impl<T> Clone for Shared<T> {
87    /// Shares immutable ownership without allocating another value or header.
88    fn clone(&self) -> Self {
89        Self(self.0.clone())
90    }
91}
92impl<T> std::ops::Deref for Shared<T> {
93    type Target = T;
94    /// Borrows the immutable shared value without copying.
95    fn deref(&self) -> &T {
96        &self.0
97    }
98}
99impl<T> AsRef<T> for Shared<T> {
100    /// Borrows the immutable shared value without copying.
101    fn as_ref(&self) -> &T {
102        self
103    }
104}
105impl<T: std::fmt::Debug> std::fmt::Debug for Shared<T> {
106    /// Delegates to the value's existing safe diagnostic projection.
107    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
108        self.0.fmt(f)
109    }
110}
111impl<T: PartialEq> PartialEq for Shared<T> {
112    /// Compares retained meaning rather than allocator identity.
113    fn eq(&self, other: &Self) -> bool {
114        **self == **other
115    }
116}
117impl<T: Eq> Eq for Shared<T> {}

119/// Copies previously bounded owned data using fallible allocation throughout the implementation.
120///
121/// Callers must preflight hostile recursive values before copying; this trait is
122/// not a validator and does not replace depth, retention or cancellation policy.
123pub trait TryClone: Sized {
124    /// Returns an exact independent copy or an allocation failure, never partial ownership.
125    ///
126    /// # Errors
127    /// Returns [`AllocationError`] when any constituent allocation cannot be reserved.
128    fn try_clone(&self) -> Result<Self, AllocationError>;
129}
130
131/// Copies previously bounded UTF-8 bytes without an infallible string growth operation.
132///
133/// # Errors
134/// Returns [`AllocationError`] when capacity or allocator limits prevent the copy.
135pub fn text(value: &str) -> Result<String, AllocationError> {
136    let mut result = String::new();
137    result
138        .try_retain_exact(value.len())
139        .map_err(|_| AllocationError)?;
140    result.push_str(value);
141    Ok(result)
142}

144/// Moves a value into a fallible standard box without changing public data representations.
145///
146/// The reviewed dependency checks the global allocation result for null; errors
147/// are discarded directly, never formatted or converted to allocating I/O errors.
148///
149/// # Errors
150/// Returns [`AllocationError`] when the box allocation fails.
151pub fn boxed<T>(value: T) -> Result<Box<T>, AllocationError> {
152    checkpoint()?;
153    trybox::or_drop(value).map_err(|_| AllocationError)
154}
155
156impl TryClone for String {
157    /// Copies the exact UTF-8 bytes after one fallible reservation.
158    fn try_clone(&self) -> Result<Self, AllocationError> {
159        text(self)
160    }
161}
162impl TryClone for u64 {
163    /// Copies a nonallocating scalar used by optional length restrictions.
164    fn try_clone(&self) -> Result<Self, AllocationError> {
165        Ok(*self)
166    }
167}
168impl<T: ?Sized> TryClone for &T {
169    /// Copies a borrowed reference without allocating or copying the referent.
170    fn try_clone(&self) -> Result<Self, AllocationError> {
171        Ok(*self)
172    }
173}
174impl<T: TryClone> TryClone for Vec<T> {
175    /// Reserves the outer collection and copies every owned child fallibly.
176    fn try_clone(&self) -> Result<Self, AllocationError> {
177        copy_slice(self)
178    }
179}

181/// Copies a preflighted borrowed slice with fallible outer and constituent retention.
182///
183/// # Errors
184/// Returns [`AllocationError`] when any outer reservation or child copy fails.
185pub fn copy_slice<T: TryClone>(values: &[T]) -> Result<Vec<T>, AllocationError> {
186    let mut result = Vec::new();
187    result
188        .try_retain_exact(values.len())
189        .map_err(|_| AllocationError)?;
190    for value in values {
191        result.push(value.try_clone()?);
192    }
193    Ok(result)
194}
195impl<T: TryClone> TryClone for Box<T> {
196    /// Copies the child and reserves its box, propagating either allocation failure.
197    fn try_clone(&self) -> Result<Self, AllocationError> {
198        boxed((**self).try_clone()?)
199    }
200}
201impl<T: TryClone> TryClone for Option<T> {
202    /// Preserves absence without allocating and copies present values fallibly.
203    fn try_clone(&self) -> Result<Self, AllocationError> {
204        self.as_ref().map(TryClone::try_clone).transpose()
205    }
206}
207impl<A: TryClone, B: TryClone> TryClone for (A, B) {
208    /// Copies both tuple fields without exposing an incomplete tuple.
209    fn try_clone(&self) -> Result<Self, AllocationError> {
210        Ok((self.0.try_clone()?, self.1.try_clone()?))
211    }
212}
213impl<A: TryClone, B: TryClone, C: TryClone> TryClone for (A, B, C) {
214    /// Copies a complete nominal key fallibly, dropping any successful prefix on failure.
215    fn try_clone(&self) -> Result<Self, AllocationError> {
216        Ok((
217            self.0.try_clone()?,
218            self.1.try_clone()?,
219            self.2.try_clone()?,
220        ))
221    }
222}
223impl TryClone for std::convert::Infallible {
224    /// Eliminates the uninhabited closed-reference representation.
225    fn try_clone(&self) -> Result<Self, AllocationError> {
226        match *self {}
227    }
