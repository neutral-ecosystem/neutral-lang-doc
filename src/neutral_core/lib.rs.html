3//! Foundational contracts shared across the Neutral implementation.
4//!
5//! This crate owns source identity, source spans, diagnostics, resource limits,
6//! cancellation, and result classification. It must remain independent of the
7//! compiler, reader, command-line hosts, and ambient host services.
8//!
9//! # Contract boundaries
10//! Source and vocabulary digests identify exact bytes; semantic digests identify
11//! explicitly framed meaning. They are distinct types so callers cannot quietly
12//! replace one identity layer with another. Locations likewise retain original
13//! byte offsets, not decoded-character or editor-display coordinates.
14//!
15//! This crate supplies primitives, not a validator for complete programs. Hosts
16//! own acquisition, higher layers enforce their work budgets, and consumers must
17//! validate external values before treating these contracts as authoritative.

31/// Frozen prefix for v0 SHA-256 digest text.
32const SHA256_TEXT_PREFIX: &str = "sha256:";
33/// Exact character length of one prefixed SHA-256 digest.
34const SHA256_TEXT_LENGTH: usize = SHA256_TEXT_PREFIX.len() + 64;
35/// Frozen outer domain-separation envelope for every Neutral Hash Transcript v1.
36pub const HASH_TRANSCRIPT_ENVELOPE: &str = "neutral-nht-v1";

38/// A typed SHA-256 digest of exact captured source bytes.
39#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
40pub struct SourceContentDigest([u8; 32]);
41
42impl SourceContentDigest {
43    /// Reconstructs a typed digest from an externally validated 32-byte value.
44    #[must_use]
45    pub const fn from_raw_bytes(bytes: [u8; 32]) -> Self {
46        Self(bytes)
47    }
48
49    /// Computes the digest over exactly `bytes`, without normalization.
50    ///
51    /// Whitespace, BOMs, and line-ending changes therefore change source identity
52    /// even when a later compiler considers the program logically equivalent.
53    #[must_use]
54    pub fn from_bytes(bytes: &[u8]) -> Self {
55        let bytes: [u8; 32] = Sha256::digest(bytes).into();
56        Self(bytes)
57    }
58
59    /// Returns the raw SHA-256 bytes for integrity comparison or storage.
60    #[must_use]
61    pub const fn as_bytes(self) -> [u8; 32] {
62        self.0
63    }
64}

85/// A typed SHA-256 digest of exact captured vocabulary-bundle bytes.
86#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
87pub struct VocabularyContentDigest([u8; 32]);
88
89impl VocabularyContentDigest {
90    /// Reconstructs a typed digest from an externally validated 32-byte value.
91    #[must_use]
92    pub const fn from_raw_bytes(bytes: [u8; 32]) -> Self {
93        Self(bytes)
94    }
95
96    /// Computes the digest over exactly `bytes`, without JSON normalization.
97    #[must_use]
98    pub fn from_bytes(bytes: &[u8]) -> Self {
99        let bytes: [u8; 32] = Sha256::digest(bytes).into();
100        Self(bytes)
101    }
102
103    /// Parses the frozen lowercase `sha256:` textual representation.
104    ///
105    /// # Errors
106    ///
107    /// Returns [`DigestTextError::InvalidSha256Text`] for any prefix, length,
108    /// case, or hexadecimal deviation.
109    pub fn parse_text(value: &str) -> Result<Self, DigestTextError> {
110        if value.len() != SHA256_TEXT_LENGTH || !value.starts_with(SHA256_TEXT_PREFIX) {
111            return Err(DigestTextError::InvalidSha256Text);
112        }
113        let mut bytes = [0_u8; 32];
114        let hexadecimal = &value.as_bytes()[SHA256_TEXT_PREFIX.len()..];
115        for (index, byte) in bytes.iter_mut().enumerate() {
116            let offset = index * 2;
117            let high = lowercase_hex_value(hexadecimal[offset])
118                .ok_or(DigestTextError::InvalidSha256Text)?;
119            let low = lowercase_hex_value(hexadecimal[offset + 1])
120                .ok_or(DigestTextError::InvalidSha256Text)?;
121            *byte = (high << 4) | low;
122        }
123        Ok(Self(bytes))
124    }
125
126    /// Returns the raw SHA-256 bytes for typed integrity comparison.
127    #[must_use]
128    pub const fn as_bytes(self) -> [u8; 32] {
129        self.0
130    }
131
132    /// Compares digest bytes without data-dependent early exit.
133    #[must_use]
134    pub fn securely_matches(self, other: Self) -> bool {
135        self.0
136            .iter()
137            .zip(other.0)
138            .fold(0_u8, |difference, (left, right)| {
139                difference | (left ^ right)
140            })
141            == 0
142    }
143}

156/// A malformed frozen digest textual representation.
157#[derive(Clone, Copy, Debug, Eq, PartialEq)]
158pub enum DigestTextError {
159    /// Text was not exactly `sha256:` followed by 64 lowercase hexadecimal digits.
160    InvalidSha256Text,
161}

163/// Converts one lowercase ASCII hexadecimal digit to its numeric value.
164fn lowercase_hex_value(value: u8) -> Option<u8> {
165    match value {
166        b'0'..=b'9' => Some(value - b'0'),
167        b'a'..=b'f' => Some(value - b'a' + 10),
168        _ => None,
169    }
170}

172/// A typed SHA-256 digest produced by a Neutral Hash Transcript v1 domain.
173#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
174pub struct SemanticDigest([u8; 32]);
175
176impl SemanticDigest {
177    /// Hashes a caller-constructed complete NHT transcript without framing it again.
178    /// The caller owns transcript validation, domain separation, and resource bounds.
179    ///
180    /// Use this when an encoder has already written the envelope and domain.
181    /// Passing that transcript to `from_nht` instead would add a second envelope
182    /// and produce a different digest. Hashing alone does not validate meaning.
183    #[must_use]
184    pub fn from_transcript(transcript: &[u8]) -> Self {
185        Self(Sha256::digest(transcript).into())
186    }
187    /// Reconstructs a typed semantic digest from validated transcript bytes.
188    #[must_use]
189    pub const fn from_raw_bytes(bytes: [u8; 32]) -> Self {
190        Self(bytes)
191    }
192
193    /// Hashes `payload` using the frozen NHT-v1 framing and supplied ASCII domain.
194    ///
195    /// # Errors
196    ///
197    /// Returns an error when a tag is non-ASCII or a framed length cannot be
198    /// represented by the frozen transcript integer widths.
199    pub fn from_nht(domain: &str, payload: &[u8]) -> Result<Self, CoreError> {
200        let domain = nht_frame(domain, payload)?;
201        let transcript = nht_frame(HASH_TRANSCRIPT_ENVELOPE, &domain)?;
202        let bytes: [u8; 32] = Sha256::digest(transcript).into();
203        Ok(Self(bytes))
204    }
205
206    /// Returns the raw SHA-256 bytes for integrity comparison or storage.
207    #[must_use]
208    pub const fn as_bytes(self) -> [u8; 32] {
209        self.0
210    }
211}

223/// Frames one NHT-v1 tagged payload using unsigned big-endian lengths.
224///
225/// The layout is `u16(tag length) || ASCII tag || u64(payload length) || payload`.
226/// Explicit lengths distinguish tuples that collide under plain concatenation.
227/// This low-level helper checks representable lengths, not application budgets;
228/// large or hostile inputs need a bounded encoder before reaching this boundary.
229///
230/// # Errors
231///
232/// Returns an error for non-ASCII tags or lengths outside the frozen widths.
233pub fn nht_frame(tag: &str, payload: &[u8]) -> Result<Vec<u8>, CoreError> {
234    if !tag.is_ascii() {
235        return Err(CoreError::InvalidTranscriptTag);
236    }
237    let tag_length = u16::try_from(tag.len()).map_err(|_| CoreError::TranscriptLengthExceeded)?;
238    let payload_length =
239        u64::try_from(payload.len()).map_err(|_| CoreError::TranscriptLengthExceeded)?;
240    let capacity = 2_usize
241        .checked_add(tag.len())
242        .and_then(|length| length.checked_add(8))
243        .and_then(|length| length.checked_add(payload.len()))
244        .ok_or(CoreError::TranscriptLengthExceeded)?;
245    let mut frame = Vec::with_capacity(capacity);
246    frame.extend_from_slice(&tag_length.to_be_bytes());
247    frame.extend_from_slice(tag.as_bytes());
248    frame.extend_from_slice(&payload_length.to_be_bytes());
249    frame.extend_from_slice(payload);
250    Ok(frame)
251}

264/// A checked half-open byte range within one original source byte sequence.
265#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
266pub struct ByteSpan {
267    /// The first included original-byte offset.
268    start: u64,
269    /// The first excluded original-byte offset.
270    end: u64,
271}
272
273impl ByteSpan {
274    /// Creates a half-open span when `start` does not follow `end`.
275    ///
276    /// # Errors
277    ///
278    /// Returns [`CoreError::InvertedSpan`] when `start` follows `end`.
279    pub const fn new(start: u64, end: u64) -> Result<Self, CoreError> {
280        if start > end {
281            return Err(CoreError::InvertedSpan { start, end });
282        }
283        Ok(Self { start, end })
284    }
285
286    /// Returns the first included original-byte offset.
287    #[must_use]
288    pub const fn start(self) -> u64 {
289        self.start
290    }
291
292    /// Returns the first excluded original-byte offset.
293    #[must_use]
294    pub const fn end(self) -> u64 {
295        self.end
296    }
297
298    /// Returns the number of bytes in this span.
299    #[must_use]
300    pub const fn len(self) -> u64 {
301        self.end - self.start
302    }
303
304    /// Returns whether this span contains no bytes.
305    #[must_use]
306    pub const fn is_empty(self) -> bool {
307        self.start == self.end
308    }
309}

367/// Derives a one-based line and original-byte column at `offset`.
368///
369/// CRLF is one line ending, while a lone CR and a lone LF are each one line
370/// ending. The LF byte in a CRLF pair maps to the following line's first column.
371/// Columns count original bytes, so a multibyte UTF-8 character occupies multiple
372/// columns. The end-of-input offset is valid; no decoding or tab expansion occurs.
373/// The scan is linear in `offset`, so repeated bulk lookups belong in an indexed
374/// source-map layer rather than repeated calls to this primitive.
375///
376/// # Errors
377///
378/// Returns [`CoreError::OffsetOutOfBounds`] when `offset` exceeds `bytes`.
379pub fn line_column_at(bytes: &[u8], offset: u64) -> Result<LineColumn, CoreError> {
380    let offset = usize::try_from(offset).unwrap_or(usize::MAX);
381    if offset > bytes.len() {
382        return Err(CoreError::OffsetOutOfBounds {
383            offset: u64::try_from(offset).unwrap_or(u64::MAX),
384            source_bytes: u64::try_from(bytes.len()).unwrap_or(u64::MAX),
385        });
386    }
387
388    let mut line = 1_u64;
389    let mut column = 1_u64;
390    let mut index = 0_usize;
391    while index < offset {
392        match bytes[index] {
393            b'\r' => {
394                line = line.saturating_add(1);
395                column = 1;
396                index += 1;
397                if index < offset && bytes.get(index) == Some(&b'\n') {
398                    index += 1;
399                }
400            }
401            b'\n' => {
402                line = line.saturating_add(1);
403                column = 1;
404                index += 1;
405            }
406            _ => {
407                column = column.saturating_add(1);
408                index += 1;
409            }
410        }
411    }
412    Ok(LineColumn { line, column })
413}

470/// A deterministically ordered, safe diagnostic projection.
471#[derive(Clone, Debug, Eq, PartialEq)]
472pub struct Diagnostic {
473    /// The stable diagnostic identifier.
474    code: DiagnosticCode,
475    /// The owning compiler or validation layer.
476    layer: DiagnosticLayer,
477    /// The user-facing severity classification.
478    severity: DiagnosticSeverity,
479    /// The primary exact source location.
480    primary: SourceLocation,
481    /// Deterministically ordered source locations related to the primary failure.
482    related: Vec<SourceLocation>,
483    /// Deterministic safe values used to render the diagnostic.
484    parameters: Vec<String>,
485    /// Whether detail was bounded by a configured limit.
486    truncated: bool,
487}
488
489impl Ord for Diagnostic {
490    /// Orders diagnostics by source identity, byte range, code, then safe parameters.
491    ///
492    /// Remaining fields break ties to keep ordering consistent with equality.
493    /// Sorting never depends on emission timing or localized rendered messages.
494    fn cmp(&self, other: &Self) -> CompareOrdering {
495        self.primary
496            .cmp(&other.primary)
497            .then_with(|| self.code.cmp(&other.code))
498            .then_with(|| self.related.cmp(&other.related))
499            .then_with(|| self.parameters.cmp(&other.parameters))
500            .then_with(|| self.layer.cmp(&other.layer))
501            .then_with(|| self.severity.cmp(&other.severity))
502            .then_with(|| self.truncated.cmp(&other.truncated))
503    }
504}
505
506impl PartialOrd for Diagnostic {
507    /// Delegates partial comparison to the total canonical diagnostic ordering.
508    fn partial_cmp(&self, other: &Self) -> Option<CompareOrdering> {
509        Some(self.cmp(other))
510    }
511}

805/// A shareable cancellation signal for bounded compilation work.
806///
807/// Clones share one monotonic signal; there is deliberately no reset operation.
808/// This is cooperative cancellation, not thread interruption or rollback. Each
809/// phase must poll at its work boundaries and before authoritative publication.
810#[derive(Clone, Debug, Default)]
811pub struct CancellationToken(Arc<AtomicBool>);
812
813impl CancellationToken {
814    /// Creates a cancellation signal that is initially clear.
815    #[must_use]
816    pub fn new() -> Self {
817        Self::default()
818    }
819
820    /// Requests cancellation for every holder of this signal.
821    ///
822    /// Release/acquire ordering synchronizes observation of the request; callers
823    /// must still explicitly poll and must not infer successful work from a token.
824    pub fn cancel(&self) {
825        self.0.store(true, Ordering::Release);
826    }
827
828    /// Returns whether cancellation has been requested.
829    #[must_use]
830    pub fn is_cancelled(&self) -> bool {
831        self.0.load(Ordering::Acquire)
832    }
833}

835/// A stable non-authoritative outcome class.
836#[derive(Clone, Copy, Debug, Eq, PartialEq)]
837pub enum ResultClass {
838    /// Capture input could not be accepted.
839    Capture,
840    /// A structural limit was exceeded.
841    Resource,
842    /// Captured source was rejected by decoding, lexing, layout, or parsing.
843    Syntax,
844    /// Parsed source was rejected by name, type, or value semantics.
845    Semantics,
846    /// A typed identity-reference target or edge was invalid.
847    Reference,
848    /// A captured vocabulary contract or payload was invalid.
849    Vocabulary,
850    /// The caller cancelled bounded work.
851    Cancellation,
852    /// A compiler invariant failed without authoritative output.
853    Internal,
854}

856/// A failure while constructing foundational contracts.
857#[derive(Clone, Debug, Eq, PartialEq)]
858pub enum CoreError {
859    /// A half-open span ended before it began.
860    InvertedSpan {
861        /// The supplied first included byte offset.
862        start: u64,
863        /// The supplied first excluded byte offset.
864        end: u64,
865    },
866    /// A requested original-byte offset was outside the captured source.
867    OffsetOutOfBounds {
868        /// The requested original-byte offset.
869        offset: u64,
870        /// The exact captured source byte count.
871        source_bytes: u64,
872    },
873    /// A diagnostic code was empty or contained non-ASCII text.
874    InvalidDiagnosticCode,
875    /// A required deterministic limit was zero.
876    ZeroLimit,
877    /// An NHT tag contained non-ASCII text.
878    InvalidTranscriptTag,
879    /// An NHT tag, payload, or combined frame exceeded its fixed-width length.
880    TranscriptLengthExceeded,
881}
